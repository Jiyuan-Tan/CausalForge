Encode a stratum label by \(k=0\) at rank zero and by \(k=(r,s)\) on \(\mathcal M_r(s)\).  Let
\[
 \mathbb K=\{0\}\cup\bigl(\{1,2\}\times(0,B_{\max}]\bigr),
 \qquad
 \overline{\mathbb K}=\{0\}\cup\bigl(\{1,2\}\times[0,B_{\max}]\bigr),
\]
where the three rank coordinates are topologically disjoint, and define the complete labeled legal graph
\[
 \mathcal X^\circ
 =\{(0,0,0)\}\cup
 \left\{(P_\Omega M,\tau(M),(r,s)):
 r\in\{1,2\},\ s\in(0,B_{\max}],\ M\in\mathcal M_r(s)\right\}.
\]
Its compactification is
\[
 \overline{\mathcal X}=\operatorname{cl}_{\mathbb R^{12}\times[-B_{\max},B_{\max}]\times\overline{\mathbb K}}
 \mathcal X^\circ.
\]
Thus the closure retains the rank label and the strength coordinate; in particular, limits with \(s\downarrow0\) are not identified with rank zero or with a different rank label.  For \(k\in\mathbb K\), write \(\mathcal X_k^\circ\) for the corresponding original legal fiber.

Fix once and for all an ordering of dyadic boxes and a cylindrical-algebraic-decomposition sampler that returns one lifted witness \((M,P_\Omega M,\tau(M))\) from every nonempty semialgebraic box intersection.  Put
\[
 \eta_m=2B_{\max}2^{-m},
 \qquad
 \mathcal S_m=\{jB_{\max}2^{-m}:1\leq j\leq2^m\}.
\]
For every \(k=0\) and every \(k=(r,s)\) with \(s\in\mathcal S_m\), partition the \((\mu,t)\)-box \([-B_{\max},B_{\max}]^{13}\) into the ordered dyadic boxes of side at most \(\eta_m\).  From every box meeting \(\mathcal X_k^\circ\), retain the projected point returned by the fixed sampler, and retain all points chosen at earlier levels.  Denote this nested finite fiber net by \(\mathcal N_m(k)\), and set
\[
 \Theta_m=\mathcal N_m(0)\cup
 \bigcup_{r\in\{1,2\}}\ \bigcup_{s\in\mathcal S_m}\mathcal N_m((r,s)).
\]
For an arbitrary query \(k=(r,s)\in\mathbb K\), the query-augmented grid is
\[
 \Theta_m[k]=\Theta_m\cup\mathcal N_m(k),
\]
where \(\mathcal N_m(k)\) is obtained by applying the same box sampler to the exact fiber \(\mathcal X_k^\circ\).

For any finite labeled legal grid \(\Theta_N=\{(\mu_i,t_i,k(i)):1\leq i\leq N\}\), let
\[
 \mathcal T_N=\{t_i:1\leq i\leq N\},
 \qquad
 \mathcal A_N=\{[a,b]:a,b\in\mathcal T_N,\ a\leq b\},
\]
and let
\[
 f_i(y)=\left(\frac{n}{2\pi\sigma^2}\right)^6
 \exp\!\left\{-\frac{n}{2\sigma^2}\lVert y-\mu_i\rVert_2^2\right\}.
\]
Let \(\mathcal P_N\) be the Borel probability kernels on \(\mathcal A_N\), and define
\[
 Q_i(p)=\int\sum_{A\in\mathcal A_N}\mathbf1\{t_i\in A\}p_A(y)f_i(y)\,dy,
 \qquad
 R_i(p)=\int\sum_{A\in\mathcal A_N}\operatorname{len}(A)p_A(y)f_i(y)\,dy.
\]
For \(\beta\in(0,1/2)\) and each represented label \(k\), put
\[
 L_{k,N}(\beta)=
 \inf_{p\in\mathcal P_N:\,Q_j(p)\geq1-\beta\ \forall j}
 \max_{i:k(i)=k}R_i(p),
 \qquad L_{k,N}=L_{k,N}(\alpha).
\]
The grid is normalization-legal at level \(\beta\) when \(L_{k,N}(\beta)>0\) for every represented \(k\).  On such a grid define
\[
 \kappa_N(\beta)=\inf_{\substack{\kappa\geq0,\ p\in\mathcal P_N}}
 \left\{\kappa:Q_i(p)\geq1-\beta,\quad
 R_i(p)\leq\kappa L_{k(i),N}(\beta)\quad(1\leq i\leq N)\right\},
 \qquad \kappa_N^*=\kappa_N(\alpha).
\]

The Gaussian total-variation transfer allowance and the stricter canonical level are
\[
 v_m=\frac{\sqrt{3n}\,\eta_m}{\sigma},
 \qquad
 m_0=\min\{m:v_m<\alpha\},
 \qquad
 \alpha_m=\alpha-v_m\quad(m\geq m_0).
\]
For \(k\in\mathbb K\), let \(L_{k,m}^{\mathrm{can}}\) be the \(k\)-frontier on \(\Theta_m[k]\) computed at noncoverage \(\alpha_m\).  Let \(\kappa_m^{\mathrm{can}}\) be the simultaneous factor on \(\Theta_m\), also computed at \(\alpha_m\).  These are finite-dimensional convex programs, and their likelihood duals are obtained by replacing \(\alpha\) by \(\alpha_m\) in the finite-grid dual.

For a deterministic connected interval \(C\), define
\[
 R_0(C)=E_{P_0}[\operatorname{len}C(Y)],
 \qquad
 R_{r,s}(C)=\sup_{M\in\mathcal M_r(s)}E_{P_M}[\operatorname{len}C(Y)].
\]
The sharp deterministic simultaneous excess is
\[
 \kappa_\infty^*=
 \inf_{C\in\mathfrak C_\alpha(\mathcal U)}
 \max\left\{
 \frac{R_0(C)}{\mathfrak L_0^{\mathcal U}(n)},
 \sup_{r\in\{1,2\}}\sup_{s\in(0,B_{\max}]}
 \frac{R_{r,s}(C)}{\mathfrak L_{r,s}^{\mathcal U}(n)}
 \right\}.
\]
Define the lower-semicontinuous boundary extension of the stratum frontier by
\[
 \overline L(0)=\mathfrak L_0^{\mathcal U}(n),
 \qquad
 \overline L(r,s)=\lim_{\varepsilon\downarrow0}
 \inf_{\substack{u\in(0,B_{\max}]\\ |u-s|<\varepsilon}}
 \mathfrak L_{r,u}^{\mathcal U}(n),
 \quad s\in[0,B_{\max}].
\]
For \(x=(\mu,t,k)\in\overline{\mathcal X}\), write
\[
 f_x(y)=\left(\frac{n}{2\pi\sigma^2}\right)^6
 \exp\!\left\{-\frac{n}{2\sigma^2}\lVert y-\mu\rVert_2^2\right\}.
\]
Let \(\mathcal M_+(\overline{\mathcal X})\) denote the finite nonnegative regular Borel measures on \(\overline{\mathcal X}\).  The continuum measure dual is
\[
\begin{split}
 \mathfrak D_\alpha^\infty
 =\sup_{\substack{\lambda,w\in\mathcal M_+(\overline{\mathcal X})\\
             \int\overline L(k(x))\,dw(x)=1}}
 \bigg[&(1-\alpha)\lambda(\overline{\mathcal X})\\
 &+\int_{\mathbb R^{12}}
 \inf_{-B_{\max}\leq a\leq b\leq B_{\max}}
 \left\{(b-a)\int f_x(y)\,dw(x)
 -\int f_x(y)\mathbf1\{t(x)\in[a,b]\}\,d\lambda(x)\right\}dy\bigg].
\end{split}
\]

For the direct profile baseline, put
\[
 \overline{\mathcal K}=\{(\mu,t):(\mu,t,k)\in\overline{\mathcal X}\text{ for some }k\},
 \qquad
 \rho_\alpha=\frac{\sigma}{\sqrt n}\sqrt{q_{12,1-\alpha}},
\]
and define
\[
 T(y)=\{t:(\mu,t)\in\overline{\mathcal K},\ \lVert y-\mu\rVert_2\leq\rho_\alpha\},
 \qquad
 C_{\mathrm{prof}}(y)=
 \begin{cases}
 [\inf T(y),\sup T(y)],&T(y)\neq\varnothing,\\
 \{0\},&T(y)=\varnothing.
 \end{cases}
\]
The simultaneous-inversion handle consists of the canonical grids \(\Theta_m\), the query-augmented frontier programs \(L_{k,m}^{\mathrm{can}}\), the finite factors \(\kappa_m^{\mathrm{can}}\), the sharp deterministic excess \(\kappa_\infty^*\), its measure-dual candidate \(\mathfrak D_\alpha^\infty\), and the profile baseline \(C_{\mathrm{prof}}\).
