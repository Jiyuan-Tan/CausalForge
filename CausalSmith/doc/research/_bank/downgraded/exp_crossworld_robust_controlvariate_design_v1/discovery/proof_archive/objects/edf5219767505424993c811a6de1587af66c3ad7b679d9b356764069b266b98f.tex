\[
\inf_{B\in\mathcal A_{\mathrm{mm}}(\xi_0)\cup\mathcal A_0(\xi_0)}
\left\{\lambda_{\min}(n^{-1}S_B)-\kappa\right\}\ge\eta.
\]
For every sequence \((\xi_j)_{j\ge1}\subset\mathfrak X_\kappa\) converging to \(\xi_0\) in the relative topology of def:lawful-instance-class,
\[
\liminf_{j\to\infty}\Delta(\xi_j)\ge\Delta(\xi_0).
