\[
 \sup_{(P,\bar g_n)\in\mathcal Q_{\mathrm{spec},n}}
 P(\mathcal E_{1n}^c\cup\mathcal E_{2n}^c)=o(r_n).
\]
On \(\mathcal E_{1n}\), the localized selector is nonempty,
\[
 \inf_{v\in\mathcal I,z\in C_{\widehat j_n}}|F_{v,h_n}(z)|
 \ge\rho_Mc_V/2,

\[
 \widehat\tau_n(v)-\Theta^{\widehat j_n}_{v,h_n}
 =n_2^{-1}\sum_{i\in I_2}\psi^{\widehat j_n}_{v,h_n}(O_i)

In addition, the two actual fallback probabilities are \(o(r_n)\) uniformly and
\[
 E_P\!\left[\widehat c_{1-\alpha}\sup_{v\in\mathcal I}\widehat s_n(v)\,; \text{both actual success events}\right]\le C\sqrt{\log n}.
