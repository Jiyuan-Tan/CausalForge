Fix an arbitrary \(P\in\mathfrak P_{\mathcal D}\) and write \(\theta_d(P)=\mu_{d,1}-\mu_{d,0}\). Conditional on the independent full-catalogue pilot, suppose that the purchased experimental and observational samples are mutually independent and consist of independent and identically distributed observations from \(P(\cdot\mid G=E)\) and \(P(\cdot\mid G=O)\), respectively.

Use a fixed integer \(L\ge2\) of observational-only cross-fitting folds. For deterministic \(d_B\in\mathcal D\) and \(\alpha_B\in(0,1)\), put
\[
 e_B=\left\lfloor\frac{\alpha_BB}{c_{E,d_B}}\right\rfloor,
 \qquad
 o_B=\left\lfloor\frac{(1-\alpha_B)B}{c_{O,d_B}}\right\rfloor,
 \qquad n_B=e_B+o_B,
\]
and let \(P_{B,d_B,\alpha_B}^{\circ}\) have source probabilities \(e_B/n_B\) and \(o_B/n_B\) and source-conditional laws \(P(\cdot\mid G=E)\) and \(P(\cdot\mid G=O)\). Write \(\|\cdot\|_{2,B}^{\circ}=\|\cdot\|_{L_2(P_{B,d_B,\alpha_B}^{\circ})}\). Along every deterministic sequence satisfying
\[
 |\alpha_B-\alpha_{d_B}^*(P)|\longrightarrow0, \tag{1}
\]
suppose, uniformly over \(1\le\ell\le L\), that with \(\Delta h_{B,\ell}=\widehat h_{d_B,B}^{(-\ell)}-h_{d_B}\) and \(\Delta q_{B,\ell}=\widehat q_{d_B,B}^{(-\ell)}-q_{d_B}\),
\[
 \|T_{d_B}\Delta h_{B,\ell}\|_{2,B}^{\circ}=O_P(\delta_{h,B}),
 \qquad
 \|\Delta h_{B,\ell}\|_{2,B}^{\circ}=O_P(\rho_{h,B}), \tag{2}
\]
\[
 \|T_{d_B}^{\star}\Delta q_{B,\ell}\|_{2,B}^{\circ}=O_P(\delta_{q,B}),
 \qquad
 \|\Delta q_{B,\ell}\|_{2,B}^{\circ}=O_P(\rho_{q,B}), \tag{3}
\]
where all four deterministic rates are \(o(1)\) and
\[
 \min\{\delta_{h,B}\rho_{q,B},\rho_{h,B}\delta_{q,B}\}=o(B^{-1/2}). \tag{4}
\]
Suppose also that the foldwise empirical multipliers obey
\[
\begin{aligned}
 \max_{1\le\ell\le L}\bigl(&
 \|\mathbf 1\{G=O\}\Delta q_{B,\ell}(Y-h_{d_B})\|_{2,B}^{\circ}
 +\|\mathbf 1\{G=O\}q_{d_B}\Delta h_{B,\ell}\|_{2,B}^{\circ}\\
 &+\|\mathbf 1\{G=O\}\Delta q_{B,\ell}\Delta h_{B,\ell}\|_{2,B}^{\circ}
 \bigr)=o_P(1),
\end{aligned} \tag{5}
\]
with every bridge evaluated at its design-specific arguments. For each \(o_B\), partition the observational indices deterministically into balanced folds \(O_{B,1},\ldots,O_{B,L}\). Train \(\widehat h_{d_B,B}^{(-\ell)}\) on \(O_B\setminus O_{B,\ell}\) and \(\widehat q_{d_B,B}^{(-\ell)}\) on \(E_B\cup(O_B\setminus O_{B,\ell})\), with any auxiliary training randomness independent of evaluation observations; assign every degenerate fit a fixed measurable value. Evaluate every experimental observation with every one of the \(L\) outcome-bridge fits.

For \(a\in\{0,1\}\), put
\[
 e_{B,a}=\sum_{i\in E_B}\mathbf 1\{A_i=a\},
 \qquad
 o_{B,\ell,a}=\sum_{i\in O_{B,\ell}}\mathbf 1\{A_i=a\},
\]
and, on positive denominators, define
\[
 \widehat p_{E,a}=e_{B,a}/e_B,
 \qquad
 \widehat p_{O,\ell,a}=o_{B,\ell,a}/|O_{B,\ell}|,
\]
\[
 \widehat\mu_{d_B,a}^{(-\ell)}
 =\frac1{e_{B,a}}\sum_{i\in E_B}\mathbf 1\{A_i=a\}
 \widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i),
\]
\[
\begin{aligned}
 \widehat\mu_{B,d_B}(a)
 &=\frac1L\sum_{\ell=1}^L\widehat\mu_{d_B,a}^{(-\ell)}\\
 &\quad+\frac1L\sum_{\ell=1}^L\frac1{o_{B,\ell,a}}
 \sum_{i\in O_{B,\ell}}\mathbf 1\{A_i=a\}
 \widehat q_{d_B,B}^{(-\ell)}(S_{2d_B,i},S_{1d_B,i},a,X_i)\\
 &\hspace{36mm}\times
 \{Y_i-\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)\},
\end{aligned} \tag{6}
\]
and \(\widehat\tau_B(d_B,\alpha_B)=\widehat\mu_{B,d_B}(1)-\widehat\mu_{B,d_B}(0)\). Define the exact arm-normalized plug-in scores by
\[
\begin{aligned}
 \widehat\phi_{E,d_B,B,i}
 =\frac1L\sum_{\ell=1}^L\sum_{a=0}^1(-1)^{1-a}
 \frac{\mathbf 1\{A_i=a\}}{\widehat p_{E,a}}
 \{&\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)
 -\widehat\mu_{d_B,a}^{(-\ell)}\},
\end{aligned} \tag{7}
\]
for \(i\in E_B\), and
\[
\begin{aligned}
 \widehat\phi_{O,d_B,B,i}
 &=\sum_{a=0}^1(-1)^{1-a}
 \frac{\mathbf 1\{A_i=a\}}{\widehat p_{O,\ell,a}}
 \widehat q_{d_B,B}^{(-\ell)}(S_{2d_B,i},S_{1d_B,i},a,X_i)\\
 &\hspace{22mm}\times
 \{Y_i-\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)\},
 \qquad i\in O_{B,\ell}.
\end{aligned} \tag{8}
\]
Set \(\widetilde V_{E,d_B,B}=e_B^{-1}\sum_{i\in E_B}\widehat\phi_{E,d_B,B,i}^2\) and \(\widetilde V_{O,d_B,B}=o_B^{-1}\sum_{i\in O_B}\widehat\phi_{O,d_B,B,i}^2\). Every quantity in (6)--(8), including the two variance estimators, is assigned zero whenever one of its displayed denominators is zero.

Let \(m\to\infty\) and \(B\to\infty\), assume \(\underline v_m=o(1)\), and assume that \(\mathcal P_m\) is independent of the main study. Suppose
\[
 \max_{d\in\mathcal D}\max\{|\widehat V_{E,d}-V_{E,d}|,
 |\widehat V_{O,d}-V_{O,d}|\}=O_P(r_{V,m}),
 \qquad r_{V,m}=o(1), \tag{9}
\]
and
\[
 \max_{d\in\mathcal D,\,g\in\{E,O\}}
 |\widetilde V_{g,d,B}-V_{g,d}|=O_P(r_{\widetilde V,B}),
 \qquad r_{\widetilde V,B}=o(1). \tag{10}
\]
Implement (6)--(8) at \((\widehat d_m,\widehat\alpha_m)\), and define
\[
\begin{aligned}
 \mathsf D_B
 ={}&\{N_{E,B}\ge1,N_{O,B}\ge1\}
 \cap\bigcap_{a=0}^1\{e_{B,a}\ge1\}\\
 &{}\cap\bigcap_{\ell=1}^L\bigcap_{a=0}^1\{o_{B,\ell,a}\ge1\}.
\end{aligned} \tag{11}
\]
With all zero-denominator fractions assigned zero, set
\[
 \widehat s_B^2
 =\mathbf 1\{\mathsf D_B\}B\left\{
 \frac{\widetilde V_{E,\widehat d_m,B}}{N_{E,B}}
 +\frac{\widetilde V_{O,\widehat d_m,B}}{N_{O,B}}
 \right\},
 \qquad
 \widehat s_B=(\widehat s_B^2)^{1/2}, \tag{12}
\]
and \(\mathsf G_{\mathrm{tot}}=\mathsf D_B\cap\{\widehat s_B^2>0\}\). Let \(s_{0,B}\) be the nonnegative square root of
\[
 s_{0,B}^2
 =B\left\{\frac{V_{E,\widehat d_m}}{N_{E,B}}
 +\frac{V_{O,\widehat d_m}}{N_{O,B}}\right\} \tag{13}
\]
when both source counts are positive, and set \(s_{0,B}=0\) otherwise. On \(\mathsf G_{\mathrm{tot}}\), set \(\widehat\tau_B=\widehat\tau_B(\widehat d_m,\widehat\alpha_m)\), define
\[
 \mathsf T_B=\frac{\sqrt B\{\widehat\tau_B-\theta_{\widehat d_m}(P)\}}{\widehat s_B},
 \qquad
 I_B=\left[\widehat\tau_B-z_{1-\eta/2}\frac{\widehat s_B}{\sqrt B},
 \widehat\tau_B+z_{1-\eta/2}\frac{\widehat s_B}{\sqrt B}\right], \tag{14}
\]
and on \(\mathsf G_{\mathrm{tot}}^c\) set \(\widehat\tau_B=0\), \(\widehat s_B=0\), \(\mathsf T_B=0\), and \(I_B=[0,0]\). Then
\[
 P(\mathsf G_{\mathrm{tot}})\longrightarrow1, \tag{15}
\]
\[
 s_{0,B}^2-\mathcal V_{\widehat d_m}^*(P)=o_P(1),
 \qquad
 \widehat s_B^2-\mathcal V_{\widehat d_m}^*(P)=o_P(1), \tag{16}
\]
\[
 \mathsf T_B\rightsquigarrow\mathcal N(0,1),
 \qquad
 P\{\theta_{\widehat d_m}(P)\in I_B\}\longrightarrow1-\eta. \tag{17}
\]
If, in addition, \(\theta_d(P)=\tau\) for every \(d\in\mathcal D\), then \(\mathsf T_B\) is the same statistic centered at \(\tau\), \(\sqrt B(\widehat\tau_B-\tau)/\widehat s_B\rightsquigarrow\mathcal N(0,1)\) with the stipulated zero extension, and \(P\{\tau\in I_B\}\to1-\eta\).
