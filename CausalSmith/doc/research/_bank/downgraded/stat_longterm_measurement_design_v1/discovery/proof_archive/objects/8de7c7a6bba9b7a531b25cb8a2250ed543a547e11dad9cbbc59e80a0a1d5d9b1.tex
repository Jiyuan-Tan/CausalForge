Fix \(P\in\mathfrak P_{\mathcal D}\) in the common-identification subclass on which the two latent bridge equations used by the two Imbens--Kallus--Mao--Wang identification routes hold for every \(d\in\mathcal D\), and on which
\[
 \mu_{d,1}-\mu_{d,0}=\tau,\qquad d\in\mathcal D.             \tag{1}
\]
Use a fixed integer \(L\ge2\) of observational-only cross-fitting folds.  For a deterministic design \(d_B\in\mathcal D\) and allocation \(\alpha_B\in(0,1)\), define
\[
 e_B=\left\lfloor\frac{\alpha_BB}{c_{E,d_B}}\right\rfloor,
 \qquad
 o_B=\left\lfloor\frac{(1-\alpha_B)B}{c_{O,d_B}}\right\rfloor,
 \qquad n_B=e_B+o_B,                                        \tag{2}
\]
and let \(P_{B,d_B,\alpha_B}^{\circ}\) have source probabilities \(e_B/n_B\) and \(o_B/n_B\) and source-conditional laws \(P(\cdot\mid G=E)\) and \(P(\cdot\mid G=O)\).  Write \(\|\cdot\|_{2,B}^{\circ}=\|\cdot\|_{L_2(P_{B,d_B,\alpha_B}^{\circ})}\).  Suppose that, along every deterministic sequence satisfying
\[
 |\alpha_B-\alpha_{d_B}^*(P)|\longrightarrow0,               \tag{3}
\]
the foldwise errors \(\Delta h_{B,\ell}=\widehat h_{d_B,B}^{(-\ell)}-h_{d_B}\) and \(\Delta q_{B,\ell}=\widehat q_{d_B,B}^{(-\ell)}-q_{d_B}\) satisfy, uniformly over \(1\le\ell\le L\),
\[
 \|T_{d_B}\Delta h_{B,\ell}\|_{2,B}^{\circ}=O_P(\delta_{h,B}),
 \qquad
 \|\Delta h_{B,\ell}\|_{2,B}^{\circ}=O_P(\rho_{h,B}),       \tag{4}
\]
\[
 \|T_{d_B}^{\star}\Delta q_{B,\ell}\|_{2,B}^{\circ}=O_P(\delta_{q,B}),
 \qquad
 \|\Delta q_{B,\ell}\|_{2,B}^{\circ}=O_P(\rho_{q,B}),       \tag{5}
\]
where all four deterministic rates are \(o(1)\) and
\[
 \min\{\delta_{h,B}\rho_{q,B},\rho_{h,B}\delta_{q,B}\}
 =o(B^{-1/2}).                                                \tag{6}
\]
Suppose also that the foldwise empirical multipliers satisfy
\[
\begin{aligned}
 \max_{1\le\ell\le L}\bigl(&
 \|\mathbf 1\{G=O\}\Delta q_{B,\ell}(Y-h_{d_B})\|_{2,B}^{\circ}
 +\|\mathbf 1\{G=O\}q_{d_B}\Delta h_{B,\ell}\|_{2,B}^{\circ}\\
 &+\|\mathbf 1\{G=O\}\Delta q_{B,\ell}\Delta h_{B,\ell}\|_{2,B}^{\circ}
 \bigr)=o_P(1).
\end{aligned}                                               \tag{7}
\]
where every bridge in (7) is evaluated at its design-specific arguments.  Condition (7) is only an empirical multiplier-stability condition; it does not replace the weak--strong product condition (6).

For each \(o_B\), partition the observational indices deterministically into \(L\) folds \(O_{B,1},\ldots,O_{B,L}\) whose sizes differ by at most one.  The fit \(\widehat h_{d_B,B}^{(-\ell)}\) is trained on \(O_B\setminus O_{B,\ell}\), and \(\widehat q_{d_B,B}^{(-\ell)}\) is trained on \(E_B\cup(O_B\setminus O_{B,\ell})\); any auxiliary training randomness is independent of the evaluation observations.  Every experimental observation is evaluated with every one of the \(L\) outcome-bridge fits.  Put
\[
 e_{B,a}=\sum_{i\in E_B}\mathbf 1\{A_i=a\},
 \qquad
 o_{B,\ell,a}=\sum_{i\in O_{B,\ell}}\mathbf 1\{A_i=a\},     \tag{8}
\]
and, when their denominators are positive, define
\[
 \widehat p_{E,a}=\frac{e_{B,a}}{e_B},
 \qquad
 \widehat p_{O,\ell,a}=\frac{o_{B,\ell,a}}{|O_{B,\ell}|},   \tag{9}
\]
\[
 \widehat\mu_{d_B,a}^{(-\ell)}
 =\frac1{e_{B,a}}\sum_{i\in E_B}\mathbf 1\{A_i=a\}
 \widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i). \tag{10}
\]
The arm-normalized estimator is
\begin{align}
 \widehat\mu_{B,d_B}(a)
 &=\frac1L\sum_{\ell=1}^L\widehat\mu_{d_B,a}^{(-\ell)}\notag\\
 &\quad+\frac1L\sum_{\ell=1}^L\frac1{o_{B,\ell,a}}
 \sum_{i\in O_{B,\ell}}\mathbf 1\{A_i=a\}
 \widehat q_{d_B,B}^{(-\ell)}(S_{2d_B,i},S_{1d_B,i},a,X_i)\notag\\
 &\hspace{44mm}\times
 \{Y_i-\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)\},
                                                                  \tag{11}\\
 \widehat\tau_B(d_B,\alpha_B)&=\widehat\mu_{B,d_B}(1)-\widehat\mu_{B,d_B}(0).
                                                                  \tag{12}
\end{align}
For \(i\in E_B\) and \(i\in O_{B,\ell}\), respectively, define the exact source-indexed plug-in scores
\begin{align}
 \widehat\phi_{E,d_B,i}
 &=\frac1L\sum_{\ell=1}^L\sum_{a=0}^1(-1)^{1-a}
 \frac{\mathbf 1\{A_i=a\}}{\widehat p_{E,a}}
 \{\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)
   -\widehat\mu_{d_B,a}^{(-\ell)}\},                       \tag{13}\\
 \widehat\phi_{O,d_B,i}
 &=\sum_{a=0}^1(-1)^{1-a}
 \frac{\mathbf 1\{A_i=a\}}{\widehat p_{O,\ell,a}}
 \widehat q_{d_B,B}^{(-\ell)}(S_{2d_B,i},S_{1d_B,i},a,X_i)\notag\\
 &\hspace{22mm}\times
 \{Y_i-\widehat h_{d_B,B}^{(-\ell)}(S_{3d_B,i},S_{2d_B,i},a,X_i)\}.
                                                                  \tag{14}
\end{align}
Set \(\widetilde V_{E,d_B,B}=e_B^{-1}\sum_{i\in E_B}\widehat\phi_{E,d_B,i}^2\) and \(\widetilde V_{O,d_B,B}=o_B^{-1}\sum_{i\in O_B}\widehat\phi_{O,d_B,i}^2\).  Every quantity in (9)--(14), including the two variance estimators, is assigned the value zero whenever one of its displayed denominators is zero.

Let \(m\to\infty\) and \(B\to\infty\), assume \(\underline v_m=o(1)\), and assume that \(\mathcal P_m\) is independent of the main study.  Suppose
\[
 \max_{d\in\mathcal D}\max\{|\widehat V_{E,d}-V_{E,d}|,
 |\widehat V_{O,d}-V_{O,d}|\}=O_P(r_{V,m}),
 \qquad r_{V,m}=o(1),                                      \tag{15}
\]
and
\[
 \max_{d\in\mathcal D,\,g\in\{E,O\}}
 |\widetilde V_{g,d,B}-V_{g,d}|=O_P(r_{\widetilde V,B}),
 \qquad r_{\widetilde V,B}=o(1).                           \tag{16}
\]
Implement (8)--(14) at \((\widehat d_m,\widehat\alpha_m)\), and let
\[
\begin{aligned}
 \mathsf D_B
 ={}&\{N_{E,B}\ge1,\ N_{O,B}\ge1\}\\
 &{}\cap\bigcap_{a=0}^1\{e_{B,a}\ge1\}
 \cap\bigcap_{\ell=1}^L\bigcap_{a=0}^1\{o_{B,\ell,a}\ge1\}.
\end{aligned}
\]
Under the zero-denominator convention, define the total variance estimator by
\[
 \widehat s_B^2
 =\mathbf 1\{\mathsf D_B\}B\left\{
 \frac{\widetilde V_{E,\widehat d_m,B}}{N_{E,B}}
 +\frac{\widetilde V_{O,\widehat d_m,B}}{N_{O,B}}
 \right\},                                                  \tag{17}
\]
and define
\[
 \mathsf G_{\mathrm{tot}}
 =\mathsf D_B\cap\{\widehat s_B^2>0\}.                      \tag{18}
\]
and set
\[
 s_{0,B}^2
 =B\left\{
 \frac{V_{E,\widehat d_m}}{N_{E,B}}
 +\frac{V_{O,\widehat d_m}}{N_{O,B}}
 \right\}                                                   \tag{19}
\]
when the two source counts are positive and set \(s_{0,B}^2=0\) otherwise.  The event \(\mathsf D_B\) contains every random-denominator event introduced by (9)--(14).  On \(\mathsf G_{\mathrm{tot}}\), take \(\widehat s_B=(\widehat s_B^2)^{1/2}\) to be the nonnegative root and use the estimator (12); on \(\mathsf G_{\mathrm{tot}}^c\), set \(\widehat\tau_B=0\), \(\widehat s_B=0\), the studentized statistic equal to zero, and \(I_B=[0,0]\).  Then
\[
 P(\mathsf G_{\mathrm{tot}})\longrightarrow1,               \tag{20}
\]
\[
 s_{0,B}^2-\mathcal V_{\widehat d_m}^*(P)=o_P(1),
 \qquad
 \widehat s_B^2-\mathcal V_{\widehat d_m}^*(P)=o_P(1),      \tag{21}
\]
and
\[
 \frac{\sqrt B(\widehat\tau_B-\tau)}{\widehat s_B}
 \rightsquigarrow\mathcal N(0,1),                           \tag{22}
\]
where the ratio in (22) has the stated zero extension.  If \(z_{1-\eta/2}\) is the \((1-\eta/2)\)-quantile of the standard normal distribution, define on \(\mathsf G_{\mathrm{tot}}\)
\[
 I_B=\left[
 \widehat\tau_B-z_{1-\eta/2}\frac{\widehat s_B}{\sqrt B},
 \widehat\tau_B+z_{1-\eta/2}\frac{\widehat s_B}{\sqrt B}
 \right]                                                    \tag{23}
\]
and use the extension above on its complement.  Then \(P\{\tau\in I_B\}\to1-\eta\).
