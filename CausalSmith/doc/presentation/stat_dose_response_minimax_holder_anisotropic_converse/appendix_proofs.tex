\begin{proof}[Proof of \cref{obj:thm:sharp-pointwise-lower-bound}]
\begin{enumerate}
\item \emph{Instantiate the oracle floor.} The theorem hypotheses provide the inputs required by \cref{obj:lem:oracle-dose-regression-lower-all-beta} for the same covariate dimension \(d\) and the same constants \(\alpha,\beta,s,M,c_0,\varepsilon_0,t_0\). The positive-smoothness assumptions give \(\alpha>0\), \(\beta>0\), and \(s>0\). The regime-constants assumptions give \(M>0\), \(c_0>0\), \(t_0\in(0,1)\), \(\varepsilon_0\in(0,1/2)\), and the interior-window condition recorded in \cref{obj:ass:interior-dose}. The strict-slack baseline assumption is \cref{obj:ass:baseline-submodel-slack} for the tuple \((d,\beta,s,M,c_0,\varepsilon_0,t_0)\), matching the slack-baseline input of the lemma.
% lean: sharp_pointwise_lower_bound

\item \emph{Apply the oracle floor.} Applying \cref{obj:lem:oracle-dose-regression-lower-all-beta} with these inputs yields a constant \(c_{\mathrm{or}}>0\) such that, for all sufficiently large \(n\in\mathbb N\),
\[
R_n\!\left(
    \mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),
    t_0
  \right)
\ge
c_{\mathrm{or}}\,n^{-2\alpha/(2\alpha+1)} .
\]
Here the class is the model class of \cref{obj:def:holder-dose-class}, and \(R_n\) is the minimax risk of \cref{obj:def:minimax-risk}. The displayed tail bound is the conclusion of the cited oracle floor for the fixed treatment-density smoothness parameter \(\beta>0\).
% lean: oracle_dose_regression_lower_all_beta

\item \emph{Conclusion.} Set \(c=c_{\mathrm{or}}\). Then \(c>0\), and the inequality from Step 2 is equivalently
\[
  c\, n^{-2\alpha/(2\alpha+1)}
  \le
  R_n\!\left(
    \mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),
    t_0
  \right)
\]
for all sufficiently large sample sizes \(n\). This is the asserted positive constant and eventual lower bound over the model class of \cref{obj:def:holder-dose-class} for the minimax risk of \cref{obj:def:minimax-risk}.
% lean: sharp_pointwise_lower_bound
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:oracle-regime-reduction}]
\begin{enumerate}
\item The positive-smoothness, regime-constant, and baseline-slack hypotheses assumed here --
\(\alpha>0\), \(\beta>0\), \(s>0\), \(M>0\), \(c_0>0\), \(t_0\in(0,1)\),
\(\varepsilon_0\in(0,1/2)\), \([t_0-\varepsilon_0,t_0+\varepsilon_0]\subset(0,1)\),
and \cref{obj:ass:baseline-submodel-slack} for \((d,\beta,s,M,c_0,\varepsilon_0,t_0)\) --
are the hypotheses required by \cref{obj:thm:sharp-pointwise-lower-bound}. Applying that result gives
a constant \(c>0\) such that, for all sufficiently large sample sizes \(n\),
\[
c\, n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
Fix this \(c\) for the rest of the proof. % lean: sharp_pointwise_lower_bound

\item Intersect the eventual lower-bound event supplied by \cref{obj:thm:sharp-pointwise-lower-bound}
with the eventual sample-size event \(n\ge1\). Equivalently, after increasing the sufficiently-large
threshold if necessary, every sample size under consideration satisfies both \(n\ge1\) and the
lower-bound inequality supplied by \cref{obj:thm:sharp-pointwise-lower-bound}.
% lean: Filter.eventually_ge_atTop

\item For any such \(n\), the hypotheses \(\alpha>0\), \(s>0\), \(d\le4s\), and \(n\ge1\)
are the hypotheses of \cref{obj:lem:rho-oracle-regime-algebra}. Hence the published benchmark rate
satisfies
\[
\rho_n
=
n^{-2\alpha/(2\alpha+1)} .
\]
% lean: rho_oracle_regime_algebra

\item For the same \(n\), the lower-bound inequality supplied by
\cref{obj:thm:sharp-pointwise-lower-bound} gives
\[
c\, n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
The identity supplied by \cref{obj:lem:rho-oracle-regime-algebra} rewrites \(c\rho_n\) as
\(c\,n^{-2\alpha/(2\alpha+1)}\), and therefore
\[
c\,\rho_n
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
This holds on the same eventual set of sample sizes. Together with \(c>0\) and the displayed identity
for \(\rho_n\), it is the assertion of \cref{obj:prop:oracle-regime-reduction}.
% lean: oracle_regime_reduction
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:sharp-minimax-smooth-covariate}]
\begin{enumerate}
\item The smoothness, regime-constant and strict-slack hypotheses assumed here — \(\alpha>0\),
\(\beta>0\), \(s>0\), \(M>0\), \(c_0>0\), \(t_0\in(0,1)\), \(\varepsilon_0\in(0,1/2)\),
\([t_0-\varepsilon_0,t_0+\varepsilon_0]\subset(0,1)\), and \cref{obj:ass:baseline-submodel-slack}
for \((d,\beta,s,M,c_0,\varepsilon_0,t_0)\) — are exactly the hypotheses of
\cref{obj:thm:sharp-pointwise-lower-bound}; the smooth-covariate condition \(d\le 4s\) is not needed
for this step and is used only in Step 3. That theorem therefore supplies a constant \(c>0\) and a
threshold \(n_1\) with
\[
c\,n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(
\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0
\right)
\qquad\text{for every }n\ge n_1 .
\]
Fix this \(c\); it is the constant asserted in the statement. % lean: sharp_pointwise_lower_bound

\item Set \(n_0=\max\{n_1,1\}\). For every \(n\ge n_0\) both the displayed lower bound of Step 1 and
the elementary condition \(n\ge1\) hold; this is the intersection of two eventual conditions, and
the conjunction asserted by the theorem is claimed only for such \(n\).
% lean: Filter.eventually_ge_atTop

\item Fix \(n\ge n_0\). The hypotheses \(\alpha>0\), \(s>0\), \(d\le 4s\) and \(n\ge1\) are exactly
those of \cref{obj:lem:rho-oracle-regime-algebra}, so
\[
\rho_n
=
n^{-2\alpha/(2\alpha+1)},
\qquad\text{equivalently}\qquad
n^{-2\alpha/(2\alpha+1)}=\rho_n ,
\]
which is the second assertion of the theorem, in the orientation in which it is stated. Holding this
identity together with the lower bound of Step 1 on the same range \(n\ge n_0\) gives, for the same
constant \(c>0\), the asserted eventual conjunction of the lower bound and the exponent identity.
% lean: rho_oracle_regime_algebra
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:frontier-bracket-deficient}]
\begin{enumerate}
\item The regime-constant and strict-slack hypotheses assumed here — \(\alpha>0\), \(\beta>0\),
\(s>0\), \(M>0\), \(c_0>0\), \(t_0\in(0,1)\), \(\varepsilon_0\in(0,1/2)\),
\([t_0-\varepsilon_0,t_0+\varepsilon_0]\subset(0,1)\), and \cref{obj:ass:baseline-submodel-slack}
for \((d,\beta,s,M,c_0,\varepsilon_0,t_0)\) — are exactly the hypotheses of
\cref{obj:lem:oracle-dose-regression-lower-all-beta}; the deficient-covariate condition \(4s<d\) is
not used for this step and enters only in Step 3. That lemma therefore supplies a constant \(c>0\)
and a threshold \(n_1\) with
\[
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right)
\ge
c\,n^{-2\alpha/(2\alpha+1)}
\qquad\text{for every }n\ge n_1 .
\]
Fix this \(c\); it is the constant asserted in the statement. Note that the lower floor is
established for the given fixed \(\beta>0\) and its exponent does not involve \(\beta\).
% lean: oracle_dose_regression_lower_all_beta

\item Set \(n_0=\max\{n_1,1\}\). For every \(n\ge n_0\) both the display of Step 1 and the
elementary condition \(n\ge1\) hold; this is the intersection of two eventual conditions, and the
conjunction asserted by the theorem is claimed only for such \(n\).
% lean: Filter.eventually_ge_atTop

\item Fix \(n\ge n_0\). The hypotheses \(\alpha>0\), \(s>0\), \(4s<d\) and \(n\ge1\) are exactly
those of \cref{obj:lem:rho-deficient-regime-algebra}, which gives both the deficient-regime formula
\[
\rho_n
=
n^{-\,2/(1+d/(4s)+1/\alpha)}
\]
and the strict comparison of exponents
\[
\frac{2}{1+d/(4s)+1/\alpha}
<
\frac{2\alpha}{2\alpha+1}.
\]
% lean: rho_deficient_regime_algebra

\item On the common range \(n\ge n_0\), the lower bound of Step 1, rewritten in the orientation of
the statement as
\[
c\,n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(
\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0
\right),
\]
holds together with the two conclusions of Step 3. That conjunction, with the constant \(c>0\) fixed
in Step 1, is exactly the assertion of the theorem: the certified lower floor sits at the
treatment-regression exponent \(2\alpha/(2\alpha+1)\), while the published benchmark sequence is the
covariate-smoothness power, whose exponent is strictly smaller.
% lean: frontier_bracket_deficient
\end{enumerate}
\end{proof}
