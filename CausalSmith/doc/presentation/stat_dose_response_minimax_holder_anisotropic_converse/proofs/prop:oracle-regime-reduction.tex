\begin{proof}[Proof of \cref{obj:prop:oracle-regime-reduction}]
\begin{enumerate}
\item The positive-smoothness, regime-constant, and baseline-slack hypotheses assumed here --
\(\alpha>0\), \(\beta>0\), \(s>0\), \(M>0\), \(c_0>0\), \(t_0\in(0,1)\),
\(\varepsilon_0\in(0,1/2)\), \([t_0-\varepsilon_0,t_0+\varepsilon_0]\subset(0,1)\),
and \cref{obj:ass:baseline-submodel-slack} for \((d,\beta,s,M,c_0,\varepsilon_0,t_0)\) --
are the hypotheses required by \cref{obj:thm:sharp-pointwise-lower-bound}. Applying that result gives
a constant \(c>0\) such that, for all sufficiently large sample sizes \(n\),
\[
c\, n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
Fix this \(c\) for the rest of the proof. % lean: sharp_pointwise_lower_bound

\item Intersect the eventual lower-bound event supplied by \cref{obj:thm:sharp-pointwise-lower-bound}
with the eventual sample-size event \(n\ge1\). Equivalently, after increasing the sufficiently-large
threshold if necessary, every sample size under consideration satisfies both \(n\ge1\) and the
lower-bound inequality supplied by \cref{obj:thm:sharp-pointwise-lower-bound}.
% lean: Filter.eventually_ge_atTop

\item For any such \(n\), the hypotheses \(\alpha>0\), \(s>0\), \(d\le4s\), and \(n\ge1\)
are the hypotheses of \cref{obj:lem:rho-oracle-regime-algebra}. Hence the published benchmark rate
satisfies
\[
\rho_n
=
n^{-2\alpha/(2\alpha+1)} .
\]
% lean: rho_oracle_regime_algebra

\item For the same \(n\), the lower-bound inequality supplied by
\cref{obj:thm:sharp-pointwise-lower-bound} gives
\[
c\, n^{-2\alpha/(2\alpha+1)}
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
The identity supplied by \cref{obj:lem:rho-oracle-regime-algebra} rewrites \(c\rho_n\) as
\(c\,n^{-2\alpha/(2\alpha+1)}\), and therefore
\[
c\,\rho_n
\le
R_n\!\left(\mathcal P_{\alpha,\beta,s}(M,c_0,\varepsilon_0,t_0),t_0\right).
\]
This holds on the same eventual set of sample sizes. Together with \(c>0\) and the displayed identity
for \(\rho_n\), it is the assertion of \cref{obj:prop:oracle-regime-reduction}.
% lean: oracle_regime_reduction
\end{enumerate}
\end{proof}
