\begin{proof}[Proof of \cref{obj:thm:exceptional-locus-codimension-one}]
Fix \(m\ge 1\) and set \(K=2m+2\).  All sets below live in the retained cumulant coordinates \(t_{r,a}\), \(2\le r\le K\), \(0\le a\le r\); both arrow maps of \cref{obj:def:forward-cumulant-map} and \cref{obj:def:reverse-cumulant-map} vanish outside that range, so \(\overline E_m\) and the two image varieties \(C^{\mathrm{right}}_{m,K}\), \(C^{\mathrm{left}}_{m,K}\) of \cref{obj:def:image-varieties} all consist of vectors supported there, the projection onto \(\mathbb C^{q_K}\) is injective on them, and every chain, dimension and codimension statement below may be read in \(\mathbb C^{q_K}\).  Throughout, ``dimension \(d\)'' means: there is a strictly increasing chain of \(d+1\) nonempty irreducible closed subsets, and none of length \(d+2\); and ``\(\overline E_m\) has codimension \(d\) in \(X\)'' means: every irreducible component \(D\) of \(\overline E_m\) admits a strictly increasing chain of \(d+1\) irreducible closed sets running from \(D\) to \(X\), and some component admits no such chain of length \(d+2\).  Put
\[
D_m:=\frac{3(m+2)^2+(m+2)}{2}-5 .
\]
% lean: HasCodimensionIn HasAffineZariskiDimension commonAxisExpectedDimension

\begin{enumerate}
\item \textbf{The common-axis family.}  Let
\[
\Delta_m:=\bigl\{\theta=(\gamma,\rho,c)\in\Theta^{\mathrm{right},\circ}_{m,K}:\ \rho_1=0\bigr\},
\qquad
W_m:=\operatorname{ZariskiClosure}\bigl(\Phi^{\mathrm{right}}_{m,K}(\Delta_m)\bigr).
\]
For \(\theta\in\Delta_m\) genericity gives \(\gamma\ne0\) and pairwise distinct slopes, so \(\rho_i\ne\rho_1=0\) for \(2\le i\le m\).  Define the reverse twin \(\theta^{\mathrm{tw}}=(\delta,\sigma,d)\) by
\[
\delta=\gamma^{-1},\qquad
\sigma_1=0,\qquad
\sigma_i=\rho_i^{-1}\ (2\le i\le m),
\]
\[
d_{0r}=c_{1r},\qquad
d_{1r}=c_{m+1,r},\qquad
d_{jr}=c_{jr}\,\rho_j^{\,r}\ (2\le j\le m),\qquad
d_{m+1,r}=c_{0r}\,\gamma^{\,r},
\qquad 2\le r\le K .
\]
Then the two loading families agree up to scalars, direction by direction:
\[
u_1=(1,0)=v_0,\qquad
u_{m+1}=(0,1)=v_1,\qquad
u_0=(1,\gamma)=\gamma\,v_{m+1},\qquad
u_j=(1,\rho_j)=\rho_j\,v_j\ (2\le j\le m).
\]
The \((r,a)\) summand \(c_{jr}(u_{j1})^{r-a}(u_{j2})^{a}\) is homogeneous of degree \(r\) in the loading vector, so rescaling a loading by \(\lambda\) is absorbed by multiplying its weight by \(\lambda^{r}\); this is exactly the weight prescription above.  Summing the four cases over the reindexed source labels gives
\[
\Phi^{\mathrm{left}}_{m,K}(\theta^{\mathrm{tw}})=\Phi^{\mathrm{right}}_{m,K}(\theta),
\qquad \theta\in\Delta_m .
\]
% lean: commonAxisReverseTwin commonAxisReverseTwin_map_eq

\item \textbf{\(W_m\subseteq\overline E_m\).}  Let \(\theta\in\Delta_m\) and \(t=\Phi^{\mathrm{right}}_{m,K}(\theta)\).  Then \(\theta\) itself lies in \(R^{\mathrm{right}}_{m,K}(t)\cap\Theta^{\mathrm{right},\circ}_{m,K}\), and by the previous display \(\theta^{\mathrm{tw}}\in R^{\mathrm{left}}_{m,K}(t)\), so
\[
R^{\mathrm{right}}_{m,K}(t)\cap\Theta^{\mathrm{right},\circ}_{m,K}\ne\varnothing
\quad\text{and}\quad
R^{\mathrm{left}}_{m,K}(t)\ne\varnothing,
\]
which is the first alternative in the definition of \(E_m\) in \cref{obj:def:generic-full-fiber-compatibility}.  Hence \(\Phi^{\mathrm{right}}_{m,K}(\Delta_m)\subseteq E_m\), and taking Zariski closures, \(W_m\subseteq\overline E_m\).  Moreover \(E_m\subseteq C^{\mathrm{right}}_{m,K}\) and \(E_m\subseteq C^{\mathrm{left}}_{m,K}\), because a point of \(E_m\) has a nonempty fiber under \emph{both} arrow maps and therefore lies in both images; closing up,
\[
W_m\subseteq\overline E_m\subseteq C^{\mathrm{right}}_{m,K}\cap C^{\mathrm{left}}_{m,K}.
\]
% lean: forwardCommonAxis_image_mem_exceptional forwardCommonAxisImageClosure_subset_exceptional genericCompatibilityClosure_subset_forwardVariety genericCompatibilityClosure_subset_reverseVariety

\item \textbf{\(W_m\) is irreducible.}  Deleting the pinned coordinate \(\rho_1=0\) presents \(\Delta_m\) as the nonvanishing locus of a single nonzero polynomial (the generic-locus product of \cref{obj:def:generic-parameter-loci} with \(\rho_1\) set to \(0\)) inside the remaining affine coordinates; such a locus is Zariski dense in that affine space, and the Zariski closure of the image of a dense set under a polynomial map is irreducible.  Hence \(W_m\) is a nonempty irreducible closed set.  The same argument applied to the whole affine parameter space shows that \(C^{\mathrm{right}}_{m,K}\) and \(C^{\mathrm{left}}_{m,K}\) are irreducible.
% lean: restrict_forwardCommonAxisImageClosure_isIrreducible forwardCumulantImageVariety_isIrreducible reverseCumulantImageVariety_isIrreducible

\item \textbf{Upper bounds on the two dimensions.}  Every retained coordinate of \(\Phi^{\mathrm{right}}_{m,K}\) is a polynomial in the finite family
\[
\mathcal G=\{\gamma,\rho_1,\ldots,\rho_m\}
\ \cup\ \{t_{r,a}:2\le r\le m,\ 0\le a\le r\}
\ \cup\ \{c_{jr}:0\le j\le m+1,\ m+1\le r\le K\}:
\]
a coordinate of order \(r\le m\) is itself a member of \(\mathcal G\), and a coordinate of order \(r\ge m+1\) equals \(\sum_j c_{jr}(u_{j1})^{r-a}(u_{j2})^{a}\), a polynomial in the slopes and the high-order weights listed in \(\mathcal G\).  Counting,
\[
\#\mathcal G=(m+1)+q_m+(m+2)^2=D_m+1,
\qquad q_m=\frac{m(m+3)}{2}-2 ,
\]
so the coordinate algebra of the forward image has transcendence degree at most \(D_m+1\) and therefore
\[
\dim C^{\mathrm{right}}_{m,K}\le D_m+1 .
\]
On \(\Delta_m\) the coordinate \(\rho_1\) is the constant \(0\), so the same family with \(\rho_1\) deleted generates; it has \(D_m\) members, whence \(\dim W_m\le D_m\).
% lean: forwardBandCoordinateSubalgebra_trdeg_le_generatorCard forwardBandCoordinateSubalgebra_trdeg_le_expected forwardCommonAxisCoordinateSubalgebra_trdeg_le_expected

\item \textbf{Matching lower bound for the forward image: a nonvanishing \((D_m+1)\times(D_m+1)\) Jacobian minor.}  Select the following rows (retained cumulant coordinates) and columns (parameter coordinates):
\begin{itemize}
\item for each order \(2\le r\le m\): the rows \(t_{r,a}\), \(0\le a\le r\), and the columns \(c_{a,r}\), \(0\le a\le r\);
\item for each order \(m+1\le r\le 2m+1\): the rows \(t_{r,a}\), \(0\le a\le m\), together with \(t_{r,r}\), and the columns \(c_{j,r}\), \(0\le j\le m\), together with \(c_{m+1,r}\);
\item at the top order \(r=K\): the rows \(t_{K,e}\), \(0\le e\le 2m+1\), and the columns consisting of the \(m+1\) top weights \(c_{j,K}\), \(0\le j\le m\), and the \(m+1\) finite slopes \(\gamma,\rho_1,\ldots,\rho_m\); augmented by the extra row \(t_{K,K}\) and the extra column \(c_{m+1,K}\).
\end{itemize}
The total count is
\[
\sum_{r=2}^{m}(r+1)+(m+1)(m+2)+(2m+2)+1=D_m+1 .
\]
Evaluate the minor at the parameter \(\theta^{J}\) with
\[
\gamma=1,\qquad \rho_i=i+1\ (1\le i\le m),\qquad c_{jr}=1\ \text{for all }j\text{ and all }2\le r\le K,
\]
whose finite loading directions are \(u_j=(1,j+1)\) for \(0\le j\le m\) and \(u_{m+1}=(0,1)\).  A weight column of order \(r\) has zero derivative in every cumulant coordinate of an order other than \(r\), because that coordinate does not involve \(c_{\cdot,r}\); placing the slope columns last, the matrix is therefore block upper triangular with the following diagonal blocks.
\begin{itemize}
\item Order \(r\le m\): \(\partial t_{r,a}/\partial c_{j,r}=(u_{j1})^{r-a}(u_{j2})^{a}=(j+1)^{a}\), the transposed Vandermonde matrix in the \(r+1\) distinct nodes \(1,\ldots,r+1\); its determinant is nonzero.
\item Order \(m+1\le r\le 2m+1\): for \(j\le m\) and \(a\le m\) the entry is \((j+1)^{a}\), the Vandermonde matrix in the \(m+1\) distinct nodes \(1,\ldots,m+1\); the column \(c_{m+1,r}\) has entries \((u_{m+1,1})^{r-a}(u_{m+1,2})^{a}=0\) for \(a\le m<r\) and \(1\) in the row \(t_{r,r}\).  The block is triangular with determinant equal to that Vandermonde determinant, hence nonzero.
\item Order \(K\): the weight columns give \(\partial t_{K,e}/\partial c_{j,K}=(j+1)^{e}\) and the slope columns give \(\partial t_{K,e}/\partial\rho_j=e\,c_{j,K}(j+1)^{e-1}=e\,(j+1)^{e-1}\), so this is the confluent (Hermite) Vandermonde matrix with value and first-derivative columns at the \(m+1\) distinct nodes \(1,\ldots,m+1\) and monomial rows of degrees \(0,\ldots,2m+1\); its determinant is nonzero.  The augmenting column \(c_{m+1,K}\) contributes \(0^{K-e}1^{e}=0\) in every row \(e\le 2m+1\) and \(1\) in the row \(t_{K,K}\), so the augmented block is triangular with the same nonzero determinant.
\end{itemize}
The product of the diagonal determinants is nonzero, so the selected minor is a nonzero polynomial and
\[
\dim C^{\mathrm{right}}_{m,K}\ge D_m+1 .
\]
With the upper bound of the previous step, \(\dim C^{\mathrm{right}}_{m,K}=D_m+1\).

\item \textbf{Matching lower bound for the common-axis image.}  On \(\Delta_m\) the slope \(\rho_1\) is frozen at \(0\), so it supplies no column; delete it and use the same rows with the top row \(t_{K,2m+1}\) deleted as well, leaving \(D_m\) rows and \(D_m\) columns.  Evaluate at the pinned witness with
\[
\rho_1=0,\qquad \gamma=1,\qquad \rho_i=i\ (2\le i\le m),\qquad c_{jr}=1 ,
\]
whose finite nodes are the distinct values \(0,1,\ldots,m\).  The ordinary blocks are Vandermonde and augmented-Vandermonde exactly as before, and the top block is now the pinned Hermite matrix carrying values at all \(m+1\) nodes and first derivatives at the \(m\) unpinned nodes, with monomial rows of degrees \(0,\ldots,2m\); its determinant is nonzero because the nodes are distinct.  Hence \(\dim W_m\ge D_m\), and with the upper bound
\[
\dim W_m=D_m .
\]
% lean: forwardCommonAxisPolynomialJacobianMinor_ne_zero det_canonicalCommonAxisFullJacobianAtWitness_ne_zero canonicalCommonAxisTopJacobianAtWitness_eq_pinned restrict_forwardCommonAxisImageClosure_dimension_expected

\item \textbf{No intermediate irreducible set between \(W_m\) and the forward variety.}  Suppose \(Y\) were irreducible and closed with \(W_m\subsetneq Y\subsetneq C^{\mathrm{right}}_{m,K}\).  Appending \(Y\) and \(C^{\mathrm{right}}_{m,K}\) to a maximal chain of \(D_m+1\) irreducible closed subsets of \(W_m\) produces a strictly increasing chain of \(D_m+3\) irreducible closed subsets of \(C^{\mathrm{right}}_{m,K}\), contradicting \(\dim C^{\mathrm{right}}_{m,K}=D_m+1\).  Hence
\[
\nexists\,Y\ \text{irreducible closed with}\ W_m\subsetneq Y\subsetneq C^{\mathrm{right}}_{m,K}.
\]
% lean: no_intermediate_of_exact_affine_dimensions

\item \textbf{The same for the reverse variety, by coordinate reversal.}  Let \(\iota\) be the involution of cumulant space that exchanges the two observed coordinates,
\[
(\iota t)_{r,a}=t_{r,r-a}
\qquad (2\le r\le K,\ 0\le a\le r).
\]
Exchanging the roles of the two axes in \cref{obj:def:forward-cumulant-map} and \cref{obj:def:reverse-cumulant-map} gives \(\Phi^{\mathrm{left}}_{m,K}(\theta^{\mathrm{rev}})=\iota\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta)\bigr)\), where \(\theta^{\mathrm{rev}}\) keeps the slopes of \(\theta\) and exchanges the weight families of the two axis source labels \(j=0\) and \(j=m+1\).  That relabelling is an involution of the parameter space, hence onto, so
\[
\iota\bigl(C^{\mathrm{right}}_{m,K}\bigr)=C^{\mathrm{left}}_{m,K}.
\]
For \(W_m\) the exchange is not applied to \(\theta\) itself but to the reverse twin of Step 1.  Let \(\theta\in\Delta_m\) and put \(\theta^{\dagger}=(\theta^{\mathrm{tw}})^{\mathrm{rev}}\), the same axis relabelling read from the reverse chart back to the forward one; explicitly, \(\theta^{\dagger}\) carries the reciprocated slopes
\[
\gamma^{\dagger}=\gamma^{-1},
\qquad
\rho^{\dagger}_1=0,
\qquad
\rho^{\dagger}_i=\rho_i^{-1}\ (2\le i\le m),
\]
together with the weight family \(d\) of \(\theta^{\mathrm{tw}}\), its two axis labels \(0\) and \(m+1\) exchanged.  Genericity of \(\theta\) gives \(\gamma\ne0\) and pairwise distinct nonzero \(\rho_2,\ldots,\rho_m\), so the reciprocals \(\gamma^{-1},\rho_2^{-1},\ldots,\rho_m^{-1}\) are again pairwise distinct and nonzero, hence distinct from \(\rho^{\dagger}_1=0\); the weights of \(\theta^{\mathrm{tw}}\) are the weights of \(\theta\) times powers of nonzero slopes, hence nonzero.  Therefore \(\theta^{\dagger}\in\Delta_m\).  Because the axis relabelling is an involution, \((\theta^{\dagger})^{\mathrm{rev}}=\theta^{\mathrm{tw}}\), so the axis-exchange display applied to \(\theta^{\dagger}\) reads \(\Phi^{\mathrm{left}}_{m,K}(\theta^{\mathrm{tw}})=\iota\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta^{\dagger})\bigr)\); combining with \(\Phi^{\mathrm{left}}_{m,K}(\theta^{\mathrm{tw}})=\Phi^{\mathrm{right}}_{m,K}(\theta)\) from Step 1,
\[
\iota\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta^{\dagger})\bigr)=\Phi^{\mathrm{right}}_{m,K}(\theta),
\qquad\theta\in\Delta_m .
\]
So \(\iota\) carries \(\Phi^{\mathrm{right}}_{m,K}(\Delta_m)\) into itself, and being an involution it carries it onto itself; closing up,
\[
\iota\bigl(W_m\bigr)=W_m .
\]
What is stable under the exchange is thus the image \(\Phi^{\mathrm{right}}_{m,K}(\Delta_m)\) under a reciprocal-slope reparametrization of \(\Delta_m\), not the individual parameter.  Since \(\iota\) is an injective polynomial involution it preserves irreducibility and strict inclusions, so an intermediate irreducible closed set between \(W_m\) and \(C^{\mathrm{left}}_{m,K}\) would be carried to one between \(W_m\) and \(C^{\mathrm{right}}_{m,K}\).  By the previous step there is none.

\item \textbf{Both inclusions \(\overline E_m\subseteq C^{\mathrm{right}}_{m,K}\), \(\overline E_m\subseteq C^{\mathrm{left}}_{m,K}\) are proper.}  For a cumulant vector \(t\) form the \((m+2)\times(m+2)\) matrix
\[
N^{\mathrm{hor}}(t)_{0,b}=t_{m+2,\,b+1},
\qquad
N^{\mathrm{hor}}(t)_{a+1,b}=\binom{m}{a}\,t_{2m+2,\,a+b+1}
\qquad (0\le a,b\le m+1,\ a\le m),
\]
and set \(P^{\mathrm{hor}}(t)=\det N^{\mathrm{hor}}(t)\), a polynomial in the retained coordinates.  If \(t\) is the image of a parameter with loading family \((w_j)_{j=0}^{m+1}\) and weights \((c_{jr})\), then substituting the coordinate formula of \cref{obj:def:forward-cumulant-map} (resp. \cref{obj:def:reverse-cumulant-map}) and splitting the exponents factors the matrix as \(N^{\mathrm{hor}}(t)=A\,B\) with
\[
A_{0j}=c_{j,m+2},
\qquad
A_{a+1,j}=\binom{m}{a}c_{j,2m+2}(w_{j1})^{m-a}(w_{j2})^{a},
\qquad
B_{jb}=(w_{j1})^{m+1-b}(w_{j2})^{b+1}.
\]
Every entry of the \(j\)th row of \(B\) carries the factor \((w_{j2})^{b+1}\) with \(b+1\ge1\); hence if some loading direction has vanishing second coordinate, that row of \(B\) is zero and \(P^{\mathrm{hor}}(t)=0\).  Every reverse loading family contains \(v_0=(1,0)\), so \(P^{\mathrm{hor}}\) vanishes on the whole reverse image and therefore on its Zariski closure \(C^{\mathrm{left}}_{m,K}\), hence on \(\overline E_m\) by Step 2.  On the other hand take the forward parameter \(\theta^{\star}\) with
\[
\gamma=1,\quad \rho_i=i+1,\qquad
c_{j,m+2}=\mathbb 1_{\{j=m+1\}},\quad
c_{j,2m+2}=\mathbb 1_{\{j\le m\}},\quad
c_{jr}=0\ \text{for }r\notin\{m+2,2m+2\}.
\]
At \(\theta^\star\) the matrix \(A\) has last column \((1,0,\ldots,0)^{\top}\) and, in its first \(m+1\) columns and last \(m+1\) rows, the entries \(\binom{m}{a}(j+1)^{a}\); the matrix \(B\) has last row \((0,\ldots,0,1)\) and, in its first \(m+1\) rows, the entries \((j+1)^{b+1}\).  Both are triangular with a Vandermonde determinant in the distinct nodes \(1,\ldots,m+1\) as the surviving factor, so \(\det A\ne0\ne\det B\) and
\[
P^{\mathrm{hor}}\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta^\star)\bigr)\ne0 .
\]
Thus \(\Phi^{\mathrm{right}}_{m,K}(\theta^\star)\in C^{\mathrm{right}}_{m,K}\setminus\overline E_m\) and \(\overline E_m\ne C^{\mathrm{right}}_{m,K}\).  Applying \(\iota\) to the whole construction — equivalently, using \(P^{\mathrm{ver}}=P^{\mathrm{hor}}\circ\iota\), which vanishes on \(C^{\mathrm{right}}_{m,K}\) because every forward loading family contains \(u_{m+1}=(0,1)\), and is nonzero at the reversed witness — gives \(\overline E_m\ne C^{\mathrm{left}}_{m,K}\).

\item \textbf{\(W_m\) is an irreducible component of \(\overline E_m\).}  \(W_m\) is irreducible and closed and \(W_m\subseteq\overline E_m\).  If \(Y\) is irreducible and closed with \(W_m\subseteq Y\subseteq\overline E_m\) and \(Y\ne W_m\), then \(W_m\subsetneq Y\subseteq\overline E_m\subsetneq C^{\mathrm{right}}_{m,K}\) by Step 9, so \(Y\) is an intermediate set of the kind excluded in Step 7 — a contradiction.  Hence \(W_m\) is maximal among irreducible closed subsets of \(\overline E_m\), i.e. an irreducible component.

\item \textbf{Codimension exactly one.}  Let \(X\) be either \(C^{\mathrm{right}}_{m,K}\) or \(C^{\mathrm{left}}_{m,K}\) and let \(P\) be the corresponding contraction minor \(P^{\mathrm{hor}}\) or \(P^{\mathrm{ver}}\) from Step 9.  First, every irreducible component \(D\) of \(\overline E_m\) satisfies \(D\subseteq\overline E_m\subsetneq X\), so \(D\subsetneq X\) and the two-term chain \(D\subsetneq X\) of irreducible closed sets exists; this is the required chain of length \(2=1+1\) from \(D\) to \(X\).  Second, \(P\) vanishes on \(W_m\) and not on \(X\), so the prime ideal of \(W_m\) contains the ideal of \(X\) together with \(P\) and, by Step 7 and Step 8, no prime of the coordinate ring lies strictly between them: the ideal of \(W_m\) is a minimal prime over the ideal of \(X\) enlarged by \(P\).  Equivalently, no irreducible closed \(Y\) satisfies \(W_m\subsetneq Y\subsetneq X\), so the component \(W_m\) admits no chain of length \(3\) from \(W_m\) to \(X\).  The two clauses are exactly codimension \(1\), so
\[
\operatorname{codim}\bigl(\overline E_m,\ C^{\mathrm{right}}_{m,K}\bigr)=1,
\qquad
\operatorname{codim}\bigl(\overline E_m,\ C^{\mathrm{left}}_{m,K}\bigr)=1 .
\]

\item \textbf{Exclusion of codimension \(m\) for \(m\ge2\).}  Suppose \(\overline E_m\) had codimension \(m\) in \(X\) with \(m\ge2\).  Applied to the component \(W_m\), the first clause of that statement supplies a strictly increasing chain \(Y_0\subsetneq Y_1\subsetneq\cdots\subsetneq Y_m\) of irreducible closed sets with \(Y_0=W_m\) and \(Y_m=X\).  Keeping only the indices \(0,1,m\) leaves a strictly increasing chain of three irreducible closed sets running from \(W_m\) to \(X\), which the second clause of Step 11 forbids.  Hence
\[
\operatorname{codim}\bigl(\overline E_m,\ C^{\mathrm{right}}_{m,K}\bigr)\ne m,
\qquad
\operatorname{codim}\bigl(\overline E_m,\ C^{\mathrm{left}}_{m,K}\bigr)\ne m
\qquad (m\ge2).
\]

\item \textbf{Forward preimage.}  By \cref{obj:def:generic-full-fiber-compatibility}, \(\theta\in H^{\mathrm{right}}_m\) means \(\theta\in\Theta^{\mathrm{right},\circ}_{m,K}\) and \(\Phi^{\mathrm{right}}_{m,K}(\theta)\in E_m\).  Write \(t=\Phi^{\mathrm{right}}_{m,K}(\theta)\), with the fibers of \cref{obj:def:fiber-correspondences}.  If \(\theta\) is generic then \(\theta\in R^{\mathrm{right}}_{m,K}(t)\cap\Theta^{\mathrm{right},\circ}_{m,K}\), so the first alternative in the definition of \(E_m\) reduces to \(R^{\mathrm{left}}_{m,K}(t)\ne\varnothing\); and the second alternative also forces \(R^{\mathrm{left}}_{m,K}(t)\ne\varnothing\), since it asserts in particular that the reverse fiber over \(t\) contains a generic point.  Conversely, if \(\theta\) is generic and \(R^{\mathrm{left}}_{m,K}(t)\ne\varnothing\) then the first alternative holds outright.  Therefore
\[
\theta\in H^{\mathrm{right}}_m
\quad\Longleftrightarrow\quad
\theta\in\Theta^{\mathrm{right},\circ}_{m,K}
\ \text{and}\
R^{\mathrm{left}}_{m,K}\!\left(\Phi^{\mathrm{right}}_{m,K}(\theta)\right)\neq\varnothing .
\]
Coordinates outside the range \(2\le r\le K\) vanish under the truncation convention, so they impose no further condition on either side.

\item \textbf{Low-dimensional specializations.}  Let \(m\in\{1,2\}\).  A pair \(p=(p^{\mathrm{right}},p^{\mathrm{left}})\) solves the incidence system \(A_m\) of \cref{obj:def:worked-compatibility-instances} precisely when both members lie in the retained coordinate space, their retained cumulants agree,
\[
\Phi^{\mathrm{right}}_{m,K}(p^{\mathrm{right}})=\Phi^{\mathrm{left}}_{m,K}(p^{\mathrm{left}}),
\]
and at least one member lies in its generic locus.  If \(t\in E_m\), then by definition one arrow has a generic fiber point over \(t\) and the other a fiber point over \(t\); calling them \(p^{\mathrm{right}}\) and \(p^{\mathrm{left}}\), their retained cumulants both equal \(t\), so the displayed equality holds, the generic-locus disjunction holds, and \(t=\Phi^{\mathrm{right}}_{m,K}(p^{\mathrm{right}})\); thus \(p\) solves \(A_m\) and projects to \(t\).  Conversely, if \(p\) solves \(A_m\) and \(t=\Phi^{\mathrm{right}}_{m,K}(p^{\mathrm{right}})\), then \(p^{\mathrm{right}}\in R^{\mathrm{right}}_{m,K}(t)\) and, by the displayed equality, \(p^{\mathrm{left}}\in R^{\mathrm{left}}_{m,K}(t)\); whichever member is generic supplies the corresponding alternative in the definition of \(E_m\), so \(t\in E_m\).  Coordinates beyond the truncation order vanish on both sides by convention.  Taking \(m=1\) (with \(K=4\)) and \(m=2\) (with \(K=6\)) gives
\[
E_1=\bigl\{t:\ \exists\,p\ \text{satisfying}\ A_1,\ \Phi^{\mathrm{right}}_{1,4}(p^{\mathrm{right}})=t\bigr\},
\qquad
E_2=\bigl\{t:\ \exists\,p\ \text{satisfying}\ A_2,\ \Phi^{\mathrm{right}}_{2,6}(p^{\mathrm{right}})=t\bigr\}.
\]

\end{enumerate}
\end{proof}
