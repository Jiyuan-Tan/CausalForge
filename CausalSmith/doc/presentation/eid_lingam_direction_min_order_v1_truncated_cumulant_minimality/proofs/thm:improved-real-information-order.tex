\begin{proof}[Proof of \cref{obj:thm:improved-real-information-order}]
Fix \(m\ge 3\) and put \(L=2m+1\) and \(n=m+2\).  As in the apolar setting write \(\ell_u(x,y)=u_1x+u_2y\), write \(q(u)\) for the evaluation of a binary form at a direction \(u\), and, for a loading family \((w_j)_{j=0}^{m+1}\), write its support annihilator in the normalization of \cref{obj:def:apolar-notation} as
\[
Q_D=\prod_{j=0}^{m+1}\bigl(w_{j2}x-w_{j1}y\bigr),
\]
a nonzero homogeneous form of degree \(n\) vanishing exactly on the \(n\) loading directions.  Expanding the binomial in \(\ell_{u_j}^{\,r}\) and comparing with \cref{obj:def:forward-cumulant-map} gives, for \(2\le r\le L\),
\[
f_r(x,y)=\sum_{a=0}^{r}\binom{r}{a}\bigl[\Phi^{\mathrm{right}}_{m,L}(\theta)\bigr]_{r,a}x^{r-a}y^{a}
=\sum_{j=0}^{m+1}c_{jr}\,\ell_{u_j}(x,y)^{r},
\]
and likewise on the reverse side; and for \(q\) homogeneous of degree \(n\),
\[
q(\partial)f_{n+k}=\frac{(n+k)!}{k!}\sum_{j=0}^{m+1}c_{j,n+k}\,q(u_j)\,\ell_{u_j}^{\,k},
\qquad 0\le k\le m-1 ,
\]
the range of \(k\) being limited by \(n+k\le L\).  This is one block fewer than at order \(2m+2\).
% lean: dividedPowerBlock dividedPowerBlock_forward_eq_sum_linForm_pow diffApply_linForm_pow supportAnnihilator

\begin{enumerate}
\item \textbf{The lower-order rank determinant.}  Out of the \(m\) available contraction equations select \(m+2\) scalar rows: the single equation at \(k=0\), the coefficient of \(x\) in the equation at \(k=1\), and all \(m\) coefficients of the degree-\((m-1)\) form at \(k=m-1\).  Explicitly, define the \((m+2)\times(m+2)\) matrix \(N^{\mathrm{right}}(\theta)\), with columns indexed by the source labels \(j=0,\ldots,m+1\), by
\[
N^{\mathrm{right}}_{0,j}=c_{j,m+2},
\qquad
N^{\mathrm{right}}_{1,j}=c_{j,m+3}\,u_{j1},
\qquad
N^{\mathrm{right}}_{i+2,j}=\binom{m-1}{i}c_{j,2m+1}\,(u_{j1})^{m-1-i}(u_{j2})^{i}
\ \ (0\le i\le m-1),
\]
and let \(\Pi_{\mathrm r}(\theta)=\det N^{\mathrm{right}}(\theta)\).  The three row blocks come from the three distinct available orders \(m+2\), \(m+3\) and \(2m+1\), which are pairwise distinct precisely because \(m\ge3\).  By construction, \(N^{\mathrm{right}}(\theta)e\) collects the selected rows of the contraction equations applied to a vector \(e=(e_j)_j\); hence
\[
\Pi_{\mathrm r}(\theta)\ne0
\quad\Longrightarrow\quad
\Bigl[\ \sum_{j}c_{j,n+k}\,e_j\,\ell_{u_j}^{\,k}=0\ \text{for all }0\le k\le m-1\ \Bigr]\Longrightarrow e=0 .
\]
The determinant is a nonzero polynomial: at the forward parameter
\[
\gamma=1,\quad \rho_i=i+1,\quad
c_{j,m+2}=\mathbb 1_{\{j=m+1\}},\quad
c_{j,m+3}=\mathbb 1_{\{j=0\}},\quad
c_{j,2m+1}=\mathbb 1_{\{1\le j\le m\}},\quad c_{jr}=0\ \text{otherwise},
\]
the first row is supported on the column \(j=m+1\), the second on the column \(j=0\), and the last \(m\) rows on the columns \(j=1,\ldots,m\), where they form the matrix \(\binom{m-1}{i}\rho_j^{\,i}\); the matrix is therefore block triangular after a permutation of columns, with a Vandermonde determinant in the \(m\) distinct nodes \(\rho_1,\ldots,\rho_m\) as its only nonconstant factor, so it is nonzero.  All weights of this witness vanish outside the orders \(2\le r\le L\).  Define \(N^{\mathrm{left}}\) and \(\Pi_{\mathrm l}\) by the coordinate-reversed formulas, with the reverse loadings \(v_j\) and weights \(d_{jr}\), and note that the mirror witness gives \(\Pi_{\mathrm l}\ne0\) likewise.
% lean: lowerForwardContractionMinorPolynomial lowerForwardWeightedContraction lowerForwardExplicitRankData lowerReverseContractionMinorPolynomial lowerReverseExplicitRankData

\item \textbf{The exceptional sets.}  On the real forward parameter space put
\[
\Delta^{\mathrm{right}}_m=\Bigl(\prod_{i=1}^m\rho_i\Bigr)\,\Pi_{\mathrm r},
\qquad
E^{\mathrm{right}}=\{\theta:\Delta^{\mathrm{right}}_m(\theta)=0\},
\]
and on the real reverse parameter space
\[
\Delta^{\mathrm{left}}_m=\Bigl(\prod_{i=1}^m\sigma_i\Bigr)\,\Pi_{\mathrm l},
\qquad
E^{\mathrm{left}}=\{\eta:\Delta^{\mathrm{left}}_m(\eta)=0\}.
\]
Both \(\Delta^{\mathrm{right}}_m\) and \(\Delta^{\mathrm{left}}_m\) are nonzero real polynomials by Step 1, so \(E^{\mathrm{right}}\) and \(E^{\mathrm{left}}\) are proper real algebraic subsets of their parameter spaces.  Their complements meet the feasible regions: the truncated moment problem at order \(L\) supplies a real cumulant vector \(c^{0}\) and a radius \(\varepsilon>0\) such that
\[
|c'_r-c^{0}_r|<\varepsilon\ (2\le r\le L)
\ \Longrightarrow\
\exists\ \text{a centered non-Gaussian law }\nu,\ \mathbb E|\,\cdot\,|^{L}<\infty,\ \operatorname{Cum}_r(\nu)=c'_r\ (2\le r\le L).
\]
Consider the box in which every slope coordinate ranges over \(\mathbb R\) and every retained weight coordinate \(c_{jr}\) ranges over \((c^{0}_r-\varepsilon,c^{0}_r+\varepsilon)\).  Each factor is infinite, and the product of \(\Delta^{\mathrm{right}}_m\) with the genericity polynomial of \cref{obj:def:generic-parameter-loci} is a nonzero polynomial in the retained coordinates, hence cannot vanish at every point of the box.  A point where it does not vanish has \(\gamma\ne0\), pairwise distinct slopes, and every weight family realized by a centered non-Gaussian real source law with finite \(L\)-th moment, so it lies in \(F^{\mathrm{right}}_{m,L}\) by \cref{obj:def:real-feasible-regions}, and it lies outside \(E^{\mathrm{right}}\).  Therefore
\[
F^{\mathrm{right}}_{m,L}\setminus E^{\mathrm{right}}\ne\varnothing,
\qquad
F^{\mathrm{left}}_{m,L}\setminus E^{\mathrm{left}}\ne\varnothing .
\]
% lean: lowerForwardRealExceptionalPolynomial lowerForwardRealExceptionalPolynomial_ne_zero IsProperRealAlgebraicSubset TruncatedMomentInterior truncatedMomentInterior exists_feasible_nonvanishing

\item \textbf{The lower-order kernel identity.}  Let \(\theta\in F^{\mathrm{right}}_{m,L}\setminus E^{\mathrm{right}}\) and read \(\theta\) as a complex parameter; the cumulant maps have real polynomial coordinates, so complexification changes no value.  From \(\Delta^{\mathrm{right}}_m(\theta)\ne0\),
\[
\rho_i\ne0\ (1\le i\le m),
\qquad
\Pi_{\mathrm r}(\theta)\ne0 ,
\]
and feasibility gives \(\gamma\ne0\) together with pairwise distinct \(\gamma,\rho_1,\ldots,\rho_m\); so the \(m+1\) finite forward loading directions \(u_0=(1,\gamma)\), \(u_j=(1,\rho_j)\) have distinct nonzero second coordinates, and together with \(u_{m+1}=(0,1)\) the \(n\) directions are pairwise non-proportional.  Now let \(q\) be homogeneous of degree \(n\) with
\[
q(\partial)f_{n+k}=0
\qquad\text{for every }0\le k\le m-1 .
\]
Dividing the contraction display by the nonzero factorial and applying the selected rows of Step 1, the vector \(e=(q(u_j))_{j=0}^{m+1}\) is annihilated by \(N^{\mathrm{right}}(\theta)\), whose determinant is nonzero; hence \(q(u_j)=0\) for every \(j\).  A binary form of degree \(n\) vanishing at \(n\) pairwise non-proportional directions is divisible by the corresponding product of linear forms, so, comparing degrees, \(q=c\,Q_{\mathrm{right}}\) for some \(c\in\mathbb C\), where
\[
Q_{\mathrm{right}}=x\,(\gamma x-y)\prod_{j=1}^{m}(\rho_jx-y)
\]
is the forward support annihilator.  Conversely every multiple of \(Q_{\mathrm{right}}\) vanishes at all directions and hence annihilates every block.  Thus
\[
\bigl[q(\partial)f_{n+k}=0\ \text{for every }0\le k\le m-1\bigr]
\quad\Longleftrightarrow\quad
\exists c\in\mathbb C:\ q=c\,Q_{\mathrm{right}} .
\]
% lean: lowerForwardApolarKernelIdentity lowerForwardEvaluationsVanish forward_points_imply_supportAnnihilator_multiple lowerForwardSupportAnnihilatorInKernel

\item \textbf{Forward generic separation.}  Suppose some \(\eta=(\delta,\sigma,d)\in F^{\mathrm{left}}_{m,L}\) satisfied
\[
\Phi^{\mathrm{right}}_{m,L}(\theta)=\Phi^{\mathrm{left}}_{m,L}(\eta),
\]
so that all divided-power blocks of the two parameters coincide.  Let
\[
Q_{\mathrm{left}}=-\,y\prod_{j=1}^{m}(x-\sigma_jy)\,(x-\delta y)
\]
be the reverse support annihilator of \(\eta\), homogeneous of degree \(n\) and nonzero.  It vanishes at every reverse direction, so it annihilates every reverse block of orders \(n,\ldots,L\), hence, by the displayed equality, every forward block \(f_{n+k}\), \(0\le k\le m-1\).  Step 3 therefore gives \(c\in\mathbb C\) with
\[
Q_{\mathrm{left}}=c\,Q_{\mathrm{right}} .
\]
If \(c=0\) then \(Q_{\mathrm{left}}=0\), which is false.  If \(c\ne0\), then \(y\mid Q_{\mathrm{left}}\) — the reverse family contains \(v_0=(1,0)\), contributing the factor \(-y\) — forces \(y\mid Q_{\mathrm{right}}\); but no factor of \(Q_{\mathrm{right}}\) is divisible by \(y\), since \(\gamma\ne0\) and every \(\rho_j\ne0\).  Both cases are impossible, so no such \(\eta\) exists.  This proves the forward generic-separation clause with the theorem-local exceptional set \(E^{\mathrm{right}}\).
% lean: lowerForwardReverseImpossible lowerReverseSupportAnnihilatorInKernel X1_dvd_supportAnnihilator_reverse not_X1_dvd_supportAnnihilator_forward

\item \textbf{Reverse generic separation.}  Let \(\eta\in F^{\mathrm{left}}_{m,L}\setminus E^{\mathrm{left}}\).  Exactly as in Step 3, \(\Delta^{\mathrm{left}}_m(\eta)\ne0\) gives \(\sigma_i\ne0\) for all \(i\) and \(\Pi_{\mathrm l}(\eta)\ne0\), while feasibility gives \(\delta\ne0\) and pairwise distinct \(\delta,\sigma_1,\ldots,\sigma_m\); hence the \(m+1\) finite reverse directions \(v_j=(\sigma_j,1)\), \(v_{m+1}=(\delta,1)\) have distinct nonzero first coordinates and, with \(v_0=(1,0)\), the \(n\) reverse directions are pairwise non-proportional.  The same two steps give the reverse kernel identity
\[
\bigl[q(\partial)f_{n+k}=0\ \text{for every }0\le k\le m-1\bigr]
\quad\Longleftrightarrow\quad
\exists c\in\mathbb C:\ q=c\,Q_{\mathrm{left}},
\qquad t=\Phi^{\mathrm{left}}_{m,L}(\eta).
\]
If some \(\theta\in F^{\mathrm{right}}_{m,L}\) satisfied \(\Phi^{\mathrm{left}}_{m,L}(\eta)=\Phi^{\mathrm{right}}_{m,L}(\theta)\), its forward support annihilator \(Q_{\mathrm{right}}\) would annihilate the reverse blocks and hence equal \(c\,Q_{\mathrm{left}}\); \(c=0\) is impossible since \(Q_{\mathrm{right}}\ne0\), and \(c\ne0\) transfers the divisibility \(x\mid Q_{\mathrm{right}}\) (the forward family contains \(u_{m+1}=(0,1)\), contributing the factor \(x\)) to \(x\mid Q_{\mathrm{left}}\), contradicting \(\delta\ne0\) and \(\sigma_j\ne0\).  Hence no such \(\theta\) exists.
% lean: lowerReverseApolarKernelIdentity lowerReverseForwardImpossible X0_dvd_supportAnnihilator_forward not_X0_dvd_supportAnnihilator_reverse

\item \textbf{The information order.}  Steps 2, 4 and 5 establish, for each arrow, a proper real algebraic exceptional subset whose complement meets the corresponding feasible region and off which the opposite arrow admits no representation with the same truncated cumulants; that is exactly the generic real separation criterion of \cref{obj:def:information-order} at truncation order \(L=2m+1\).  Since \(m\ge3\) we have \(2\le 2m+1\), so \(2m+1\) belongs to the set of admissible orders whose infimum defines \(K^\star(m)\), whence
\[
K^\star(m)\le 2m+1 .
\]
If \(K^\star(m)\) were equal to \(2m+2\), the displayed inequality would read \(2m+2\le 2m+1\), which is false; hence
\[
K^\star(m)\ne 2m+2 .
\]
% lean: separatesAtOrder lowerOrderApolarSeparation informationOrder improvedRealInformationOrder
\end{enumerate}
\end{proof}
