\begin{proof}[Proof of \cref{obj:thm:generic-arrow-recovery-and-fiber-obstruction}]
Fix \(m\ge 1\), put \(K=2m+2\) and \(n=m+2\), and take for \(U^{\mathrm{right}}_m\) and \(U^{\mathrm{left}}_m\) the two loci produced by \cref{obj:thm:generic-apolar-arrow-recovery}.  That theorem already supplies the relative genericity clause (each locus is contained in the corresponding generic locus of \cref{obj:def:generic-parameter-loci} and is relatively Zariski open and dense) and the real-incidence clause (the Euclidean-open sets meeting the feasible regions, with complexifications inside the loci).  It also supplies, for each \(\theta\in U^{\mathrm{right}}_m\), three further items used below: the empty opposite-arrow fiber, the univariate polynomial \(Q\) with \(Q\ne 0\), \(Q(0)\ne0\) and root multiset \(\{\gamma,\rho_1,\ldots,\rho_m\}\), and the kernel identity: writing
\[
f_r(x,y)=\sum_{a=0}^{r}\binom{r}{a}\bigl[\Phi^{\mathrm{right}}_{m,K}(\theta)\bigr]_{r,a}x^{r-a}y^{a},
\qquad
Q_D=\prod_{j=0}^{m+1}\bigl(u_{j2}x-u_{j1}y\bigr),
\]
one has, for every homogeneous binary form \(q\) of degree \(n\),
\[
\bigl[q(\partial)f_{n+k}=0\ \text{for every }0\le k\le m\bigr]
\quad\Longleftrightarrow\quad
\exists c\in\mathbb C:\ q=cQ_D .
\]
We refer to this last display as the kernel identity.
The mirror statements hold on \(U^{\mathrm{left}}_m\).
% lean: generic_apolar_arrow_recovery

\begin{enumerate}
\item \textbf{Forward multiset recovery on the whole fiber.}  Let \(\theta=(\gamma,\rho,c)\in U^{\mathrm{right}}_m\) and let \(\theta'=(\gamma',\rho',c')\) lie in \(R^{\mathrm{right}}_{m,K}\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta)\bigr)\).  Membership in the fiber compares the two cumulant vectors on the retained coordinates \(2\le r\le K\), \(a\le r\), and both maps vanish outside that range, so
\[
\Phi^{\mathrm{right}}_{m,K}(\theta')=\Phi^{\mathrm{right}}_{m,K}(\theta).
\]
Let \(Q_D'=\prod_{j}(u'_{j2}x-u'_{j1}y)\) be the support annihilator of the loading family of \(\theta'\).  It is homogeneous of degree \(n\) and nonzero, because each loading direction \(u'_j\) is either \((1,\cdot)\) or \((0,1)\) and hence contributes a nonzero linear factor.  Since \(Q_D'\) vanishes at every direction \(u'_j\), and each divided-power block of \(\theta'\) is \(\sum_j c'_{jr}\,(u'_{j1}x+u'_{j2}y)^{r}\), the operator \(Q_D'(\partial)\) kills every block of \(\theta'\), hence — by the displayed equality of cumulant vectors — every block \(f_{n+k}\) of \(\theta\).  By the kernel identity there is \(c\) with \(Q_D'=c\,Q_D\), and \(c\ne0\) since \(Q_D'\ne0\).  Dehomogenizing at \(x=1\) turns \(Q_D\) into \((-1)^{m+1}(z-\gamma)\prod_{i=1}^{m}(z-\rho_i)\) and \(Q_D'\) into the corresponding product \((-1)^{m+1}(z-\gamma')\prod_{i=1}^{m}(z-\rho'_i)\) for \(\theta'\); equal up to the nonzero scalar \(c\), the two have the same roots, so
\[
\{\gamma',\rho'_1,\ldots,\rho'_m\}=\{\gamma,\rho_1,\ldots,\rho_m\}
\]
as multisets.  This is the forward same-arrow recovery clause.
% lean: forward_slopes_determined_by_kernel_identity forward_supportAnnihilator_in_contraction_kernel loadingSlopeMultiset

\item \textbf{Forward opposite-arrow exclusion, and nonzero latent slopes.}  For the same \(\theta\), \cref{obj:thm:generic-apolar-arrow-recovery} gives
\[
R^{\mathrm{left}}_{m,K}\!\left(\Phi^{\mathrm{right}}_{m,K}(\theta)\right)=\varnothing .
\]
Moreover, if some \(\rho_i\) were \(0\), then \(0\) would be a root of the polynomial \(Q\), whose root multiset is \(\{\gamma,\rho_1,\ldots,\rho_m\}\), contradicting \(Q(0)\ne0\).  Hence
\[
\rho_i\ne0\qquad(1\le i\le m).
\]
% lean: latent_slopes_ne_zero_of_root_polynomial

\item \textbf{The direct/latent swap.}  Fix \(i\in\{1,\ldots,m\}\) and define \(\theta'=(\gamma',\rho',c')\) by exchanging the direct source with the \(i\)th latent source, slope and weight family together:
\[
\gamma'=\rho_i,
\qquad
\rho'_i=\gamma,\qquad \rho'_\ell=\rho_\ell\ (\ell\ne i),
\qquad
c'_{0r}=c_{ir},\qquad c'_{ir}=c_{0r},\qquad c'_{jr}=c_{jr}\ (j\notin\{0,i\}),
\]
for all \(2\le r\le K\).  This is exactly the swap of the full pairs \((\gamma,(c_{0r})_{r=2}^K)\) and \((\rho_i,(c_{ir})_{r=2}^K)\) required in the statement, with all other coordinates fixed.
% lean: forwardDirectLatentSwap IsForwardDirectLatentSwap forwardDirectLatentSwap_spec

\item \textbf{The swap stays generic.}  The entries of \(\theta'\) are those of \(\theta\) permuted, so its retained weights are again all nonzero and its slope list \(\gamma',\rho'_1,\ldots,\rho'_m\) is the list \(\gamma,\rho_1,\ldots,\rho_m\) with two entries exchanged, hence again pairwise distinct; and \(\gamma'=\rho_i\ne0\) by Step 2.  Thus \(\theta'\in\Theta^{\mathrm{right},\circ}_{m,K}\).
% lean: forwardDirectLatentSwap_generic gamma_ne_rho_of_generic

\item \textbf{The swap preserves the cumulant vector.}  Both exchanged loading directions have first coordinate \(1\):
\[
u_0(\theta)=(1,\gamma),\qquad u_i(\theta)=(1,\rho_i),
\]
so exchanging the two slopes exchanges the two directions, \(u_0(\theta')=u_i(\theta)\) and \(u_i(\theta')=u_0(\theta)\), while \(u_j(\theta')=u_j(\theta)\) for all other \(j\); the weights were exchanged along with them.  Hence for every retained \((r,a)\) the defining sum of \cref{obj:def:forward-cumulant-map} is the original sum reindexed by the transposition of the source labels \(0\) and \(i\), and finite sums are invariant under such reindexing:
\[
\Phi^{\mathrm{right}}_{m,K}(\theta')=\Phi^{\mathrm{right}}_{m,K}(\theta).
\]
% lean: forwardCumulantMap_directLatentSwap

\item \textbf{The swap is outside the admissible orbit.}  By \cref{obj:def:admissible-source-swaps} every admissible relabeling \(\pi\in G_m\) leaves the direct slope untouched, so every member of the \(G_m\)-orbit of \(\theta\) has direct slope \(\gamma\).  But \(\theta'\) has direct slope \(\rho_i\), and genericity gives \(\gamma\ne\rho_i\).  Hence \(\theta'\notin G_m\cdot\theta\).
% lean: forwardDirectLatentSwap_not_mem_admissibleOrbit admissibleOrbit

\item \textbf{Exact dimension of the forward fiber for \(m\ge2\).}  Write \(Z=R^{\mathrm{right}}_{m,K}\bigl(\Phi^{\mathrm{right}}_{m,K}(\theta)\bigr)\).  For a parameter \(\vartheta\) in the retained-coordinate space with loading family \(u(\vartheta)\), let
\[
Z^{\mathrm{fix}}(\vartheta)=\{\theta'\in Z:\ \gamma'=\vartheta_\gamma,\ \rho'=\vartheta_\rho\}
\]
be the part of the fiber with the loading coordinates frozen at those of \(\vartheta\).  If \(\theta'\) has the same loadings as \(\vartheta\), then \(\theta'\in Z^{\mathrm{fix}}(\vartheta)\) exactly when its weight difference \(e_{jr}=c'_{jr}-\vartheta_{c,jr}\) lies in
\[
\mathcal K(\vartheta)=\Bigl\{(e_{jr}):\ \sum_{j=0}^{m+1}e_{jr}\,(u_{j1})^{r-a}(u_{j2})^{a}=0
\ \ \text{for all }2\le r\le K,\ 0\le a\le r\Bigr\},
\]
because the cumulant map is linear in the weights at fixed loadings.  So \(Z^{\mathrm{fix}}(\vartheta)\), when nonempty, is a translate of the linear space \(\mathcal K(\vartheta)\), hence an irreducible closed set of dimension \(\dim\mathcal K(\vartheta)\).

The conditions in \(\mathcal K(\vartheta)\) decouple across orders, and at order \(r\), writing \(s_0=\vartheta_\gamma\), \(s_j=\vartheta_{\rho,j}\) for the \(m+1\) finite slopes and using \(u_j=(1,s_j)\), \(u_{m+1}=(0,1)\), they read
\[
\sum_{j=0}^{m}e_{jr}\,s_j^{\,a}=0\ \ (0\le a\le r-1),
\qquad
\sum_{j=0}^{m}e_{jr}\,s_j^{\,r}+e_{m+1,r}=0 .
\]
The last equation determines \(e_{m+1,r}\) from the others and imposes no further restriction; the first block is the Vandermonde system in the \(m+1\) nodes \(s_0,\ldots,s_m\) with \(r\) rows, which has rank \(\min(r,m+1)\) when the nodes are pairwise distinct.  Hence the order-\(r\) kernel has dimension \((m+1-r)^{+}\), and
\[
\dim\mathcal K(\vartheta)=\sum_{r=2}^{K}(m+1-r)^{+}=\sum_{r=2}^{m}(m+1-r)=\frac{m(m-1)}2 .
\]
Applying this to \(\vartheta=\theta\), whose slopes are pairwise distinct, \(Z^{\mathrm{fix}}(\theta)\) contains \(\theta\), so it is a nonempty irreducible closed subset of \(Z\) of dimension \(m(m-1)/2\); this gives a strictly increasing chain of \(m(m-1)/2+1\) irreducible closed subsets of \(Z\).

For the reverse inequality, let \(T\subseteq Z\) be irreducible and closed.  By Step 1 every point of \(Z\) has loading-slope multiset \(\{\gamma,\rho_1,\ldots,\rho_m\}\), so each of the \(m+1\) slope coordinates takes values in that finite set on \(T\); a coordinate function taking finitely many values on an irreducible set is constant, since otherwise \(T\) would split as a finite union of proper closed pieces.  Hence all loading coordinates are constant on \(T\), so \(T\subseteq Z^{\mathrm{fix}}(\theta_0)\) for some \(\theta_0\in Z\).  The slopes of \(\theta_0\) are again pairwise distinct (their multiset is that of \(\theta\), which has \(m+1\) distinct entries), so \(Z^{\mathrm{fix}}(\theta_0)\) has dimension \(m(m-1)/2\) and contains no chain of \(m(m-1)/2+2\) irreducible closed subsets; therefore neither does \(Z\).  Combining the two bounds, for \(m\ge2\) the forward fiber has exact relative Zariski dimension
\[
\frac{m(m-1)}{2}.
\]
% lean: forwardFixedLoadingFiber loadingBandWeightKernel endpointOrderSynthesis endpointOrderSynthesis_ker_finrank bandWeightKernel_finrank_flagship forward_fixedLoadingFiber_dimension_of_injective forward_irreducible_subset_fixedLoading forward_full_fiber_dimension

\item \textbf{The reverse locus.}  Let \(\eta=(\delta,\sigma,d)\in U^{\mathrm{left}}_m\).  The four preceding arguments run verbatim after exchanging the two coordinate axes, with the reverse loading family \(v_0=(1,0)\), \(v_j=(\sigma_j,1)\), \(v_{m+1}=(\delta,1)\) in place of the forward one and dehomogenization at \(y=1\) in place of \(x=1\).  They give: every \(\eta'\in R^{\mathrm{left}}_{m,K}\bigl(\Phi^{\mathrm{left}}_{m,K}(\eta)\bigr)\) satisfies
\[
\{\delta',\sigma'_1,\ldots,\sigma'_m\}=\{\delta,\sigma_1,\ldots,\sigma_m\};
\]
the opposite-arrow fiber
\[
R^{\mathrm{right}}_{m,K}\!\left(\Phi^{\mathrm{left}}_{m,K}(\eta)\right)=\varnothing
\]
is empty, and \(\sigma_i\ne0\) for every \(i\) because the reverse root polynomial does not vanish at \(0\); for each \(i\) the parameter \(\eta'\) defined by
\[
\delta'=\sigma_i,\qquad \sigma'_i=\delta,\qquad \sigma'_\ell=\sigma_\ell\ (\ell\ne i),
\qquad
d'_{m+1,r}=d_{ir},\qquad d'_{ir}=d_{m+1,r},\qquad d'_{jr}=d_{jr}\ \text{otherwise},
\]
lies in \(\Theta^{\mathrm{left},\circ}_{m,K}\), satisfies
\[
\Phi^{\mathrm{left}}_{m,K}(\eta')=\Phi^{\mathrm{left}}_{m,K}(\eta)
\]
— here the exchanged directions \(v_{m+1}=(\delta,1)\) and \(v_i=(\sigma_i,1)\) both have second coordinate \(1\), so exchanging the two slopes exchanges the two directions and the sum is reindexed by the transposition of the source labels \(m+1\) and \(i\) — and is not in the admissible \(G_m\)-orbit of \(\eta\), since admissible relabelings fix the reverse direct slope \(\delta\) while \(\eta'\) has direct slope \(\sigma_i\ne\delta\).  Finally, for \(m\ge2\) the same fixed-loading decomposition, now with the \(m+1\) distinct reverse finite loading coordinates \(\delta,\sigma_1,\ldots,\sigma_m\) as Vandermonde nodes, gives exact relative Zariski dimension
\[
\frac{m(m-1)}{2}
\]
for \(R^{\mathrm{left}}_{m,K}\bigl(\Phi^{\mathrm{left}}_{m,K}(\eta)\bigr)\).
% lean: reverse_slopes_determined_by_kernel_identity reverseDirectLatentSwap reverseDirectLatentSwap_generic reverseCumulantMap_directLatentSwap reverseDirectLatentSwap_not_mem_admissibleOrbit reverse_full_fiber_dimension
\end{enumerate}
\end{proof}
