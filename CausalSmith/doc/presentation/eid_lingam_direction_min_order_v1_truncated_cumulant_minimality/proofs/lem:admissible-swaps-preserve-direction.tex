\begin{proof}[Proof of \cref{obj:lem:admissible-swaps-preserve-direction}]
Throughout, fix \(\pi\in G_m\) and let \(\tilde\pi\) be its extension to the full source-label set \(\{0,1,\ldots,m+1\}\) obtained by fixing the two axis labels:
\[
\tilde\pi(0)=0,
\qquad
\tilde\pi(i)=\pi(i)\quad(1\le i\le m),
\qquad
\tilde\pi(m+1)=m+1 .
\]
Since \(\pi\) is a bijection of \(\{1,\ldots,m\}\), the map \(\tilde\pi\) is a bijection of \(\{0,1,\ldots,m+1\}\).  With this notation the forward action of \cref{obj:def:admissible-source-swaps} reads
\[
\pi\theta=\bigl(\gamma,(\rho_{\pi(i)})_{i=1}^m,(c_{\tilde\pi(j),r})_{0\le j\le m+1,\ 2\le r\le L}\bigr),
\]
and the reverse action is the same formula with \((\delta,\sigma,d)\) in place of \((\gamma,\rho,c)\); in particular the direct slopes \(\gamma\) and \(\delta\) are untouched.
% lean: permMiddle permMiddleEquiv admissibleSourceSwap

\begin{enumerate}
\item \textbf{Loading vectors move with the label.}  Write \(u_j(\theta)=(u_{j1},u_{j2})\) for the forward loading vector attached to label \(j\) by the slopes carried by \(\theta\), as in \cref{obj:def:forward-cumulant-map}, and \(v_j(\eta)=(v_{j1},v_{j2})\) for the reverse loading vector attached to label \(j\) by the slopes carried by \(\eta\), as in \cref{obj:def:reverse-cumulant-map}.  For every source label \(j\),
\[
u_j(\pi\theta)=u_{\tilde\pi(j)}(\theta),
\qquad
v_j(\pi\eta)=v_{\tilde\pi(j)}(\eta).
\]
Indeed, at \(j=0\) both sides of the first identity are \((1,\gamma)\) and at \(j=m+1\) both are \((0,1)\), because \(\tilde\pi\) fixes those two labels and the action does not change \(\gamma\); for \(1\le j\le m\) the left side is \((1,\rho_{\pi(j)})\), which is the loading of \(\theta\) at label \(\pi(j)=\tilde\pi(j)\).  The reverse identity is the same computation with \(v_0=(1,0)\), \(v_{m+1}=(\delta,1)\), and \(v_j=(\sigma_j,1)\).
% lean: forwardLoading_admissibleSourceSwap reverseLoading_admissibleSourceSwap

\item \textbf{Complex cumulant maps.}  Fix a retained coordinate \((r,a)\) with \(2\le r\le L\) and \(0\le a\le r\).  Using the two displays above,
\[
\bigl[\Phi^{\mathrm{right}}_{m,L}(\pi\theta^{\mathbb C})\bigr]_{r,a}
=\sum_{j=0}^{m+1} c^{\mathbb C}_{\tilde\pi(j),r}\,
   \bigl(u_{\tilde\pi(j),1}\bigr)^{r-a}\bigl(u_{\tilde\pi(j),2}\bigr)^{a}
=\sum_{j=0}^{m+1} c^{\mathbb C}_{jr}\,(u_{j1})^{r-a}(u_{j2})^{a}
=\bigl[\Phi^{\mathrm{right}}_{m,L}(\theta^{\mathbb C})\bigr]_{r,a},
\]
the middle equality because \(\tilde\pi\) is a bijection of the summation index set \(\{0,\ldots,m+1\}\) and a finite sum is invariant under bijective reindexing.  Outside the retained range both sides vanish by the coordinate convention of \cref{obj:def:forward-cumulant-map}.  The identical computation with \(v_j\) and \(d\) gives the reverse case, so
\[
\Phi^{\mathrm{right}}_{m,L}(\pi\theta^{\mathbb C})=\Phi^{\mathrm{right}}_{m,L}(\theta^{\mathbb C}),
\qquad
\Phi^{\mathrm{left}}_{m,L}(\pi\eta^{\mathbb C})=\Phi^{\mathrm{left}}_{m,L}(\eta^{\mathbb C}).
\]
% lean: sum_permMiddle forwardCumulantMap_admissibleSourceSwap reverseCumulantMap_admissibleSourceSwap

\item \textbf{Real feasible regions.}  Let \(\theta=(\gamma,\rho,c)\in F^{\mathrm{right}}_{m,L}\).  \Cref{obj:def:real-feasible-regions} requires three things of \(\pi\theta\), and each is transported.

First, the direct slope of \(\pi\theta\) is \(\gamma\ne0\).

Second, write \(\hat\pi\) for the bijection of the slope index set \(\{0,1,\ldots,m\}\) with \(\hat\pi(0)=0\) and \(\hat\pi(i)=\pi(i)\) for \(1\le i\le m\), and write \(s(\theta):\{0,\ldots,m\}\to\mathbb R\) for the slope tuple \(s(\theta)_0=\gamma\), \(s(\theta)_i=\rho_i\).  Then
\[
s(\pi\theta)_k=s(\theta)_{\hat\pi(k)}
\qquad(0\le k\le m),
\]
so \(s(\pi\theta)=s(\theta)\circ\hat\pi\) is injective whenever \(s(\theta)\) is; that is, \(\gamma,\rho_{\pi(1)},\ldots,\rho_{\pi(m)}\) are again pairwise distinct.
% lean: permLeadingEquiv finCons_comp_perm_injective

Third, for the source-realizability clause, let \(\nu_0,\ldots,\nu_{m+1}\) be centered non-Gaussian real laws with finite \(L\)-th absolute moment such that \(\operatorname{Cum}_r(\nu_j)=c_{jr}\) for \(2\le r\le L\).  For the label \(j\) of \(\pi\theta\) take the law \(\nu_{\tilde\pi(j)}\): it is centered, non-Gaussian, has finite \(L\)-th absolute moment, and
\[
\operatorname{Cum}_r\bigl(\nu_{\tilde\pi(j)}\bigr)=c_{\tilde\pi(j),r}=(\pi c)_{jr},
\qquad 2\le r\le L .
\]
Since the retained coordinates are indexed by \(2\le r\le L\), the reindexed weight family \((\pi c)_{jr}=c_{\tilde\pi(j),r}\) also vanishes for \(r<2\) and \(r>L\) whenever \((c_{jr})\) does, so \(\pi\theta\) stays in the same finite coordinate space.  Hence \(\pi\theta\in F^{\mathrm{right}}_{m,L}\).  The reverse case is the same argument with \((\delta,\sigma,d)\), giving \(\pi\eta\in F^{\mathrm{left}}_{m,L}\).
% lean: realFeasibleRegion_admissibleSourceSwap

\item \textbf{Forward witnesses.}  Let \(\mu\), \(S_0,\ldots,S_{m+1}\), and \((X,Y)\) be as in the statement, with \(\theta\in F^{\mathrm{right}}_{m,L}\), \(P_{\mathrm{right}}=\mu\circ(X,Y)^{-1}\), and \(\operatorname{Cum}_r(S_j)=c_{jr}\) for \(2\le r\le L\).  Define the relabeled sources
\[
S'_j=S_{\tilde\pi(j)},\qquad 0\le j\le m+1 .
\]
Because \(\tilde\pi\) is injective, the family \((S'_j)_{j=0}^{m+1}\) is the original family precomposed with an injective index map, so mutual independence in \cref{obj:ass:independent-sources} is inherited.  Centering, non-Gaussianity, and the finite moments of \cref{obj:ass:finite-cumulants} through order \(2m+2\) are the corresponding hypotheses evaluated at the label \(\tilde\pi(j)\):
\[
\int_\Omega S'_j\,d\mu=\int_\Omega S_{\tilde\pi(j)}\,d\mu=0,
\qquad
\mu\circ (S'_j)^{-1}=\mu\circ (S_{\tilde\pi(j)})^{-1}\ \text{is non-Gaussian},
\qquad
\mathbb E|S'_j|^{2m+2}=\mathbb E|S_{\tilde\pi(j)}|^{2m+2}<\infty .
\]
% lean: permMiddleEquiv

The forward axis equations of \cref{obj:ass:forward-axis-model} are preserved with the swapped parameter: for every \(\omega\in\Omega\),
\[
\sum_{j=0}^{m+1}u_{j1}(\pi\theta)\,S'_j(\omega)
=\sum_{j=0}^{m+1}u_{\tilde\pi(j),1}\,S_{\tilde\pi(j)}(\omega)
=\sum_{j=0}^{m+1}u_{j1}\,S_j(\omega)
=X(\omega),
\]
and identically with the second loading coordinate and \(Y\); the middle equality is again bijective reindexing of a finite sum by \(\tilde\pi\).  Forward noncollinearity for \(\pi\theta\) is the injectivity statement already proved, and the nonzero-edge condition of \cref{obj:ass:forward-nonzero-edge} holds because the direct slope is still \(\gamma\).  Finally, for every \(j\) and every \(2\le r\le L\),
\[
\operatorname{Cum}_r(S'_j)=\operatorname{Cum}_r(S_{\tilde\pi(j)})=c_{\tilde\pi(j),r}=(\pi c)_{jr}.
\]
Thus \(\bigl(\Omega,\mu,(S'_j)_j,X,Y,\pi\theta\bigr)\) satisfies every clause of \cref{obj:def:forward-lvlingam-class} with the same observed pair, hence with the same pushforward law, and \(P_{\mathrm{right}}\in\mathcal P^{\mathrm{right}}_{m,2m+2}\).
% lean: forwardAxisModel_admissibleSourceSwap finCons_comp_perm_injective

\item \textbf{Reverse witnesses.}  With the same relabeling \(S'_j=S_{\tilde\pi(j)}\), independence, centering, non-Gaussianity, and finiteness through order \(2m+2\) transport exactly as above.  The reverse axis equations of \cref{obj:ass:reverse-axis-model} are preserved because, for every \(\omega\in\Omega\),
\[
\sum_{j=0}^{m+1}v_{j1}(\pi\eta)\,S'_j(\omega)
=\sum_{j=0}^{m+1}v_{\tilde\pi(j),1}\,S_{\tilde\pi(j)}(\omega)
=\sum_{j=0}^{m+1}v_{j1}\,S_j(\omega)=X(\omega),
\]
and likewise for the second coordinate and \(Y\).  Reverse noncollinearity holds because \(s(\pi\eta)=s(\eta)\circ\hat\pi\) with \(s(\eta)_0=\delta\), \(s(\eta)_i=\sigma_i\), and the reverse nonzero-edge condition holds because \(\delta\) is unchanged.  The cumulant relation becomes
\[
\operatorname{Cum}_r(S'_j)=\operatorname{Cum}_r(S_{\tilde\pi(j)})=d_{\tilde\pi(j),r}=(\pi d)_{jr},
\qquad 2\le r\le L .
\]
Hence the same law satisfies \cref{obj:def:reverse-lvlingam-class}: \(P_{\mathrm{left}}\in\mathcal P^{\mathrm{left}}_{m,2m+2}\).
% lean: reverseAxisModel_admissibleSourceSwap finCons_comp_perm_injective

\item \textbf{Direction of structural models.}  Let \(M\) be a structural model with \(D(M)=X\to Y\) and parameter \(\theta\), in the sense of \cref{obj:def:separated-model-domain}.  Being a forward model with parameter \(\theta\) supplies two things: a forward representation of \(P_M\) by centered independent non-Gaussian sources whose loadings and order-\(r\) cumulants are exactly the coordinates of \(\theta\), and the cumulant identity
\[
\Phi^{\mathrm{right}}_{m,L}(\theta)=T_L(P_M).
\]
Relabel the sources of that representation by \(\tilde\pi\).  The forward-witness step turns them into a forward representation of the same law \(P_M\) with parameter \(\pi\theta\), the feasibility step gives \(\pi\theta\in F^{\mathrm{right}}_{m,L}\), and the map-invariance step gives
\[
\Phi^{\mathrm{right}}_{m,L}(\pi\theta)=\Phi^{\mathrm{right}}_{m,L}(\theta)=T_L(P_M).
\]
Let \(M'\) be the structural model with observational law \(P_M\), arrow \(X\to Y\), and parameter \(\pi\theta\); the three displays just established are exactly what \(M'\) has to carry, and the reverse-arrow clause is vacuous for it.  Then \(P_{M'}=P_M\), \(D(M')=X\to Y\), and \(M'\) has parameter \(\pi\theta\).  The reverse case is the same construction from the reverse representation of \(M\), producing \(M'\) with \(P_{M'}=P_M\), \(D(M')=Y\to X\), and parameter \(\pi\eta\).
% lean: realFeasibleRegion_admissibleSourceSwap forwardAxisModel_admissibleSourceSwap reverseAxisModel_admissibleSourceSwap forwardCumulantMap_admissibleSourceSwap reverseCumulantMap_admissibleSourceSwap

\item \textbf{Arrow-tagged orbits.}  On complex parameters the action is coordinatewise reindexing, so by the display defining \(\pi\theta\),
\[
(\pi c^{\mathbb C})_{jr}=c^{\mathbb C}_{\tilde\pi(j),r},
\qquad
(\pi d^{\mathbb C})_{jr}=d^{\mathbb C}_{\tilde\pi(j),r},
\]
which for \(1\le j\le m\) is the stated identity with \(\pi(j)\) and for the two axis labels is the identity on those coordinates.  Taking \(j\in\{0,m+1\}\) in the loading display gives
\[
u_0(\pi\theta^{\mathbb C})=u_0(\theta^{\mathbb C}),
\qquad
u_{m+1}(\pi\theta^{\mathbb C})=u_{m+1}(\theta^{\mathbb C}),
\]
so the boundary forward loading vectors are fixed.  Now let the group act on arrow-tagged parameters by
\[
\pi\cdot(b,\theta^{\mathbb C})=(b,\pi\theta^{\mathbb C}),
\qquad b\in\{\mathrm{right},\mathrm{left}\},
\]
so that the tagged orbits are
\[
\mathcal O(\mathrm{right},\theta^{\mathbb C})
=\bigl\{(\mathrm{right},\pi'\theta^{\mathbb C}):\pi'\in G_m\bigr\},
\qquad
\mathcal O(\mathrm{left},\eta^{\mathbb C})
=\bigl\{(\mathrm{left},\pi'\eta^{\mathbb C}):\pi'\in G_m\bigr\}.
\]
The tag component is carried unchanged by every \(\pi'\in G_m\), so every element of the first orbit has tag \(\mathrm{right}\) and every element of the second has tag \(\mathrm{left}\).  A common element would have to carry both tags at once, which is impossible; hence
\[
\mathcal O(\mathrm{right},\theta^{\mathbb C})\cap\mathcal O(\mathrm{left},\eta^{\mathbb C})=\varnothing .
\]
% lean: forwardLoading_admissibleSourceSwap admissibleSourceSwapArrow admissibleSourceSwapTagged arrowTaggedOrbit arrowTaggedOrbit_right_left_disjoint

\item \textbf{All admissible relabelings.}  The cumulant-map computation used only that the extension \(\tilde\pi\) of \(\pi\) is a bijection of \(\{0,\ldots,m+1\}\) fixing the two axis labels, which holds for every element of \(G_m\).  Repeating it verbatim with an arbitrary \(\tau\in G_m\) gives
\[
\Phi^{\mathrm{right}}_{m,L}(\tau\theta^{\mathbb C})
=\Phi^{\mathrm{right}}_{m,L}(\theta^{\mathbb C}),
\qquad
\Phi^{\mathrm{left}}_{m,L}(\tau\eta^{\mathbb C})
=\Phi^{\mathrm{left}}_{m,L}(\eta^{\mathbb C}).
\]
% lean: forwardCumulantMap_admissibleSourceSwap reverseCumulantMap_admissibleSourceSwap
\end{enumerate}
\end{proof}
