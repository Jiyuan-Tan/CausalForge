\begin{proof}[Proof of \cref{obj:thm:known-zero-instrument}]
Put
\[
D_n(X)=g_0(X)-\bar g_n(X),\qquad
b_n(X)=q_0(X)-\theta_0D_n(X),
\]
as in \cref{obj:def:outcome-contamination}, and write
\[
J(w)=J_{z_0,\ell}(w)=w^{\ell-1}e^{z_0w},\qquad w\in\mathbb C,
\]
as in \cref{obj:def:zero-instrument}.  From the definitions of \(\eta\) and \(Z_n\) in \cref{obj:def:plm-model,obj:def:contour-functional},
\[
Z_n=T-\bar g_n(X)=\eta+D_n(X).
\]
The definitions of \(\xi\) in \cref{obj:def:plm-model} and \(b_n\) above also give, by algebra,
\[
Y=\theta_0Z_n+b_n(X)+\xi .
\]
Membership in \(\mathcal P_{\mathrm{NG},n}\), together with the clipping convention in \cref{obj:def:plm-model}, supplies the integrability and boundedness used below.  Pulled back to the observation space, \cref{obj:def:non-gaussian-class} gives
\[
|g_0(X)|\le C_g,\qquad |q_0(X)|\le C_q,\qquad |\theta_0|\le C_{\theta}
\qquad P\text{-a.s.},
\]
and the clipping gives \(|\bar g_n(X)|\le C_g\).  Hence
\[
|D_n(X)|\le 2C_g,
\qquad
|b_n(X)|\le C_q+C_{\theta}(2C_g)
\qquad P\text{-a.s.}
\]
The treatment and outcome sub-Gaussian clauses in \cref{obj:def:non-gaussian-class} imply real exponential moments of every order for \(\eta\) and \(\xi\); the displayed bound on \(D_n\) then gives the corresponding real exponential moments for \(Z_n=\eta+D_n(X)\).  These facts justify the differentiated transforms, conditional expectations, and products appearing below.

\begin{enumerate}
\item Since \(M\) is analytic at \(z_0\) and has analytic order \(\ell\ge1\) there,
\[
M^{(\alpha)}(z_0)=0\qquad\text{for every }0\le\alpha<\ell .
\]
Moreover, for each such \(\alpha\),
\[
M^{(\alpha)}(z_0)
=
E_P\!\left[\eta^\alpha e^{z_0\eta}\right],
\]
by differentiating the moment-generating function under the expectation. Hence, for any \(d\in\mathbb R\),
\[
\begin{aligned}
E_P[J(\eta+d)]
&=
E_P\!\left[(\eta+d)^{\ell-1}e^{z_0(\eta+d)}\right]  \\
&=
e^{z_0d}
\sum_{\alpha=0}^{\ell-1}
\binom{\ell-1}{\alpha}d^{\ell-1-\alpha}
E_P\!\left[\eta^\alpha e^{z_0\eta}\right] \\
&=
e^{z_0d}
\sum_{\alpha=0}^{\ell-1}
\binom{\ell-1}{\alpha}d^{\ell-1-\alpha}
M^{(\alpha)}(z_0)
=0 .
\end{aligned}
\]
This is the shifted annihilation identity. % lean: zeroInstrument_integral_add_eq_zero

\item We next show the conditional identity. Let \(\mathcal A_X\) be any measurable covariate event. Since \(Z_n=\eta+D_n(X)\), the binomial expansion gives
\[
\begin{aligned}
&\int_{\{X\in\mathcal A_X\}} J(Z_n)\,dP  \\
&\quad =
\sum_{\alpha=0}^{\ell-1}
\binom{\ell-1}{\alpha}
\int
\eta^\alpha e^{z_0\eta}
D_n(X)^{\ell-1-\alpha}e^{z_0D_n(X)}
\mathbf 1_{\mathcal A_X}(X)\,dP .
\end{aligned}
\]
The treatment-noise independence in \cref{obj:def:non-gaussian-class} factors each summand as
\[
\binom{\ell-1}{\alpha}
E_P\!\left[\eta^\alpha e^{z_0\eta}\right]\,
E_P\!\left[
D_n(X)^{\ell-1-\alpha}e^{z_0D_n(X)}
\mathbf 1_{\mathcal A_X}(X)
\right],
\]
and the first factor is \(M^{(\alpha)}(z_0)=0\). Thus the integral over every covariate event is zero, which characterizes
\[
E_P[J(Z_n)\mid X]=0
\qquad P\text{-a.s.}
\]
% lean: zeroInstrument_condExp_learnedResidual_eq_zero

\item Define, for \(z\in\mathbb C\),
\[
F_n(z)=E_P[e^{zZ_n}].
\]
Since \(Z_n=\eta+D_n(X)\), the same independence gives the factorization
\[
F_n(z)=E_P[e^{z\eta}]\,E_P[e^{zD_n(X)}]=M(z)H_n(z),
\]
with \(H_n\) as in \cref{obj:def:contour-functional}. Also,
\[
E_P[Z_nJ(Z_n)]
=
E_P[Z_n^\ell e^{z_0Z_n}]
=
F_n^{(\ell)}(z_0).
\]
Differentiating \(F_n=MH_n\) \(\ell\) times at \(z_0\) yields
\[
F_n^{(\ell)}(z_0)
=
\sum_{\alpha=0}^{\ell}
\binom{\ell}{\alpha}
M^{(\alpha)}(z_0)H_n^{(\ell-\alpha)}(z_0).
\]
All terms with \(\alpha<\ell\) vanish by the analytic-order calculation above, and the remaining term is
\[
F_n^{(\ell)}(z_0)=M^{(\ell)}(z_0)H_n(z_0).
\]
Therefore
\[
E_P[Z_nJ(Z_n)]
=
M^{(\ell)}(z_0)H_n(z_0).
\]
% lean: zeroInstrument_learnedResidual_integral_eq

\item Finally, using the algebraic decomposition \(Y=\theta_0Z_n+b_n(X)+\xi\),
\[
E_P[YJ(Z_n)]
=
\theta_0E_P[Z_nJ(Z_n)]
+
E_P[b_n(X)J(Z_n)]
+
E_P[\xi J(Z_n)] .
\]
The second term is zero because \(b_n(X)\) is covariate-measurable and the conditional identity above gives
\[
E_P[b_n(X)J(Z_n)]
=
E_P\!\left[b_n(X)E_P[J(Z_n)\mid X]\right]
=0 .
\]
The third term is zero because \(J(Z_n)\) is measurable with respect to \((X,T)\), while \cref{obj:def:non-gaussian-class} gives \(E_P[\xi\mid X,T]=0\):
\[
E_P[\xi J(Z_n)]
=
E_P\!\left[J(Z_n)E_P[\xi\mid X,T]\right]
=0 .
\]
Thus
\[
E_P[YJ(Z_n)]
=
\theta_0E_P[Z_nJ(Z_n)] .
\]
Whenever \(E_P[Z_nJ(Z_n)]\ne0\), division gives
\[
\theta_0
=
\frac{E_P[YJ(Z_n)]}{E_P[Z_nJ(Z_n)]}.
\]
% lean: outcome_zeroInstrument_integral_eq_theta_mul
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:exact-contour-identification}]
Write \(\rho_j\) for the radius of the selected positively oriented bank circle \(C_j=\{z\in\mathbb C:|z|=\rho_j\}\).  The bank construction gives \(\rho_j>0\).  Put
\[
D_n(X)=g_0(X)-\bar g_n(X),\qquad
b_n(X)=q_0(X)-\theta_0(P)D_n(X),
\]
and
\[
B_n(z)=E_P\!\left[b_n(X)e^{zD_n(X)}\right],\qquad z\in\mathbb C .
\]
By \cref{obj:lem:l1-nuisance-zero-free},
\[
|D_n(X)|\le 2C_g\qquad P\text{-a.s.}
\]
The clipping defining \(\bar g_n\), the range conditions in \(\mathcal P_{\mathrm{NG},n}\), and \(|\theta_0(P)|\le C_\theta\) give
\[
|b_n(X)|
\le |q_0(X)|+|\theta_0(P)|\,|D_n(X)|
\le C_q+2C_\theta C_g
\qquad P\text{-a.s.}
\]
Consequently \(H_n\) and \(B_n\) are complex differentiable at every point.  Indeed, on the unit neighborhood of any \(z_0\in\mathbb C\),
\[
|D_n(X)e^{zD_n(X)}|
\le 2C_g\exp\{(|z_0|+1)2C_g\},
\]
and
\[
|b_n(X)D_n(X)e^{zD_n(X)}|
\le (C_q+2C_\theta C_g)(2C_g)
   \exp\{(|z_0|+1)2C_g\},
\]
so dominated differentiation under the integral applies to the weighted transforms.  Separately, the sub-Gaussian exponential-moment condition for the treatment innovation \(\eta\), together with \(Z_n=\eta+D_n(X)\) and the almost-sure bound on \(D_n\), gives \(E_P e^{tZ_n}<\infty\) for every real \(t\).  The standard moment-generating-function analyticity criterion therefore gives analyticity of \(F_n\) on a neighborhood of the closed disk \(\{|z|\le \rho_j\}\).

On the closed disk \(\{|z|\le\rho_j\}\), the nuisance zero-freeness assumption gives \(H_n(z)\ne0\).  Hence the quotient \(B_n/H_n\) is well-defined on the closed disk, differentiable in the open disk, and continuous on the closed disk.  Cauchy's theorem on the circle gives
\[
\oint_{C_j}\frac{B_n(z)}{H_n(z)}\,dz=0 .
\]

For \(z\in C_j\), \cref{obj:lem:observable-factorization} gives
\[
F_n(z)=M(z)H_n(z),
\qquad
G_n(z)=\theta_0(P)F_n'(z)+M(z)B_n(z),
\]
where \(M(z)=E_P[e^{z\eta}]\).  Since \(F_n(z)\ne0\) and \(H_n(z)\ne0\) on \(C_j\), also \(M(z)\ne0\) on \(C_j\).  Therefore, pointwise for \(z\in C_j\),
\[
\frac{G_n(z)}{F_n(z)}
=
\theta_0(P)\frac{F_n'(z)}{F_n(z)}
+
\frac{B_n(z)}{H_n(z)} .
\]
Integrating around \(C_j\) and using the Cauchy identity for the quotient \(B_n/H_n\) yields
\[
\oint_{C_j}\frac{G_n(z)}{F_n(z)}\,dz
=
\theta_0(P)
\oint_{C_j}\frac{F_n'(z)}{F_n(z)}\,dz .
\]

The function \(F_n\) is analytic near the closed disk and zero-free on \(C_j\), so the argument principle gives
\[
\frac{1}{2\pi i}
\oint_{C_j}\frac{F_n'(z)}{F_n(z)}\,dz
=
N_{C_j}(F_n).
\]
Thus
\[
\oint_{C_j}\frac{G_n(z)}{F_n(z)}\,dz
=
\theta_0(P)\,N_{C_j}(F_n)\,2\pi i .
\]
The positive-count assumption gives \(N_{C_j}(F_n)\ge1\), so \(N_{C_j}(F_n)\,2\pi i\ne0\).  Dividing by this nonzero factor gives
\[
\theta_0(P)
=
\{N_{C_j}(F_n)\,2\pi i\}^{-1}
\oint_{C_j}\frac{G_n(z)}{F_n(z)}\,dz .
\]

By \cref{obj:def:contour-functional}, the right-hand side is \(\Theta_{C_j}(F_n,G_n)\).  Therefore
\[
(\theta_0(P):\mathbb C)
=
\Theta_{C_j}(F_n,G_n)
=
\{N_{C_j}(F_n)\,2\pi i\}^{-1}
\oint_{C_j}\frac{G_n(z)}{F_n(z)}\,dz .
\]
\end{proof}

Le Cam's two-point MSE inequality therefore supplies a constant \(\lambda_{\mathrm{LC}}>0\), depending only on \(V_{\mathrm{KL}}b_{\theta}^2\), such that every measurable estimator \(\widetilde\theta\) satisfies
\[
\max_{\upsilon\in\{-,+\}}
E_{m_{\upsilon}}\!\left[
(\widetilde\theta-\theta_0(m_{\upsilon}))^2
\right]
\ge
\lambda_{\mathrm{LC}}\frac{b_{\theta}^2}{n}.
\]
Thus the fixed-code non-Gaussian minimax risk is at least
\[
\frac{\mathfrak c_{\mathrm{NG}}}{n},
\qquad
\mathfrak c_{\mathrm{NG}}=\lambda_{\mathrm{LC}}b_{\theta}^2>0.
\]
This is the fixed-code non-Gaussian converse. % lean: CausalSmith.Stat.SaPlmCumulantConverse.fixed_code_non_gaussian_minimax_lower

For a nonempty fixed-code JMS ACE class, the separate fixed-code ACE converse applies the same two-point argument starting from a law in that JMS ACE class.  The affine perturbations preserve JMS ACE membership, the target values \(\theta_-\) and \(\theta_+\), the law of \((X,T)\), the nuisance functions \(g_0,q_0\), and both clipped supplied codes.  The same product relative-entropy calculation and Le Cam inequality give a constant \(\mathfrak c_{\mathrm{JMS}}>0\), depending only on the primitive tuple, such that
\[
\frac{\mathfrak c_{\mathrm{JMS}}}{n}
\le
\inf_{\widetilde\theta}
\sup_{P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}(p;\mathrm{base})}
E_P\!\left[(\widetilde\theta-\theta_0(P))^2\right].
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.fixed_code_ace_minimax_lower

\item For the fixed-code non-Gaussian class, assume the hypotheses in the theorem.  The lower bound from the preceding step and \(c\le\mathfrak c_{\mathrm{NG}}\) give
\[
\frac{c}{n}\le \mathfrak R_n^{\mathrm{NG}}.
\]
The statistic \(\widehat\theta_{\mathrm{spec},n}\) is measurable by Step 3, so it is an admissible rule in the minimax infimum, and therefore
\[
\mathfrak R_n^{\mathrm{NG}}
\le
\sup_{P\in\mathcal P_{\mathrm{NG},n}(p;\mathrm{base})}
E_P\!\left[
(\widehat\theta_{\mathrm{spec},n}-\theta_0(P))^2
\right].
\]
For every \(P\) in this fixed-code class, the current clipped treatment code agrees with the base clipped treatment code.  Since the canonical inputs in \cref{obj:def:adaptive-contour-estimator} use the current clipped treatment code, the statistic assembled from the base treatment-code sequence agrees pointwise with the statistic assembled from the class member's treatment-code sequence.  Applying Step 6 to the latter statistic and then using this pointwise equality gives
\[
\sup_{P\in\mathcal P_{\mathrm{NG},n}(p;\mathrm{base})}
E_P\!\left[
(\widehat\theta_{\mathrm{spec},n}-\theta_0(P))^2
\right]
\le
\frac{C}{n}.
\]
The same pointwise equality reduces the absolute-error statistic in the generalized-quantile display to the statistic covered by Step 7, giving the displayed generalized-quantile bound over the same supremum. % lean: CausalSmith.Stat.SaPlmCumulantConverse.adaptive_rootn_minimax

\item For the fixed-code JMS ACE class, assume its nonemptiness, \(n\ge2\), and the same smallness condition on \(\varepsilon_{1,n}\).  The ACE lower bound from Step 8 and \(c\le\mathfrak c_{\mathrm{JMS}}\) give
\[
\frac{c}{n}
\le
\inf_{\widetilde\theta}
\sup_{P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}(p;\mathrm{base})}
E_P\!\left[
(\widetilde\theta-\theta_0(P))^2
\right].
\]
The estimator is measurable, hence admissible, so the infimum is bounded above by its worst-case risk.  By Step 5, every JMS ACE law lies in \(\mathcal P_{\mathrm{NG},n}\).  The fixed-code restriction gives the same current clipped treatment code as the base model, so the pointwise congruence from \cref{obj:def:adaptive-contour-estimator} identifies the statistic built from the base treatment-code sequence with the statistic built from the class member's treatment-code sequence.  Step 6 therefore yields
\[
\sup_{P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}(p;\mathrm{base})}
E_P\!\left[
(\widehat\theta_{\mathrm{spec},n}-\theta_0(P))^2
\right]
\le
\frac{C}{n}.
\]
The same pointwise equality and Step 7 give the asserted generalized-quantile bound on this class as well. % lean: CausalSmith.Stat.SaPlmCumulantConverse.adaptive_rootn_minimax
\end{enumerate}
\end{proof}

\(r,\delta,C_{\theta},C_g,C_q,\psi_{\eta},\psi_{\xi}\).

\begin{enumerate}
\item For the class relations, apply \cref{obj:lem:jms-ace-class-relations} at
the displayed parameter record and the displayed sample size \(n\).  It gives,
law by law,
\[
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
  \Longrightarrow
  P\in\mathcal P_{\mathrm{NG},n}
\]
and
\[
  P\in\mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}}
  \Longrightarrow
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}} .

 \mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
 \subseteq \mathcal P_{\mathrm{NG},n},
 \qquad
 \mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}}
 \subseteq \mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}} .
\]
The same result also gives, when \(s=r\),
\[
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
  \Longleftrightarrow
  P\in\mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}},
\]
hence the asserted equality of the two classes under the common clipped-code
convention. % lean: CausalSmith.Stat.SaPlmCumulantConverse.jms_ace_class_relations

\item Fix \(P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\) using the clipped
supplied codes \(\bar g_n,\bar q_n\), and suppose
\(\mathsf E_n^{\mathrm{JMS}}\), \(\varepsilon_{1,n}>0\), and
\(\varepsilon_{2,n}\ge0\).  If \(\varepsilon_{2,n}>0\), the positive-radius
part of the assumed published order-\(r\) ACE generalized-quantile guarantee,
with the objects and gates stated in \cref{obj:prop:jms-ace-alignment}, applied
with the supplied handle and the same clipped codes, gives
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le B_{\mathrm{ACE},n,\gamma}.

\(\chi>0\), let \(p^{(\chi)}\) be the parameter record obtained from \(p\) by
replacing the whole second radius sequence by
\[
  \varepsilon_{2,m}^{(\chi)}=\max\{\varepsilon_{2,m},\chi\},
  \qquad m\ge1,
\]
and leaving every other coordinate unchanged.  This sequence is again
nonnegative and nonincreasing.  Read the same observed-data law, regressions,
and supplied code sequences as a law \(P^{(\chi)}\) for \(p^{(\chi)}\).  If
\(0<\chi<\varepsilon_{1,n}\), then the class membership is preserved: every
condition in \(\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\) is unchanged
except the outcome-code radius, and at index \(n\)
\[
  \varepsilon_{2,n}^{(\chi)}=\chi
  \quad\text{with}\quad
  0\le \varepsilon_{2,n}\le \chi .
\]
Moreover, with
\[
  \omega_n=(\gamma n)^{-1/2}(\psi_{\xi}+C_{\theta}\psi_{\eta}),
\]
the denominator entering the second JMS eligibility quantity is unchanged:
\[
 \max\{\varepsilon_{1,n},\max\{\varepsilon_{2,n}^{(\chi)},\omega_n\}\}
 =
 \max\{\varepsilon_{1,n},\max\{\varepsilon_{2,n},\omega_n\}\},
\]
because \(\varepsilon_{2,n}=0\) and \(0<\chi<\varepsilon_{1,n}\).  All other
eligibility quantities are unchanged by construction, so
\(\mathsf E_n^{\mathrm{JMS}}\) holds for \(p^{(\chi)}\) at the same index.

Let
\[
  \Lambda_n
  =
  C_{\gamma} r!16^r\delta^{-1}\varepsilon_{1,n}^r .
\]
The assumed ACE guarantee includes \(C_{\gamma}>0\), and the displayed hypotheses
give \(\Lambda_n>0\).  For any real \(\beta>B_{\mathrm{ACE},n,\gamma}\), set
\[
  \chi_{\beta}
  =
  \min\!\left\{
    \frac{\varepsilon_{1,n}}{2},
    \frac{\beta-B_{\mathrm{ACE},n,\gamma}}{2\Lambda_n}
  \right\}.
\]
Then \(0<\chi_{\beta}<\varepsilon_{1,n}\), and the positive-radius result
applied to \(p^{(\chi_{\beta})}\) gives
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le
 B_{\mathrm{ACE},n,\gamma}+\Lambda_n\chi_{\beta}.
\]
By the definition of \(\chi_{\beta}\),
\[
  \Lambda_n\chi_{\beta}
  \le
  \frac{\beta-B_{\mathrm{ACE},n,\gamma}}{2},
\]
so the same quantile is at most \(\beta\).  Since this holds for every
\(\beta>B_{\mathrm{ACE},n,\gamma}\), the desired boundary inequality follows:
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le B_{\mathrm{ACE},n,\gamma}.
\] % lean: CausalSmith.Stat.SaPlmCumulantConverse.jms_ace_alignment

\item For the spectral guarantee, assume the displayed nonemptiness condition
and choose a law \(P^{\circ}\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\)
using the clipped supplied codes \(\bar g_n,\bar q_n\).  Define the fixed-code
ACE class
\[
 \mathcal A_n^{\star}
 =
 \left\{
 P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}:
 \bar g_n(P)=\bar g_n,\ \bar q_n(P)=\bar q_n
 \right\}.
\]
This is the ACE-class set associated with the base law \(P^{\circ}\) in
\cref{obj:thm:adaptive-rootn-minimax}.  The statistic computed from the
supplied treatment-code sequence agrees pointwise with the statistic computed
from the base law's supplied treatment-code sequence, because both constructions
use the current treatment code only through the residuals
\[
  T_i-\bar g_n(X_i),\qquad i=1,\ldots,n .
\]
The ACE-class part of \cref{obj:thm:adaptive-rootn-minimax}, applied with the
fixed primitive and range records, \(n\ge2\), and
\[
 \varepsilon_{1,n}
 \le
 \{4R_1\exp(2C_gR_1)\}^{-1},
\]
therefore yields
\[
 \sup_{P\in\mathcal A_n^{\star}}
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{spec},n}-\theta_0(P)|\bigr)
 \le
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma n}} .
\]
Since \(\mathcal A_n^{\star}\) is exactly the class of laws in the statement
with the clipped supplied codes \(\bar g_n,\bar q_n\), this is the asserted
spectral upper guarantee. % lean: CausalSmith.Stat.SaPlmCumulantConverse.adaptive_rootn_minimax

\item Finally consider a sequence \((p_n)\) satisfying the hypotheses in the

\[
  \mathcal L_n
  =
  \varepsilon_{1,n}^r\varepsilon_{2,n}
  +C_{\theta}\varepsilon_{1,n}^{r+1}
\]
and
\[
  \lambda_{\mathrm{ACE}}
  =
  C_{\gamma}r!16^r\delta^{-1}.
\]
The assumed ACE guarantee includes \(C_{\gamma}>0\), so
\(\lambda_{\mathrm{ACE}}>0\).  Along the sequence, the fixed-constant
identities in the hypotheses rewrite the ACE bound as
\[
 B_{\mathrm{ACE},n,\gamma}
 =
 \lambda_{\mathrm{ACE}}
 \left[
   \mathcal L_n
   +
   64(C_g+\psi_{\eta})^r
   \{r^2(C_g+\psi_{\eta})+\psi_{\xi}+C_{\theta}\psi_{\eta}\}
   (\gamma n)^{-1/2}
 \right].
\]
The second summand in brackets is nonnegative, hence eventually
\[
  \lambda_{\mathrm{ACE}}\mathcal L_n
  \le B_{\mathrm{ACE},n,\gamma}.
\]

\[
  \frac{\mathcal L_n}{n^{-1/2}}\longrightarrow\infty .
\]

\[
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma n}}
 =
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma}}\;n^{-1/2}.
\]
Once \(\mathcal L_n>0\) and
\(\lambda_{\mathrm{ACE}}\mathcal L_n\le B_{\mathrm{ACE},n,\gamma}\), we have

 0\le
 \frac{\sqrt{C_{\mathrm{spec}}/(\gamma n)}}{B_{\mathrm{ACE},n,\gamma}}
 \le
 \frac{\sqrt{C_{\mathrm{spec}}/\gamma}}{\lambda_{\mathrm{ACE}}}
 \left(\frac{\mathcal L_n}{n^{-1/2}}\right)^{-1}.
\]
The right-hand side tends to \(0\), so the squeeze theorem gives
\[
 \frac{
   \sqrt{C_{\mathrm{spec}}/(\gamma n)}
 }{
   B_{\mathrm{ACE},n,\gamma}
 }
 \longrightarrow 0 .
\] % lean: CausalSmith.Stat.SaPlmCumulantConverse.tendsto_sqrt_div_over_of_dominates
\end{enumerate}

\begin{proof}[Proof of \cref{obj:prop:symmetric-mixture-reduction}]
Set
\[
t=\frac{\pi}{2},\qquad
A_{\sin}=\exp\!\left(-\frac{\pi^2}{8}\right),
\]
and
\[
K_{\sin}
=
4(C_q+2C_{\theta}C_g)^2+4\psi_{\xi}^2
+C_{\theta}^2(8+16C_g^2).
\]
Choose
\[
C=1+\frac{16}{A_{\sin}^2}K_{\sin}.
\]
Since \(A_{\sin}>0\) and \(K_{\sin}\ge0\), this constant is positive and depends only on
\(C_{\theta},C_g,C_q,\psi_{\xi}\). Fix \(p\), \(P\), and the model objects satisfying the
hypotheses. Define, on the observation space,
\[
Z_n=T-\bar g_n(X),\qquad
D_n(X)=g_0(X)-\bar g_n(X),
\]
so that \(Z_n=\eta+D_n(X)\). % lean: symmetric_mixture_reduction

\begin{enumerate}
\item The stipulated law gives the treatment-innovation transform explicitly. Indeed, for
every \(z\in\mathbb C\),
\[
\begin{aligned}
M(z)
&=\frac12\exp\!\left(-z+\frac{z^2}{2}\right)
  +\frac12\exp\!\left(z+\frac{z^2}{2}\right)  \\
&=\exp\!\left(\frac{z^2}{2}\right)\cosh z .
\end{aligned}
\]
Therefore
\[
M(it)=\exp\!\left(-\frac{\pi^2}{8}\right)\cos\!\left(\frac{\pi}{2}\right)=0.
\]
Near zero,
\[
\frac{d}{dz}\log M(z)=z+\tanh z.
\]
The third derivative at zero of \(z\) is zero, while
\[
(\tanh z)'=\operatorname{sech}^2 z,\qquad
(\tanh z)''=-2\tanh z\,\operatorname{sech}^2 z,
\qquad
(\tanh z)'''(0)=-2.
\]
Hence the fourth logarithmic derivative of \(M\) at zero is \(-2\), which is the asserted
fourth cumulant:
\[
\kappa_4(\eta)=-2 .
\]
All differentiations are justified by the Gaussian mixture's finite exponential moments. % lean: treatmentMGF_eq_symmetricGaussianMixture

\item The same transform computation gives the weighted characteristic moment used for the
denominator. Since
\[
M'(z)=\exp(z^2/2)\,z\cosh z+\exp(z^2/2)\sinh z,
\]
evaluating at \(z=it\) yields
\[
E_P\!\left[\eta\,e^{it\eta}\right]
=M'(it)
=iA_{\sin}.
\]
Using \(Z_n=\eta+D_n(X)\), the independence of \(\eta\) and \(X\), and the identity
\(M(it)=0\),
\[
\begin{aligned}
E_P[Z_ne^{itZ_n}]
&=E_P[\eta e^{it\eta}]\,E_P[e^{itD_n(X)}]
  +E_P[e^{it\eta}]\,E_P[D_n(X)e^{itD_n(X)}] \\
&=iA_{\sin}\,E_P[e^{itD_n(X)}].
\end{aligned}
\]
Taking imaginary parts gives
\[
E_P\!\left[Z_n\sin(tZ_n)\right]
=
A_{\sin}E_P\!\left[\cos(tD_n(X))\right],
\]
which is the displayed identity after substituting \(t=\pi/2\). % lean: learnedResidual_sine_denominator_identity

\item The \(L^1(P_X)\) treatment-code condition gives
\[
E_P|D_n(X)|
=
\|\bar g_n-g_0\|_{L^1(P_X)}
\le \varepsilon_{1,n}.
\]
Since \(\varepsilon_{1,n}\le1/\pi\),
\[
t\varepsilon_{1,n}\le \frac12.
\]
Using \(|1-\cos u|\le |u|\),
\[
E_P[\cos(tD_n(X))]
\ge
1-tE_P|D_n(X)|
\ge
1-t\varepsilon_{1,n}
\ge \frac12.
\]
Combining this with the denominator identity gives
\[
\frac{A_{\sin}}{2}
\le
E_P\!\left[Z_n\sin(tZ_n)\right].
\]
% lean: learnedResidual_sine_denominator_lower

\item Define the residual sine score
\[
\operatorname{Rem}_n(O)
=
\{Y-\theta_0(P)Z_n\}\sin(tZ_n)
\]
and the denominator sine score
\[
\operatorname{Den}_n(O)
=
Z_n\sin(tZ_n).
\]
With
\[
b_n(X)=q_0(X)-\theta_0(P)D_n(X),
\]
the outcome equation gives
\[
Y-\theta_0(P)Z_n=b_n(X)+\xi.
\]
The zero \(M(it)=0\) implies
\[
E_P[\sin(t\eta)]=0,\qquad E_P[\cos(t\eta)]=0.
\]
By independence of \(\eta\) and \(X\),
\[
E_P[b_n(X)\sin\{t(\eta+D_n(X))\}]=0.
\]
The conditional mean restriction for the outcome innovation gives
\[
E_P[\xi\sin(tZ_n)]
=
E_P\!\left[E_P(\xi\mid X,T)\sin(tZ_n)\right]
=0.
\]
Thus
\[
E_P[\operatorname{Rem}_n(O)]=0.
\]
% lean: sine_remainder_centered_of_mgf_zero

\item The mixture second moment is bounded by
\[
E_P[\eta^2]\le4.
\]
The range condition for \(g_0\), together with clipping of \(\bar g_n\), gives
\[
|D_n(X)|\le2C_g\qquad P\text{-a.s.}
\]
Since \(|\sin u|\le1\),
\[
E_P[\operatorname{Den}_n(O)^2]
\le
E_P[Z_n^2]
\le
2E_P[\eta^2]+2E_P[D_n(X)^2]
\le
8+16C_g^2.
\]
Also,
\[
|b_n(X)|\le C_q+2C_{\theta}C_g.
\]
The Luxemburg sub-Gaussian condition in \cref{obj:ass:xi-subgaussian} gives
\[
E_P[\xi^2]\le2\psi_{\xi}^2.
\]
Therefore
\[
E_P[\operatorname{Rem}_n(O)^2]
\le
4(C_q+2C_{\theta}C_g)^2+4\psi_{\xi}^2.
\]
% lean: sine_score_memLp_of_eta_second_lintegral_le

\item By \cref{obj:def:sine-estimator}, the statistic \(\widehat\theta_{\sin,n}\) is the
clipped ratio with denominator score \(\operatorname{Den}_n\), remainder score
\(\operatorname{Rem}_n\), frequency \(t=\pi/2\), and threshold \(A_{\sin}/4\), because
\[
\mathbb P_n\{Y\sin(tZ_n)\}
=
\theta_0(P)\,\mathbb P_n\operatorname{Den}_n
+\mathbb P_n\operatorname{Rem}_n.
\]
Let
\[
\mu_{\operatorname{Den}}=E_P[\operatorname{Den}_n(O)].
\]
The preceding lower bound gives \(\mu_{\operatorname{Den}}\ge A_{\sin}/2\). Applying
\cref{obj:lem:clipped-ratio-risk-decomposition} pathwise and then averaging under the
i.i.d. product law yields
\[
E_P\!\left[
(\widehat\theta_{\sin,n}-\theta_0(P))^2
\right]
\le
\frac{16}{A_{\sin}^2 n}
\left\{
E_P[\operatorname{Rem}_n(O)^2]
+
C_{\theta}^2E_P[\operatorname{Den}_n(O)^2]
\right\}.
\]
Using the two score bounds,
\[
E_P\!\left[
(\widehat\theta_{\sin,n}-\theta_0(P))^2
\right]
\le
\frac{16K_{\sin}}{A_{\sin}^2 n}
\le
\frac{C}{n}.
\]
Together with the identities and denominator lower bound proved above, this proves all claims. % lean: clippedRatioFromScores_lintegral_le
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:local-to-gaussian-partial-benchmarks}]
1.  Fix \(C_{\theta},C_g,C_q,\psi_{\xi}\).  By
\cref{obj:lem:gaussian-rademacher-l1-benchmark}, there is a constant
\(C>0\), depending only on these four displayed constants, such that whenever
an admissible parameter record has primitive entries
\(C_{\theta},C_g,C_q,\psi_{\xi}\) equal to the displayed constants and
cumulant order \(k=4\), every \(a\in(0,1]\) satisfying the
Gaussian--Rademacher law assumption on \(\eta\), the sampling and model
restrictions, and the \(L^1(P_X)\) treatment-code radius satisfies the full
Gaussian--Rademacher path conclusion with constant \(C\).  Applying this rule
to the theorem's records with \(r=3\) and \(k=4\) gives the first assertion. % lean: CausalSmith.Stat.SaPlmCumulantConverse.gaussian_rademacher_l1_benchmark

2.  Now fix a published ACE handle, \(\gamma\in\mathbb R\), and
\(C_{\gamma}\in\mathbb R\), and assume that the handle satisfies the theorem's
published ACE interface at \((\gamma,C_{\gamma})\).  Unpacking that hypothesis
gives first
\[
C_{\gamma}>0.
\]
It also gives the following specialization rule: for every parameter record
\(p\) whose probability-level coordinate is \(\gamma\), every real
\(\Delta>0\), and every pair of strictly positive current code radii
\[
\varepsilon_{1,n}>0,
\qquad
\varepsilon_{2,n}>0,
\]
every supplied code pair \((\widehat g,\widehat q)\) and every model satisfying
the clipped-code identities, the local ACE class conditions at \(\Delta\), and
the local JMS eligibility inequalities at \(\Delta\) obey
\[
Q_{P,1-\gamma}
\!\left(\left|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)\right|\right)
\le
B_{\mathrm{ACE},n,\gamma}(\Delta;C_{\gamma}).
\]
For the theorem, take
\[
\Delta=\delta_n .
\]
The standing assumption \(\delta_n>0\) supplies the required positivity of
\(\Delta\).  Together with the displayed strict positivity conditions
\(\varepsilon_{1,n}>0\) and \(\varepsilon_{2,n}>0\), the theorem's clipped-code
identities, local ACE class conditions, and local JMS eligibility inequalities
are exactly the premises of the specialization rule at this \(\Delta\).
Substitution therefore yields
\[
Q_{P,1-\gamma}
\!\left(\left|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)\right|\right)
\le
B_{\mathrm{ACE},n,\gamma}(\delta_n;C_{\gamma}),
\]
where
\[
B_{\mathrm{ACE},n,\gamma}(\Delta;C_{\gamma})
=
C_{\gamma}\,r!\,16^r\,\Delta^{-1}
\left(
\varepsilon_{1,n}^r\varepsilon_{2,n}
+C_{\theta}\varepsilon_{1,n}^{r+1}
+64(C_g+\psi_{\eta})^r
\bigl(r^2(C_g+\psi_{\eta})+\psi_{\xi}+C_{\theta}\psi_{\eta}\bigr)
(\gamma n)^{-1/2}
\right).
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.local_to_gaussian_partial_benchmarks
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:bounded-outcome-gaussian-degeneracy}]
\begin{enumerate}
\item Fix a model \(m\) in the displayed Gaussian class, and write \(P\) for its observed-data law. Define
\[
\eta=T-g_0(X),
\qquad
\xi=Y-q_0(X)-\theta_0(P)\eta .
\]
The conditional mean restrictions in \cref{obj:def:gaussian-class,obj:def:plm-model} give
\[
E_P[\xi\mid X,T]=0,
\]
and \(q_0(X)\) and \(\theta_0(P)\eta\) are measurable with respect to \((X,T)\). Hence conditional-expectation linearity yields
\[
E_P[Y\mid X,T]
=
q_0(X)+\theta_0(P)\eta
\qquad P\text{-a.s.}
\]
The bounded-outcome condition gives \(-C_q\le Y\le C_q\) \(P\)-a.s. Conditioning the two inequalities on \((X,T)\) gives
\[
-C_q\le E_P[Y\mid X,T]\le C_q
\qquad P\text{-a.s.},
\]
and therefore
\[
\bigl|q_0(X)+\theta_0(P)\eta\bigr|\le C_q
\qquad P\text{-a.s.}
\]
Conditioning the same bounded-outcome inequalities on \(X\), and using
\[
E_P[Y\mid X]=q_0(X),
\]
gives
\[
|q_0(X)|\le C_q
\qquad P\text{-a.s.}
\]
Combining the two displayed bounds,
\[
|\theta_0(P)\eta|
=
\bigl|(q_0(X)+\theta_0(P)\eta)-q_0(X)\bigr|
\le
\bigl|q_0(X)+\theta_0(P)\eta\bigr|+|q_0(X)|
\le 2C_q
\qquad P\text{-a.s.}
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.gaussianClass_theta_eq_zero

\item Suppose, for contradiction, that \(\theta_0(P)\ne0\), and define
\[
\Lambda_\theta=\frac{2C_q}{|\theta_0(P)|}\in\mathbb R .
\]
The previous step implies
\[
|\eta|\le \Lambda_\theta
\qquad P\text{-a.s.},
\]
so
\[
P\{\eta>\Lambda_\theta\}=0.
\]
By \cref{obj:def:gaussian-class}, \(\eta\sim N(0,\sigma^2)\) with \(\sigma>0\). Thus
\[
N(0,\sigma^2)\bigl((\Lambda_\theta,\infty)\bigr)=0.
\]
A nondegenerate Gaussian law is equivalent to Lebesgue measure on \(\mathbb R\), while
\[
\lambda\bigl((\Lambda_\theta,\infty)\bigr)=\infty .
\]
This contradiction proves
\[
\theta_0(P)=0 .
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.gaussianClass_theta_eq_zero

\item Now fix deterministic code sequences \(\widehat g,\widehat q\) and assume the fixed-code Gaussian class in the statement is inhabited. Consider the estimator
\[
\widetilde\theta_0(O_1,\ldots,O_n;\widehat g_n,\widehat q_n)=0 .
\]
For every model in that fixed-code class, the first part gives \(\theta_0(P)=0\), and hence
\[
E_{P^{\otimes n}}\!\left[
\{\widetilde\theta_0(O_1,\ldots,O_n;\widehat g_n,\widehat q_n)-\theta_0(P)\}^2
\right]=0 .
\]
Taking the supremum over the fixed-code class and then the infimum over all estimators in \cref{obj:def:minimax-risks} gives
\[
\mathfrak R_n^{\mathrm G}(p;\widehat g,\widehat q)\le0.
\]
Since this risk is nonnegative by definition,
\[
\mathfrak R_n^{\mathrm G}(p;\widehat g,\widehat q)=0 .
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.bounded_outcome_gaussian_degeneracy

\item It remains to evaluate the generalized-quantile criterion for the same zero estimator. Since \(\gamma\in(1/2,1)\),
\[
0<1-\gamma<1.
\]
For every model in the fixed-code Gaussian class,
\[
\bigl|\widetilde\theta_0(O_1,\ldots,O_n;\widehat g_n,\widehat q_n)-\theta_0(P)\bigr|=0
\qquad P^{\otimes n}\text{-a.s.}
\]
Thus the law of this loss is the point mass at zero. Its distribution function is \(0\) for negative thresholds and \(1\) at every nonnegative threshold, so the generalized lower \((1-\gamma)\)-quantile of \cref{obj:def:generalized-quantile} is
\[
Q_{P,1-\gamma}(0)=0 .
\]
Consequently the supremum of the fixed-code quantile loss for the zero estimator is \(0\), and the minimax infimum is at most \(0\). Nonnegativity in \cref{obj:def:minimax-risks} gives
\[
\mathfrak M_{n,1-\gamma}^{\mathrm G}(p;\widehat g,\widehat q)=0 .
\]
% lean: CausalSmith.Stat.SaPlmCumulantConverse.generalizedQuantile_zero
\end{enumerate}
\end{proof}
