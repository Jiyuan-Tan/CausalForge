\begin{proof}[Proof of \cref{obj:lem:non-gaussian-hard-submodel}]
Set
\[
c_0=
\begin{cases}
\min\{C_{\theta},1\},& C_{\theta}>0,\\
1,& C_{\theta}\le 0,
\end{cases}
\qquad
\tau=\frac{\psi_{\xi}}4,
\qquad
C_0=\max\left\{1,\frac{16\psi_{\eta}^2}{\psi_{\xi}^2}\right\}.
\]
Then \(c_0>0\), \(C_0>0\), and \(0<\tau\le\psi_{\xi}\).  For any admissible choice of parameters whose displayed constants agree with the fixed constants, \cref{obj:def:model-parameters} gives \(C_{\theta}>0\), and hence \(c_0=\min\{C_{\theta},1\}\le C_{\theta}\). % lean: non_gaussian_hard_submodel

The two-point testing bound at relative-entropy budget \(C_0c_0^2\) supplies a constant \(c_{\mathrm{LC}}>0\), depending only on that budget, such that whenever two \(n\)-sample laws have relative entropy at most \(C_0c_0^2\), every measurable sample statistic \(\widehat\alpha\) with finite quadratic risks satisfies
\[
c_{\mathrm{LC}}(\theta_1-\theta_0)^2
\le
\max\left\{
E_0[(\widehat\alpha-\theta_0)^2],
E_1[(\widehat\alpha-\theta_1)^2]
\right\}.
\]
The minimax reduction applies this bound after symmetrically clipping an arbitrary estimator to a range containing both target values.  Define
\[
c_{\mathrm{ACE}}=c_{\mathrm{LC}}c_0^2>0 .
\]
% lean: minimaxRiskOn_two_point_lower

Fix now an admissible choice of parameters \(p\) with the displayed primitive constants and a class index \(n\ge1\).  Assume first that \(\mathcal P_{\mathrm{NG},n}\) is nonempty, and choose \(P_{\mathrm{NG}}^\circ\in\mathcal P_{\mathrm{NG},n}\), represented by a base model \(m_{\mathrm{NG}}^\circ\).  For each \(\vartheta\in\mathbb R\), define \(P_{n,\vartheta}\) by preserving the joint law of \((X,T)\) under \(P_{\mathrm{NG}}^\circ\), drawing an independent
\[
\xi_\star\sim N(0,\tau^2),
\]
and setting
\[
Y=q_0(X)+\vartheta\{T-g_0(X)\}+\xi_\star .
\]
Equivalently, the path is the product of the retained \((X,T)\)-law and the centered Gaussian innovation, transported through this affine outcome map. % lean: affineGaussianLaw; affineGaussianModel

Along this path,
\[
\theta_0(P_{n,\vartheta})=\vartheta,
\qquad
g_0(P_{n,\vartheta})=g_0(P_{\mathrm{NG}}^\circ),
\qquad
q_0(P_{n,\vartheta})=q_0(P_{\mathrm{NG}}^\circ),
\]
and the clipped supplied codes satisfy
\[
\bar g_n(P_{n,\vartheta})=\bar g_n(P_{\mathrm{NG}}^\circ),
\qquad
\bar q_n(P_{n,\vartheta})=\bar q_n(P_{\mathrm{NG}}^\circ).
\]
The covariate-treatment marginal is also retained:
\[
(P_{n,\vartheta})_{X,T}=(P_{\mathrm{NG}}^\circ)_{X,T}.
\]
% lean: affineGaussianModel_theta0; affineGaussianModel_g0; affineGaussianModel_q0; affineGaussianModel_map_xt

The treatment innovation is unchanged along the path:
\[
\eta_{P_{n,\vartheta}}=\eta_{P_{\mathrm{NG}}^\circ}.
\]
The outcome innovation is the fresh Gaussian coordinate:
\[
\xi_{P_{n,\vartheta}}
=
Y-q_0(X)-\vartheta\eta
=
\xi_\star,
\]
so
\[
(P_{n,\vartheta})\circ \xi_{P_{n,\vartheta}}^{-1}
=
N(0,\tau^2),
\qquad
\xi_{P_{n,\vartheta}}\perp (X,T).
\]
% lean: affineGaussianModel_eta; affineGaussianModel_map_xi; affineGaussianModel_indep_xi_xt

For \(|\vartheta|\le c_0\), the coefficient range holds because \(c_0\le C_{\theta}\).  All treatment-side assumptions, the regression ranges, the cumulant separation, and the treatment \(L^1(P_X)\) code-radius condition are inherited from \(P_{\mathrm{NG}}^\circ\).  The conditional mean condition follows from the displayed outcome equation, the independence of \(\xi_\star\) from \((X,T)\), and its centering.  Finally, \cref{obj:lem:gaussian-exponential-square-quarter-scale} applied at scale \(\psi_\xi\) gives
\[
E\exp\{\xi_\star^2/\psi_\xi^2\}\le2.
\]
Since \(\xi_{P_{n,\vartheta}}=\xi_\star\) is independent of \(X\), the conditional Luxemburg part of \cref{obj:ass:xi-subgaussian} is
\[
\exp\{\xi_{P_{n,\vartheta}}^2/\psi_\xi^2\}\in L^1(P_{n,\vartheta}),
\qquad
E_{P_{n,\vartheta}}\!\left[
\exp\{\xi_{P_{n,\vartheta}}^2/\psi_\xi^2\}\mid X
\right]
\le 2
\quad P_{n,\vartheta}\text{-a.s.}
\]
Hence
\[
|\vartheta|\le c_0
\quad\Longrightarrow\quad
P_{n,\vartheta}\in\mathcal P_{\mathrm{NG},n}.
\]
% lean: affineGaussianModel_nonGaussianClass_quarter

The same construction gives the asserted outcome equation:
\[
Y=q_0(X)+\vartheta\,\eta+\xi
\qquad P_{n,\vartheta}\text{-a.s.}
\]
% lean: non_gaussian_hard_submodel

It remains, for this branch, to record the product relative-entropy bound.  For the affine Gaussian path, one observation satisfies
\[
D_{\mathrm{KL}}(P_{n,\vartheta_0}\|P_{n,\vartheta_1})
\le
\frac{(\vartheta_1-\vartheta_0)^2\,2\psi_{\eta}^2}{2\tau^2}.
\]
The moment bound used here is the Luxemburg consequence
\[
E_{P_{\mathrm{NG}}^\circ}[\eta^2]\le 2\psi_{\eta}^2 .
\]
Tensorization over \(n\) independent observations gives
\[
D_{\mathrm{KL}}\!\left(P_{n,\vartheta_0}^{\otimes n}\,\middle\|\,P_{n,\vartheta_1}^{\otimes n}\right)
\le
n\,\frac{(\vartheta_1-\vartheta_0)^2\,2\psi_{\eta}^2}{2\tau^2}.
\]
If \(|\vartheta_1-\vartheta_0|=h/\sqrt n\) and \(\tau=\psi_\xi/4\), then
\[
n\,\frac{(\vartheta_1-\vartheta_0)^2\,2\psi_{\eta}^2}{2\tau^2}
=
\frac{16\psi_{\eta}^2}{\psi_\xi^2}h^2
\le C_0h^2 .
\]
Thus the required entropy display holds. % lean: affineGaussianModel_iid_kl_le; non_gaussian_hard_submodel

Now let \(P_{\mathrm{ACE}}^\circ\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\).  Define \(P_{n,\vartheta}^{\mathrm{ACE}}\) by the same affine Gaussian outcome construction from this starting law:
\[
Y=q_0(X)+\vartheta\{T-g_0(X)\}+\xi_\star,
\qquad
\xi_\star\sim N(0,\tau^2),\quad \xi_\star\perp(X,T).
\]
The displayed preservation identities remain valid, and the target coordinate is the path parameter:
\[
\theta_0(P_{n,\vartheta}^{\mathrm{ACE}})=\vartheta,
\qquad
(P_{n,\vartheta}^{\mathrm{ACE}})_{X,T}=(P_{\mathrm{ACE}}^\circ)_{X,T},
\]
\[
g_0(P_{n,\vartheta}^{\mathrm{ACE}})=g_0(P_{\mathrm{ACE}}^\circ),\quad
q_0(P_{n,\vartheta}^{\mathrm{ACE}})=q_0(P_{\mathrm{ACE}}^\circ),\quad
\eta_{P_{n,\vartheta}^{\mathrm{ACE}}}=\eta_{P_{\mathrm{ACE}}^\circ},
\]
and
\[
\bar g_n(P_{n,\vartheta}^{\mathrm{ACE}})=\bar g_n(P_{\mathrm{ACE}}^\circ),
\qquad
\bar q_n(P_{n,\vartheta}^{\mathrm{ACE}})=\bar q_n(P_{\mathrm{ACE}}^\circ).
\]
% lean: affineGaussianModel_theta0; affineGaussianModel_map_xt; affineGaussianModel_eta

Consequently the two JMS ACE code-radius inequalities are transported unchanged:
\[
\|\bar g_n-g_0\|_{L^r(P_X)}\le\varepsilon_{1,n},
\qquad
\|\bar q_n-q_0\|_{L^r(P_X)}\le\varepsilon_{2,n}.
\]
Together with the transported treatment-side and range conditions, the coefficient bound \(|\vartheta|\le c_0\le C_{\theta}\), and the Gaussian conditional Luxemburg calculation above, this gives
\[
|\vartheta|\le c_0
\quad\Longrightarrow\quad
P_{n,\vartheta}^{\mathrm{ACE}}\in
\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}} .
\]
The Gaussian residual law, residual independence, outcome equation, and product entropy bound are the same as in the non-Gaussian branch:
\[
D_{\mathrm{KL}}\!\left((P_{n,\vartheta_0}^{\mathrm{ACE}})^{\otimes n}\,\middle\|\,(P_{n,\vartheta_1}^{\mathrm{ACE}})^{\otimes n}\right)
\le C_0h^2
\]
whenever \(|\vartheta_1-\vartheta_0|=h/\sqrt n\). % lean: affineGaussianModel_jmsAceClass_quarter; affineGaussianModel_iid_kl_le

It remains to prove the fixed-code ACE lower bound.  Suppose a starting law \(P_\star^\circ\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\) has clipped codes \(g_\star,q_\star\).  Set
\[
\vartheta_-=-\frac{c_0}{2\sqrt n},
\qquad
\vartheta_+=\frac{c_0}{2\sqrt n},
\]
and let \(P_-:=P_{n,\vartheta_-}^{\mathrm{ACE}}\), \(P_+:=P_{n,\vartheta_+}^{\mathrm{ACE}}\), using \(P_\star^\circ\) as the starting law.  Since \(n\ge1\),
\[
|\vartheta_-|\le c_0,
\qquad |\vartheta_+|\le c_0,
\qquad
|\vartheta_+-\vartheta_-|=\frac{c_0}{\sqrt n}.
\]
Both laws belong to the fixed-code JMS ACE class with clipped codes \(g_\star,q_\star\), and the entropy bound gives
\[
D_{\mathrm{KL}}(P_-^{\otimes n}\|P_+^{\otimes n})
\le C_0c_0^2.
\]
% lean: non_gaussian_hard_submodel

Applying the two-point lower bound with target values \(\vartheta_-\) and \(\vartheta_+\) yields, for every measurable estimator whose two quadratic risks are finite,
\[
\max_{\nu\in\{-,+\}}
E_{P_\nu^{\otimes n}}\!\left[(\widetilde\theta-\theta_0(P_\nu))^2\right]
\ge
c_{\mathrm{LC}}(\vartheta_+-\vartheta_-)^2
=
\frac{c_{\mathrm{LC}}c_0^2}{n}
=
\frac{c_{\mathrm{ACE}}}{n}.
\]
The minimax reduction clips arbitrary measurable estimators to \([-C_{\theta},C_{\theta}]\), so the preceding finite-risk display applies inside the fixed-code risk.  Taking the supremum over the fixed-code JMS ACE class and then the infimum over estimators gives
\[
\frac{c_{\mathrm{ACE}}}{n}
\le
\inf_{\widetilde\theta}
\sup_{\substack{P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}:\\
\bar g_n=g_\star,\ \bar q_n=q_\star}}
E_P\!\left[\{\widetilde\theta-\theta_0(P)\}^2\right].
\]
% lean: minimaxRiskOn_two_point_lower; non_gaussian_hard_submodel
\end{proof}
