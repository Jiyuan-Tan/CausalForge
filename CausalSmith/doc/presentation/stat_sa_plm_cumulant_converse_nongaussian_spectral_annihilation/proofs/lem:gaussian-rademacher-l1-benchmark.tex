\begin{proof}[Proof of \cref{obj:lem:gaussian-rademacher-l1-benchmark}]
Choose
\[
K=
4(C_q+2C_{\theta}C_g)^2+4\psi_\xi^2
+
C_{\theta}^2(8+16C_g^2),
\qquad
C=1+64K+4C_{\theta}^2 .
\]
Then \(C>0\).  Fix \(p,a,m\) satisfying the assumptions, and write
\[
t=t_a=\frac{\pi}{2a},
\qquad
A=A_a=a\exp\!\left\{-\frac{(1-a^2)\pi^2}{8a^2}\right\}.
\]
Since \(0<a\le1\), \(t>0\) and \(A>0\). % lean: gaussian_rademacher_l1_benchmark

Independence of \(G_\circ\) and \(S\) gives the product transform
\[
M_a(z)
=
E\exp\{z\sqrt{1-a^2}G_\circ\}\,
E\exp\{zaS\}
=
\exp\!\left(\frac{1-a^2}{2}z^2\right)\cosh(az).
\]
The fourth logarithmic derivative at zero of this transform is the Rademacher contribution, hence
\[
\kappa_4(\eta_a)=-2a^4 .
\]
% lean: treatmentMGF_eq_gaussianRademacher; model_gaussianRademacher_kappa_four

On the imaginary axis,
\[
M_a(iu)=
\exp\!\left(-\frac{1-a^2}{2}u^2\right)\cos(au).
\]
The exponential factor is nonzero.  Therefore the first positive zero occurs when \(au=\pi/2\), namely
\[
t_a=\frac{\pi}{2a},
\qquad
M_a(it_a)=0,
\qquad
M_a(iu)\ne0\quad(0<u<t_a).
\]
% lean: gaussianRademacher_first_characteristic_zero

For any \(d\in\mathbb R\),
\[
E_P[\sin\{t_a(\eta+d)\}]
=
\Im\left(e^{it_ad}E_P[e^{it_a\eta}]\right)
=
\Im\left(e^{it_ad}M_a(it_a)\right)
=
0 .
\]
% lean: integral_sin_shift_eq_zero

Let
\[
Z_n=T-\bar g_n(X),
\qquad
D_n=g_0(X)-\bar g_n(X),
\]
so that
\[
Z_n=\eta+D_n(X).
\]
The derivative of the transform at the first zero is
\[
M_a'(it_a)
=
iA_a,
\]
equivalently
\[
E_P\!\left[\eta\,e^{it_a\eta}\right]=iA_a .
\]
% lean: eta_cexp_moment_at_first_zero

Using \(Z_n=\eta+D_n(X)\), independence of \(\eta\) and \(X\), and \(M_a(it_a)=0\),
\[
\begin{aligned}
E_P[Z_ne^{it_aZ_n}]
&=
E_P[\eta e^{it_a\eta}]\,E_P[e^{it_aD_n(X)}]
+
E_P[e^{it_a\eta}]\,E_P[D_n(X)e^{it_aD_n(X)}]  \\
&=
iA_a\,E_P[e^{it_aD_n(X)}].
\end{aligned}
\]
Taking imaginary parts gives
\[
E_P[Z_n\sin(t_aZ_n)]
=
A_aE_P[\cos(t_aD_n)] .
\]
% lean: learnedResidual_sine_denominator_identity

Assume now that \(t_a\varepsilon_{1,n}\le1/2\).  The \(L^1(P_X)\) condition gives
\[
E_P|D_n(X)|\le\varepsilon_{1,n}.
\]
Since \(1-\cos u\le |u|\),
\[
E_P[\cos(t_aD_n)]
\ge
1-t_aE_P|D_n|
\ge
1-t_a\varepsilon_{1,n}
\ge
\frac12 .
\]
Together with the preceding identity and \(A_a>0\), this yields
\[
E_P[Z_n\sin(t_aZ_n)]\ge A_a/2 .
\]
% lean: learnedResidual_sine_denominator_lower

Define the two scores
\[
W=Z_n\sin(t_aZ_n),
\qquad
R=(Y-\theta_0Z_n)\sin(t_aZ_n).
\]
The zero-shift identity and the conditional mean condition imply
\[
E_P[R]=0.
\]
Indeed, writing
\[
Y-\theta_0Z_n
=
q_0(X)-\theta_0D_n(X)+\xi,
\]
the covariate term is killed by the sine-shift identity after conditioning on \(X\), while
\[
E_P[\xi\sin(t_aZ_n)]
=
E_P\!\left[\sin(t_aZ_n)E_P(\xi\mid X,T)\right]
=
0 .
\]
Thus
\[
E_P[Y\sin(t_aZ_n)]
=
\theta_0\,E_P[Z_n\sin(t_aZ_n)] .
\]
% lean: sine_remainder_centered_of_mgf_zero

The Gaussian--Rademacher path has the second-moment envelope
\[
E_P[\eta^2]\le4.
\]
With \(|g_0|\le C_g\) and \(|\bar g_n|\le C_g\), we have
\[
|D_n|\le2C_g,
\]
and therefore
\[
E_P[W^2]\le 8+16C_g^2.
\]
% lean: gaussianRademacher_second_lintegral_le; sine_score_memLp_of_eta_second_lintegral_le

Similarly, the range conditions give
\[
|q_0(X)-\theta_0D_n(X)|
\le
C_q+2C_\theta C_g,
\]
and the conditional Luxemburg bound for \(\xi\) gives
\[
E_P[\xi^2]\le 2\psi_\xi^2.
\]
Consequently
\[
E_P[R^2]
\le
4(C_q+2C_\theta C_g)^2+4\psi_\xi^2.
\]
The two displayed square-moment bounds are exactly the components entering \(K\). % lean: sine_score_memLp_of_eta_second_lintegral_le

Let
\[
\overline W=\mathbb P_n W,
\qquad
\overline R=\mathbb P_n R,
\qquad
\mu_W=E_PW .
\]
The estimator in the statement is the clipped ratio based on
\[
\mathbb P_n[Y\sin(t_aZ_n)]
=
\theta_0\overline W+\overline R
\]
with threshold \(A_a/4\).  Since \(\mu_W\ge A_a/2\), \cref{obj:lem:clipped-ratio-risk-decomposition} gives the pathwise inequality
\[
(\widehat\theta_{a,n}-\theta_0)^2
\le
\frac{16}{A_a^2}
\left\{\overline R^2+C_\theta^2(\overline W-\mu_W)^2\right\}.
\]
Taking expectation under the i.i.d. product law and using \(E_PR=0\) gives
\[
E_P[(\widehat\theta_{a,n}-\theta_0)^2]
\le
\frac{16}{nA_a^2}
\left(E_P[R^2]+C_\theta^2E_P[W^2]\right)
\le
\frac{16K}{nA_a^2}.
\]
% lean: thetaHatAt_eq_clippedRatioFromScores; clippedRatioFromScores_lintegral_le

The clipping range also gives the deterministic bound
\[
(\widehat\theta_{a,n}-\theta_0)^2\le4C_\theta^2 .
\]
Combining this with the preceding display and using
\[
A_a^{-2}
=
a^{-2}
\exp\!\left(\frac{(1-a^2)\pi^2}{4a^2}\right)
\]
yields
\[
E_P[(\widehat\theta_{a,n}-\theta_0)^2]
\le
C\min\!\left\{
1,\frac1n a^{-2}
\exp\!\left(\frac{(1-a^2)\pi^2}{4a^2}\right)
\right\}.
\]
% lean: clippedRatioFromScores_sq_le_global; gaussian_rademacher_l1_benchmark

The smallness condition is equivalent to the stated radius bound.  Since
\[
t_a\varepsilon_{1,n}\le\frac12
\quad\Longleftrightarrow\quad
\varepsilon_{1,n}\le\frac{a}{\pi},
\]
and
\[
\Delta_a=2a^4,
\qquad
\Delta_a^{1/4}=2^{1/4}a,
\]
we obtain
\[
\varepsilon_{1,n}
\le
\frac{\Delta_a^{1/4}}{2^{1/4}\pi}.
\]
% lean: gaussian_rademacher_l1_benchmark

Finally,
\[
\Delta_a^{-1/2}
=
(2a^4)^{-1/2}
=
(\sqrt2\,a^2)^{-1}.
\]
The first risk factor is bounded by the cumulant-parametrized factor:
\[
a^{-2}\exp\!\left(\frac{(1-a^2)\pi^2}{4a^2}\right)
\le
\Delta_a^{-1/2}
\exp\!\left(\frac{\pi^2\sqrt2}{4}\Delta_a^{-1/2}\right).
\]
Substituting this inequality into the previous risk display gives
\[
E_P[(\widehat\theta_{a,n}-\theta_0)^2]
\le
C\min\!\left\{
1,\frac1n\Delta_a^{-1/2}
\exp\!\left(\frac{\pi^2\sqrt2}{4}\Delta_a^{-1/2}\right)
\right\}.
\]
% lean: gaussian_rademacher_l1_benchmark
\end{proof}
