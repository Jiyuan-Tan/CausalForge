\begin{proof}[Proof of \cref{obj:prop:jms-ace-alignment}]
Fix the displayed constants and choose the constants \(c\) and \(C\) supplied by
\cref{obj:thm:adaptive-rootn-minimax}.  Set
\[
  C_{\mathrm{spec}}=C .
\]
Then \(C_{\mathrm{spec}}>0\), and its dependence is only through
\(r,\delta,C_{\theta},C_g,C_q,\psi_{\eta},\psi_{\xi}\).

\begin{enumerate}
\item For the class relations, apply \cref{obj:lem:jms-ace-class-relations} at
the displayed parameter record and the displayed sample size \(n\).  It gives,
law by law,
\[
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
  \Longrightarrow
  P\in\mathcal P_{\mathrm{NG},n}
\]
and
\[
  P\in\mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}}
  \Longrightarrow
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}} .
\]
Since these pointwise implications are exactly the membership predicates for
the displayed law classes, they yield
\[
 \mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
 \subseteq \mathcal P_{\mathrm{NG},n},
 \qquad
 \mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}}
 \subseteq \mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}} .
\]
The same result also gives, when \(s=r\),
\[
  P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}
  \Longleftrightarrow
  P\in\mathcal P_{\mathrm{NG},n}^{\mathrm{ACE}},
\]
hence the asserted equality of the two classes under the common clipped-code
convention. % lean: CausalSmith.Stat.SaPlmCumulantConverse.jms_ace_class_relations

\item Fix \(P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\) using the clipped
supplied codes \(\bar g_n,\bar q_n\), and suppose
\(\mathsf E_n^{\mathrm{JMS}}\), \(\varepsilon_{1,n}>0\), and
\(\varepsilon_{2,n}\ge0\).  If \(\varepsilon_{2,n}>0\), the positive-radius
part of the assumed published order-\(r\) ACE generalized-quantile guarantee,
with the objects and gates stated in \cref{obj:prop:jms-ace-alignment}, applied
with the supplied handle and the same clipped codes, gives
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le B_{\mathrm{ACE},n,\gamma}.
\]

It remains to treat the boundary case \(\varepsilon_{2,n}=0\).  For any
\(\chi>0\), let \(p^{(\chi)}\) be the parameter record obtained from \(p\) by
replacing the whole second radius sequence by
\[
  \varepsilon_{2,m}^{(\chi)}=\max\{\varepsilon_{2,m},\chi\},
  \qquad m\ge1,
\]
and leaving every other coordinate unchanged.  This sequence is again
nonnegative and nonincreasing.  Read the same observed-data law, regressions,
and supplied code sequences as a law \(P^{(\chi)}\) for \(p^{(\chi)}\).  If
\(0<\chi<\varepsilon_{1,n}\), then the class membership is preserved: every
condition in \(\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\) is unchanged
except the outcome-code radius, and at index \(n\)
\[
  \varepsilon_{2,n}^{(\chi)}=\chi
  \quad\text{with}\quad
  0\le \varepsilon_{2,n}\le \chi .
\]
Moreover, with
\[
  \omega_n=(\gamma n)^{-1/2}(\psi_{\xi}+C_{\theta}\psi_{\eta}),
\]
the denominator entering the second JMS eligibility quantity is unchanged:
\[
 \max\{\varepsilon_{1,n},\max\{\varepsilon_{2,n}^{(\chi)},\omega_n\}\}
 =
 \max\{\varepsilon_{1,n},\max\{\varepsilon_{2,n},\omega_n\}\},
\]
because \(\varepsilon_{2,n}=0\) and \(0<\chi<\varepsilon_{1,n}\).  All other
eligibility quantities are unchanged by construction, so
\(\mathsf E_n^{\mathrm{JMS}}\) holds for \(p^{(\chi)}\) at the same index.

Let
\[
  \Lambda_n
  =
  C_{\gamma} r!16^r\delta^{-1}\varepsilon_{1,n}^r .
\]
The assumed ACE guarantee includes \(C_{\gamma}>0\), and the displayed hypotheses
give \(\Lambda_n>0\).  For any real \(\beta>B_{\mathrm{ACE},n,\gamma}\), set
\[
  \chi_{\beta}
  =
  \min\!\left\{
    \frac{\varepsilon_{1,n}}{2},
    \frac{\beta-B_{\mathrm{ACE},n,\gamma}}{2\Lambda_n}
  \right\}.
\]
Then \(0<\chi_{\beta}<\varepsilon_{1,n}\), and the positive-radius result
applied to \(p^{(\chi_{\beta})}\) gives
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le
 B_{\mathrm{ACE},n,\gamma}+\Lambda_n\chi_{\beta}.
\]
By the definition of \(\chi_{\beta}\),
\[
  \Lambda_n\chi_{\beta}
  \le
  \frac{\beta-B_{\mathrm{ACE},n,\gamma}}{2},
\]
so the same quantile is at most \(\beta\).  Since this holds for every
\(\beta>B_{\mathrm{ACE},n,\gamma}\), the desired boundary inequality follows:
\[
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{ACE},r,n}-\theta_0(P)|\bigr)
 \le B_{\mathrm{ACE},n,\gamma}.
\] % lean: CausalSmith.Stat.SaPlmCumulantConverse.jms_ace_alignment

\item For the spectral guarantee, assume the displayed nonemptiness condition
and choose a law \(P^{\circ}\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}\)
using the clipped supplied codes \(\bar g_n,\bar q_n\).  Define the fixed-code
ACE class
\[
 \mathcal A_n^{\star}
 =
 \left\{
 P\in\mathcal P_{\mathrm{ACE},n}^{\mathrm{JMS}}:
 \bar g_n(P)=\bar g_n,\ \bar q_n(P)=\bar q_n
 \right\}.
\]
This is the ACE-class set associated with the base law \(P^{\circ}\) in
\cref{obj:thm:adaptive-rootn-minimax}.  The statistic computed from the
supplied treatment-code sequence agrees pointwise with the statistic computed
from the base law's supplied treatment-code sequence, because both constructions
use the current treatment code only through the residuals
\[
  T_i-\bar g_n(X_i),\qquad i=1,\ldots,n .
\]
The ACE-class part of \cref{obj:thm:adaptive-rootn-minimax}, applied with the
fixed primitive and range records, \(n\ge2\), and
\[
 \varepsilon_{1,n}
 \le
 \{4R_1\exp(2C_gR_1)\}^{-1},
\]
therefore yields
\[
 \sup_{P\in\mathcal A_n^{\star}}
 Q_{P,1-\gamma}
 \bigl(|\widehat\theta_{\mathrm{spec},n}-\theta_0(P)|\bigr)
 \le
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma n}} .
\]
Since \(\mathcal A_n^{\star}\) is exactly the class of laws in the statement
with the clipped supplied codes \(\bar g_n,\bar q_n\), this is the asserted
spectral upper guarantee. % lean: CausalSmith.Stat.SaPlmCumulantConverse.adaptive_rootn_minimax

\item Finally consider a sequence \((p_n)\) satisfying the hypotheses in the
last item of the proposition.  Define
\[
  \mathcal L_n
  =
  \varepsilon_{1,n}^r\varepsilon_{2,n}
  +C_{\theta}\varepsilon_{1,n}^{r+1}
\]
and
\[
  \lambda_{\mathrm{ACE}}
  =
  C_{\gamma}r!16^r\delta^{-1}.
\]
The assumed ACE guarantee includes \(C_{\gamma}>0\), so
\(\lambda_{\mathrm{ACE}}>0\).  Along the sequence, the fixed-constant
identities in the hypotheses rewrite the ACE bound as
\[
 B_{\mathrm{ACE},n,\gamma}
 =
 \lambda_{\mathrm{ACE}}
 \left[
   \mathcal L_n
   +
   64(C_g+\psi_{\eta})^r
   \{r^2(C_g+\psi_{\eta})+\psi_{\xi}+C_{\theta}\psi_{\eta}\}
   (\gamma n)^{-1/2}
 \right].
\]
The second summand in brackets is nonnegative, hence eventually
\[
  \lambda_{\mathrm{ACE}}\mathcal L_n
  \le B_{\mathrm{ACE},n,\gamma}.
\]
The assumed divergence is
\[
  \frac{\mathcal L_n}{n^{-1/2}}\longrightarrow\infty .
\]
For \(n\ge1\),
\[
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma n}}
 =
 \sqrt{\frac{C_{\mathrm{spec}}}{\gamma}}\;n^{-1/2}.
\]
Once \(\mathcal L_n>0\) and
\(\lambda_{\mathrm{ACE}}\mathcal L_n\le B_{\mathrm{ACE},n,\gamma}\), we have
\[
 0\le
 \frac{\sqrt{C_{\mathrm{spec}}/(\gamma n)}}{B_{\mathrm{ACE},n,\gamma}}
 \le
 \frac{\sqrt{C_{\mathrm{spec}}/\gamma}}{\lambda_{\mathrm{ACE}}}
 \left(\frac{\mathcal L_n}{n^{-1/2}}\right)^{-1}.
\]
The right-hand side tends to \(0\), so the squeeze theorem gives
\[
 \frac{
   \sqrt{C_{\mathrm{spec}}/(\gamma n)}
 }{
   B_{\mathrm{ACE},n,\gamma}
 }
 \longrightarrow 0 .
\] % lean: CausalSmith.Stat.SaPlmCumulantConverse.tendsto_sqrt_div_over_of_dominates
\end{enumerate}
\end{proof}
