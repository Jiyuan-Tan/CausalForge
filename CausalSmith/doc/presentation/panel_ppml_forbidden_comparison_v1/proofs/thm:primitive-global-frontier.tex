\begin{proof}[Proof of \cref{obj:thm:primitive-global-frontier}]
Write \(q_{gt}=\pi_g/T\) and \(m_{gt}=m_{gt}(\delta)=B_{gt}\exp(D_{gt}\delta_{gt})\) with \(B_{gt}=\bar b_g\exp(\gamma_{t0})\), so that the quantities of \cref{obj:def:frontier-elimination-handle} satisfy
\[
h_{gt}=\pi_g m_{gt}=T\,q_{gt}m_{gt},
\qquad
R_g=\sum_{t=1}^T h_{gt},
\qquad
C_t=\sum_{g\in\mathcal C}h_{gt},
\qquad
M=\sum_{g\in\mathcal C}R_g,
\qquad
A=\sum_{g\in\mathcal C}\sum_{t=1}^T D_{gt}h_{gt}.
\]
A collapsed parameter is written \(\theta=(a,\beta)\) with nuisance block \(a\) and treatment coordinate \(\beta\), and \(r_{gt}'\theta=\sum_jX_{gt,j}a_j+D_{gt}\beta\). Finally set
\[
B^{obs}_{gt}=\frac{m_{g1}m_{\infty t}}{m_{\infty 1}},
\qquad
H=\{(g,t):g\in\mathcal C,\ D_{gt}=1\},
\qquad
Z=\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt},
\]
\[
\tau_{gt}=\frac{m_{gt}}{B^{obs}_{gt}}-1,
\qquad
\omega_{gt}=\frac{q_{gt}B^{obs}_{gt}}{Z},
\qquad
PTT=\sum_{(g,t)\in H}\omega_{gt}\tau_{gt} .
\]
% lean: primitive_global_frontier, observedCounterfactualBaseline, granularTau, pttNormalizer, pttWeight, counterfactualSharePTT

\textit{Step 1: positivity of the primitive margins.} Since \(T\ge4\), in particular \(T>0\), and \(\infty\in\mathcal C\), so \(\mathcal C\) is nonempty. Every \(h_{gt}=\pi_g\bar b_g\exp(\gamma_{t0})\exp(D_{gt}\delta_{gt})\) is a product of strictly positive factors, hence
\[
h_{gt}>0,\qquad R_g>0,\qquad C_t>0,\qquad M>0 .
\]
% lean: frontierPrimitiveH_pos, frontierPrimitiveRow_pos, frontierPrimitiveColumn_pos, frontierPrimitiveTotal_pos
Also, summing \(h_{gt}\) in the two possible orders,
\[
\sum_{t=1}^T C_t=M=\sum_{g\in\mathcal C}R_g .
\]
% lean: frontierPrimitiveColumn_sum

\textit{Step 2: the zero-treatment row-column fit.} Define the collapsed parameter \(\theta^0=(a^0,0)\), whose treatment coordinate is zero and whose nuisance levels are
\[
a^0_{\text{int}}=\log\frac{R_\infty}{\pi_\infty}+\log C_1-\log M,
\qquad
a^0_{\text{coh},g}=\log\frac{R_g}{\pi_g}-\log\frac{R_\infty}{\pi_\infty}\quad(g\neq\infty),
\qquad
a^0_{\text{time},t}=\log C_t-\log C_1\quad(t\neq1).
\]
Adding the intercept, cohort, and period contributions — the cohort dummy being absent for \(g=\infty\) and the period dummy for \(t=1\) — gives, for every supported cell,
\[
r_{gt}'\theta^0=\log\frac{R_g}{\pi_g}+\log C_t-\log M,
\qquad\text{hence}\qquad
\exp(r_{gt}'\theta^0)=\frac{R_gC_t}{\pi_g M},
\]
the exponentiation being legitimate because \(R_g/\pi_g\), \(C_t\), and \(M\) are strictly positive by Step 1.
% lean: frontierNuisanceParameter, frontierNuisanceParameter_index, frontierNuisanceParameter_exp

\textit{Step 3: the conditional residuals and their margins.} Define the mass-weighted residual of \(\theta^0\) in cell \((g,t)\),
\[
e_{gt}
=
q_{gt}\left\{m_{gt}-\exp(r_{gt}'\theta^0)\right\}
=
\frac{\pi_g}{T}\left\{m_{gt}-\frac{R_gC_t}{\pi_gM}\right\}
=
\frac{1}{T}\left\{h_{gt}-\frac{R_gC_t}{M}\right\}.
\]
% lean: frontierConditionalResidual, frontierConditionalResidual_eq
Its row and column sums vanish: using \(\sum_tC_t=M\) and \(\sum_{g\in\mathcal C}R_g=M\) from Step 1,
\[
\sum_{t=1}^T e_{gt}=\frac{1}{T}\left\{R_g-\frac{R_g}{M}\sum_{t=1}^TC_t\right\}=0
\qquad(g\in\mathcal C),
\]
\[
\sum_{g\in\mathcal C} e_{gt}=\frac{1}{T}\left\{C_t-\frac{C_t}{M}\sum_{g\in\mathcal C}R_g\right\}=0
\qquad(t=1,\ldots,T).
\]
% lean: frontierConditionalResidual_row_sum, frontierConditionalResidual_column_sum

\textit{Step 4: the scores at \(\theta^0\).} Every collapsed nuisance regressor \(X_{\cdot,j}\) is either the constant \(1\), the indicator of one cohort \(g_0\neq\infty\), or the indicator of one period \(t_0\neq1\). The corresponding score at \(\theta^0\) is therefore \(\sum_{g,t}e_{gt}\), \(\sum_te_{g_0t}\), or \(\sum_ge_{gt_0}\), and all three vanish by Step 3:
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}X_{gt,j}\left\{m_{gt}-\exp(r_{gt}'\theta^0)\right\}=0
\qquad\text{for every nuisance coordinate }j .
\]
% lean: frontierConditionalNuisanceScore
The treatment score at \(\theta^0\) is, by Step 3 and the definitions of \(A\) and \(\Phi\),
\[
S
=
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}D_{gt}\left\{m_{gt}-\exp(r_{gt}'\theta^0)\right\}
=
\frac{1}{T}\left\{A-\frac{1}{M}\sum_{g\in\mathcal C}\sum_{t=1}^T D_{gt}R_gC_t\right\}
=
\frac{\Phi}{TM},
\]
and \(TM>0\) by Step 1.
% lean: frontierConditionalScalarScore

\textit{Step 5: the pseudo-true coefficient has the sign of \(S\).} By \cref{obj:lem:pseudo-true-ppml-projection}, \(\theta^\star(\delta)\) is the unique global maximizer of \(L(\cdot;\delta)\) and every directional score vanishes there:
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}(r_{gt}'d)\left\{m_{gt}-\mu^\star_{gt}(\delta)\right\}=0
\qquad\text{for every direction }d .
\]
Write \(v=\theta^\star(\delta)-\theta^0\), whose treatment coordinate is \(v_\beta=\beta^\star(\delta)\) because \(\theta^0\) has treatment coordinate zero. Decomposing an arbitrary direction \(d=(d_a,d_\beta)\) into its nuisance and treatment parts and using Step 4, the score at \(\theta^0\) in direction \(d\) is
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}(r_{gt}'d)\left\{m_{gt}-\exp(r_{gt}'\theta^0)\right\}
=
d_\beta\,S .
\]
% lean: Causalean.Stat.finitePoissonObjective_snd_sign_of_nuisance_score
Suppose first \(v\neq0\). Subtracting the vanishing score at \(\theta^\star(\delta)\) in direction \(v\) from the score at \(\theta^0\) in direction \(v\),
\[
\beta^\star(\delta)\,S
=
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}(r_{gt}'v)\left\{\exp(r_{gt}'\theta^\star(\delta))-\exp(r_{gt}'\theta^0)\right\}.
\]
Each summand is nonnegative: \(q_{gt}>0\), and since \(r_{gt}'\theta^\star(\delta)-r_{gt}'\theta^0=r_{gt}'v\) and the exponential is increasing, \(r_{gt}'v\) and \(\exp(r_{gt}'\theta^\star(\delta))-\exp(r_{gt}'\theta^0)\) have the same sign. Moreover \(v\neq0\) and the injectivity of \(\theta\mapsto(r_{gt}'\theta)_{(g,t)}\) established from \cref{obj:ass:collapsed-design-rank} in \cref{obj:lem:pseudo-true-ppml-projection} give a supported cell with \(r_{gt}'v\neq0\), where the summand is strictly positive by strict monotonicity. Hence
\[
\beta^\star(\delta)\,S>0
\qquad\text{whenever }\theta^\star(\delta)\neq\theta^0 .
\]
If instead \(v=0\), then \(\beta^\star(\delta)=0\), and applying the vanishing score at \(\theta^\star(\delta)=\theta^0\) to the pure treatment direction gives \(S=0\). Conversely, if \(S=0\) then by the previous display every score at \(\theta^0\) vanishes in every direction, so \(\theta^0\) is a global maximizer of \(L(\cdot;\delta)\) — for a Poisson criterion with nonnegative weights the tangent-line bound \(\exp(\eta')\ge\exp(\eta)\{1+(\eta'-\eta)\}\) makes vanishing scores sufficient for global maximality — and uniqueness forces \(\theta^\star(\delta)=\theta^0\), i.e. \(\beta^\star(\delta)=0\).
% lean: Causalean.Stat.finitePoissonObjective_isMax_of_score
Collecting the two cases: \(\beta^\star(\delta)=0\iff S=0\), and when both are nonzero their product is positive, so they have the same sign. Since \(S=\Phi/(TM)\) with \(TM>0\), the sign of \(S\) is the sign of \(\Phi\), and therefore
\[
\beta^\star(\delta)<0\iff \Phi<0,\qquad
\beta^\star(\delta)=0\iff \Phi=0,\qquad
0<\beta^\star(\delta)\iff 0<\Phi .
\]
% lean: betaStar_sign_frontierEliminationHandle

\textit{Step 6: the observed baseline proxy reproduces the untreated mean.} For every supported cohort \(g\) the first-period treatment indicator is zero: if \(g=\infty\) this is the never-treated path, and if \(g\) is finite then \(D_{g1}=1\) would force the adoption date to be the first period, which the support excludes. Also \(D_{\infty t}=0\) for every \(t\), and \cref{obj:ass:unit-untreated-exponential-mean} gives \(\gamma_{10}=0\). Hence, for every \(g\in\mathcal C\) and every \(t\),
\[
B^{obs}_{gt}
=
\frac{m_{g1}m_{\infty t}}{m_{\infty 1}}
=
\frac{\bar b_ge^{\gamma_{10}}\cdot\bar b_\infty e^{\gamma_{t0}}}{\bar b_\infty e^{\gamma_{10}}}
=
\bar b_ge^{\gamma_{t0}}
=
B_{gt},
\]
the cancellation being legitimate because \(m_{\infty1}=\bar b_\infty>0\).
% lean: primitive_global_frontier

\textit{Step 7: \(H\) is nonempty and \(Z>0\).} \Cref{obj:ass:multicohort-frontier-scope} supplies a supported finite cohort adopting in the second period; in that period its treatment indicator equals one, so \(H\) is nonempty. For every \((g,t)\in H\) we have \(q_{gt}>0\) and, by Step 6, \(B^{obs}_{gt}=B_{gt}>0\). Hence every summand of \(Z\) is positive and at least one exists:
\[
Z=\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}>0 .
\]
% lean: primitive_global_frontier

\textit{Step 8: membership in \(\mathcal R_T\) is equivalent to \(\Phi<0\).} If the induced primitive tuple lies in \(\mathcal R_T\), then the last clause of \cref{obj:def:global-sign-reversal-region} asserts negativity of its treatment coordinate; that coordinate is the treatment coordinate of the maximizer of the criterion written in the restricted primitive coordinates, which is the same function of \(\theta\) as \(L(\theta;\delta)\), so it equals \(\beta^\star(\delta)\), and Step 5 gives \(\Phi<0\).
% lean: primitiveLimitingCriterion_restrict, primitiveBetaStar_restrict
Conversely suppose \(\Phi<0\). The horizon, support, and frontier-scope clauses of \cref{obj:def:global-sign-reversal-region} are hypotheses of the present theorem. The simplex clause holds because the finite-array shares sum to one for every \(N\ge|\mathcal C|\),
\[
\sum_{g\in\mathcal C}\frac{n_{gN}}{N}=1,
\]
every unit label lying in \(\mathcal C\), and passing to the limit with \cref{obj:ass:cohort-share-limit} gives \(\sum_{g\in\mathcal C}\pi_g=1\).
% lean: cohortShareLimit_sum_eq_one
The untreated-mean clause is witnessed by \(g\mapsto\bar b_g\), since the primitive untreated coordinate in cell \((g,t)\) is \(B_{gt}=\bar b_ge^{\gamma_{t0}}\). The positive-effect clause is \cref{obj:ass:strict-positive-effects}, and the rank clause is \cref{obj:ass:collapsed-design-rank} after rewriting the sum over supported cells as the double sum over \(g\in\mathcal C\) and \(t\). Finally Step 5 gives \(\beta^\star(\delta)<0\), which is the tuple's treatment coordinate. Hence the tuple lies in \(\mathcal R_T\).

\textit{Step 9: the four-cohort polynomial.} Let \(x>1\), \(y>1\), and consider the four-period support \(\{2,3,4,\infty\}\) with \(T=4\), equal shares \(\pi_g=1/4\), unit limiting baselines, zero untreated time effects (so \(B_{gt}=1\)), multiplier \(x\) on every treated cell except \((2,4)\), and multiplier \(y\) at \((2,4)\); that is, \(\exp(\delta_{gt})=x\) on treated cells other than \((2,4)\) and \(\exp(\delta_{2,4})=y\). With treatment absorbing from the adoption date on, so that \(D_{gt}=1\) exactly when \(g\le t\) and \(D_{\infty t}=0\), the array \(h_{gt}=\pi_gB_{gt}\exp(D_{gt}\delta_{gt})\) is
\[
4\,(h_{gt})=
\begin{pmatrix}
1&x&x&y\\
1&1&x&x\\
1&1&1&x\\
1&1&1&1
\end{pmatrix},
\]
with rows \(g=2,3,4,\infty\) and columns \(t=1,2,3,4\). Summing rows and columns,
\[
4R_2=1+2x+y,\quad 4R_3=2+2x,\quad 4R_4=3+x,\quad 4R_\infty=4,
\]
\[
4C_1=4,\quad 4C_2=x+3,\quad 4C_3=2x+2,\quad 4C_4=2x+y+1,
\]
\[
4M=10+5x+y,
\qquad
4A=5x+y .
\]
The treated cells of row \(g=2\) are \(t=2,3,4\), of row \(g=3\) are \(t=3,4\), and of row \(g=4\) is \(t=4\), so
\[
16\sum_{g,t}D_{gt}R_gC_t
=
(1+2x+y)(5x+y+6)+(2+2x)(4x+y+3)+(3+x)(2x+y+1),
\]
which expands to \(20x^2+10xy+y^2+38x+12y+15\), while
\[
16\,MA=(10+5x+y)(5x+y)=25x^2+10xy+y^2+50x+10y .
\]
Subtracting,
\[
\Phi=MA-\sum_{g,t}D_{gt}R_gC_t=\frac{5x^2+12x-2y-15}{16}.
\]
% lean: primitive_global_frontier, fourMultiplierEffects, frontierEliminationHandle

\textit{Step 10: the numerical identity.} With \(5(101/100)^2=51005/10000\) and \(12(101/100)=1212/100\),
\[
\frac{5(101/100)^2+12(101/100)-15}{2}=\frac{4441}{4000}.
\]

\textit{Step 11: the value at the witness.} For \(W_4\) of \cref{obj:def:four-cohort-witness} the treated multipliers are \(\exp(\log(101/100))=101/100\) off cell \((2,4)\) and \(\exp(\log4)=4\) at \((2,4)\), so Step 9 applies with \(x=101/100\) and \(y=4\):
\[
\Phi=\frac{5(101/100)^2+12(101/100)-2\cdot4-15}{16}=-\frac{11559}{32000}.
\]
% lean: fourCohortDelta_eq_multiplier, fourFrontierEliminationHandle_exact

\textit{Step 12: the counterfactual-share objects on treated cells.} Let \((g,t)\in H\), so \(D_{gt}=1\). Step 6 gives \(B^{obs}_{gt}=B_{gt}\), and \(B_{gt}>0\), so
\[
\tau_{gt}=\frac{m_{gt}}{B^{obs}_{gt}}-1=\frac{B_{gt}\exp(\delta_{gt})}{B_{gt}}-1=\exp(\delta_{gt})-1 ,
\]
and, using \(q_{gt}>0\), \(B^{obs}_{gt}>0\), and \(Z>0\) from Step 7,
\[
\omega_{gt}=\frac{q_{gt}B^{obs}_{gt}}{Z}>0 .
\]

\textit{Step 13: the weights sum to one.} By the definition of \(Z\) and \(Z>0\),
\[
\sum_{(g,t)\in H}\omega_{gt}
=
\frac{1}{Z}\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}
=
\frac{Z}{Z}
=
1 .
\]

\textit{Step 14: the ratio form of \(PTT\).} Expanding the definition and using \(B^{obs}_{gt}>0\) to cancel it inside each term,
\[
PTT
=
\sum_{(g,t)\in H}\frac{q_{gt}B^{obs}_{gt}}{Z}\left(\frac{m_{gt}}{B^{obs}_{gt}}-1\right)
=
\frac{\sum_{(g,t)\in H}q_{gt}m_{gt}}{Z}-\frac{\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}}{Z},
\]
and the second fraction equals \(1\) by Step 13, so
\[
PTT=\frac{\sum_{(g,t)\in H}q_{gt}m_{gt}}{\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}}-1 .
\]

\textit{Step 15: the sign contrast.} Suppose \(\Phi<0\). Step 5 gives \(\beta^\star(\delta)<0\). For \(PTT\), each treated cell has \(\omega_{gt}>0\) by Step 12, and \cref{obj:ass:strict-positive-effects} gives \(\delta_{gt}>0\), hence \(\exp(\delta_{gt})>1\) and \(\tau_{gt}>0\). The sum defining \(PTT\) thus has no negative term, and by Step 7 it has at least one term, which is strictly positive. Therefore \(PTT>0\), and
\[
\Phi<0\quad\Longrightarrow\quad \beta^\star(\delta)<0<PTT .
\]
% lean: primitive_global_frontier
\end{proof}
