\begin{proof}[Proof of \cref{obj:thm:primitive-global-frontier}]
Write \(q_{gt}=\pi_g/T\) and
\(m_{gt}=m_{gt}(\delta)=B_{gt}\exp(D_{gt}\delta_{gt})\), with
\(B_{gt}=\bar b_g\exp(\gamma_{t0})\). Then the quantities of
\cref{obj:def:frontier-elimination-handle} are
\[
h_{gt}=\pi_g m_{gt}=Tq_{gt}m_{gt},
\qquad
R_g=\sum_{t=1}^T h_{gt},
\qquad
C_t=\sum_{g\in\mathcal C}h_{gt},
\]
\[
M=\sum_{g\in\mathcal C}R_g,
\qquad
A=\sum_{g\in\mathcal C}\sum_{t=1}^T D_{gt}h_{gt},
\qquad
\Phi=MA-\sum_{g\in\mathcal C}\sum_{t=1}^T D_{gt}R_gC_t .
\]
A collapsed parameter is written \(\theta=(a,\beta)\), with nuisance block \(a\) and treatment coordinate \(\beta\), and
\(r_{gt}'\theta=X_{gt}'a+D_{gt}\beta\). Finally set
\[
B^{obs}_{gt}=\frac{m_{g1}m_{\infty t}}{m_{\infty 1}},
\qquad
H=\{(g,t):g\in\mathcal C,\ D_{gt}=1\},
\qquad
Z=\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt},
\]
\[
\tau_{gt}=\frac{m_{gt}}{B^{obs}_{gt}}-1,
\qquad
\omega_{gt}=\frac{q_{gt}B^{obs}_{gt}}{Z},
\qquad
PTT=\sum_{(g,t)\in H}\omega_{gt}\tau_{gt} .
\]
% lean: primitive_global_frontier, observedCounterfactualBaseline, granularTau, pttNormalizer, pttWeight, counterfactualSharePTT

\textit{Step 1: positivity of the primitive margins.} Since \(T\ge4\), we have \(T>0\), and since \(\infty\in\mathcal C\), the support \(\mathcal C\) is nonempty. Every
\(h_{gt}=\pi_g\bar b_g\exp(\gamma_{t0})\exp(D_{gt}\delta_{gt})\) is a product of strictly positive factors. Hence
\[
h_{gt}>0,
\qquad
R_g>0,
\qquad
C_t>0,
\qquad
M>0 .
\]
% lean: frontierPrimitiveH_pos, frontierPrimitiveRow_pos, frontierPrimitiveColumn_pos, frontierPrimitiveTotal_pos
Also, summing \(h_{gt}\) in the two possible orders gives
\[
\sum_{t=1}^T C_t=M=\sum_{g\in\mathcal C}R_g .
\]
% lean: frontierPrimitiveColumn_sum

\textit{Step 2: the zero-treatment row-column fit.} Define the collapsed parameter \(\theta^0=(a^0,0)\), whose treatment coordinate is zero and whose nuisance levels are
\[
a^0_{\mathrm{int}}=\log\frac{R_\infty}{\pi_\infty}+\log C_1-\log M,
\qquad
a^0_{\mathrm{coh},g}=\log\frac{R_g}{\pi_g}-\log\frac{R_\infty}{\pi_\infty}\quad(g\neq\infty),
\]
\[
a^0_{\mathrm{time},t}=\log C_t-\log C_1\quad(t\neq1).
\]
Adding the intercept, cohort, and period contributions, with the cohort dummy absent for \(g=\infty\) and the period dummy absent for \(t=1\), gives, for every supported cell,
\[
r_{gt}'\theta^0=\log\frac{R_g}{\pi_g}+\log C_t-\log M,
\qquad
\exp(r_{gt}'\theta^0)=\frac{R_gC_t}{\pi_g M},
\]
where the logarithms and division use the positivity in Step 1.
% lean: frontierNuisanceParameter, frontierNuisanceParameter_index, frontierNuisanceParameter_exp

\textit{Step 3: the conditional residuals and their margins.} Define the mass-weighted residual of \(\theta^0\) in cell \((g,t)\) by
\[
e_{gt}
=q_{gt}\{m_{gt}-\exp(r_{gt}'\theta^0)\}
=\frac{\pi_g}{T}\left\{m_{gt}-\frac{R_gC_t}{\pi_gM}\right\}
=\frac{1}{T}\left\{h_{gt}-\frac{R_gC_t}{M}\right\}.
\]
% lean: frontierConditionalResidual, frontierConditionalResidual_eq
Using \(\sum_tC_t=M\) and \(\sum_{g\in\mathcal C}R_g=M\) from Step 1, its row and column sums vanish:
\[
\sum_{t=1}^T e_{gt}=\frac{1}{T}\left\{R_g-\frac{R_g}{M}\sum_{t=1}^TC_t\right\}=0
\qquad(g\in\mathcal C),
\]
\[
\sum_{g\in\mathcal C} e_{gt}=\frac{1}{T}\left\{C_t-\frac{C_t}{M}\sum_{g\in\mathcal C}R_g\right\}=0
\qquad(t=1,\ldots,T).
\]
% lean: frontierConditionalResidual_row_sum, frontierConditionalResidual_column_sum

\textit{Step 4: the scores at \(\theta^0\).} Every collapsed nuisance regressor \(X_{\cdot,j}\) is either the constant \(1\), the indicator of one cohort \(g_0\neq\infty\), or the indicator of one period \(t_0\neq1\). The corresponding nuisance score at \(\theta^0\) is therefore \(\sum_{g,t}e_{gt}\), \(\sum_te_{g_0t}\), or \(\sum_ge_{gt_0}\), and these vanish by Step 3:
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T
q_{gt}X_{gt,j}\{m_{gt}-\exp(r_{gt}'\theta^0)\}=0
\qquad\text{for every nuisance coordinate }j .
\]
Because the nuisance block is finite-dimensional and \(X_{gt}'u\) is the corresponding linear combination of the coordinate regressors, the same conclusion holds for every nuisance direction \(u\):
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T
q_{gt}(X_{gt}'u)\{m_{gt}-\exp(r_{gt}'\theta^0)\}
=
\sum_j u_j
\sum_{g\in\mathcal C}\sum_{t=1}^T
q_{gt}X_{gt,j}\{m_{gt}-\exp(r_{gt}'\theta^0)\}
=0 .
\]
% lean: frontierConditionalNuisanceScore
The treatment score at \(\theta^0\) is
\[
S
=
\sum_{g\in\mathcal C}\sum_{t=1}^T
q_{gt}D_{gt}\{m_{gt}-\exp(r_{gt}'\theta^0)\}
=
\frac{1}{T}\left\{A-\frac{1}{M}\sum_{g\in\mathcal C}\sum_{t=1}^T D_{gt}R_gC_t\right\}
=
\frac{\Phi}{TM},
\]
and \(TM>0\) by Step 1.
% lean: frontierConditionalScalarScore

\textit{Step 5: the pseudo-true coefficient has the sign of \(S\).} The finite-dimensional Poisson sign comparison used here is the following. For a finite index set with strictly positive weights \(q_i\), strictly positive means \(m_i\), and an injective linear design map \(\mathscr A\), suppose that a zero-treatment point \((u^0,0)\) clears every nuisance-direction score,
\[
\sum_i q_i \mathscr A(u,0)_i\{m_i-\exp(\mathscr A(u^0,0)_i)\}=0
\qquad\text{for every nuisance direction }u.
\]
If
\[
s_0=\sum_i q_i \mathscr A(0,1)_i\{m_i-\exp(\mathscr A(u^0,0)_i)\}
\]
and \(\beta\) is the treatment coordinate of the unique maximizer of
\(\sum_i q_i\{m_i(\mathscr A\theta)_i-\exp((\mathscr A\theta)_i)\}\), then
\[
\beta<0\iff s_0<0,
\qquad
\beta=0\iff s_0=0,
\qquad
0<\beta\iff 0<s_0 .
\]
Indeed, the score at a maximizer vanishes in every direction. Writing \(\widehat\theta\) for the maximizer, \(\theta^0=(u^0,0)\), and \(v=\widehat\theta-\theta^0\), the score identities and the nuisance-direction condition give
\[
\widehat\beta s_0
=
\sum_i q_i(\mathscr A v)_i\{\exp((\mathscr A\widehat\theta)_i)-\exp((\mathscr A\theta^0)_i)\} .
\]
The summands on the right are nonnegative, and at least one is strictly positive when \(v\neq0\), by injectivity of \(\mathscr A\) and strict monotonicity of the exponential. If \(s_0=0\), all scores at \(\theta^0\) vanish, and the tangent-line inequality for the exponential makes \(\theta^0\) globally maximizing; uniqueness gives the zero treatment coordinate.

Apply this comparison with index set \(\mathcal C\times\{1,\ldots,T\}\), design \(\mathscr A\theta=(r_{gt}'\theta)_{g,t}\), and \((u^0,0)=\theta^0\). The weights and means are positive by Step 1, Step 4 supplies the nuisance-direction score condition, and \cref{obj:ass:collapsed-design-rank} gives injectivity: if two collapsed parameters have the same fitted-index array, their difference has weighted sum of squared indices equal to zero, so the rank condition forces the difference to be zero. Therefore the treatment coordinate \(\beta^\star(\delta)\) of \(\theta^\star(\delta)\) has the same three-way sign as \(S\). Since \(S=\Phi/(TM)\) with \(TM>0\),
\[
\beta^\star(\delta)<0\iff \Phi<0,
\qquad
\beta^\star(\delta)=0\iff \Phi=0,
\qquad
0<\beta^\star(\delta)\iff 0<\Phi .
\]
% lean: Causalean.Stat.finitePoissonObjective_snd_sign_of_nuisance_score, Causalean.Stat.finitePoissonObjective_isMax_of_score, collapsedDesignMap_injective, betaStar_sign_frontierEliminationHandle

\textit{Step 6: the observed baseline proxy reproduces the untreated mean.} For every supported cohort \(g\), the first-period treatment indicator is zero: if \(g=\infty\) this is the never-treated path, and if \(g\) is finite then \(D_{g1}=1\) would force the adoption date to be the first period, which the support excludes. Also \(D_{\infty t}=0\) for every \(t\), and \cref{obj:ass:unit-untreated-exponential-mean} gives \(\gamma_{10}=0\). Hence, for every \(g\in\mathcal C\) and every \(t\),
\[
B^{obs}_{gt}
=
\frac{m_{g1}m_{\infty t}}{m_{\infty 1}}
=
\frac{\bar b_ge^{\gamma_{10}}\cdot\bar b_\infty e^{\gamma_{t0}}}{\bar b_\infty e^{\gamma_{10}}}
=
\bar b_ge^{\gamma_{t0}}
=
B_{gt},
\]
where \(m_{\infty1}=\bar b_\infty>0\) justifies the cancellation.
% lean: primitive_global_frontier

\textit{Step 7: \(H\) is nonempty and \(Z>0\).} \Cref{obj:ass:multicohort-frontier-scope} supplies a supported finite cohort adopting in the second period; in that period its treatment indicator equals one, so \(H\) is nonempty. For every \((g,t)\in H\), \(q_{gt}>0\), and Step 6 gives \(B^{obs}_{gt}=B_{gt}>0\). Hence every summand of \(Z\) is nonnegative and at least one summand is strictly positive:
\[
Z=\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}>0 .
\]
% lean: primitive_global_frontier

\textit{Step 8: membership in \(\mathcal R_T\) is equivalent to \(\Phi<0\).} If the induced primitive tuple lies in \(\mathcal R_T\), then the last clause of \cref{obj:def:global-sign-reversal-region} gives negativity of its treatment coordinate. In the restricted primitive coordinates the limiting criterion is the same collapsed criterion \(L(\theta;\delta)\), so this coordinate equals \(\beta^\star(\delta)\), and Step 5 gives \(\Phi<0\).
% lean: primitiveLimitingCriterion_restrict, primitiveBetaStar_restrict
Conversely suppose \(\Phi<0\). The horizon, support, and frontier-scope clauses of \cref{obj:def:global-sign-reversal-region} are hypotheses of the present theorem. The simplex clause holds because the finite-array shares sum to one for every \(N\ge|\mathcal C|\),
\[
\sum_{g\in\mathcal C}\frac{n_{gN}}{N}=1,
\]
every unit label lies in \(\mathcal C\), and \cref{obj:ass:cohort-share-limit} gives
\(\sum_{g\in\mathcal C}\pi_g=1\) after passage to the limit.
% lean: cohortShareLimit_sum_eq_one
The untreated-mean clause is witnessed by \(g\mapsto\bar b_g\), since the primitive untreated coordinate is \(B_{gt}=\bar b_g e^{\gamma_{t0}}\). The positive-effect clause is \cref{obj:ass:strict-positive-effects}. The rank clause is \cref{obj:ass:collapsed-design-rank}, after rewriting the sum over supported cells as the double sum over \(g\in\mathcal C\) and \(t\). Finally Step 5 gives \(\beta^\star(\delta)<0\), which is the tuple's treatment coordinate. Hence the tuple lies in \(\mathcal R_T\).

\textit{Step 9: the four-cohort polynomial.} Let \(x>1\), \(y>1\), and consider the four-period support \(\{2,3,4,\infty\}\) with \(T=4\), equal shares \(\pi_g=1/4\), unit limiting baselines, zero untreated time effects, multiplier \(x\) on every treated cell except \((2,4)\), and multiplier \(y\) at \((2,4)\). With treatment absorbing from the adoption date on, \(D_{gt}=1\) exactly when \(g\le t\) and \(D_{\infty t}=0\), so
\[
4\,(h_{gt})=
\begin{pmatrix}
1&x&x&y\\
1&1&x&x\\
1&1&1&x\\
1&1&1&1
\end{pmatrix},
\]
with rows \(g=2,3,4,\infty\) and columns \(t=1,2,3,4\). Summing rows and columns,
\[
4R_2=1+2x+y,
\quad
4R_3=2+2x,
\quad
4R_4=3+x,
\quad
4R_\infty=4,
\]
\[
4C_1=4,
\quad
4C_2=x+3,
\quad
4C_3=2x+2,
\quad
4C_4=2x+y+1,
\]
\[
4M=10+5x+y,
\qquad
4A=5x+y .
\]
The treated cells of row \(g=2\) are \(t=2,3,4\), those of row \(g=3\) are \(t=3,4\), and the treated cell of row \(g=4\) is \(t=4\), so
\[
16\sum_{g,t}D_{gt}R_gC_t
=
(1+2x+y)(5x+y+6)+(2+2x)(4x+y+3)+(3+x)(2x+y+1),
\]
which expands to \(20x^2+10xy+y^2+38x+12y+15\), while
\[
16\,MA=(10+5x+y)(5x+y)=25x^2+10xy+y^2+50x+10y .
\]
Subtracting gives
\[
\Phi=MA-\sum_{g,t}D_{gt}R_gC_t=\frac{5x^2+12x-2y-15}{16}.
\]
% lean: primitive_global_frontier, fourMultiplierEffects, frontierEliminationHandle

\textit{Step 10: the numerical identity.} With \(5(101/100)^2=51005/10000\) and \(12(101/100)=1212/100\),
\[
\frac{5(101/100)^2+12(101/100)-15}{2}=\frac{4441}{4000}.
\]
% lean: primitive_global_frontier

\textit{Step 11: the value at the witness.} For \(W_4\) of \cref{obj:def:four-cohort-witness}, the treated multipliers are \(\exp(\log(101/100))=101/100\) off cell \((2,4)\) and \(\exp(\log4)=4\) at \((2,4)\), so Step 9 applies with \(x=101/100\) and \(y=4\):
\[
\Phi=\frac{5(101/100)^2+12(101/100)-2\cdot4-15}{16}=-\frac{11559}{32000}.
\]
% lean: fourCohortDelta_eq_multiplier, fourFrontierEliminationHandle_exact

\textit{Step 12: the counterfactual-share objects on treated cells.} Let \((g,t)\in H\), so \(D_{gt}=1\). Step 6 gives \(B^{obs}_{gt}=B_{gt}\), and \(B_{gt}>0\), so
\[
\tau_{gt}=\frac{m_{gt}}{B^{obs}_{gt}}-1
=\frac{B_{gt}\exp(\delta_{gt})}{B_{gt}}-1
=\exp(\delta_{gt})-1 .
\]
Moreover \(q_{gt}>0\), \(B^{obs}_{gt}>0\), and \(Z>0\) by Step 7, hence
\[
\omega_{gt}=\frac{q_{gt}B^{obs}_{gt}}{Z}>0 .
\]
% lean: primitive_global_frontier, granularTau, pttWeight

\textit{Step 13: the weights sum to one.} By the definition of \(Z\) and the positivity \(Z>0\),
\[
\sum_{(g,t)\in H}\omega_{gt}
=
\frac{1}{Z}\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}
=
\frac{Z}{Z}
=
1 .
\]
% lean: primitive_global_frontier, pttWeight, pttNormalizer

\textit{Step 14: the ratio form of \(PTT\).} Expanding the definition and using \(B^{obs}_{gt}>0\) on every treated cell,
\[
PTT
=
\sum_{(g,t)\in H}\frac{q_{gt}B^{obs}_{gt}}{Z}\left(\frac{m_{gt}}{B^{obs}_{gt}}-1\right)
=
\frac{\sum_{(g,t)\in H}q_{gt}m_{gt}}{Z}
-
\frac{\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}}{Z}.
\]
The second fraction equals \(1\), so
\[
PTT=\frac{\sum_{(g,t)\in H}q_{gt}m_{gt}}{\sum_{(g,t)\in H}q_{gt}B^{obs}_{gt}}-1 .
\]
% lean: primitive_global_frontier, counterfactualSharePTT

\textit{Step 15: the sign contrast.} Suppose \(\Phi<0\). Step 5 gives \(\beta^\star(\delta)<0\). For \(PTT\), each treated cell has \(\omega_{gt}>0\) by Step 12, and \cref{obj:ass:strict-positive-effects} gives \(\delta_{gt}>0\), hence \(\exp(\delta_{gt})>1\) and \(\tau_{gt}>0\). The sum defining \(PTT\) has nonnegative terms and, by Step 7, at least one strictly positive term. Therefore \(PTT>0\), and
\[
\Phi<0\quad\Longrightarrow\quad \beta^\star(\delta)<0<PTT .
\]
% lean: primitive_global_frontier
\end{proof}
