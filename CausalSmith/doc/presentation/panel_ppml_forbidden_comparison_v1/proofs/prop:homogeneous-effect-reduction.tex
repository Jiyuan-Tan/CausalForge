\begin{proof}[Proof of \cref{obj:prop:homogeneous-effect-reduction}]
For a collapsed parameter \(\theta\in\mathbb R^{|\mathcal C|+T}\) and a supported cell \((g,t)\), write the collapsed linear index as
\[
\eta_{gt}(\theta)=r_{gt}'\theta,
\]
so that, by \cref{obj:def:collapsed-population-projection},
\[
L(\theta;\delta)
=
\sum_{g\in\mathcal C}\sum_{t=1}^T
q_{gt}\left\{m_{gt}(\delta)\,\eta_{gt}(\theta)-\exp\!\left(\eta_{gt}(\theta)\right)\right\},
\qquad
\mu^\star_{gt}(\delta)=\exp\!\left\{\eta_{gt}(\theta^\star(\delta))\right\}.
\]
The map \(\theta\mapsto\eta_{gt}(\theta)\) is linear, so \(\eta_{gt}(\theta)-\eta_{gt}(\theta')=\eta_{gt}(\theta-\theta')\) for all \(\theta,\theta'\).

\textit{Step 1: the candidate parameter.} Let \(\infty\) be the never-treated reference cohort and define \(\theta^0=(a^0,\delta_0)\) coordinatewise by
\[
a^0_{\text{int}}=\log\bar b_\infty,
\qquad
a^0_{\text{coh},c}=\log\bar b_c-\log\bar b_\infty\quad(c\in\mathcal C,\ c\neq\infty),
\qquad
a^0_{\text{time},u}=\gamma_{u0}\quad(u\neq1),
\]
with treatment coordinate \(\delta_0\).
% lean: homogeneous_effect_reduction
\Cref{obj:ass:unit-untreated-exponential-mean} carries the normalization \(\gamma_{10}=0\). Hence, for every period \(t\),
\[
\begin{cases}
0, & t=1,\\
a^0_{\text{time},t}, & t\neq1,
\end{cases}
\quad=\quad\gamma_{t0},
\]
since for \(t=1\) all period dummies are switched off and \(\gamma_{10}=0\), while for \(t\neq1\) exactly the dummy of period \(t\) contributes. Likewise, for every \(g\in\mathcal C\),
\[
\begin{cases}
0, & g=\infty,\\
a^0_{\text{coh},g}, & g\neq\infty,
\end{cases}
\quad=\quad
\begin{cases}
0, & g=\infty,\\
\log\bar b_g-\log\bar b_\infty, & g\neq\infty,
\end{cases}
\]
because exactly the dummy of cohort \(g\) contributes when \(g\neq\infty\), and none when \(g=\infty\).

\textit{Step 2: the index at \(\theta^0\).} Adding the intercept, cohort, period, and treatment contributions in the two cases \(g=\infty\) and \(g\neq\infty\) gives, for every \(g\in\mathcal C\) and every period \(t\),
\[
\eta_{gt}(\theta^0)=\log\bar b_g+\gamma_{t0}+D_{gt}\delta_0 .
\]
% lean: homogeneous_effect_reduction

\textit{Step 3: the homogeneous-effect hypothesis in product form.} For every \(g\in\mathcal C\) and every \(t\),
\[
D_{gt}\delta_{gt}=D_{gt}\delta_0 :
\]
if \(D_{gt}=1\) the common treated-cell hypothesis gives \(\delta_{gt}=\delta_0\), and if \(D_{gt}=0\) both sides vanish.

\textit{Step 4: the criterion is exactly specified at \(\theta^0\).} Using \(B_{gt}=\bar b_g\exp(\gamma_{t0})\) together with Steps 2 and 3, the observed collapsed mean satisfies
\[
m_{gt}(\delta)
=
B_{gt}\exp(D_{gt}\delta_{gt})
=
\exp\{\log\bar b_g+\gamma_{t0}+D_{gt}\delta_0\}
=
\exp\{\eta_{gt}(\theta^0)\}.
\]
In particular \(m_{gt}(\delta)>0\) and
\[
\log m_{gt}(\delta)=\eta_{gt}(\theta^0)
\qquad\text{for every supported cell.}
\]

\textit{Step 5: the scalar Poisson cell inequality.} For every \(m>0\) and every real \(\eta\),
\[
m\eta-\exp(\eta)\le m\log m-\exp(\log m),
\]
with strict inequality when \(\eta\neq\log m\). Apply \(1+x\le\exp(x)\), strict for \(x\neq0\), at \(x=\eta-\log m\), multiply by \(m>0\), and rearrange using \(\exp(\eta)=m\exp(\eta-\log m)\).
% lean: homogeneous_effect_reduction

\textit{Step 6: the cell masses are strictly positive.} First \(T>0\): if \(T=0\), then for the nonzero collapsed parameter with zero nuisance block and treatment coordinate one the weighted sum \(\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}(r_{gt}'a)^2\) is an empty sum, hence zero, contradicting \cref{obj:ass:collapsed-design-rank}. Since \(\pi_g\in(0,1)\), it follows that
\[
q_{gt}=\frac{\pi_g}{T}>0
\qquad\text{for every supported cell.}
\]

\textit{Step 7: \(\theta^0\) is a global maximizer.} Fix any \(\theta\). Applying Step 5 cell by cell with \(m=m_{gt}(\delta)>0\) and \(\eta=\eta_{gt}(\theta)\), and using \(\log m_{gt}(\delta)=\eta_{gt}(\theta^0)\) from Step 4,
\[
m_{gt}(\delta)\eta_{gt}(\theta)-\exp\!\left(\eta_{gt}(\theta)\right)
\le
m_{gt}(\delta)\eta_{gt}(\theta^0)-\exp\!\left(\eta_{gt}(\theta^0)\right).
\]
Multiplying by \(q_{gt}>0\) and summing over supported cells,
\[
L(\theta;\delta)\le L(\theta^0;\delta).
\]

\textit{Step 8: the maximizer is unique.} Suppose \(L(\theta;\delta)=L(\theta^0;\delta)\) and \(\theta\neq\theta^0\). Then \(\theta-\theta^0\neq0\), so \cref{obj:ass:collapsed-design-rank} gives
\[
\sum_{g\in\mathcal C}\sum_{t=1}^T q_{gt}\left(\eta_{gt}(\theta-\theta^0)\right)^2>0 ,
\]
and by linearity \(\eta_{gt}(\theta-\theta^0)=\eta_{gt}(\theta)-\eta_{gt}(\theta^0)\), so some supported cell \((g_0,t_0)\) has
\[
\eta_{g_0t_0}(\theta)\neq\eta_{g_0t_0}(\theta^0)=\log m_{g_0t_0}(\delta).
\]
At that cell the strict half of Step 5 applies, and all other cells obey the weak inequality of Step 7. Since \(q_{g_0t_0}>0\), summing gives
\[
L(\theta;\delta)<L(\theta^0;\delta),
\]
contradicting the assumed equality. Hence \(\theta=\theta^0\).

\textit{Step 9: identification of the projection.} By Steps 7 and 8 the criterion \(L(\cdot;\delta)\) has \(\theta^0\) as its unique global maximizer, and \cref{obj:def:collapsed-population-projection} selects \(\theta^\star(\delta)\) among the global maximizers, so
\[
\theta^\star(\delta)=\theta^0 .
\]
% lean: Causalean.Stat.maximizerOrZero, homogeneous_effect_reduction

\textit{Step 10: the two conclusions.} The treatment coordinate of \(\theta^0\) is \(\delta_0\), so
\[
\beta^\star(\delta)=\delta_0 ,
\]
and for every \(g\in\mathcal C\) and every period \(t\), by Step 2,
\[
\mu^\star_{gt}(\delta)
=
\exp\{\eta_{gt}(\theta^\star(\delta))\}
=
\exp\{\log\bar b_g+\gamma_{t0}+D_{gt}\delta_0\}
=
B_{gt}\exp(D_{gt}\delta_0).
\]
% lean: homogeneous_effect_reduction
\end{proof}
