\begin{proof}[Proof of \cref{obj:thm:regular-cell-unknown-weight-attainment}]
Throughout, \(L\) is the fixed constant of the statement, so that
\[
B_{\varepsilon,c}=8(\varepsilon^{-2}+c^{-1}),
\qquad
L\ge\left(\frac{2B_{\varepsilon,c}}{\alpha}\right)^{1/2},
\qquad
C_0=\max\left\{2,\ 4\sqrt2\,L+8B_{\varepsilon,c}\right\}.
\]
Because \(\varepsilon>0\) and \(c>0\) we have \(B_{\varepsilon,c}\ge0\), hence \(L\ge0\), and since \(C_0\) is a maximum with \(2\) as one of its arguments,
\[
0<2\le C_0 .
\]
% lean: regular_cell_unknown_weight_attainment

\begin{enumerate}
\item \textbf{Every row of the class is nonempty.}  The carrier hypothesis supplies, for each \(n\), distinct points \(x_{1,n},\dots,x_{k_n,n}\in\mathcal X\) whose singletons are measurable; write
\[
\mathcal X_n=\{x_{1,n},\dots,x_{k_n,n}\}
\]
for the resulting realized support.  Under \(N_n/n\to c>0\), \(k_n\to\infty\), and \(k_n/\sqrt n\to0\), the uniform finite-cell array carried on \(\mathcal X_n\), with source cell masses \(1/k_n\) and propensity \(1/2\), satisfies all conditions of \(\mathcal P_n\) in \cref{obj:def:model-class} together with \cref{obj:ass:finite-cell-source}.  Since \(0<c_-\le1\le c_+\),
\[
\frac{c_-}{k_n}\le\frac1{k_n}\le\frac{c_+}{k_n},
\]
so this array also satisfies the cell-mass bounds of \cref{obj:def:regular-finite-cell-class} and therefore
\[
\mathcal N_n^{\mathrm{reg}}\neq\varnothing
\qquad\text{for every }n .
\]
Consequently the empty-row convention in \cref{obj:def:regular-cell-feasible-honesty} is never invoked, and the coverage requirement there reads
\[
1-\alpha\le
\liminf_{n\to\infty}\ \inf_{P\in\mathcal N_n^{\mathrm{reg}}} P\left(\theta_T\in C_n^{\mathrm{reg}}\right).
\]
% lean: regularFiniteCellClass_inhabited

\item \textbf{The rule, and its almost-sure agreement with the displayed statistics.}  The procedure receives at row \(n\) the source sample \((O_i^S)_{i=1}^n\), the target sample \((X_j^T)_{j=1}^{N_n}\), the realized support \(\mathcal X_n\), and the two known \(k_n\)-vectors \((q_{x,n})_{x\in\mathcal X_n}\) and \((e_{x,n})_{x\in\mathcal X_n}\).  It extends those two vectors off \(\mathcal X_n\) by zero, i.e.\ it works with
\[
\widetilde q_n(x)=q_{x,n}\mathbf 1\{x\in\mathcal X_n\},
\qquad
\widetilde e_n(x)=e_{x,n}\mathbf 1\{x\in\mathcal X_n\},
\]
and returns the set displayed in the statement, namely \(\Theta\) when \(N_n<2\) and otherwise
\[
\left\{\vartheta\in\Theta:
\left|\widehat M_{Y,n}-\vartheta\widehat M_{D,n}\right|
\le L\sqrt{\widehat K_n/n}\right\},
\]
with \(H_i\), \(\widehat p_{x,n}\), \(\widehat m_{G,n}\), \(\widehat M_{G,n}\), \(\widehat\kappa_{U,n}\), and \(\widehat K_n\) formed from \(\widetilde q_n\) and \(\widetilde e_n\) in place of \(q_{x,n}\) and \(e(\cdot)\).  This rule uses only the source and target samples, the known cell masses, and the known propensity values, and its sample-by-\(\vartheta\) graph on the realized support is measurable because \(\mathcal X_n\) is a finite union of measurable singletons and the maps \(s\mapsto\widehat M_{Y,n}\), \(s\mapsto\widehat M_{D,n}\), \(s\mapsto\widehat K_n\) are measurable.  Hence it is a regular-cell procedure in the sense of \cref{obj:def:regular-cell-feasible-honesty}.
% lean: regularCellProcedure

Fix \(n\) and \(P\in\mathcal N_n^{\mathrm{reg}}\).  By \cref{obj:def:regular-finite-cell-class} the source covariate marginal is carried by the realized support, \(P_S^X(\mathcal X_n)=1\), and by \cref{obj:ass:transport-domination} \(P_T\ll P_S^X\), so also \(P_T(\mathcal X_n)=1\).  Therefore
\[
P\left\{X_i\in\mathcal X_n\ \text{for all }i\le n,\ \text{and}\ X_j^T\in\mathcal X_n\ \text{for all }j\le N_n\right\}=1,
\]
and on that event \(\widetilde q_n\) and \(\widetilde e_n\) agree with \(q_{x,n}=P_S^X(X=x)\) and \(e(\cdot)\) at every realized covariate.  Since \(\widehat m_{G,n}\), \(\widehat M_{G,n}\), and \(\widehat\kappa_{U,n}\) depend on the mass and propensity functions only through their values at realized covariates, the set returned by the procedure coincides \(P\)-almost surely with the set built from \(q_{x,n}\) and \(e(\cdot)\):
\[
P\left\{C_n^{\mathrm{reg}}\ \text{equals the rule of the statement evaluated at }q_{x,n}\text{ and }e\right\}=1 .
\]
% lean: regularCellProcedure_set_eq_ambient_ae

\item \textbf{Honesty.}  Since \(N_n/n\to c>0\), there is an index beyond which \(n\ge1\) and \(N_n\ge2\); fix such an \(n\) and let \(P\in\mathcal N_n^{\mathrm{reg}}\).  Write the centered inversion statistic at the true value,
\[
\zeta_n=\widehat M_{Y,n}-\theta_T\widehat M_{D,n},
\]
which is well defined because \(\theta_T\in\Theta\) by \cref{obj:prop:compact-causal-range}.  Three facts control \(\zeta_n\).

First, \(\zeta_n\) is exactly centered:
\[
\mathbb E_P[\zeta_n]=0 .
\]
Indeed, the inverse-propensity weighting makes \(\mathbb E_P[\widehat m_{G,n}(x)]=\Delta_G(x)\) for each cell \(x\), the independent target sample gives \(\mathbb E_P[\widehat p_{x,n}]=P_T(X=x)=q_{x,n}w(x)\), so \(\mathbb E_P[\widehat M_{G,n}]\) equals the transported mean of \(\Delta_G\), and \(\mu_{Y,n}-\theta_T\mu_n=0\) by \cref{obj:prop:compact-causal-range}.
% lean: regularCell_contrastMoment_mean_zero

Second, uniformly over \(P\in\mathcal N_n^{\mathrm{reg}}\) for all large \(n\),
\[
\operatorname{Var}_P(\zeta_n)\le\frac{B_{\varepsilon,c}\,\kappa_n}{n},
\qquad
\operatorname{Var}_P\!\left(\widehat M_{D,n}\right)\le\frac{B_{\varepsilon,c}\,\kappa_n}{n},
\qquad
\mathbb E_P\!\left[\widehat M_{D,n}\right]=\mu_n .
\]
The two variance bounds hold because the cross-average splits into a source-sample part and a target-sample part: conditionally on the target sample the source part is an average of \(n\) independent terms whose inverse-propensity factor is bounded by \(\varepsilon^{-1}\) under \cref{obj:ass:instrument-overlap} and whose inverse-mass weighting contributes the dispersion \(\kappa_n\), giving at most \(8\varepsilon^{-2}\kappa_n/n\); the target part is an average of \(N_n\) independent bounded terms, giving at most \(4/N_n\le8/(cn)\le8c^{-1}\kappa_n/n\) once \(N_n/n\to c>0\) and \(\kappa_n\ge1\).  The two contributions add to \(B_{\varepsilon,c}\kappa_n/n\) with \(B_{\varepsilon,c}=8(\varepsilon^{-2}+c^{-1})\).
% lean: regularCell_eventually_uniform_moment_variance

Third, the transport weight is a probability density with respect to \(P_S^X\), so by Jensen's inequality its second moment satisfies
\[
1\le\kappa_n .
\]
% lean: one_le_regularCell_kish

Now decompose the noncoverage event.  Off the full-measure support event of the previous step, and off the event \(\{\widehat K_n\ge\kappa_n/2\}\), we discard; on the intersection of both, \(L\sqrt{\widehat K_n/n}\ge L\sqrt{\kappa_n/(2n)}\), so \(|\zeta_n|\le L\sqrt{\kappa_n/(2n)}\) already forces \(\theta_T\in C_n^{\mathrm{reg}}\).  Chebyshev's inequality applied with the variance bound gives
\[
P\left\{|\zeta_n|>L\sqrt{\frac{\kappa_n}{2n}}\right\}
\le
\frac{B_{\varepsilon,c}\kappa_n/n}{L^2\kappa_n/(2n)}
=\frac{2B_{\varepsilon,c}}{L^2}
\le\alpha,
\]
the last step because \(L^2\ge2B_{\varepsilon,c}/\alpha\).  For the remaining event put
\[
\delta_n=\sup_{P\in\mathcal N_n^{\mathrm{reg}}}
P\left\{\widehat K_n<\kappa_n/2\right\}.
\]
A Chebyshev bound for the target-sample U-statistic \(\widehat\kappa_{U,n}\), whose variance is at most a constant multiple of \(k_n^2/N_n\) under the cell-mass lower bound \(q_{x,n}\ge c_-/k_n\), gives
\[
\delta_n\le\frac{128\,k_n^2}{c_-^2\,N_n}
\xrightarrow[n\to\infty]{}0,
\]
because \(N_n/n\to c>0\) and \(k_n/\sqrt n\to0\).
% lean: regularCell_Khat_lower_tail_uniform

Combining the three events, for all large \(n\) and every \(P\in\mathcal N_n^{\mathrm{reg}}\),
\[
P\left(\theta_T\in C_n^{\mathrm{reg}}\right)\ge1-\alpha-\delta_n,
\]
hence
\[
1-\alpha\le
\liminf_{n\to\infty}\ \inf_{P\in\mathcal N_n^{\mathrm{reg}}}
P\left(\theta_T\in C_n^{\mathrm{reg}}\right),
\]
and \((C_n^{\mathrm{reg}})\in\mathfrak C^{\mathrm{reg}}\).
% lean: regularCellProcedure_coverage_liminf

\item \textbf{Pointwise expected-length bound.}  Fix \(t_0>0\).  Choose \(n_0\) so that for every \(n\ge n_0\) one has \(n\ge1\), \(N_n\ge2\), and, for every \(P\in\mathcal N_n^{\mathrm{reg}}\),
\[
P\left\{\left|\widehat M_{D,n}-\mu_n\right|>\frac{\mu_n}{2}\right\}
\le\frac{4B_{\varepsilon,c}}{t_n},
\qquad
t_n=\frac{n\mu_n^2}{\kappa_n};
\]
this is Chebyshev's inequality applied to \(\widehat M_{D,n}\) at radius \(\mu_n/2\), using the mean and variance recorded in the previous step, since \((B_{\varepsilon,c}\kappa_n/n)/(\mu_n/2)^2=4B_{\varepsilon,c}/t_n\).
% lean: regularCell_firstStage_bad_probability

Fix \(n\ge n_0\) and \(P\in\mathcal N_n^{\mathrm{reg}}\).  By \cref{obj:prop:compact-causal-range} the transported first stage equals the target complier share,
\[
\mu_n=P_n^F\{D(1)=1,\ D(0)=0\mid S=0\},
\]
which is positive by \cref{obj:ass:target-complier-positivity}; thus \(\mu_n>0\).  The collision terms are nonnegative because \(q_{x,n}>0\) on \(\mathcal X_n\), so
\[
\widehat\kappa_{U,n}\ge0,
\qquad
\widehat K_n=1+\widehat\kappa_{U,n}\ge1 ,
\]
and \(\widehat K_n\) is integrable under the two-sample law.  Its mean is
\[
\mathbb E_P\!\left[\widehat K_n\right]=1+\kappa_n\le2\kappa_n,
\]
the identity because \(\widehat\kappa_{U,n}\) is an unbiased U-statistic for \(\kappa_n\) over the \(N_n(N_n-1)\) ordered distinct target pairs, and the inequality because \(1\le\kappa_n\).
% lean: regularCell_Khat_mean, regularCell_moments_memLp

By the previous step, \(C_n^{\mathrm{reg}}\) agrees almost surely with the inversion of one affine inequality intersected with \(\Theta\).  If \(\widehat M_{D,n}\neq0\), that inversion is the interval of centre \(\widehat M_{Y,n}/\widehat M_{D,n}\) and half-length \(L\sqrt{\widehat K_n/n}\,/\,|\widehat M_{D,n}|\), so
\[
\lambda\!\left(C_n^{\mathrm{reg}}\right)
\le
\min\left\{2,\ \frac{2L\sqrt{\widehat K_n/n}}{\left|\widehat M_{D,n}\right|}\right\},
\]
and in all cases \(\lambda(C_n^{\mathrm{reg}})\le\lambda(\Theta)=2\).
% lean: regularCellInversion_eq_affineInversionSet, affineInversionSet_length_le

On the event \(\{|\widehat M_{D,n}-\mu_n|\le\mu_n/2\}\) we have \(|\widehat M_{D,n}|\ge\mu_n/2\), hence
\[
\lambda\!\left(C_n^{\mathrm{reg}}\right)\le\frac{4L}{\mu_n}\sqrt{\frac{\widehat K_n}{n}} ,
\]
while on the complementary event we use \(\lambda(C_n^{\mathrm{reg}})\le2\).  Taking expectations and bounding the complementary event by the displayed deviation probability,
\[
\mathbb E_P\!\left[\lambda\!\left(C_n^{\mathrm{reg}}\right)\right]
\le
\frac{4L}{\mu_n}\,\mathbb E_P\!\left[\sqrt{\frac{\widehat K_n}{n}}\right]
+2\cdot\frac{4B_{\varepsilon,c}}{t_n}
\le
\frac{4L}{\mu_n}\sqrt{\frac{\mathbb E_P[\widehat K_n]}{n}}+\frac{8B_{\varepsilon,c}}{t_n},
\]
the second inequality by concavity of the square root.  Inserting \(\mathbb E_P[\widehat K_n]\le2\kappa_n\) and \(t_n=n\mu_n^2/\kappa_n\),
\[
\frac{4L}{\mu_n}\sqrt{\frac{2\kappa_n}{n}}=4\sqrt2\,L\,t_n^{-1/2},
\qquad\text{so}\qquad
\mathbb E_P\!\left[\lambda\!\left(C_n^{\mathrm{reg}}\right)\right]
\le
\min\left\{2,\ 4\sqrt2\,L\,t_n^{-1/2}+\frac{8B_{\varepsilon,c}}{t_n}\right\}.
\]
% lean: expectedLength_affineInversion_le, expectedLength_affineInversion_frontier_le_inflated

Finally, for \(t>0\), \(u\ge0\), and \(v\ge0\),
\[
\min\left\{2,\ u\,t^{-1/2}+v\,t^{-1}\right\}
\le
\max\{2,u+v\}\,\min\{1,t^{-1/2}\}:
\]
for \(t\le1\) the left side is at most \(2\le\max\{2,u+v\}=\max\{2,u+v\}\min\{1,t^{-1/2}\}\), and for \(t\ge1\) one has \(t^{-1}\le t^{-1/2}\), so the left side is at most \((u+v)t^{-1/2}\).  With \(u=4\sqrt2\,L\), \(v=8B_{\varepsilon,c}\), and \(t=t_n\) this gives, for every \(n\ge n_0\) and every \(P\in\mathcal N_n^{\mathrm{reg}}\),
\[
\mathbb E_P\!\left[\lambda\!\left(C_n^{\mathrm{reg}}\right)\right]
\le
C_0\min\left\{1,t_n^{-1/2}\right\}.
\]
% lean: inverse_strength_to_frontier

\item \textbf{The frontier upper bound.}  Expected lengths are nonnegative and the rows \(n<n_0\) do not affect a \(\limsup\).  Since \(t\mapsto\min\{1,t^{-1/2}\}\) is nonincreasing on \((0,\infty)\), the pointwise bound of the previous step gives, for every \(n\ge n_0\),
\[
\sup_{P\in\mathcal N_n^{\mathrm{reg}}:\,t_n\ge t_0}
\mathbb E_P\!\left[\lambda\!\left(C_n^{\mathrm{reg}}\right)\right]
\le
C_0\min\left\{1,t_0^{-1/2}\right\},
\]
and therefore
\[
\limsup_{n\to\infty}\ \sup_{P\in\mathcal N_n^{\mathrm{reg}}:\,t_n\ge t_0}
\mathbb E_P\!\left[\lambda\!\left(C_n^{\mathrm{reg}}\right)\right]
\le
C_0\min\left\{1,t_0^{-1/2}\right\}
=\max\left\{2,4\sqrt2\,L+8B_{\varepsilon,c}\right\}\min\left\{1,t_0^{-1/2}\right\}.
\]
This is the asserted upper bound.
% lean: abstractClassFrontierRisk_le

\item \textbf{A deterministic geometry inside the regular-cell class.}  Fix \(t_0>0\).  Using the points \(x_{1,n},\dots,x_{k_n,n}\) of the first step, define the deterministic array \(\mathfrak g=(\nu_n,w_n,e_n)_{n\ge1}\) by
\[
\nu_n=\sum_{i=1}^{k_n}\frac1{k_n}\,\delta_{x_{i,n}},
\qquad
w_n\equiv1,
\qquad
e_n\equiv\frac12,
\]
with both the source covariate law and the target covariate law taken equal to \(\nu_n\).  Since \(k_n\cdot k_n^{-1}=1\), each \(\nu_n\) is a probability measure, and
\[
\nu_n(\mathcal X_n)=1,
\qquad
\nu_n(\{x_{i,n}\})=\frac1{k_n}\quad\text{for every }i .
\]
The four clauses of \cref{obj:def:admissible-geometry-class} hold:
\[
\varepsilon\le\frac12\le1-\varepsilon
\quad\text{(from }0<\varepsilon<1/2\text{)},
\qquad
0\le1\le2k_n
\quad\text{(from }k_n\ge1\text{)},
\]
\[
\int w_n\,d\nu_n=1,
\qquad
\kappa_n=\int w_n^2\,d\nu_n=1\le k_n,
\qquad
\nu_n(A)=\int_A w_n\,d\nu_n\ \ \text{for every measurable }A .
\]
Hence \(\mathfrak g\in\mathcal G\).  Moreover, since \(0<c_-\le1\le c_+\) gives \(c_-/k_n\le1/k_n\le c_+/k_n\), every law in the fixed-geometry slice \(\mathcal P_n(\mathfrak g)\) of \cref{obj:def:fixed-geometry-slice} has source covariate marginal \(\nu_n\), which is carried by the \(k_n\) measurable singletons of \(\mathcal X_n\) with masses in \([c_-/k_n,c_+/k_n]\); therefore
\[
\mathcal P_n(\mathfrak g)\subseteq\mathcal N_n^{\mathrm{reg}}
\qquad\text{for every }n .
\]
Through this inclusion the model restrictions assumed in the statement supply \cref{obj:ass:two-sample-array,obj:ass:instrument-overlap,obj:ass:degrading-array} on every slice row, and \cref{obj:ass:weight-envelope,obj:ass:weight-second-moment} hold there because slice rows belong to \(\mathcal P_n\).  \Cref{obj:thm:fixed-geometry-frontier} therefore applies and gives
\[
\frac{3(1-\alpha)^2}{16}\min\left\{1,t_0^{-1/2}\right\}
\le
V_{\mathfrak g}^{\star}(t_0).
\]
% lean: fixed_geometry_frontier

We also record, from the least-favorable construction of \cref{obj:def:least-favorable-witness} at this geometry, that the slices are eventually inhabited at strength \(t_0\):
\[
\text{for all large }n,\quad
\left\{P\in\mathcal P_n(\mathfrak g):t_n\ge t_0\right\}\neq\varnothing .
\]
% lean: fixedGeometrySlice_eventually_inhabited

\item \textbf{Domination of the fixed-geometry problem by the regular-cell problem.}  Let \(C=(C_n)\in\mathfrak C^{\mathrm{reg}}\) be arbitrary.  Define a fixed-geometry oracle procedure \(D=(D_n)\) by
\[
D_n\!\left((O_i^S)_{i=1}^n,(X_j^T)_{j=1}^{N_n},w,e\right)
=
\begin{cases}
C_n\!\left((O_i^S)_{i=1}^n,(X_j^T)_{j=1}^{N_n},\mathcal X_n,(1/k_n)_{x\in\mathcal X_n},(e(x))_{x\in\mathcal X_n}\right),
& E_n\ \text{holds},\\[0.5em]
\Theta, & \text{otherwise},
\end{cases}
\]
where
\[
E_n=\left\{X_i\in\mathcal X_n\ \text{for all }i\le n,\ \text{and}\ X_j^T\in\mathcal X_n\ \text{for all }j\le N_n\right\}.
\]
Then \(D_n\subseteq\Theta\) because \(C_n\subseteq\Theta\); the event \(E_n\) is measurable, being a finite intersection of preimages of the measurable set \(\mathcal X_n\), and the sample-by-\(\vartheta\) graph of \(D_n\) is the union of the graph of \(C_n\) restricted to \(E_n\) with \(E_n^{\,c}\times\Theta\), hence measurable by the graph-measurability clause of \cref{obj:def:regular-cell-feasible-honesty}.  Finally, \(D_n\) does not read the weight input \(w\) at all, so it is trivially invariant across versions of \(dP_T/dP_S^X\).  Thus \(D\) is an admissible oracle procedure for the class in \cref{obj:def:fixed-geometry-honesty}.

Let \(n\) be arbitrary and \(P\in\mathcal P_n(\mathfrak g)\).  Both the source and the target covariate law of \(P\) equal \(\nu_n\), and \(\nu_n(\mathcal X_n)=1\), so
\[
P(E_n)=1 .
\]
On \(E_n\) the arguments passed to \(C_n\) by \(D_n\) are exactly the class-supplied ones, because the source cell masses of \(P\) are \(q_{x,n}=\nu_n(\{x\})=1/k_n\) on \(\mathcal X_n\) and the propensity of \(P\) is \(e_n\equiv1/2\).  Hence \(D_n\) and \(C_n\) return the same set \(P\)-almost surely, and therefore, for every \(n\) and every \(P\in\mathcal P_n(\mathfrak g)\),
\[
P\left(\theta_T\in D_n\right)=P\left(\theta_T\in C_n\right),
\qquad
\mathbb E_P[\lambda(D_n)]=\mathbb E_P[\lambda(C_n)].
\]

Because \(\mathcal P_n(\mathfrak g)\subseteq\mathcal N_n^{\mathrm{reg}}\) and both index sets are nonempty for all large \(n\) by the previous step, taking infima over the two classes and using the displayed coverage identity gives, for all large \(n\),
\[
\inf_{P\in\mathcal N_n^{\mathrm{reg}}}P\left(\theta_T\in C_n\right)
\le
\inf_{P\in\mathcal P_n(\mathfrak g)}P\left(\theta_T\in D_n\right).
\]
Coverages lie in \([0,1]\), so both sequences are bounded and the \(\liminf\) comparison is legitimate; combined with \(C\in\mathfrak C^{\mathrm{reg}}\) this yields
\[
1-\alpha
\le
\liminf_{n\to\infty}\inf_{P\in\mathcal N_n^{\mathrm{reg}}}P\left(\theta_T\in C_n\right)
\le
\liminf_{n\to\infty}\inf_{P\in\mathcal P_n(\mathfrak g)}P\left(\theta_T\in D_n\right),
\]
so \(D\in\mathfrak C^{\mathrm{or}}(\mathfrak g)\).

The same almost-sure equality gives the length comparison.  Every reported set lies in \(\Theta\), so all expected lengths lie in \([0,\lambda(\Theta)]=[0,2]\); using the inclusion \(\{P\in\mathcal P_n(\mathfrak g):t_n\ge t_0\}\subseteq\{P\in\mathcal N_n^{\mathrm{reg}}:t_n\ge t_0\}\) rowwise and the boundedness for the \(\limsup\) comparison,
\[
\limsup_{n\to\infty}\sup_{P\in\mathcal P_n(\mathfrak g):\,t_n\ge t_0}
\mathbb E_P[\lambda(D_n)]
\le
\limsup_{n\to\infty}\sup_{P\in\mathcal N_n^{\mathrm{reg}}:\,t_n\ge t_0}
\mathbb E_P[\lambda(C_n)].
\]
Fixed-geometry risks are nonnegative, so the infimum defining \(V_{\mathfrak g}^{\star}(t_0)\) in \cref{obj:def:fixed-geometry-value} is bounded below and, \(D\) being admissible for it,
\[
V_{\mathfrak g}^{\star}(t_0)
\le
\limsup_{n\to\infty}\sup_{P\in\mathcal P_n(\mathfrak g):\,t_n\ge t_0}
\mathbb E_P[\lambda(D_n)]
\le
\limsup_{n\to\infty}\sup_{P\in\mathcal N_n^{\mathrm{reg}}:\,t_n\ge t_0}
\mathbb E_P[\lambda(C_n)].
\]
Chaining with the lower bound of the previous step and taking the infimum over \(C\in\mathfrak C^{\mathrm{reg}}\) gives
\[
\frac{3(1-\alpha)^2}{16}\min\left\{1,t_0^{-1/2}\right\}
\le
\inf_{C\in\mathfrak C^{\mathrm{reg}}}
\limsup_{n\to\infty}\sup_{P\in\mathcal N_n^{\mathrm{reg}}:\,t_n\ge t_0}
\mathbb E_P[\lambda(C_n)] .
\]
% lean: regular_cell_unknown_weight_attainment
\end{enumerate}

Together with \(C_0>0\), the almost-sure representation, the membership \((C_n^{\mathrm{reg}})\in\mathfrak C^{\mathrm{reg}}\), the upper bound of the fifth step, and this lower bound, the stated regular-cell attainment conclusion follows for every \(t_0>0\).
% lean: regular_cell_unknown_weight_attainment
\end{proof}
