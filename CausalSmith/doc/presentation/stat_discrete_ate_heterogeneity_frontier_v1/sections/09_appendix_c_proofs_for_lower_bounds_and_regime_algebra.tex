\section{Appendix C. Proofs for lower bounds and regime algebra}

This appendix records the auxiliary lower-bound and algebraic statements that support the main bracket. The proof has four mechanisms. The parametric lower bound supplies the \(M^2/n\) sampling term. The exact-homogeneity transport supplies the capped collision baseline \(M^2\{n^{-1}+\min(1,d/n^2)\}\). The radius-channel transport supplies the rare-cell term \(M^2\sigma^2\min\{1,u_{n,d}\}\). The final rate algebra compares these lower benchmarks with the selector benchmark and identifies the matched regimes and residual wedge.

\begin{lemmav}{lem:parametric-lower}[Parametric minimax lower bound]
For every overlap parameter \(\epsilon\) satisfying \(0<\epsilon<1/2\), there exists a constant \(c_{\epsilon}>0\) such that the following holds. For every \(n,d\in\mathbb N\) and every \(M,\sigma\in\mathbb R\), suppose that
\begin{itemize}
\item \textbf{(Sample and alphabet sizes.)} \(n\ge 1\) and \(d\ge 1\);
\item \textbf{(Outcome radius.)} \(M\ge 1\);
\item \textbf{(Noise radius.)} \(0\le \sigma\le 2\).
\end{itemize}
Then the minimax mean-squared risk \(\mathsf R_{n,d,\epsilon,M,\sigma}\) of \cref{obj:def:minimax-risk} satisfies
\[
\mathsf R_{n,d,\epsilon,M,\sigma}\ge \frac{c_{\epsilon}M^2}{n}.
\]
\end{lemmav}

\Cref{obj:lem:parametric-lower} anchors the lower benchmark at the scale of estimating a bounded real mean from \(n\) observations. This component is combined below with the exact-homogeneity collision scale and the radius-channel rare-cell scale.

The radius-sensitive converse is organized through a channel from binary source experiments into the real-outcome class. The channel preserves the cell and treatment-assignment structure, contracts binary response contrasts by \(\sigma/2\), and then rescales by \(M\).

\begin{definitionv}{synth_5}[Radius-channel hard-family embedding]
For \(0\le\sigma\le2\), set \(\lambda:=\sigma/2\). The embedding \(\Psi\) sends a binary source law from the radius-channel hard family, after padding unused cells with zero mass in the \(d\)-alphabet, to the real-outcome law obtained by the hypothesis-independent channel
\[
B(a)\longmapsto B'(a)\sim\operatorname{Bernoulli}\!\left\{\frac12+\lambda\left(B(a)-\frac12\right)\right\},
\qquad
Y(a):=M\{B'(a)-1/2\},
\qquad
Y=Y(A).
\]
For source laws \(P_0\) and \(P_1\) in the realized family, the embedded targets satisfy
\[
\tau(\Psi(P_1))-\tau(\Psi(P_0))
=
M\frac{\sigma}{2}\{\psi(P_1)-\psi(P_0)\}.
\]
\end{definitionv}

\Cref{obj:synth_5} is the finite-sample transport map used in the radius-channel lower bound. The target identity shows that a binary separation for \(\psi\) becomes an average-treatment-effect separation with scale \(M\sigma\) in the embedded real-outcome experiment.

The next two transport lemmas calibrate the capped alphabet sizes for the exact-homogeneity and radius-channel components. Their role is purely quantitative: they convert the published binary lower-risk levels into the normalized expressions used by the finite-range bracket.

\begin{lemmav}{lem:capped-exact-transport-package}[Capped exact transport scale]
For every overlap level \(0<\epsilon<1/2\), there exist constants
\(b_{\epsilon}^{\mathrm{ex}},c_{\epsilon}^{\mathrm{ex}}>0\) and a cutoff
\(N_{\epsilon}^{\mathrm{ex}}\in\mathbb N\) such that the following properties hold.
\begin{itemize}
\item \textbf{(Capped alphabet.)} For every pair of integers \(n,d\ge1\), define the exact capped alphabet size from \cref{obj:def:least-favorable-handle} by
\[
d_{\mathrm{ex}}
=
\min\left\{d,\max\left\{1,\left\lfloor b_{\epsilon}^{\mathrm{ex}}n^2\right\rfloor\right\}\right\}.
\]
Then \(0<d_{\mathrm{ex}}\le d\), and whenever \(N_{\epsilon}^{\mathrm{ex}}\le n\),
\[
d_{\mathrm{ex}}\le b_{\epsilon}^{\mathrm{ex}}n^2 .
\]

\item \textbf{(Estimator-wise source risk.)} For every \(n,d\ge1\), with \(d_{\mathrm{ex}}\) defined above, there is a real number \(L_{\mathrm{ex}}=L_{\mathrm{ex}}(n,d)\) such that every measurable estimator
\[
\widehat\psi:\bigl(\{1,\ldots,n\}\to \bigl(\{1,\ldots,d_{\mathrm{ex}}\}\times\{0,1\}\times\{0,1\}\bigr)\bigr)\to\mathbb R
\]
admits an exactly homogeneous binary law \(R\) over \(d_{\mathrm{ex}}\) cells at overlap \(\epsilon\) for which
\[
L_{\mathrm{ex}}
\le
\mathbb E_{Q_R^{(n)}}\!\left[
\{\widehat\psi-\psi(R)\}^2
\right].
\]

\item \textbf{(Transport normalization.)} For every \(n,d\ge1\), with the same \(L_{\mathrm{ex}}(n,d)\), and for every \(M\in\mathbb R\),
\[
c_{\epsilon}^{\mathrm{ex}}M^2
\left\{\frac1n+\min\left(1,\frac d{n^2}\right)\right\}
\le
M^2L_{\mathrm{ex}} .
\]
\end{itemize}
\end{lemmav}

The exact transport package expresses the binary exactly homogeneous collision lower bound at the scale of the real-outcome target. The capped alphabet \(d_{\mathrm{ex}}\) keeps the embedded source within the available cells, while the normalization clause yields the baseline \(M^2\{n^{-1}+\min(1,d/n^2)\}\).

\begin{lemmav}{lem:capped-radial-transport-scale}[Capped radial transport scale]
Let \(n,d\) be integers and let \(a,b,M,\sigma\in\mathbb R\).  Suppose that
\begin{itemize}
\item \textbf{(Sample size.)} \(n\ge 3\).
\item \textbf{(Alphabet size.)} \(d\ge 1\).
\item \textbf{(Scale.)} \(a\ge 0\).
\item \textbf{(Radial cap.)} \(0<b\le 4\).
\item \textbf{(Range.)} \(1\le (b/2)n\log(en)\).
\end{itemize}
Define the capped radial alphabet size from \cref{obj:def:least-favorable-handle} by
\[
d_{\mathrm{rad}}
:=
\min\left\{
d,\,
\max\left\{1,\left\lfloor (b/2)n\log(en)\right\rfloor\right\}
\right\}.
\]
Then
\[
\frac{ab^2}{128}M^2\sigma^2
\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}
\le
\left(\frac{M\sigma}{2}\right)^2
\frac a2
\left\{
\frac1n+
\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}
\right\}.
\]
\end{lemmav}

\Cref{obj:lem:capped-radial-transport-scale} translates the source lower-risk scale into the rare-cell expression \(M^2\sigma^2\min\{1,d^2/(n^2\log^2(en))\}\). The capped \(d_{\mathrm{rad}}\) matches the radius-channel construction in \cref{obj:def:least-favorable-handle}.

Small sample sizes are absorbed by a separate finite-sample comparison. This keeps the radius-channel lower bound calibrated uniformly over the range in which the asymptotic radial cap is inactive.

\begin{lemmav}{lem:finite-sample-radial-transport-scale}[Finite-sample radial scale]
Let \(n,d,N\in\mathbb N\) satisfy \(1\le n\le N\), and let \(M,\sigma\in\mathbb R\).  Define
\[
u_{n,d}=\frac{d^2}{n^2\log^2(en)}.
\]
Then
\[
\frac{1}{800N}\,M^2\sigma^2\min\{1,u_{n,d}\}
\le
\left(\frac{M\sigma}{2}\right)^2\frac{1}{200n}.
\]
\end{lemmav}

Together, \cref{obj:lem:capped-radial-transport-scale,obj:lem:finite-sample-radial-transport-scale} supply the scale algebra behind the finite-range radius-channel converse. The converse itself combines the transported radial component with the parametric and exact-homogeneity components.

\begin{theoremv}{thm:radius-channel-converse}[Radius-channel converse]
For every overlap parameter \(\epsilon\) satisfying \(0<\epsilon<1/2\), there exist constants \(c_{\epsilon}\) and \(b_{\epsilon}^{\mathrm{rad}}\) such that
\[
0<c_{\epsilon}\le 1,
\qquad
0<b_{\epsilon}^{\mathrm{rad}}.
\]
For every \(n,d\in\mathbb N\), \(M\in\mathbb R\), and \(\sigma\in\mathbb R\), suppose that
\begin{itemize}
\item \textbf{(Sample and alphabet.)} \(0<n\) and \(0<d\).
\item \textbf{(Scale and radius.)} \(1\le M\) and \(0\le\sigma\le2\).
\item \textbf{(Restricted range.)}
\[
d\le c_{\epsilon}\frac{n^2}{\log(en)}.
\]
\end{itemize}
Then the minimax risk from \cref{obj:def:minimax-risk} satisfies
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\ge
c_{\epsilon}M^2
\left\{
\frac1n+\frac d{n^2}
+\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\]
Moreover, the radius-channel hard family can be realized with the same constants: there are an integer cap, a source class, a padding map into the \(d\)-alphabet, an embedding \(\Psi\) of the source laws into real-outcome laws, and a coupling for the binary full-observation experiment such that the realized radial-channel data have radius parameter \(b_{\epsilon}^{\mathrm{rad}}\), and for all source laws \(P_0\) and \(P_1\),
\[
\tau(\Psi(P_1))-\tau(\Psi(P_0))
=
M\frac{\sigma}{2}
\left\{\psi(P_1)-\psi(P_0)\right\}.
\]
For every total measurable \([-M,M]\)-valued estimator \(\widehat\tau\), some source law \(P\) in this realized family satisfies
\[
c_{\epsilon}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\mathbb E_{\Psi(P)}
\left[
\left\{\widehat\tau-\tau(\Psi(P))\right\}^2
\right].
\]
\end{theoremv}

\Cref{obj:thm:radius-channel-converse} establishes the finite-range lower benchmark used in the original calibrated bracket. Its first two terms give the parametric and collision baseline, and the final term is the radius-channel contribution. The displayed source-target identity records the mechanism of the transport: source contrasts in \(\psi\) become real-outcome contrasts in \(\tau\) through the attenuation \(\sigma/2\) and the outcome scale \(M\), as in classical two-point and fuzzy-hypothesis lower-bound arguments \citep{LeCam1986,Tsybakov2009,Cai2011,Donoho1990}.

Combining the finite-range converse with the constructive selector upper bound gives the restricted-range bracket. This statement is the calibrated predecessor of the all-alphabet bracket in \cref{obj:thm:two-sided-minimax-bracket-all-d}.

\begin{theoremv}{thm:two-sided-minimax-bracket}[Two-sided minimax bracket]
For every overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\), there exist real constants \(c_{\epsilon}\) and \(C_{\epsilon}\) such that
\[
0<c_{\epsilon}\le 1,\qquad 1\le C_{\epsilon},\qquad c_{\epsilon}\le C_{\epsilon}.
\]
For every pair of positive integers \(n,d\) and every \(M,\sigma\in\mathbb R\), assume:
\begin{itemize}
\item \textbf{(Scale.)} \(1\le M\).
\item \textbf{(Radius.)} \(0\le \sigma\le 2\).
\item \textbf{(Range.)}
\[
d\le \frac{c_{\epsilon}n^2}{\log(en)}.
\]
\end{itemize}
Let \(\mathsf R_{n,d,\epsilon,M,\sigma}\) be the minimax mean-squared risk in \cref{obj:def:minimax-risk}. Then
\[
c_{\epsilon}M^2\left\{
\frac{1}{n}+\frac{d}{n^2}
+\sigma^2\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_{\epsilon}M^2\left\{
\frac{1}{n}
+\min\left(1,\min\left(\frac{d^2}{n^2\log^2(en)},\,\sigma^2+\frac{d}{n^2}\right)\right)
\right\}.
\]
\end{theoremv}

\Cref{obj:thm:two-sided-minimax-bracket} places the finite-range upper and lower benchmarks on the same model class and loss scale. The upper expression is the selector rate, while the lower expression adds the exact-homogeneity baseline to the radius-sensitive rare-cell component. This form is useful for comparisons that keep \(d\) below the displayed calibrated range.

The endpoint reductions identify the two boundary cases of the finite-range bracket. Exact homogeneity reduces the selector and converse expressions to the collision rate, while radius two aligns the radius-indexed class with the unrestricted-radius class.

\begin{propositionv}{prop:endpoint-reductions}[Endpoint rate reductions]
For every overlap parameter \(\epsilon\in(0,1/2)\), there exist constants \(c_\epsilon,c,C\in\mathbb R\) such that
\[
0<c_\epsilon\le 1,
\qquad
0<c\le C .
\]
Write
\[
L_n=\log(en),
\qquad
u_{n,d}=\frac{d^2}{n^2L_n^2}.
\]
Then the following statements hold.
\begin{itemize}
\item \textbf{(Exact-homogeneity endpoint.)} For all integers \(n,d\ge1\) such that
\[
d\le c_\epsilon\,\frac{n^2}{L_n},
\]
the upper-frontier expression
\[
\frac1n+\min\left\{1,\min\left\{u_{n,d},\frac d{n^2}\right\}\right\}
\]
and the capped converse expression
\[
\frac1n+\min\left\{1,\frac d{n^2}\right\}
\]
are both comparable to \(n^{-1}+d/n^2\):
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\min\left\{u_{n,d},\frac d{n^2}\right\}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right),
\]
and
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right).
\]
\item \textbf{(Radius-two class identity.)} For every integer \(d\) and every \(M\in\mathbb R\), the radius-two class in \cref{obj:def:model-class} and the unrestricted class in \cref{obj:def:unrestricted-class} have exactly the same attainable laws: every law in \(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\) is the law of some member of \(\mathcal P_{d,\epsilon,M,2}\), and every law in \(\mathcal P_{d,\epsilon,M,2}\) is the law of some member of \(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\).
\item \textbf{(Radius-two endpoint.)} For all integers \(n,d\ge1\), the upper-frontier expression at radius two is exactly
\[
\frac1n+\min\{1,u_{n,d}\},
\]
and the capped converse expression at radius two,
\[
\frac1n+\min\left\{1,\frac d{n^2}\right\}+4\min\{1,u_{n,d}\},
\]
is comparable to the same rate:
\[
c\left(\frac1n+\min\{1,u_{n,d}\}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}+4\min\{1,u_{n,d}\}
\le
C\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
\end{itemize}
\end{propositionv}

\Cref{obj:prop:endpoint-reductions} gives the finite-range endpoint algebra behind the rate discussion. At \(\sigma=0\), the polynomial branch and the collision lower benchmark reduce to \(n^{-1}+d/n^2\) within the calibrated range. At \(\sigma=2\), the radius-indexed law class equals the unrestricted-radius class from \cref{obj:def:unrestricted-class}, and the upper expression is the polynomial large-alphabet rate.

The final result in this appendix localizes the regimes in which the finite-range bracket is order matched. It uses the baseline \(b_{n,d}\), the polynomial scale \(u_{n,d}\), the triangular selector benchmark \(q_{n,d,\sigma}\), and the product-form triangular benchmark \(\ell^{\mathrm{tri}}_{n,d,\sigma}\). The theorem statement below introduces the local shorthand \(\ell_{n,d,\sigma}\) for that product-form benchmark.

\begin{theoremv}{thm:fixed-interior-tightness-and-shrinking-radius-gap}[Fixed-interior gap characterization]
For every overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\), there is a constant \(c_\epsilon>0\) such that the following statements hold.  Write
\[
L_n:=\log(en),\qquad
b_{n,d}:=\frac1n+\frac d{n^2},\qquad
u_{n,d}:=\frac{d^2}{n^2L_n^2},
\]
and
\[
q_{n,d,\sigma}:=\frac1n+\min\left\{u_{n,d},\,\sigma^2+\frac d{n^2}\right\},
\qquad
\ell_{n,d,\sigma}:=b_{n,d}+\sigma^2u_{n,d}.
\]
Also let \(r_{n,d,\sigma}\) denote the frontier rate
\[
r_{n,d,\sigma}:=\frac1n+\min\left\{1,\,\min\{u_{n,d},\,\left(\sigma^2+\frac d{n^2}\right)\}\right\},
\]
and let \(\mathsf R_{n,d,\epsilon,M,\sigma}\) be the minimax mean-squared risk in \cref{obj:def:minimax-risk}.

\begin{itemize}
\item \textbf{(Triangular benchmark algebra.)}  For all \(n,d\in\mathbb N\) and all \(\sigma\in[0,2]\), if
\[
L_n^2<d<nL_n,
\]
then
\[
\frac12\left\{b_{n,d}+\min(u_{n,d},\sigma^2)\right\}
\le q_{n,d,\sigma}
\le b_{n,d}+\min(u_{n,d},\sigma^2),
\qquad
\ell_{n,d,\sigma}=b_{n,d}+\sigma^2u_{n,d}.
\]

\item \textbf{(Fixed interior radii.)}  For every \(\underline\sigma\in(0,2]\), there are constants \(c,C\) with \(0<c\le C\) such that, for all \(n,d\in\mathbb N\) and \(M,\sigma\in\mathbb R\), if
\[
n>0,\qquad d>0,\qquad M\ge1,\qquad
\underline\sigma\le\sigma\le2,\qquad
d\le c_\epsilon\frac{n^2}{L_n},
\]
then
\[
cM^2r_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
CM^2r_{n,d,\sigma}.
\]

\item \textbf{(Matching elbows.)}  For every \(K\ge0\), there are constants \(c,C\) with \(0<c\le C\) such that, for all \(n,d\in\mathbb N\) and \(M,\sigma\in\mathbb R\), if
\[
n>0,\qquad d>0,\qquad M\ge1,\qquad 0\le\sigma\le2,
\]
and at least one of
\[
u_{n,d}\ge1,\qquad
u_{n,d}\le Kb_{n,d},\qquad
\sigma^2\le Kb_{n,d}
\]
holds, then
\[
cM^2r_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
CM^2r_{n,d,\sigma}.
\]

\item \textbf{(Exact algebraic elbow.)}  For all \(n,d\in\mathbb N\) with \(n>0\),
\[
u_{n,d}\le b_{n,d}
\quad\Longleftrightarrow\quad
d^2\le nL_n^2+dL_n^2.
\]

\item \textbf{(Residual wedge implication.)}  For every alphabet sequence \(d_t\) and radius sequence \(\sigma_t\), if eventually
\[
d_t\ge1,\qquad 0\le\sigma_t\le2,
\]
and
\[
\frac{r_{t,d_t,\sigma_t}}{t^{-1}+\min\{1,d_t/t^2\}+\sigma_t^2\min\{1,u_{t,d_t}\}}\longrightarrow\infty,
\]
then the sequence lies in the residual wedge:
\[
\frac{b_{t,d_t}}{u_{t,d_t}}\longrightarrow0,\qquad
u_{t,d_t}\longrightarrow0,\qquad
\frac{b_{t,d_t}}{\sigma_t^2}\longrightarrow0,\qquad
\sigma_t^2\longrightarrow0.
\]

\item \textbf{(Witness sequence.)}  The sequence \(d_n=n\) and \(\sigma_n=L_n^{-1/2}\) lies in the residual wedge.  Along this sequence,
\[
\frac{q_{n,n,L_n^{-1/2}}}{\ell_{n,n,L_n^{-1/2}}}
\asymp L_n,
\qquad
\frac{q_{n,n,L_n^{-1/2}}}{\ell_{n,n,L_n^{-1/2}}}\longrightarrow\infty .
\]
\end{itemize}
\end{theoremv}

\Cref{obj:thm:fixed-interior-tightness-and-shrinking-radius-gap} gives the finite-range phase algebra supporting the matched-regime discussion in the main text. For every radius bounded below by a positive constant, the minimax risk is comparable to \(M^2r_{n,d,\sigma}\) over the calibrated alphabet range. The same comparison holds at the saturation and parametric-dominance elbows. In the triangular region \(L_n^2<d<nL_n\), the theorem separates the selector expression \(b_{n,d}+\min(u_{n,d},\sigma^2)\) from the converse expression \(b_{n,d}+\sigma^2u_{n,d}\), and the witness sequence shows that the residual shrinking-radius region has logarithmic benchmark separation.
