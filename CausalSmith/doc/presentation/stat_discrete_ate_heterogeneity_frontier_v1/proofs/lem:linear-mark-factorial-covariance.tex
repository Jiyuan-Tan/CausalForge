\begin{proof}[Proof of \cref{obj:lem:linear-mark-factorial-covariance}]
Take \(C_\epsilon=64\).  This constant is positive and depends only on \(\epsilon\).
% lean: linear_mark_factorial_covariance

Let
\[
F_k
:=
\widehat P_k
=
\sum_{j=0}^{K-2}\frac{g_{Kj}}{B^{j+1}}
\{U_{k,1,j}-U_{k,0,j}\}.
\]
All \(F_k\) have finite second moments under \(Q_P^{(m)}\), since each is a finite linear combination of ordered one-mark factorial statistics and the conditional second-central-moment envelope in \cref{obj:def:model-class} supplies the needed second moments.
% lean: memLp_allBlockLightPolynomialTerm

Expanding the variance of the finite sum gives
\[
\operatorname{Var}_{Q_P^{(m)}}\!\left(\sum_{k\in S}F_k\right)
=
\sum_{k\in S}\sum_{\ell\in S}
\operatorname{Cov}_{Q_P^{(m)}}(F_k,F_\ell).
\]
Consequently,
\[
\operatorname{Var}_{Q_P^{(m)}}\!\left(\sum_{k\in S}F_k\right)
\le
\sum_{k\in S}\sum_{\ell\in S}
\left|\operatorname{Cov}_{Q_P^{(m)}}(F_k,F_\ell)\right|.
\]
% lean: variance_fun_sum'

Fix \(k\in S\), and write \(p_k=P(X=k)\).  The light-cell condition and the shift condition give, for every \(0\le r\le K-2\),
\[
0\le p_k\le B,
\qquad
\frac54p_k\le B,
\qquad
\frac54p_k+\frac{r+2}{m}\le B.
\]
Indeed, \(p_k\le B/4\), while \(r+2\le K\) and \(4(K+2)/m\le3B/4\) control the final summand.  Also \(4(K+2)^2\le m\) gives
\[
\frac{2(K+2)^2}{m}\le \frac12.
\]
For each arm pair and each \(j,r\le K-2\), \cref{obj:lem:same-cell-factorial-cross-moment} therefore yields
\[
\begin{aligned}
\left|\operatorname{Cov}_{Q_P^{(m)}}(U_{k,a,j},U_{k,b,r})\right|
&\le
\frac12\,p_k^{j+2}p_k^{r+2} \\
&\quad+
\exp(1)\left(\frac54p_k\right)^{r+2}
\left(\frac54p_k+\frac{r+2}{m}\right)^{j+2}.
\end{aligned}
\]
Expanding \(F_k\), taking absolute values, and summing over both arm indices decomposes the same-cell contribution into the empty-matching and positive-matching weighted sums.  By \cref{obj:lem:shifted-chebyshev-coefficient-envelope}, the empty-matching part is bounded by
\[
4\cdot\frac12
\left\{
\sum_{j=0}^{K-2}
\left|\frac{g_{Kj}}{B^{j+1}}\right|p_k^{j+2}
\right\}
\left\{
\sum_{r=0}^{K-2}
\left|\frac{g_{Kr}}{B^{r+1}}\right|p_k^{r+2}
\right\}
\le
4\cdot\frac12\{B6^K\}^2.
\]
For the positive-matching part, the same coefficient envelope applies once with \((5/4)p_k\) and, for each fixed \(r\), once with \((5/4)p_k+(r+2)/m\); using \(\exp(1)\le3\),
\[
\begin{aligned}
&4\sum_{r=0}^{K-2}
\left|\frac{g_{Kr}}{B^{r+1}}\right|
\exp(1)\left(\frac54p_k\right)^{r+2}
\left\{
\sum_{j=0}^{K-2}
\left|\frac{g_{Kj}}{B^{j+1}}\right|
\left(\frac54p_k+\frac{r+2}{m}\right)^{j+2}
\right\} \\
&\hspace{4em}\le
4\cdot3\{B6^K\}^2.
\end{aligned}
\]
Thus
\[
\left|\operatorname{Cov}_{Q_P^{(m)}}(F_k,F_k)\right|
\le
\left(4\cdot\frac12+4\cdot3\right)\{B6^K\}^2
\le
16\{B6^K\}^2 .
\]
% lean: weighted_centeredCrossMoment_same_cell_sum_le

For distinct \(k,\ell\in S\), \cref{obj:lem:distinct-cell-factorial-cross-moment,obj:lem:shifted-chebyshev-coefficient-envelope} give
\[
\left|\operatorname{Cov}_{Q_P^{(m)}}(F_k,F_\ell)\right|
\le
32\,\frac{K^2}{m}\{B6^K\}^2 .
\]
Here all positive partial matchings between different cells are incompatible, so the only contribution is the empty-matching normalization correction
\[
\frac{2(K+2)^2}{m}p_k^{j+2}p_\ell^{r+2};
\]
summing over both arms contributes the factor \(4\), and \(K\ge2\) gives \((K+2)^2\le4K^2\).
% lean: weighted_centeredCrossMoment_cell_ne_sum_le

Since \(|S|\le d\), summing the diagonal and off-diagonal bounds gives
\[
\operatorname{Var}_{Q_P^{(m)}}\!\left(\sum_{k\in S}F_k\right)
\le
16d\{B6^K\}^2
+
32d^2\frac{K^2}{m}\{B6^K\}^2 .
\]
% lean: linear_mark_factorial_covariance
Using
\[
\{B6^K\}^2=B^2\,6^{2K},
\]
enlarging the two coefficients to \(64\), and adding the nonnegative term \(64/m\), we obtain
\[
\operatorname{Var}_{Q_P^{(m)}}\!\left(\sum_{k\in S}\widehat P_k\right)
\le
\frac{64}{m}
+
64\,6^{2K}
\left\{
dB^2+\frac{d^2K^2B^2}{m}
\right\}.
\]
This is the asserted bound with \(C_\epsilon=64\).
% lean: linear_mark_factorial_covariance
\end{proof}
