\begin{proof}[Proof of \cref{obj:lem:finite-sample-radial-transport-scale}]
Put
\[
u_{n,d}=\frac{d^2}{n^2\log^2(en)} .
\]
The minimum is bounded by its first entry, so
\[
\min\{1,u_{n,d}\}\le 1 .
\]
The assumptions \(1\le n\le N\) imply \(N>0\).  Hence
\[
0\le \frac{1}{800N}M^2\sigma^2,
\]
since the remaining factors are squares.  Multiplying the preceding minimum bound by this nonnegative factor gives
\[
\frac{1}{800N}M^2\sigma^2\min\{1,u_{n,d}\}
\le
\frac{1}{800N}M^2\sigma^2 .
\]
% lean: finiteSampleRadial_transportScale
The same assumptions give
\[
\frac1N\le \frac1n .
\]
Multiplying this reciprocal bound by the nonnegative factors \(1/800\), \(M^2\), and \(\sigma^2\) yields
\[
\frac{1}{800N}M^2\sigma^2
\le
\frac{1}{800n}M^2\sigma^2 .
\]
% lean: finiteSampleRadial_transportScale
Finally,
\[
\frac{1}{800n}M^2\sigma^2
=
\left(\frac{M\sigma}{2}\right)^2\frac{1}{200n},
\]
which proves the claim.
% lean: finiteSampleRadial_transportScale
\end{proof}
