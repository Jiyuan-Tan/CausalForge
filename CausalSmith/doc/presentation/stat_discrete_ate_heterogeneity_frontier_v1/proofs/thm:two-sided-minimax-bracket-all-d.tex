\begin{proof}[Proof of \cref{obj:thm:two-sided-minimax-bracket-all-d}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By
\cref{obj:thm:frontier-upper-all-d}, choose a constant
\(C_{\epsilon}^{\mathrm{up}}>0\) and a polynomial calibration handle such that, for every
\(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), the corresponding known-radius selector
\(\widehat\tau_n^{\star}\in\mathcal T_{n,M}\) satisfies
\[
\sup_{P\in\mathcal P_{d,\epsilon,M,\sigma}}
E_{Q_P^{(n)}}\!\left[
\{\widehat\tau_n^{\star}-\tau(P)\}^2
\right]
\le
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}.
\]
By \cref{obj:thm:radius-channel-converse-all-d}, choose
\(c_{\epsilon}^{\mathrm{lo}}\) and \(b_{\epsilon}^{\mathrm{rad}}\) such that
\[
0<c_{\epsilon}^{\mathrm{lo}}\le1,\qquad 0<b_{\epsilon}^{\mathrm{rad}},
\]
and, for every such \(n,d,M,\sigma\),
\[
c_{\epsilon}^{\mathrm{lo}}M^2\ell_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Set
\[
c_{\epsilon}:=c_{\epsilon}^{\mathrm{lo}},
\qquad
C_{\epsilon}:=\max\{1,C_{\epsilon}^{\mathrm{up}}\}.
\]
Then
\[
0<c_{\epsilon}\le1,\qquad 1\le C_{\epsilon},\qquad c_{\epsilon}\le C_{\epsilon}.
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.two_sided_minimax_bracket_all_d

\item Fix \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\).  The lower bound is exactly the
one supplied by \cref{obj:thm:radius-channel-converse-all-d} with the constants fixed above:
\[
c_{\epsilon}M^2\ell_{n,d,\sigma}
=
c_{\epsilon}^{\mathrm{lo}}M^2\ell_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The accompanying radius-channel certificate in
\cref{obj:thm:radius-channel-converse-all-d} also supplies a realized source family and an
embedding into the same model class.  Applying its source-risk assertion to the zero source
estimator gives a source law, and the embedding-membership part of the certificate gives an
embedded law \(P_{\bullet}\) with
\[
P_{\bullet}\in\mathcal P_{d,\epsilon,M,\sigma}.
\]
Thus the model class is nonempty for these indices. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

\item For any estimator \(\widehat\eta\in\mathcal T_{n,M}\), define its worst-case risk over
this class by
\[
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
:=
\sup_{P\in\mathcal P_{d,\epsilon,M,\sigma}}
E_{Q_P^{(n)}}\!\left[\{\widehat\eta-\tau(P)\}^2\right].
\]
We first record the lower-boundedness needed to apply the infimum in
\cref{obj:def:minimax-risk}.  Fix \(\widehat\eta\in\mathcal T_{n,M}\).  In the vacuous branch
where the model index is empty, the real supremum convention gives
\(\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)=0\).  In the nonvacuous branch, consider
the range of mean-squared errors indexed by \(P\in\mathcal P_{d,\epsilon,M,\sigma}\).  If
this range is bounded above, choose any law in the class; its mean-squared error is the
integral of a square and is therefore nonnegative, and it is bounded by the supremum defining
\(\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)\).  If the range is unbounded above, the
supremum is in its unbounded-above case and the inequality
\(0\le\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)\) is immediate.  Hence
\[
0\le\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
\qquad\text{for every }\widehat\eta\in\mathcal T_{n,M}.
\]
The collection of worst-case risks is therefore bounded below by \(0\).  Using
\cref{obj:def:minimax-risk} and the admissibility of \(\widehat\tau_n^{\star}\),
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
=
\inf_{\widehat\eta\in\mathcal T_{n,M}}
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
\le
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\tau_n^{\star}).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.minimaxRisk

\item Applying the selector guarantee from \cref{obj:thm:frontier-upper-all-d} gives
\[
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\tau_n^{\star})
\le
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}.
\]
Moreover,
\[
u_{n,d}\ge0,\qquad h_{n,d,\sigma}\ge0,
\]
so
\[
r_{n,d,\sigma}
=
\frac1n+\min\{1,u_{n,d},h_{n,d,\sigma}\}
\ge
\frac1n
>
0.
\]
Therefore \(M^2r_{n,d,\sigma}\ge0\), and \(C_{\epsilon}^{\mathrm{up}}\le C_{\epsilon}\)
implies
\[
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}
\le
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Combining the previous displays yields
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Together with the lower bound supplied by \cref{obj:thm:radius-channel-converse-all-d} and
instantiated above, this proves the asserted bracket for all \(n,d\ge1\), \(M\ge1\), and
\(0\le\sigma\le2\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d
\end{enumerate}
\end{proof}
