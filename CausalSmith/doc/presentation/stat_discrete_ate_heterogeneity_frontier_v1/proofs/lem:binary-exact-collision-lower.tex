\begin{proof}[Proof of \cref{obj:lem:binary-exact-collision-lower}]
Set
\[
u_\epsilon:=\min\{1/2,8(1/2-\epsilon)^2\},
\qquad
\Delta:=u_\epsilon\frac{\sqrt d}{n},
\qquad
\rho:=\sqrt{\Delta/8}.
\]
Write \(\eta:=1/2-\epsilon\).  From \(0<\epsilon<1/2\), \(d\le n^2\), and \(n,d\ge1\),
\[
0<u_\epsilon,
\qquad
u_\epsilon\le\frac12,
\qquad
u_\epsilon\le 8\eta^2,
\qquad
\sqrt d\le n.
\]
Hence
\[
0\le\Delta\le u_\epsilon\le\frac12,
\qquad
\rho^2=\frac{\Delta}{8},
\qquad
\rho\le\eta,
\qquad
\rho\le\frac14,
\qquad
\rho<\frac12,
\qquad
\rho+\frac{\Delta}{4}\le\frac12.
\]
These are the admissibility inequalities used below: \(\rho\le1/2-\epsilon\) gives overlap, while \(\rho<1/2\) and \(\rho+\Delta/4\le1/2\) keep the Bernoulli means in \([0,1]\). % lean: binaryExactMinimaxRisk_collision_lower

For each sign vector \(\xi\in\{\pm1\}^d\), define a uniform-mass binary law \(P_\xi\) with
\[
\pi_k^{(\xi)}=\frac12+\rho\xi_k,
\qquad
\mu_{1k}^{(\xi)}=\frac12-\rho\xi_k+\frac{\Delta}{4},
\qquad
\mu_{0k}^{(\xi)}=\frac12-\rho\xi_k-\frac{\Delta}{4}.
\]
Let \(P_0\) be the corresponding centered law with \(\Delta=\rho=0\).  The displayed inequalities make each \(P_\xi\) and \(P_0\) members of the fixed-sample binary exactly homogeneous source experiment in \cref{obj:synth_3}, and
\[
\psi(P_\xi)=\frac{\Delta}{2},
\qquad
\psi(P_0)=0.
\] % lean: exactSignExactLaw_ate

Let \(Q_0=P_0^{\otimes n}\) and let \(\overline Q=2^{-d}\sum_\xi P_\xi^{\otimes n}\) be the uniform mixture over signs.  For \(\xi,\xi'\in\{\pm1\}^d\), put \(S(\xi,\xi'):=\sum_{k=1}^d \xi_k\xi'_k\) and
\[
\Gamma:=\Delta-\frac{\Delta^2}{2}+\frac{\Delta^3}{8}.
\]
The centered one-observation mass satisfies \(p_0(k,a,y)=1/(4d)\).  For a fixed cell \(k\), write \(s=\xi_k\) and \(t=\xi'_k\).  Summing over the two arms and two outcomes gives
\[
\begin{aligned}
&\sum_{a,y}\frac{p_\xi(k,a,y)p_{\xi'}(k,a,y)}{p_0(k,a,y)} \\
&=\frac4d\Bigg[
\Big(\frac12+\rho s\Big)\Big(\frac12+\rho t\Big)
\left\{
\Big(\frac12-\rho s+\frac\Delta4\Big)
\Big(\frac12-\rho t+\frac\Delta4\Big)
+
\Big(\frac12+\rho s-\frac\Delta4\Big)
\Big(\frac12+\rho t-\frac\Delta4\Big)
\right\} \\
&\hspace{2.0cm}+
\Big(\frac12-\rho s\Big)\Big(\frac12-\rho t\Big)
\left\{
\Big(\frac12-\rho s-\frac\Delta4\Big)
\Big(\frac12-\rho t-\frac\Delta4\Big)
+
\Big(\frac12+\rho s+\frac\Delta4\Big)
\Big(\frac12+\rho t+\frac\Delta4\Big)
\right\}\Bigg].
\end{aligned}
\]
Using \(\rho^2=\Delta/8\), this expression reduces in the two sign cases to
\[
\frac1d
\begin{cases}
1+\Delta-\Delta^2/2+\Delta^3/8, & st=1,\\
1-\Delta+\Delta^2/2-\Delta^3/8, & st=-1,
\end{cases}
=\frac1d\{1+\Gamma st\}.
\]
Summing over cells gives the one-observation overlap identity
\[
\sum_z\frac{p_\xi(z)p_{\xi'}(z)}{p_0(z)}
=
1+\frac{\Gamma}{d}S(\xi,\xi').
\]
Therefore the finite product second-moment identity gives
\[
1+\chi^2(\overline Q\Vert Q_0)
=
2^{-2d}\sum_{\xi,\xi'}
\left(1+\frac{\Gamma}{d}S(\xi,\xi')\right)^n
=
2^{-2d}\sum_{\xi,\xi'}
\left(1+\frac{2(\Gamma/2)}{d}S(\xi,\xi')\right)^n .
\] % lean: exactChiSqOverlap_eq, one_add_chiSqDiv_exactQtrue_exactQfalse

Since \(0\le\Delta\le1/2\),
\[
0\le\Gamma\le\Delta\le1/2.
\]
We now prove the finite-sign average bound used at \(\gamma=\Gamma/2\).  Let \(\gamma\ge0\), \(2\gamma\le1\), and \(2n^2\gamma^2\le d\log2\).  With \(c:=2\gamma/d\), every sign-pair sum \(S\) lies in \([-d,d]\), so \(cS\ge-1\).  Hence \((1+cS)^n\le \exp(ncS)\), and
\[
\begin{aligned}
2^{-2d}\sum_{\xi,\xi'}(1+cS(\xi,\xi'))^n
&\le
2^{-2d}\sum_{\xi,\xi'}\exp\{ncS(\xi,\xi')\} \\
&=
\prod_{k=1}^d\left(\frac14\sum_{s,t\in\{\pm1\}}\exp(ncst)\right) \\
&=\{\cosh(nc)\}^d
\le \exp\left\{\frac{d n^2c^2}{2}\right\}
=\exp\left\{\frac{2n^2\gamma^2}{d}\right\}
\le 2.
\end{aligned}
\]
For \(\gamma=\Gamma/2\), the remaining regularity condition follows from
\[
2n^2(\Gamma/2)^2
=\frac{n^2\Gamma^2}{2}
\le \frac{n^2\Delta^2}{2}
=\frac{u_\epsilon^2d}{2}
\le \frac d8
\le d\log2,
\]
using \(\Delta^2=u_\epsilon^2d/n^2\), \(u_\epsilon\le1/2\), and \(1/8\le\log2\).  Hence \(1+\chi^2(\overline Q\Vert Q_0)\le2\), so \(\chi^2(\overline Q\Vert Q_0)\le1\).  The chi-square to total-variation comparison gives
\[
\operatorname{TV}(Q_0,\overline Q)\le\frac12 .
\] % lean: exact_tvDist_le_half

Fix any measurable estimator.  The two-point testing inequality for the targets \(0\) under \(Q_0\) and \(\Delta/2\) under \(\overline Q\), with separation threshold \(\Delta/4\), gives a tail probability at least \(1/4\) under one of these two laws.  If the mixture tail is the larger one, one sign \(\xi\) has at least that tail probability under \(P_\xi^{\otimes n}\).  Multiplying the resulting tail probability by \((\Delta/4)^2\) shows that, for this estimator, either \(P_0\) or some \(P_\xi\) has mean-squared error at least
\[
\left(\frac{\Delta}{4}\right)^2\frac14
=\frac{\Delta^2}{64}.
\]
Taking the supremum over the binary exactly homogeneous class and then the infimum over measurable estimators gives
\[
\frac{\Delta^2}{64}
\le
\inf_{\widehat\psi}\sup_{P\in\mathcal H_{n,d}^{\mathrm{bin}}(\epsilon)}
E_{P^{\otimes n}}\!\left[\{\widehat\psi-\psi(P)\}^2\right].
\] % lean: binaryExactMinimaxRisk_collision_aux

Finally \(\Delta^2=u_\epsilon^2d/n^2\), so the asserted lower bound follows.
\end{proof}
