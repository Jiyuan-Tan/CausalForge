\begin{proof}[Proof of \cref{obj:thm:robust-upper-construction-resolution-all-d}]
Fix an overlap parameter \(0<\epsilon<1/2\).
\begin{enumerate}
\item By \cref{obj:lem:continuous-ratio-polynomial-upper-all-d}, choose
\(C_{\epsilon}^{\mathrm{poly}}>0\) and a polynomial handle
\(h_{\epsilon}=(N_{\epsilon},\rho_{\epsilon})\).  The handle has
\(\rho_{\epsilon}>0\) and satisfies the calibration clause
\[
N_{\epsilon}\le n,\qquad d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n
\]
for all integers \(n,d\ge1\).  The same lemma gives, for every
\(n,d\ge1\), \(M\ge1\), \(0\le\sigma\le2\), and
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\), that the estimator
\(T_n^{\mathrm{poly}}\) indexed by this handle is measurable, takes values in
\([-M,M]\), and satisfies
\[
E_{Q_P^{(n)}}\!\left[
\{T_n^{\mathrm{poly}}-\tau(P)\}^2
\right]
\le
C_{\epsilon}^{\mathrm{poly}}M^2
\left\{
\frac1n+\min\!\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\]
% lean: continuous_ratio_polynomial_upper_all_d

\item By \cref{obj:lem:continuous-occupancy-collision-upper-all-d}, choose
\(C_{\epsilon}^{\mathrm{col}}>0\) such that, for every \(n,d\ge1\) and every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\{T_n^{\mathrm{col}}-\tau(P)\}^2
\right]
\le
C_{\epsilon}^{\mathrm{col}}M^2
\left(
\frac1n+\sigma^2+\frac{d}{n^2}
\right).
\]
Here the bounds \(M\ge1\) and \(0\le\sigma\le2\) are the corresponding
model-class fields in \(\mathcal P_{d,\epsilon,M,\sigma}\) from
\cref{obj:def:model-class}.
% lean: continuous_occupancy_collision_upper_all_d

\item The collision estimator also has the required admissibility property.  For a
sample \(s\), \cref{obj:def:collision-handle} defines
\[
T_n^{\mathrm{col}}(s)
=
\operatorname{clip}_{[-M,M]}
\left(
\begin{cases}
\displaystyle
\frac{
\sum_{k=1}^d
\mathbf 1\{N_{0k}(s)>0,\,N_{1k}(s)>0\}
N_k(s)
\left(
\overline Y_{1k}(s)-\overline Y_{0k}(s)
\right)
}{R_n(s)},&R_n(s)>0,\\[2ex]
0,&R_n(s)=0.
\end{cases}
\right),
\]
where each empirical arm-cell mean is the indicator-totalized ratio
\(S_{ak}(s)/N_{ak}(s)\) on \(N_{ak}(s)>0\) and is assigned value \(0\) when
\(N_{ak}(s)=0\).  The counts are finite sums of sample indicators, the totals
are finite sums of indicator-weighted outcomes, and the branches use
measurable comparisons, arithmetic operations, and a finite sum over cells.
Thus the inner statistic is a measurable function of the \(n\)-sample.  Since
\(M\ge1\) gives \(0\le M\), clipping to \([-M,M]\) is measurable and has range
contained in \([-M,M]\).  Therefore \(T_n^{\mathrm{col}}\) is a measurable map
of the sample and satisfies
\[
T_n^{\mathrm{col}}(s)\in[-M,M]
\qquad\text{for every sample }s.
\]
% lean: collisionEstimator_admissible

\item Define
\[
C_{\epsilon}:=\max\{C_{\epsilon}^{\mathrm{poly}},
C_{\epsilon}^{\mathrm{col}}\}.
\]
Then \(C_{\epsilon}>0\).  For \(n,d\ge1\) and \(0\le\sigma\le2\),
\[
0\le
\frac1n+\min\!\left(1,\frac{d^2}{n^2\log^2(en)}\right),
\qquad
0\le
\frac1n+\sigma^2+\frac{d}{n^2}.
\]
Together with \(M^2\ge0\), the inequalities
\(C_{\epsilon}^{\mathrm{poly}}\le C_{\epsilon}\) and
\(C_{\epsilon}^{\mathrm{col}}\le C_{\epsilon}\) upgrade the two pointwise risk
inequalities above to the same inequalities with \(C_{\epsilon}\).  With
\(h=h_{\epsilon}\), the calibration, measurability, range, and pointwise risk
clauses hold simultaneously for every \(n,d\ge1\), \(M\ge1\),
\(0\le\sigma\le2\), and every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\) from
\cref{obj:def:model-class}.  Since the resulting right-hand sides do not depend
on \(P\), taking the supremum over \(P\in\mathcal P_{d,\epsilon,M,\sigma}\)
gives the two displayed supremum inequalities.
% lean: robust_upper_construction_resolution_all_d
\end{enumerate}
\end{proof}
