\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions-all-d}]
Take \(c=1/10\) and \(C=10\).  Fix integers \(n,d\ge1\), and write
\[
\Lambda_n:=\log(en),\qquad
u_{n,d}:=\frac{d^2}{n^2\Lambda_n^2},\qquad
v_{n,d}:=\frac d{n^2}.
\]
Also set
\[
\theta^{\mathrm{ex}}_{n,d}:=\frac1n+\min(1,v_{n,d}),
\qquad
\theta^{\mathrm{poly}}_{n,d}:=\frac1n+\min(1,u_{n,d}).
\]
The quantities \(1/n\), \(u_{n,d}\), and \(v_{n,d}\) are nonnegative.

\begin{enumerate}
\item Since \(n\ge1\),
\[
\Lambda_n
=\log(en)=1+\log n\ge1,
\]
and hence \(\Lambda_n^2>0\).  At radius two the collision term is
\[
2^2+v_{n,d}=4+v_{n,d}\ge1.
\]
Therefore
\[
\frac1n+\min\{1,u_{n,d},4+v_{n,d}\}
=\frac1n+\min(1,u_{n,d})
=\theta^{\mathrm{poly}}_{n,d}.
\]
This proves the endpoint identity, and also gives the two-sided comparison of the
radius-two upper frontier expression with \(\theta^{\mathrm{poly}}_{n,d}\), because
\(\theta^{\mathrm{poly}}_{n,d}\ge0\) and \(1/10\le1\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item We use the elementary logarithmic estimate
\[
\Lambda_n^2\le 9n.
\]
Indeed, \(\log n\le 2\sqrt n\), while \(\sqrt n\ge1\) and \(\log n\ge0\).  Thus
\[
\Lambda_n=1+\log n\le 1+2\sqrt n\le3\sqrt n,
\]
and squaring gives \(\Lambda_n^2\le9n\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item First consider the case \(\Lambda_n^2\le d\).  Since \(d>0\),
\[
v_{n,d}=\frac d{n^2}
\le
\frac{d^2}{n^2\Lambda_n^2}
=u_{n,d}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
=
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression is also exactly
\[
\frac1n+\min(1,v_{n,d})=\theta^{\mathrm{ex}}_{n,d}.
\]
For the radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and, using \(\min(1,v_{n,d})\le\min(1,u_{n,d})\),
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac1n+5\min(1,u_{n,d})
\le
5\theta^{\mathrm{poly}}_{n,d}.
\]
All required comparisons in this case follow from \(1/10\le1\le5\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now consider the case \(d<\Lambda_n^2\).  By the logarithmic estimate,
\[
d\le9n,
\qquad\text{so}\qquad
v_{n,d}\le\frac9n.
\]
The radius-zero upper expression satisfies
\[
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d},
\]
and
\[
\theta^{\mathrm{ex}}_{n,d}
=\frac1n+\min(1,v_{n,d})
\le
\frac1n+\frac9n
=
\frac{10}{n}.
\]
Hence
\[
\frac1{10}\theta^{\mathrm{ex}}_{n,d}
\le
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\theta^{\mathrm{ex}}_{n,d}
\le
10\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression again equals \(\theta^{\mathrm{ex}}_{n,d}\).  For the
radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac{10}{n}+4\min(1,u_{n,d})
\le
10\theta^{\mathrm{poly}}_{n,d}.
\]
Together with the radius-two identity from the first step, this proves the stated
two-sided comparisons in the second case.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item It remains to identify the two radius-two classes.  By \cref{obj:lem:scale-sanity}, for
every \(d\), every \(\epsilon\in\mathbb R\), and every \(M\in\mathbb R\), each law in
\(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\) is represented by some member of
\(\mathcal P_{d,\epsilon,M,2}\) with the same law, and conversely each law in
\(\mathcal P_{d,\epsilon,M,2}\) is represented by some member of
\(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\) with the same law.  Thus the represented laws are
exactly the same at radius two.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.scale_sanity
\end{enumerate}
\end{proof}
