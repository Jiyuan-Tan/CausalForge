\begin{proof}[Proof of \cref{obj:thm:two-sided-minimax-bracket}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By \cref{obj:thm:two-sided-minimax-bracket-all-d}, choose constants \(c_{\epsilon}\) and \(C_{\epsilon}\) such that
\[
0<c_{\epsilon}\le 1,\qquad 1\le C_{\epsilon},\qquad c_{\epsilon}\le C_{\epsilon},
\]
and such that, for every \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\),
\[
c_{\epsilon}M^2
\left\{
\frac1n+\min\left(1,\frac{d}{n^2}\right)
+\sigma^2\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\]
and
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_{\epsilon}M^2
\left\{
\frac1n+
\min\left(1,\min\left(\frac{d^2}{n^2\log^2(en)},\sigma^2+\frac{d}{n^2}\right)\right)
\right\}.
\]
These constants already have the normalization required in the statement. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.two_sided_minimax_bracket_all_d

\item Now fix \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), and assume
\[
d\le \frac{c_{\epsilon}n^2}{\log(en)} .
\]
Since \(n\ge1\),
\[
\log(en)=\log(e)+\log n=1+\log n\ge1.
\]
The numerator \(c_{\epsilon}n^2\) is nonnegative, so
\[
d\le \frac{c_{\epsilon}n^2}{\log(en)}
\le c_{\epsilon}n^2
\le n^2,
\]
where the last inequality uses \(c_{\epsilon}\le1\).  Dividing by \(n^2>0\) gives
\[
\frac{d}{n^2}\le1,
\qquad\text{and hence}\qquad
\min\left(1,\frac{d}{n^2}\right)=\frac{d}{n^2}.
\]
Substituting this identity into the lower bound from the preceding step yields
\[
c_{\epsilon}M^2\left\{
\frac{1}{n}+\frac{d}{n^2}
+\sigma^2\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The upper bound is exactly the upper bound supplied in the preceding step.  Therefore the displayed lower and upper bounds hold throughout the stated restricted range. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.two_sided_minimax_bracket
\end{enumerate}
\end{proof}
