\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions}]
\begin{enumerate}
\item First record the logarithmic estimate used below.  If \(n\ge1\), then
\[
L_n=\log(en)=1+\log n .
\]
The standard bound \(\log n\le n^{1/2}/(1/2)=2\sqrt n\), together with
\(\log n\ge0\) and \(\sqrt n\ge1\), gives
\[
0\le L_n\le 1+2\sqrt n\le 3\sqrt n .
\]
Squaring yields
\[
L_n^2\le 9n .
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item We prove the all-alphabet comparison that supplies the constants.  For \(n,d\ge1\), put
\[
\eta_{n,d}:=\frac{d}{n^2},
\qquad
u_{n,d}:=\frac{d^2}{n^2L_n^2}.
\]
Since \(L_n\ge1\), both \(\eta_{n,d}\) and \(u_{n,d}\) are nonnegative.  At radius two the collision term is
\[
2^2+\eta_{n,d}=4+\eta_{n,d}\ge1,
\]
and therefore
\[
\frac1n+\min\{1,\min\{u_{n,d},4+\eta_{n,d}\}\}
=
\frac1n+\min\{1,u_{n,d}\}.
\]
At radius zero,
\[
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}
\le
\frac1n+\min\{1,\eta_{n,d}\},
\]
and the capped converse expression at radius zero is exactly the right-hand side.

It remains to compare the radius-two converse term
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\]
with
\[
\frac1n+\min\{1,u_{n,d}\}.
\]
If \(L_n^2\le d\), then
\[
\eta_{n,d}=\frac d{n^2}\le \frac{d^2}{n^2L_n^2}=u_{n,d},
\]
so
\[
\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}=\min\{1,\eta_{n,d}\}
\]
and
\[
\min\{1,\eta_{n,d}\}\le \min\{1,u_{n,d}\}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
5\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
If \(d<L_n^2\), the logarithmic estimate from the previous step gives
\[
d\le 9n,
\qquad
\eta_{n,d}\le \frac9n .
\]
Hence
\[
\frac1n+\min\{1,\eta_{n,d}\}\le \frac{10}{n}.
\]
The upper-frontier term at radius zero is at least \(1/n\), so
\[
\frac1{10}\left(\frac1n+\min\{1,\eta_{n,d}\}\right)
\le
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}.
\]
The remaining upper bound at radius zero follows from the displayed monotonicity inequality.  For radius two, nonnegativity gives the lower bound
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\},
\]
while \(\min\{1,\eta_{n,d}\}\le 9/n\) gives
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
10\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
Thus, uniformly over all \(n,d\ge1\), the all-alphabet comparisons hold with
\[
c=\frac1{10},
\qquad
C=10 .
\]
The radius-two equality of attainable laws is exactly the consequence of \cref{obj:lem:scale-sanity}.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now fix \(\epsilon\in(0,1/2)\) and choose
\[
c_\epsilon=1,
\qquad
c=\frac1{10},
\qquad
C=10 .
\]
Then \(0<c_\epsilon\le1\) and \(0<c\le C\).  Suppose \(n,d\ge1\) and
\[
d\le c_\epsilon\frac{n^2}{L_n}=\frac{n^2}{L_n}.
\]
Because \(L_n\ge1\),
\[
d\le n^2,
\qquad
\frac d{n^2}\le1,
\qquad
\min\left\{1,\frac d{n^2}\right\}=\frac d{n^2}.
\]
Applying the all-alphabet comparison from the previous step and replacing the capped term by \(d/n^2\) gives
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\min\left\{u_{n,d},\frac d{n^2}\right\}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right),
\]
and
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right).
\]
The class identity is the radius-two conclusion of \cref{obj:lem:scale-sanity}.  Finally, the all-alphabet radius-two calculation gives, for every \(n,d\ge1\),
\[
\frac1n+\min\{1,\min\{u_{n,d},4+d/n^2\}\}
=
\frac1n+\min\{1,u_{n,d}\},
\]
and its converse comparison gives
\[
c\left(\frac1n+\min\{1,u_{n,d}\}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}+4\min\{1,u_{n,d}\}
\le
C\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions
\end{enumerate}
\end{proof}
