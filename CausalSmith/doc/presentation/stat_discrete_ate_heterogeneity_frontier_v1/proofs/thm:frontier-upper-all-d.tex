\begin{proof}[Proof of \cref{obj:thm:frontier-upper-all-d}]
Fix \(0<\epsilon<1/2\).

\emph{Step 1. Common calibration and envelopes.}
By \cref{obj:thm:robust-upper-construction-resolution-all-d}, choose a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), and a constant
\(\Gamma_{\epsilon}>0\).  Put
\[
C_{\epsilon}:=\max\{1,\Gamma_{\epsilon}\}.
\]
Then \(C_{\epsilon}>0\), \(\Gamma_{\epsilon}\le C_{\epsilon}\), and
\(1\le C_{\epsilon}\).  The selected handle carries the displayed calibration implication
for \(K_n\): for all integers \(n,d\ge1\),
\[
N_{\epsilon}\le n,
\qquad
d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n .
\]
For fixed integers \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), define
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2}.
\]
The construction also gives measurable \([-M,M]\)-valued estimators
\(T_n^{\mathrm{poly}}\) and \(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+\min\{1,u_{n,d}\}\right)
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.robust_upper_construction_resolution_all_d

\emph{Step 2. The selector is admissible.}
The known-radius rule in \cref{obj:def:total-estimator} is
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},&
h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},&
u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,&\text{otherwise}.
\end{cases}
\]
The first two branches are measurable and take values in \([-M,M]\) by
\cref{obj:thm:robust-upper-construction-resolution-all-d}.  The zero branch is measurable and
also takes values in \([-M,M]\) because \(M\ge1\).  Hence
\(\widehat\tau_n^{\star}\) is measurable and
\(\widehat\tau_n^{\star}(s)\in[-M,M]\) for every sample \(s\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.totalSelector

\emph{Step 3. Collision branch.}
Assume
\[
h_{n,d,\sigma}\le \min\{1,u_{n,d}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{col}}\), and
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=h_{n,d,\sigma}.
\]
Therefore, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
where the last inequality uses \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the
nonnegativity of \(M^2(1/n+h_{n,d,\sigma})\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 4. Polynomial branch.}
Assume the collision-branch condition fails and
\[
u_{n,d}<\min\{1,h_{n,d,\sigma}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{poly}}\),
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=u_{n,d}.
\]
Hence, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+u_{n,d}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
again because \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the rate factor is nonnegative. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 5. Zero branch.}
It remains to consider the branch in which
\[
h_{n,d,\sigma}>\min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These two inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then the first inequality gives
\(h_{n,d,\sigma}>u_{n,d}\), while the second gives
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\), a contradiction whether
\(h_{n,d,\sigma}\le1\) or \(h_{n,d,\sigma}>1\).  With \(1\le u_{n,d}\), the first displayed
inequality gives \(1\le h_{n,d,\sigma}\), and consequently
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=1.
\]
On this branch \(\widehat\tau_n^{\star}=0\).  Viewing the law of
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\) as a member of the unrestricted class obtained by
forgetting the radius condition, \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2.
\]
Since \(C_{\epsilon}\ge1\), \(M^2\ge0\), and \(n\ge1\),
\[
M^2
\le
C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

Combining the three branches yields, for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Together with the measurability and range conclusion in Step 2, this proves the stated
selector guarantee. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_all_d
\end{proof}
