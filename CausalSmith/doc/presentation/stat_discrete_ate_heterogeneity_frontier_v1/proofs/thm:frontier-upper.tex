\begin{proof}[Proof of \cref{obj:thm:frontier-upper}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By
\cref{obj:thm:robust-upper-construction-resolution-all-d}, choose
\(C_{\epsilon}^{0}>0\) and a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), such that
\[
N_{\epsilon}\le n,
\qquad
 d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n
\]
for all positive integers \(n,d\).  The same construction gives, for every
\(n,d\ge1\), every \(M\ge1\), and every \(0\le\sigma\le2\), measurable
\([-M,M]\)-valued estimators \(T_n^{\mathrm{poly}}\) and
\(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\sigma^2+\frac{d}{n^2}
\right\}.
\]
Set
\[
C_{\epsilon}:=\max\{1,C_{\epsilon}^{0}\}.
\]
Then \(C_{\epsilon}>0\) and \(C_{\epsilon}\ge C_{\epsilon}^{0}\), so both
displayed risk bounds hold with \(C_{\epsilon}\) in place of
\(C_{\epsilon}^{0}\). % lean: robust_upper_construction_resolution_all_d

\item For fixed admissible \(n,d,M,\sigma\), put
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2},
\qquad
r_{n,d,\sigma}:=\frac1n+\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}.
\]
The known-radius selector of \cref{obj:def:total-estimator} is the three-branch
estimator
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},
& h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\end{cases}
\]
The first two branches are elements of \(\mathcal T_{n,M}\) by the construction
above.  The zero branch is measurable and takes values in \([-M,M]\) because
\(M\ge1\).  Hence \(\widehat\tau_n^{\star}\in\mathcal T_{n,M}\). % lean: totalSelector

\item Fix \(P\in\mathcal P_{d,\epsilon,M,\sigma}\).  On the collision branch,
\(h_{n,d,\sigma}\le \min\{1,u_{n,d}\}\), so
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=h_{n,d,\sigma}.
\]
Therefore
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+h_{n,d,\sigma}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
On the polynomial branch, \(u_{n,d}<\min\{1,h_{n,d,\sigma}\}\), so
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=u_{n,d}.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+u_{n,d}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
It remains to consider the zero branch.  There
\[
h_{n,d,\sigma}> \min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then
\(h_{n,d,\sigma}>u_{n,d}\), while
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\); if \(h_{n,d,\sigma}\le1\) this gives
\(h_{n,d,\sigma}\le u_{n,d}<h_{n,d,\sigma}\), and if \(h_{n,d,\sigma}>1\) it
gives \(1\le u_{n,d}<1\).  Hence \(1\le u_{n,d}\).  The first zero-branch
inequality then gives \(1<h_{n,d,\sigma}\), so \(1\le h_{n,d,\sigma}\), and
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=1.
\]
Viewing the same law as a member of the unrestricted-radius class, the scale
bound in \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Since the selected estimator is identically zero on this branch,
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2
\le C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma},
\]
where the penultimate inequality uses \(C_{\epsilon}\ge1\) and \(n\ge1\).
Together the three branches prove the asserted risk bound for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\). % lean: frontier_upper_bound_all_d

\item Put \(c_{\epsilon}:=1\).  Then \(c_{\epsilon}>0\).  The preceding
all-alphabet bound applies to every positive \(n,d\), so it applies in
particular under the restricted-dimensional hypothesis
\[
d\le c_{\epsilon}\frac{n^2}{\log(en)}.
\]
The constants \(C_{\epsilon},c_{\epsilon}\) and the same polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\) therefore satisfy all claims of the theorem.
% lean: frontier_upper
\end{enumerate}
\end{proof}
