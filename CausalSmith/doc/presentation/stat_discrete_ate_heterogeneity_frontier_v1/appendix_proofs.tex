\begin{proof}[Proof of \cref{obj:prop:ate-identification}]
\begin{enumerate}
\item
For each cell \(k=1,\ldots,d\), set
\[
p_k=P(X=k);
\]
when \(d=0\), all sums below are empty.  When \(p_k>0\), let \(\pi_k=P(A=1\mid X=k)\).  The finite-cell law carries an arm-cell outcome law \(\nu_{ak}\) for every arm \(a\in\{0,1\}\) and every cell, including cells with \(p_k=0\), and its totalized mean is
\[
\mu_{ak}:=\int y\,d\nu_{ak}(y).
\]
For every supported cell and every Borel set \(\mathfrak A\subseteq\mathbb R\), the observed arm-cell factorization is
\[
P_O(X=k,A=a,Y\in\mathfrak A)
=
p_k
\begin{cases}
\pi_k, & a=1,\\
1-\pi_k, & a=0
\end{cases}
\nu_{ak}(\mathfrak A).
\]
If \(p_k>0\), \cref{obj:ass:overlap} and \(\epsilon>0\) make the displayed scalar positive for both arms.  Hence, on supported cells, \(\nu_{ak}\) is the observed conditional law of \(Y\) given \((A=a,X=k)\), and \(\mu_{ak}\) is the corresponding observed conditional mean.  If \(p_k=0\), then \(P_O(X=k)=0\); the totalized law \(\nu_{ak}\) and mean \(\mu_{ak}\) enter the finite formula below only through terms multiplied by \(p_k\). % lean: observed_arm_outcome_measure

\item
Fix \(a\in\{0,1\}\) and a supported cell \(k\), so \(p_k>0\).  By \cref{obj:ass:consistency}, on \(\{X=k,A=a\}\),
\[
Y(a)=Y
\quad\text{almost surely}.
\]
Together with the observed arm-cell factorization, this gives, for every Borel \(\mathfrak A\),
\[
P(X=k,A=a,Y(a)\in\mathfrak A)
=
p_k
\begin{cases}
\pi_k, & a=1,\\
1-\pi_k, & a=0
\end{cases}
\nu_{ak}(\mathfrak A).
\]
Applying \cref{obj:ass:conditional-exchangeability} in cell \(k\) to the potential-outcome event \(\{Y(a)\in\mathfrak A\}\) yields
\[
P(X=k,A=a,Y(a)\in\mathfrak A)\,P(X=k)
=
P(X=k,A=a)\,P(X=k,Y(a)\in\mathfrak A).
\]
Here \(P(X=k)=p_k\), and \(P(X=k,A=a)\) is \(p_k\pi_k\) when \(a=1\) and \(p_k(1-\pi_k)\) when \(a=0\).  The same overlap argument makes this arm-cell probability positive, so cancellation gives
\[
P(X=k,Y(a)\in\mathfrak A)=p_k\nu_{ak}(\mathfrak A).
\]
Equivalently,
\[
\mathcal L_P(Y(a);\,X=k)=p_k\,\nu_{ak},
\]
where the left side is the pushforward of the full-data law restricted to \(\{X=k\}\). % lean: full_cell_potential_measure

\item
The preceding restricted-law identity supplies the full-data integrability.  On supported cells, \cref{obj:ass:second-central-moment} gives integrability of \(y\mapsto (y-\mu_{ak})^2\) under \(\nu_{ak}\).  Thus \(y-\mu_{ak}\) is square-integrable, hence integrable, and adding the finite constant \(\mu_{ak}\) gives first-moment integrability of \(y\mapsto y\) under \(\nu_{ak}\).  The restricted-law identity transfers this integrability to \(Y(a)\) on each supported cell.  If \(p_k=0\), the full-data mass of \(\{X=k\}\) is \(0\), so the corresponding restricted measure is the zero measure.  Since there are finitely many cells,
\[
Y(1)\in L^1(P),\qquad Y(0)\in L^1(P).
\]
% lean: full_potential_integrable

\item
For every \(a\in\{0,1\}\) and every cell \(k\),
\[
\int_{\{X=k\}}Y(a)\,dP
=
p_k\int y\,d\nu_{ak}(y)
=
p_k\mu_{ak}.
\]
On supported cells this is the integral form of the restricted-law identity in the second step and the defining integral for \(\mu_{ak}\).  On zero-mass cells, the restricted measure is zero and \(p_k=0\), so both sides vanish. % lean: full_potential_cell_integral

\item
Apply the finite-fiber decomposition of the integral to the measurable finite-valued map \(X\).  Using the integrability just established,
\[
\int Y(1)\,dP
=
\sum_{k=1}^{d}\int_{\{X=k\}}Y(1)\,dP,
\qquad
\int Y(0)\,dP
=
\sum_{k=1}^{d}\int_{\{X=k\}}Y(0)\,dP.
\]
Therefore
\[
\begin{aligned}
\int\{Y(1)-Y(0)\}\,dP
&=
\sum_{k=1}^{d}
\left\{
\int_{\{X=k\}}Y(1)\,dP
-
\int_{\{X=k\}}Y(0)\,dP
\right\}\\
&=
\sum_{k=1}^{d}p_k(\mu_{1k}-\mu_{0k}).
\end{aligned}
\]
By \cref{obj:def:ate-functional},
\[
\tau(P)=\sum_{k=1}^{d}p_k(\mu_{1k}-\mu_{0k}),
\]
so
\[
\tau(P)
=
\int\{Y(1)-Y(0)\}\,dP.
\]
On supported cells, the first step identifies \(\mu_{ak}\) with \(E_P[Y\mid A=a,X=k]\); cells with \(p_k=0\) contribute \(0\) to the totalized sum.  Thus, when conditional expectations on zero-mass arm-cell events are chosen to equal the totalized means \(\mu_{ak}\), the same equality is written as
\[
\tau(P)
=
\int\{Y(1)-Y(0)\}\,dP
=
\sum_{k=1}^{d}p_k
\left\{
E_P[Y\mid A=1,X=k]-E_P[Y\mid A=0,X=k]
\right\}.
\]
The observed margin determines this finite-cell formula: \(p_k=P_O(X=k)\), and on every supported cell \cref{obj:ass:overlap} gives positive observed mass in both arms, so the observed conditional laws determine the two arm-cell means.  Zero-mass cells have weight \(p_k=0\). % lean: ate_identification
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:frontier-upper}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By
\cref{obj:thm:robust-upper-construction-resolution-all-d}, choose
\(C_{\epsilon}^{0}>0\) and a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), such that
\[
N_{\epsilon}\le n,
\qquad
 d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n
\]
for all positive integers \(n,d\).  The same construction gives, for every
\(n,d\ge1\), every \(M\ge1\), and every \(0\le\sigma\le2\), measurable
\([-M,M]\)-valued estimators \(T_n^{\mathrm{poly}}\) and
\(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\sigma^2+\frac{d}{n^2}
\right\}.
\]
Set
\[
C_{\epsilon}:=\max\{1,C_{\epsilon}^{0}\}.
\]
Then \(C_{\epsilon}>0\) and \(C_{\epsilon}\ge C_{\epsilon}^{0}\), so both
displayed risk bounds hold with \(C_{\epsilon}\) in place of
\(C_{\epsilon}^{0}\). % lean: robust_upper_construction_resolution_all_d

\item For fixed admissible \(n,d,M,\sigma\), put
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2},
\qquad
r_{n,d,\sigma}:=\frac1n+\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}.
\]
The known-radius selector of \cref{obj:def:total-estimator} is the three-branch
estimator
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},
& h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\end{cases}
\]
The first two branches are elements of \(\mathcal T_{n,M}\) by the construction
above.  The zero branch is measurable and takes values in \([-M,M]\) because
\(M\ge1\).  Hence \(\widehat\tau_n^{\star}\in\mathcal T_{n,M}\). % lean: totalSelector

\item Fix \(P\in\mathcal P_{d,\epsilon,M,\sigma}\).  On the collision branch,
\(h_{n,d,\sigma}\le \min\{1,u_{n,d}\}\), so
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=h_{n,d,\sigma}.
\]
Therefore
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+h_{n,d,\sigma}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
On the polynomial branch, \(u_{n,d}<\min\{1,h_{n,d,\sigma}\}\), so
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=u_{n,d}.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+u_{n,d}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
It remains to consider the zero branch.  There
\[
h_{n,d,\sigma}> \min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then
\(h_{n,d,\sigma}>u_{n,d}\), while
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\); if \(h_{n,d,\sigma}\le1\) this gives
\(h_{n,d,\sigma}\le u_{n,d}<h_{n,d,\sigma}\), and if \(h_{n,d,\sigma}>1\) it
gives \(1\le u_{n,d}<1\).  Hence \(1\le u_{n,d}\).  The first zero-branch
inequality then gives \(1<h_{n,d,\sigma}\), so \(1\le h_{n,d,\sigma}\), and
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=1.
\]
Viewing the same law as a member of the unrestricted-radius class, the scale
bound in \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Since the selected estimator is identically zero on this branch,
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2
\le C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma},
\]
where the penultimate inequality uses \(C_{\epsilon}\ge1\) and \(n\ge1\).
Together the three branches prove the asserted risk bound for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\). % lean: frontier_upper_bound_all_d

\item Put \(c_{\epsilon}:=1\).  Then \(c_{\epsilon}>0\).  The preceding
all-alphabet bound applies to every positive \(n,d\), so it applies in
particular under the restricted-dimensional hypothesis
\[
d\le c_{\epsilon}\frac{n^2}{\log(en)}.
\]
The constants \(C_{\epsilon},c_{\epsilon}\) and the same polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\) therefore satisfy all claims of the theorem.
% lean: frontier_upper
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions}]
\begin{enumerate}
\item First record the logarithmic estimate used below.  If \(n\ge1\), then
\[
L_n=\log(en)=1+\log n .
\]
The standard bound \(\log n\le n^{1/2}/(1/2)=2\sqrt n\), together with
\(\log n\ge0\) and \(\sqrt n\ge1\), gives
\[
0\le L_n\le 1+2\sqrt n\le 3\sqrt n .
\]
Squaring yields
\[
L_n^2\le 9n .
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item We prove the all-alphabet comparison that supplies the constants.  For \(n,d\ge1\), put
\[
\eta_{n,d}:=\frac{d}{n^2},
\qquad
u_{n,d}:=\frac{d^2}{n^2L_n^2}.
\]
Since \(L_n\ge1\), both \(\eta_{n,d}\) and \(u_{n,d}\) are nonnegative.  At radius two the collision term is
\[
2^2+\eta_{n,d}=4+\eta_{n,d}\ge1,
\]
and therefore
\[
\frac1n+\min\{1,\min\{u_{n,d},4+\eta_{n,d}\}\}
=
\frac1n+\min\{1,u_{n,d}\}.
\]
At radius zero,
\[
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}
\le
\frac1n+\min\{1,\eta_{n,d}\},
\]
and the capped converse expression at radius zero is exactly the right-hand side.

It remains to compare the radius-two converse term
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\]
with
\[
\frac1n+\min\{1,u_{n,d}\}.
\]
If \(L_n^2\le d\), then
\[
\eta_{n,d}=\frac d{n^2}\le \frac{d^2}{n^2L_n^2}=u_{n,d},
\]
so
\[
\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}=\min\{1,\eta_{n,d}\}
\]
and
\[
\min\{1,\eta_{n,d}\}\le \min\{1,u_{n,d}\}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
5\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
If \(d<L_n^2\), the logarithmic estimate from the previous step gives
\[
d\le 9n,
\qquad
\eta_{n,d}\le \frac9n .
\]
Hence
\[
\frac1n+\min\{1,\eta_{n,d}\}\le \frac{10}{n}.
\]
The upper-frontier term at radius zero is at least \(1/n\), so
\[
\frac1{10}\left(\frac1n+\min\{1,\eta_{n,d}\}\right)
\le
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}.
\]
The remaining upper bound at radius zero follows from the displayed monotonicity inequality.  For radius two, nonnegativity gives the lower bound
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\},
\]
while \(\min\{1,\eta_{n,d}\}\le 9/n\) gives
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
10\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
Thus, uniformly over all \(n,d\ge1\), the all-alphabet comparisons hold with
\[
c=\frac1{10},
\qquad
C=10 .
\]
The radius-two equality of attainable laws is exactly the consequence of \cref{obj:lem:scale-sanity}.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now fix \(\epsilon\in(0,1/2)\) and choose
\[
c_\epsilon=1,
\qquad
c=\frac1{10},
\qquad
C=10 .
\]
Then \(0<c_\epsilon\le1\) and \(0<c\le C\).  Suppose \(n,d\ge1\) and
\[
d\le c_\epsilon\frac{n^2}{L_n}=\frac{n^2}{L_n}.
\]
Because \(L_n\ge1\),
\[
d\le n^2,
\qquad
\frac d{n^2}\le1,
\qquad
\min\left\{1,\frac d{n^2}\right\}=\frac d{n^2}.
\]
Applying the all-alphabet comparison from the previous step and replacing the capped term by \(d/n^2\) gives
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\min\left\{u_{n,d},\frac d{n^2}\right\}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right),
\]
and
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right).
\]
The class identity is the radius-two conclusion of \cref{obj:lem:scale-sanity}.  Finally, the all-alphabet radius-two calculation gives, for every \(n,d\ge1\),
\[
\frac1n+\min\{1,\min\{u_{n,d},4+d/n^2\}\}
=
\frac1n+\min\{1,u_{n,d}\},
\]
and its converse comparison gives
\[
c\left(\frac1n+\min\{1,u_{n,d}\}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}+4\min\{1,u_{n,d}\}
\le
C\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:radius-channel-converse}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  Apply \cref{obj:thm:radius-channel-converse-all-d} to obtain constants \(\bar c_{\epsilon}\) and \(\bar b_{\epsilon}^{\mathrm{rad}}\) such that
\[
0<\bar c_{\epsilon}\le 1,
\qquad
0<\bar b_{\epsilon}^{\mathrm{rad}},
\]
and, for every admissible \(n,d,M,\sigma\),

\bar c_{\epsilon}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
+\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Set \(c_{\epsilon}:=\bar c_{\epsilon}\) and
\(b_{\epsilon}^{\mathrm{rad}}:=\bar b_{\epsilon}^{\mathrm{rad}}\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

L:=\log(en).
\]
Since \(0<n\), we have \(n\ge1\), hence
\[
L=\log(e n)=1+\log n\ge1.
\]
The restricted range and \(c_{\epsilon}\le1\) give
\[
d\le c_{\epsilon}\frac{n^2}{L}
\le c_{\epsilon}n^2
\le n^2 .
\]
Because \(n>0\), division by \(n^2\) yields
\[
\frac d{n^2}\le1,
\qquad\text{so}\qquad
\min\left(1,\frac d{n^2}\right)=\frac d{n^2}.
\]
Substituting this identity into the all-alphabet bound displayed above gives
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\ge
c_{\epsilon}M^2
\left\{
\frac1n+\frac d{n^2}
+\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse

\item The radial certificate supplied by \cref{obj:thm:radius-channel-converse-all-d} uses the same constants \(c_{\epsilon}\) and \(b_{\epsilon}^{\mathrm{rad}}\).  Unpacking the certificate, there are an integer cap \(d_{\mathrm{rad}}\), a binary control-zero source class \(\mathcal S_{\mathrm{rad}}\), a padding map into the \(d\)-alphabet, an embedding \(\Psi\) that sends each source member into \(\mathcal P_{d,\epsilon,M,\sigma}\), and binary full-observation couplings such that the channel acts by
\[
B'(a)\mid B(a)
\sim
\operatorname{Bernoulli}\!\left\{
\frac12+\frac{\sigma}{2}\left(B(a)-\frac12\right)
\right\},
\qquad
Y(a):=M\{B'(a)-1/2\}.
\]
The source target in this certificate is the total treated-arm weighted functional
\[
\psi(P)
:=
\sum_{k=1}^{d_{\mathrm{rad}}}
p_k(P)\,\mu_{1k}^{\mathrm{bin}}(P),
\]
where \(\mu_{1k}^{\mathrm{bin}}(P)\) is the treated-arm binary mean coordinate carried by the source law in cell \(k\), defined for every source cell including zero-mass cells.  Thus, for all \(P_0,P_1\in\mathcal S_{\mathrm{rad}}\), the same certificate gives
\[
\tau(\Psi(P_1))-\tau(\Psi(P_0))
=
M\frac{\sigma}{2}
\left\{
\psi(P_1)-\psi(P_0)
\right\}.
\]
It also gives the estimator-wise lower bound: for every
\(\widehat\tau\in\mathcal T_{n,M}\), the class of total measurable \([-M,M]\)-valued estimators from \cref{obj:def:minimax-risk}, there is \(P\in\mathcal S_{\mathrm{rad}}\) such that
\[
c_{\epsilon}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\mathbb E_{\Psi(P)}
\left[
\{\widehat\tau-\tau(\Psi(P))\}^2
\right].
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:robust-upper-construction-resolution}]
Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).

\begin{enumerate}
\item First form the all-alphabet package.  By
\cref{obj:lem:continuous-ratio-polynomial-upper-all-d}, there are
\(C_{\mathrm{poly},\epsilon}>0\) and a polynomial handle
\[
\mathfrak h_\epsilon=(N_{\mathrm{cal},\epsilon},\rho_{\mathrm{cal},\epsilon}),
\qquad \rho_{\mathrm{cal},\epsilon}>0,
\]
such that, for all integers \(n,d\ge1\),
\[
N_{\mathrm{cal},\epsilon}\le n,\qquad
d\le \rho_{\mathrm{cal},\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n,
\]
and, for every \(M\ge1\), \(0\le\sigma\le2\), and
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
C_{\mathrm{poly},\epsilon}M^2
\left\{
\frac1n+
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\]
The same cited construction gives measurability of \(T_n^{\mathrm{poly}}\) and
\[
T_n^{\mathrm{poly}}(s)\in[-M,M]
\qquad\text{for every }s\in\mathcal O_d^n .
\]

By \cref{obj:lem:continuous-occupancy-collision-upper-all-d}, there is
\(C_{\mathrm{col},\epsilon}>0\) such that, for every \(n,d\ge1\),
\(M\ge1\), \(0\le\sigma\le2\), and
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
C_{\mathrm{col},\epsilon}M^2
\left(
\frac1n+\sigma^2+\frac{d}{n^2}
\right).
\]
For the admissibility of the collision estimator, write its unclipped inner
statistic as the total sample function
\[
\Gamma_{\mathrm{col}}(s)
:=
\begin{cases}
\displaystyle
\frac{1}{R_n(s)}
\sum_{k=1}^d
N_k(s)I_k(s)
\left[
\mathbf 1\{N_{1k}(s)>0\}\frac{S_{1k}(s)}{N_{1k}(s)\vee 1}
-
\mathbf 1\{N_{0k}(s)>0\}\frac{S_{0k}(s)}{N_{0k}(s)\vee 1}
\right],
& R_n(s)>0,\\[1.2em]
0,& R_n(s)=0.
\end{cases}
\]
Here \(m\vee1\) denotes \(\max\{m,1\}\).  The displayed arm factors are total
functions of the sample and agree with the arm-cell means of
\cref{obj:def:collision-handle} on positive arm-cell counts; multiplication by
\(I_k\) selects exactly the cells where both arm-cell counts are positive.  Hence
\cref{obj:def:collision-handle} gives
\[
T_n^{\mathrm{col}}(s)=\max\{-M,\min(M,\Gamma_{\mathrm{col}}(s))\}.
\]
The counts, indicators, outcome totals, finite sums, totalized ratios, and the

\begin{proof}[Proof of \cref{obj:thm:frontier-upper-all-d}]
Fix \(0<\epsilon<1/2\).

\emph{Step 1. Common calibration and envelopes.}
By \cref{obj:thm:robust-upper-construction-resolution-all-d}, choose a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), and a constant
\(\Gamma_{\epsilon}>0\).  Put
\[
C_{\epsilon}:=\max\{1,\Gamma_{\epsilon}\}.
\]
Then \(C_{\epsilon}>0\), \(\Gamma_{\epsilon}\le C_{\epsilon}\), and
\(1\le C_{\epsilon}\).  The selected handle carries the displayed calibration implication
for \(K_n\): for all integers \(n,d\ge1\),
\[
N_{\epsilon}\le n,
\qquad
d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n .
\]
For fixed integers \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), define
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2}.
\]
The construction also gives measurable \([-M,M]\)-valued estimators
\(T_n^{\mathrm{poly}}\) and \(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+\min\{1,u_{n,d}\}\right)
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.robust_upper_construction_resolution_all_d

\emph{Step 2. The selector is admissible.}
The known-radius rule in \cref{obj:def:total-estimator} is
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},&
h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},&
u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,&\text{otherwise}.
\end{cases}
\]
The first two branches are measurable and take values in \([-M,M]\) by
\cref{obj:thm:robust-upper-construction-resolution-all-d}.  The zero branch is measurable and
also takes values in \([-M,M]\) because \(M\ge1\).  Hence
\(\widehat\tau_n^{\star}\) is measurable and
\(\widehat\tau_n^{\star}(s)\in[-M,M]\) for every sample \(s\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.totalSelector

\emph{Step 3. Collision branch.}
Assume
\[
h_{n,d,\sigma}\le \min\{1,u_{n,d}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{col}}\), and
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=h_{n,d,\sigma}.
\]
Therefore, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
where the last inequality uses \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the
nonnegativity of \(M^2(1/n+h_{n,d,\sigma})\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 4. Polynomial branch.}
Assume the collision-branch condition fails and
\[
u_{n,d}<\min\{1,h_{n,d,\sigma}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{poly}}\),
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=u_{n,d}.
\]
Hence, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+u_{n,d}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
again because \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the rate factor is nonnegative. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 5. Zero branch.}
It remains to consider the branch in which
\[
h_{n,d,\sigma}>\min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These two inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then the first inequality gives
\(h_{n,d,\sigma}>u_{n,d}\), while the second gives
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\), a contradiction whether
\(h_{n,d,\sigma}\le1\) or \(h_{n,d,\sigma}>1\).  With \(1\le u_{n,d}\), the first displayed
inequality gives \(1\le h_{n,d,\sigma}\), and consequently
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=1.
\]
On this branch \(\widehat\tau_n^{\star}=0\).  Viewing the law of
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\) as a member of the unrestricted class obtained by
forgetting the radius condition, \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2.
\]
Since \(C_{\epsilon}\ge1\), \(M^2\ge0\), and \(n\ge1\),
\[
M^2
\le
C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

Combining the three branches yields, for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Together with the measurability and range conclusion in Step 2, this proves the stated
selector guarantee. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_all_d
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions-all-d}]
Take \(c=1/10\) and \(C=10\).  Fix integers \(n,d\ge1\), and write
\[
\Lambda_n:=\log(en),\qquad
u_{n,d}:=\frac{d^2}{n^2\Lambda_n^2},\qquad
v_{n,d}:=\frac d{n^2}.
\]
Also set
\[
\theta^{\mathrm{ex}}_{n,d}:=\frac1n+\min(1,v_{n,d}),
\qquad
\theta^{\mathrm{poly}}_{n,d}:=\frac1n+\min(1,u_{n,d}).
\]
The quantities \(1/n\), \(u_{n,d}\), and \(v_{n,d}\) are nonnegative.

\begin{enumerate}
\item Since \(n\ge1\),
\[
\Lambda_n
=\log(en)=1+\log n\ge1,
\]
and hence \(\Lambda_n^2>0\).  At radius two the collision term is
\[
2^2+v_{n,d}=4+v_{n,d}\ge1.
\]
Therefore
\[
\frac1n+\min\{1,u_{n,d},4+v_{n,d}\}
=\frac1n+\min(1,u_{n,d})
=\theta^{\mathrm{poly}}_{n,d}.
\]
This proves the endpoint identity, and also gives the two-sided comparison of the
radius-two upper frontier expression with \(\theta^{\mathrm{poly}}_{n,d}\), because
\(\theta^{\mathrm{poly}}_{n,d}\ge0\) and \(1/10\le1\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item We use the elementary logarithmic estimate
\[
\Lambda_n^2\le 9n.
\]
Indeed, \(\log n\le 2\sqrt n\), while \(\sqrt n\ge1\) and \(\log n\ge0\).  Thus
\[
\Lambda_n=1+\log n\le 1+2\sqrt n\le3\sqrt n,
\]
and squaring gives \(\Lambda_n^2\le9n\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item First consider the case \(\Lambda_n^2\le d\).  Since \(d>0\),
\[
v_{n,d}=\frac d{n^2}
\le
\frac{d^2}{n^2\Lambda_n^2}
=u_{n,d}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
=
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression is also exactly
\[
\frac1n+\min(1,v_{n,d})=\theta^{\mathrm{ex}}_{n,d}.
\]
For the radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and, using \(\min(1,v_{n,d})\le\min(1,u_{n,d})\),
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac1n+5\min(1,u_{n,d})
\le
5\theta^{\mathrm{poly}}_{n,d}.
\]
All required comparisons in this case follow from \(1/10\le1\le5\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now consider the case \(d<\Lambda_n^2\).  By the logarithmic estimate,
\[
d\le9n,
\qquad\text{so}\qquad
v_{n,d}\le\frac9n.
\]
The radius-zero upper expression satisfies
\[
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d},
\]
and
\[
\theta^{\mathrm{ex}}_{n,d}
=\frac1n+\min(1,v_{n,d})
\le
\frac1n+\frac9n
=
\frac{10}{n}.
\]
Hence
\[
\frac1{10}\theta^{\mathrm{ex}}_{n,d}
\le
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\theta^{\mathrm{ex}}_{n,d}
\le
10\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression again equals \(\theta^{\mathrm{ex}}_{n,d}\).  For the
radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac{10}{n}+4\min(1,u_{n,d})
\le
10\theta^{\mathrm{poly}}_{n,d}.
\]
Together with the radius-two identity from the first step, this proves the stated
two-sided comparisons in the second case.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item It remains to identify the two radius-two classes.  By \cref{obj:lem:scale-sanity}, for
every \(d\), every \(\epsilon\in\mathbb R\), and every \(M\in\mathbb R\), each law in
\(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\) is represented by some member of
\(\mathcal P_{d,\epsilon,M,2}\) with the same law, and conversely each law in
\(\mathcal P_{d,\epsilon,M,2}\) is represented by some member of
\(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\) with the same law.  Thus the represented laws are
exactly the same at radius two.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.scale_sanity
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:fixed-interior-tightness-and-shrinking-radius-gap-all-d}]
Let
\[
\omega_{n,d,\sigma}
:=
\frac1n+\min\left\{1,\frac d{n^2}\right\}
+\sigma^2\min\{1,u_{n,d}\}.
\]
Thus \(\omega_{n,d,\sigma}\) is the capped converse benchmark appearing in the
residual-ratio display.  We use the all-alphabet two-sided bracket
\[
c_0M^2\omega_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_0M^2r_{n,d,\sigma},
\]
valid for \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), as supplied by
\cref{obj:thm:two-sided-minimax-bracket-all-d}.

\begin{enumerate}
\item Suppose \(L_n^2<d<nL_n\), and put
\[
\eta_{n,d}:=\frac d{n^2},
\qquad
\zeta_\sigma:=\sigma^2 .
\]
Then \(n\ge1\), \(L_n>0\), and \(d>0\).  The two strict inequalities give
\[
\eta_{n,d}\le u_{n,d},
\qquad
u_{n,d}<1.
\]
Hence
\[
q_{n,d,\sigma}
=
\frac1n+\min\{u_{n,d},\zeta_\sigma+\eta_{n,d}\}.
\]
For nonnegative \(\eta_{n,d},u_{n,d},\zeta_\sigma\) with
\(\eta_{n,d}\le u_{n,d}\),
\[
\frac12\left\{
\frac1n+\eta_{n,d}+\min(u_{n,d},\zeta_\sigma)
\right\}
\le
\frac1n+\min\{u_{n,d},\zeta_\sigma+\eta_{n,d}\}
\le
\frac1n+\eta_{n,d}+\min(u_{n,d},\zeta_\sigma).
\]
The upper bound follows from the two bounds by \(u_{n,d}\) and by
\(\zeta_\sigma+\eta_{n,d}\).  For the lower bound, if
\(u_{n,d}\le\zeta_\sigma+\eta_{n,d}\), the right-hand minimum is \(u_{n,d}\)
and \(\eta_{n,d}\le u_{n,d}\); otherwise the right-hand minimum is
\(\zeta_\sigma+\eta_{n,d}\), and then
\(\min(u_{n,d},\zeta_\sigma)=\zeta_\sigma\).  Substituting
\(b_{n,d}=n^{-1}+\eta_{n,d}\) gives the triangular display.  The identity
\[
\ell_{n,d,\sigma}=b_{n,d}+\sigma^2u_{n,d}
\]
is the definition of \(\ell_{n,d,\sigma}\). % lean: selectorBenchmark_triangular

\item Fix \(\underline\sigma\in(0,2]\) and define
\[
\Gamma_{\underline\sigma}:=\max\{1,\underline\sigma^{-2}\}.
\]
If \(\underline\sigma\le\sigma\le2\), then
\[
r_{n,d,\sigma}
\le
\frac1n+\min\{1,u_{n,d}\}
\le
\Gamma_{\underline\sigma}
\left[
\frac1n+\sigma^2\min\{1,u_{n,d}\}
\right]
\le
\Gamma_{\underline\sigma}\omega_{n,d,\sigma}.
\]
Combining this comparison with the two-sided bracket gives
\[
\frac{c_0}{\Gamma_{\underline\sigma}}M^2r_{n,d,\sigma}
\le
c_0M^2\omega_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_0M^2r_{n,d,\sigma}.
\]
Thus the fixed-interior-radius assertion holds with
\(c=c_0/\Gamma_{\underline\sigma}\) and \(C=C_0\). % lean: fixed_radius_minimax_equivalence

\item Fix \(K\ge0\).  We prove
\[
r_{n,d,\sigma}\le(1+K)\omega_{n,d,\sigma}
\]
under each matching-elbow alternative.  If \(1\le u_{n,d}\), then
\[
r_{n,d,\sigma}
\le
\frac1n+\min\left\{1,\sigma^2+\frac d{n^2}\right\}.
\]
When \(d/n^2\le1\), the condition \(1\le u_{n,d}\) gives
\[
\omega_{n,d,\sigma}
=
\frac1n+\frac d{n^2}+\sigma^2,
\]
and the preceding display is bounded by \(\omega_{n,d,\sigma}\).  When
\(d/n^2>1\), the same condition gives
\[
\omega_{n,d,\sigma}
=
\frac1n+1+\sigma^2,
\]
and the preceding display is again bounded by \(\omega_{n,d,\sigma}\).
Thus \(r_{n,d,\sigma}\le\omega_{n,d,\sigma}\) in the first matching-elbow
case.

If \(u_{n,d}\le Kb_{n,d}\) and \(d/n^2\le1\), then
\[
r_{n,d,\sigma}
\le
\frac1n+u_{n,d}
\le
(1+K)b_{n,d}
\le
(1+K)\omega_{n,d,\sigma};
\]
if \(d/n^2>1\), then
\[
r_{n,d,\sigma}\le\frac1n+1\le\omega_{n,d,\sigma}
\le(1+K)\omega_{n,d,\sigma}.
\]
If \(\sigma^2\le Kb_{n,d}\) and \(d/n^2\le1\), then
\[
r_{n,d,\sigma}
\le
b_{n,d}+\sigma^2
\le
(1+K)b_{n,d}
\le
(1+K)\omega_{n,d,\sigma};
\]
the case \(d/n^2>1\) is again covered by
\(r_{n,d,\sigma}\le n^{-1}+1\le\omega_{n,d,\sigma}\).  The two-sided bracket
therefore yields
\[
\frac{c_0}{1+K}M^2r_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_0M^2r_{n,d,\sigma}.
\] % lean: matching_elbow_minimax_equivalence

\item For \(n\ge1\), \(L_n>0\).  Multiplying the inequality
\(u_{n,d}\le b_{n,d}\) by \(n^2L_n^2\) gives
\[
u_{n,d}\le b_{n,d}
\quad\Longleftrightarrow\quad
d^2
\le
\left(\frac1n+\frac d{n^2}\right)n^2L_n^2
=
nL_n^2+dL_n^2.
\] % lean: polynomialComponent_le_baseRate_iff

\item If \(d\le\sqrt n\,L_n\), then \(d^2\le nL_n^2\), and hence
\(d^2\le nL_n^2+dL_n^2\).  If \(d\le L_n^2\), then
\(d^2\le dL_n^2\le nL_n^2+dL_n^2\).  The exact algebraic equivalence above
gives \(u_{n,d}\le b_{n,d}\) in both cases. % lean: polynomialComponent_le_baseRate_of_le_sqrt; polynomialComponent_le_baseRate_of_le_log_sq

\item Let \(d_n\) and \(\sigma_n\) satisfy the admissibility hypothesis and
suppose
\[
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}\to\infty .
\]
For every fixed \(K\ge0\), divergence gives an eventual lower bound
\[
1+K<
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}.
\]
Intersect this eventual set with admissibility and with \(n>0\); then
\(d_n\ge1\), \(0\le\sigma_n\le2\), and \(\omega_{n,d_n,\sigma_n}>0\).  If one
of the matching-elbow alternatives
\[
1\le u_{n,d_n},
\qquad
u_{n,d_n}\le Kb_{n,d_n},
\qquad
\sigma_n^2\le Kb_{n,d_n}
\]
holds at such an index, the matching-elbow comparison above gives
\[
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}\le1+K,
\]
contradicting the lower bound.  Hence, for each fixed \(K\ge0\), all three
alternatives are eventually false.  Taking \(K=a^{-1}\) for any \(a>0\) gives
\[
\frac{b_{n,d_n}}{u_{n,d_n}}\to0,
\qquad
\frac{b_{n,d_n}}{\sigma_n^2}\to0.
\]
Next fix \(a>0\), set \(\rho_a=\sqrt a\), and set
\(\Gamma_a=\max\{1,\rho_a^{-2}\}\).  Divergence gives eventually
\[
\Gamma_a<
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}.
\]
At any sufficiently large index with \(\sigma_n^2\ge a\), admissibility gives
\(\rho_a\le\sigma_n\), and the fixed-radius comparison above gives
\[
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}\le\Gamma_a,
\]
a contradiction.  Therefore \(\sigma_n^2\to0\).

It remains to show \(u_{n,d_n}\to0\).  The \(K=0\) avoidance gives
\(u_{n,d_n}<1\) eventually.  For a target \(a\ge1\) this already gives
\(u_{n,d_n}<a\) eventually.  For \(0<a<1\), take \(n\) so large that
\[
\frac{b_{n,d_n}}{u_{n,d_n}}<\frac12,
\qquad
\sigma_n^2<\frac a2,
\qquad
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}>a^{-1},
\qquad
u_{n,d_n}<1.
\]
If \(u_{n,d_n}\ge a\), then
\[
\frac{d_n}{n^2}\le b_{n,d_n}<\frac12u_{n,d_n},
\qquad
\sigma_n^2<\frac12u_{n,d_n},
\]
so \(\sigma_n^2+d_n/n^2<u_{n,d_n}\) and \(d_n/n^2\le1\).  Therefore
\[
r_{n,d_n,\sigma_n}=b_{n,d_n}+\sigma_n^2,
\qquad
\omega_{n,d_n,\sigma_n}
=
b_{n,d_n}+\sigma_n^2u_{n,d_n}.
\]
Since \(u_{n,d_n}\ge a\),
\[
\frac{r_{n,d_n,\sigma_n}}{\omega_{n,d_n,\sigma_n}}
=
\frac{b_{n,d_n}+\sigma_n^2}
{b_{n,d_n}+\sigma_n^2u_{n,d_n}}
\le a^{-1},
\]
contradicting the chosen lower bound.  Thus \(u_{n,d_n}\to0\), and the
residual-wedge implication is proved. % lean: benchmark_divergence_implies_residual_wedge

\item For the diagonal sequence \(d_n=n\) and
\(\sigma_n=L_n^{-1/2}\), eventually \(n\ge1\), \(L_n\ge1\), and
\(0\le\sigma_n\le1\le2\).  Moreover
\[
b_{n,n}=\frac2n,
\qquad
u_{n,n}=\frac1{L_n^2},
\qquad
\sigma_n^2=\frac1{L_n}.
\]
Since \(L_n\to\infty\) and \(L_n^k/n\to0\) for each fixed natural power \(k\),
\[
\frac{b_{n,n}}{u_{n,n}}=\frac{2L_n^2}{n}\to0,
\qquad
u_{n,n}\to0,
\qquad
\frac{b_{n,n}}{\sigma_n^2}=\frac{2L_n}{n}\to0,
\qquad
\sigma_n^2\to0.
\]
Thus the diagonal sequence lies in the residual wedge. % lean: diagonal_residual_wedge

\item Along the same diagonal,
\[
q_{n,n,L_n^{-1/2}}
=
\frac1n+\frac1{L_n^2},
\qquad
\ell_{n,n,L_n^{-1/2}}
=
\frac2n+\frac1{L_n^3}.
\]
Because \(L_n^3/n\to0\), eventually \(L_n^3/n<1\).  On that event,
\[
\frac13L_n
\le
\frac{q_{n,n,L_n^{-1/2}}}{\ell_{n,n,L_n^{-1/2}}}
\le
2L_n.
\]
Hence the ratio is eventually of order \(L_n\).  Since \(L_n\to\infty\), the
same ratio diverges to infinity. % lean: diagonal_benchmark_eventually_same_order; diagonal_benchmark_ratio_tendsto_atTop
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:two-sided-minimax-bracket-all-d}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By
\cref{obj:thm:frontier-upper-all-d}, choose a constant
\(C_{\epsilon}^{\mathrm{up}}>0\) and a polynomial calibration handle such that, for every
\(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), the corresponding known-radius selector
\(\widehat\tau_n^{\star}\in\mathcal T_{n,M}\) satisfies
\[
\sup_{P\in\mathcal P_{d,\epsilon,M,\sigma}}
E_{Q_P^{(n)}}\!\left[
\{\widehat\tau_n^{\star}-\tau(P)\}^2
\right]
\le
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}.
\]
By \cref{obj:thm:radius-channel-converse-all-d}, choose
\(c_{\epsilon}^{\mathrm{lo}}\) and \(b_{\epsilon}^{\mathrm{rad}}\) such that
\[
0<c_{\epsilon}^{\mathrm{lo}}\le1,\qquad 0<b_{\epsilon}^{\mathrm{rad}},
\]
and, for every such \(n,d,M,\sigma\),
\[
c_{\epsilon}^{\mathrm{lo}}M^2\ell_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Set
\[
c_{\epsilon}:=c_{\epsilon}^{\mathrm{lo}},
\qquad
C_{\epsilon}:=\max\{1,C_{\epsilon}^{\mathrm{up}}\}.
\]
Then
\[
0<c_{\epsilon}\le1,\qquad 1\le C_{\epsilon},\qquad c_{\epsilon}\le C_{\epsilon}.
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.two_sided_minimax_bracket_all_d

\item Fix \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\).  The lower bound is exactly the
one supplied by \cref{obj:thm:radius-channel-converse-all-d} with the constants fixed above:
\[
c_{\epsilon}M^2\ell_{n,d,\sigma}
=
c_{\epsilon}^{\mathrm{lo}}M^2\ell_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The accompanying radius-channel certificate in
\cref{obj:thm:radius-channel-converse-all-d} also supplies a realized source family and an
embedding into the same model class.  Applying its source-risk assertion to the zero source
estimator gives a source law, and the embedding-membership part of the certificate gives an
embedded law \(P_{\bullet}\) with
\[
P_{\bullet}\in\mathcal P_{d,\epsilon,M,\sigma}.
\]
Thus the model class is nonempty for these indices. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

\item For any estimator \(\widehat\eta\in\mathcal T_{n,M}\), define its worst-case risk over
this class by
\[
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
:=
\sup_{P\in\mathcal P_{d,\epsilon,M,\sigma}}
E_{Q_P^{(n)}}\!\left[\{\widehat\eta-\tau(P)\}^2\right].
\]
We first record the lower-boundedness needed to apply the infimum in
\cref{obj:def:minimax-risk}.  Fix \(\widehat\eta\in\mathcal T_{n,M}\).  In the vacuous branch
where the model index is empty, the real supremum convention gives
\(\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)=0\).  In the nonvacuous branch, consider
the range of mean-squared errors indexed by \(P\in\mathcal P_{d,\epsilon,M,\sigma}\).  If
this range is bounded above, choose any law in the class; its mean-squared error is the
integral of a square and is therefore nonnegative, and it is bounded by the supremum defining
\(\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)\).  If the range is unbounded above, the
supremum is in its unbounded-above case and the inequality
\(0\le\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)\) is immediate.  Hence
\[
0\le\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
\qquad\text{for every }\widehat\eta\in\mathcal T_{n,M}.
\]
The collection of worst-case risks is therefore bounded below by \(0\).  Using
\cref{obj:def:minimax-risk} and the admissibility of \(\widehat\tau_n^{\star}\),
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
=
\inf_{\widehat\eta\in\mathcal T_{n,M}}
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\eta)
\le
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\tau_n^{\star}).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.minimaxRisk

\item Applying the selector guarantee from \cref{obj:thm:frontier-upper-all-d} gives
\[
\mathcal W_{n,d,\epsilon,M,\sigma}(\widehat\tau_n^{\star})
\le
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}.
\]
Moreover,
\[
u_{n,d}\ge0,\qquad h_{n,d,\sigma}\ge0,
\]
so
\[
r_{n,d,\sigma}
=
\frac1n+\min\{1,u_{n,d},h_{n,d,\sigma}\}
\ge
\frac1n
>
0.
\]
Therefore \(M^2r_{n,d,\sigma}\ge0\), and \(C_{\epsilon}^{\mathrm{up}}\le C_{\epsilon}\)
implies
\[
C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}
\le
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Combining the previous displays yields
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Together with the lower bound supplied by \cref{obj:thm:radius-channel-converse-all-d} and
instantiated above, this proves the asserted bracket for all \(n,d\ge1\), \(M\ge1\), and
\(0\le\sigma\le2\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:published-binary-collision-comparison}]
Fix \(\epsilon\in(0,1/2)\) and assume the published binary collision guarantee stated in the proposition.  We prove the simultaneous conclusion in three steps.

\begin{enumerate}
\item The assumed guarantee is exactly the first component of the simultaneous assertion: it supplies the stated constants \(C_\epsilon,b_\epsilon>0\) and the two displayed bias and variance bounds for every admissible \(n,d,P\), and \(\sigma_{\mathrm{bin}}\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.published_binary_collision_comparison

\item For the pointwise algebraic comparison, fix \(n,d\in\mathbb N\) and \(\sigma\in\mathbb R\) in the stated range.  The defining order property of the minimum gives
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}
\le
\min\{u_{n,d},h_{n,d,\sigma}\}
\le
h_{n,d,\sigma}.
\]
This is the asserted pointwise inequality. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.published_binary_collision_comparison

\item For the asymptotic comparison, let \(d_n\) and \(\sigma_n\) satisfy the stated sequence hypotheses.  Limits along \(\mathbb N\) are unchanged by discarding finitely many initial indices, so work on the cofinal set \(n>0\).  There \(d_n>0\), and hence
\[
h_{n,d_n,\sigma_n}
=
\sigma_n^2+\frac{d_n}{n^2}
>
0.
\]
Also \(u_{n,d_n}\ge0\), \(h_{n,d_n,\sigma_n}\ge0\), and \(1\ge0\).  Therefore
\[
0
\le
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}
{h_{n,d_n,\sigma_n}}.
\]
The same minimum inequalities, now using the left branch of the inner minimum, give
\[
\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}
\le
\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}
\le
u_{n,d_n}.
\]
Dividing by the positive denominator \(h_{n,d_n,\sigma_n}\) yields, for all sufficiently large \(n\),
\[
0
\le
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}
{h_{n,d_n,\sigma_n}}
\le
\frac{u_{n,d_n}}{h_{n,d_n,\sigma_n}}.
\]
By hypothesis the right-hand side converges to \(0\), so the squeeze theorem gives
\[
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}
{h_{n,d_n,\sigma_n}}
\to0 .
\]
This proves the sequence comparison and completes the proof. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.published_binary_collision_comparison
\end{enumerate}
\end{proof}

\item Fix \(d\ge1\) and \(M\ge1\).  For a binary observed record

\varphi_M(z):=\bigl(x,a,M(b-\tfrac12)\bigr),

where \(b\in\{0,1\}\) is read as a real number.  Let \(\Phi_M(P)\) be the real-outcome law whose observed law is

Q_{\Phi_M(P)}=(\varphi_M)_{\#}Q_P,

whose cell masses and propensities agree with those of \(P\), and whose positive arm-cell conditional outcome laws satisfy

\operatorname{Law}_{\Phi_M(P)}(Y\mid A=a,X=k)

\bigl(b\mapsto M(b-\tfrac12)\bigr)_{\#}

\[
(x,a,\tilde b_0,\tilde b_1)
\longmapsto
\bigl(x,a,M(\tilde b_0-\tfrac12),M(\tilde b_1-\tfrac12),M(\tilde b_a-\tfrac12)\bigr).
\]
This construction is an affine binary embedding.  We use the totalized finite-cell convention for this construction: the displayed arm-cell mean is defined for every arm and cell, while the conditional-mean assumptions in \cref{obj:ass:mean-normalization,obj:ass:second-central-moment} use it only on positive-mass cells.  With this convention,
\[
\mu_{ak}(\Phi_M(P))
=
M\{\mu_{ak}^{\mathrm{bin}}(P)-\tfrac12\}
\]
for every arm and cell.

If \(P\in\mathcal D_d^{\mathrm{bin}}(\epsilon)\), overlap is unchanged because the cell masses and propensities are unchanged.  The displayed full-data construction is consistent, and the independent missing-potential augmentation gives the conditional-exchangeability identities.  Also, on every positive-mass cell,
\[
|\mu_{ak}(\Phi_M(P))|
=
M\left|\mu_{ak}^{\mathrm{bin}}(P)-\frac12\right|
\le \frac M2
\]
and the affine binary outcome satisfies
\[
E_{\Phi_M(P)}\!\left[
\{Y-\mu_{ak}(\Phi_M(P))\}^2\mid A=a,X=k
\right]\le M^2 .
\]
These certificates put \(\Phi_M(P)\) in \(\mathcal P_{d,\epsilon,M}^{\mathrm{unr}}\).

\[
\tau_k(\Phi_M(P))
=
M\{\mu_{1k}^{\mathrm{bin}}(P)-\mu_{0k}^{\mathrm{bin}}(P)\},
\]
and
\[
\tau(\Phi_M(P))
=
M\sum_{k=1}^d p_k(P)
\{\mu_{1k}^{\mathrm{bin}}(P)-\mu_{0k}^{\mathrm{bin}}(P)\}.
\]
The binary effects are common across cells, and the cell masses sum to one, so
\[
|\tau_k(\Phi_M(P))-\tau(\Phi_M(P))|=0
\]
on every positive-mass cell.  Together with the preceding consistency, exchangeability, overlap, mean-normalization, and second-moment certificates, this gives
\(\Phi_M(P)\in\mathcal P_{d,\epsilon,M,0}\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.affineBinaryRealLaw_model_zero

\item For strictness, choose one cell \(k_\circ\).  For each \(0\le\sigma\le2\), let
\(\mathfrak W_\circ\) be the one-cell real-outcome law with all mass on \(k_\circ\), propensity \(1/2\), control outcome identically \(0\), and treated outcome distributed as the symmetric two-point law on \(\{-M/2,M/2\}\).  Its arm means are both \(0\), so
\[
\tau_{k_\circ}(\mathfrak W_\circ)=\tau(\mathfrak W_\circ)=0,
\qquad
|\tau_{k_\circ}(\mathfrak W_\circ)-\tau(\mathfrak W_\circ)|
=0\le\sigma M .
\]
The constructed full-data law is consistent and conditionally exchangeable.  Overlap holds in the only positive-mass cell because its propensity is \(1/2\) and \(0<\epsilon<1/2\).  Mean normalization and the second-central-moment bound hold from the endpoint scale \(M/2\), the zero arm means, and \(M\ge1\).  Hence
\[
\mathfrak W_\circ\in\mathcal P_{d,\epsilon,M,\sigma}.
\]
For every affine binary image \(\Phi_M(P)\), the control arm-cell outcome law at \(k_\circ\) is supported on
\(\{-M/2,M/2\}\).  Since \(M\ge1\), it assigns mass \(0\) to \(\{0\}\).  The law
\(\mathfrak W_\circ\) assigns mass \(1\) to \(\{0\}\) in that same conditional law.  Thus
\[
\Phi_M(P)\ne \mathfrak W_\circ
\]
for every binary law \(P\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.affineBinaryRealLaw_ne_testModelLaw

\item Now fix \(n\ge1\), \(0\le\sigma\le2\), and assume
\[
d\le c_{\mathrm{ex}}n^2 .
\]
Since \(c_{\mathrm{ex}}\le1\), this implies \(d/n^2\le1\).  Let
\[
\mathcal B_{n,d,\epsilon}
:=
\inf_{\widehat\psi}
\sup_{P\in\mathcal H_d^{\mathrm{bin}}(\epsilon)}
E_{P^{\otimes n}}\!\left[
\{\widehat\psi-\psi(P)\}^2
\right]
\]
be the fixed-sample binary exact-homogeneity minimax risk.  For the parametric term, take the null source law \(P_0\) with uniform cell masses, propensity \(1/2\), and arm-cell binary means \(1/2\) in both arms.  Let
\[
\Delta_n:=\frac{2/5}{\sqrt n},
\qquad
\Delta_n^2=\frac{4}{25n},
\]
and take \(P_1\) with the same cell masses, propensity, and control means, and treated arm-cell means \(1/2+\Delta_n\).  Since \(n\ge1\), \(\Delta_n\le2/5\), so both endpoint laws belong to \(\mathcal H_d^{\mathrm{bin}}(\epsilon)\), with
\[
\psi(P_0)=0,
\qquad
\psi(P_1)=\Delta_n .
\]
Moreover
\[
n\left\{\frac{(1/2)\Delta_n^2}{(1/2)(1-1/2)}\right\}
=\frac{8}{25}
\le \log 2,
\]
and the product total-variation bound for this Bernoulli endpoint experiment gives
\[
\operatorname{TV}\!\bigl(P_0^{\otimes n},P_1^{\otimes n}\bigr)\le \frac12 .
\]
Le Cam's two-point inequality therefore gives
\[
\frac{1}{100n}
=
\frac{\Delta_n^2}{16}
\le
\mathcal B_{n,d,\epsilon}.
\]
Also, by \cref{obj:lem:binary-exact-collision-lower},
\[
\frac{\omega_\epsilon^2}{64}\frac{d}{n^2}
\le
\mathcal B_{n,d,\epsilon}.
\]
Therefore, with
\(\alpha_\epsilon:=\min\{1/100,\omega_\epsilon^2/64\}\),
\[
c_{\mathrm{ex}}\left(\frac1n+\frac d{n^2}\right)
\le
\frac{\alpha_\epsilon}{4}\left(\frac1n+\frac d{n^2}\right)
\le
\frac12\,\mathcal B_{n,d,\epsilon}
<
\mathcal B_{n,d,\epsilon},
\]
where the strict final inequality uses the positive parametric endpoint lower bound.  Hence, for every measurable source estimator, some
\(P\in\mathcal H_d^{\mathrm{bin}}(\epsilon)\) attains source risk at least
\(c_{\mathrm{ex}}(n^{-1}+d/n^2)\).  Given a real-outcome estimator
\(\widehat\tau\in\mathcal T_{n,M}\), define the source estimator
\[
\widehat\psi(s)
:=
\frac1M\,\widehat\tau\bigl(\varphi_M(s_1),\ldots,\varphi_M(s_n)\bigr),
\qquad
s=(s_1,\ldots,s_n).
\]
The coordinatewise pushforward identity and the target-scaling identity give
\[
Q_{\Phi_M(P)}^{(n)}
=
(\varphi_M^{(n)})_{\#}P^{\otimes n},
\qquad
\tau(\Phi_M(P))=M\psi(P).
\]
Consequently
\[
\operatorname{MSE}_{\Phi_M(P)}(\widehat\tau)
=
M^2
E_{P^{\otimes n}}\!\left[
\{\widehat\psi-\psi(P)\}^2
\right]
\ge
c_{\mathrm{ex}}M^2
\left(\frac1n+\frac d{n^2}\right).
\]
This proves the estimator-wise exact-subclass lower transfer. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.binaryExactMinimaxRisk_hard_family_of_lt

\item The minimax exact lower bound follows from \cref{obj:lem:scaled-binary-exact-lower-transfer-all-d}.  Since \(d/n^2\le1\) and \(c_{\mathrm{ex}}\le\bar c_{\mathrm{ex}}\),
\[
c_{\mathrm{ex}}M^2\left(\frac1n+\frac d{n^2}\right)
\le
\bar c_{\mathrm{ex}}M^2
\left(\frac1n+\min\left\{1,\frac d{n^2}\right\}\right)
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
For the radius term, \cref{obj:thm:radius-channel-converse-all-d} gives both the transfer certificate with constants
\(c_{\mathrm{rad}},b_{\mathrm{rad}}\) and
\[
c_{\mathrm{rad}}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
+\sigma^2\min(1,u_{n,d})
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Because all discarded summands are nonnegative,
\[
c_{\mathrm{rad}}M^2\sigma^2\min(1,u_{n,d})
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
This is the stated radius-channel transfer and certificate. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

\item Finally, \cref{obj:lem:continuous-ratio-polynomial-upper-all-d} applied to the chosen handle gives, for every \(n,d\ge1\), \(M\ge1\), \(0\le\sigma\le2\), and every
\(Q\in\mathcal P_{d,\epsilon,M,\sigma}\), a measurable estimator
\(T_n^{\mathrm{poly}}\) with
\[
T_n^{\mathrm{poly}}(s)\in[-M,M]
\quad\text{for every sample }s,
\]
and
\[
\operatorname{MSE}_{Q}(T_n^{\mathrm{poly}})
\le
C_{\mathrm{up}}M^2
\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
The active-range implication for \(K_n\) is the handle property recorded in the first step. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.continuous_ratio_polynomial_upper_all_d
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:radius-channel-converse-all-d}]
Fix \(0<\epsilon<1/2\).

\begin{enumerate}
\item
Let \(a_{\epsilon}>0\), \(b_{\epsilon}>0\), and \(N_{\epsilon}\) be the constants from \cref{obj:lem:zeng-binary-one-arm-lower}.  Set
\[
b_{0,\epsilon}:=\min\{b_{\epsilon},4\},
\qquad
b_{\epsilon}^{\mathrm{rad}}:=\frac{b_{0,\epsilon}}2,
\qquad
\bar b_{\epsilon}^{\mathrm{rad}}:=b_{\epsilon},
\]
\[
a_{\epsilon}^{\mathrm{rad}}:=a_{\epsilon},
\qquad
c_{\epsilon}^{\mathrm{rad}}:=\frac{a_{\epsilon}b_{0,\epsilon}^{2}}{128},
\qquad
N_{\epsilon}^{\mathrm{rad},0}:=
\max\left\{3,N_{\epsilon},\left\lceil\frac{2}{b_{0,\epsilon}}\right\rceil\right\}.
\]
Then \(a_{\epsilon}^{\mathrm{rad}}>0\), \(b_{\epsilon}^{\mathrm{rad}}>0\), \(\bar b_{\epsilon}^{\mathrm{rad}}>b_{\epsilon}^{\mathrm{rad}}\), and \(c_{\epsilon}^{\mathrm{rad}}>0\).  For every \(n,d\ge1\), define the capped radial alphabet
\[
d_{\mathrm{rad}}
:=
\min\left\{
d,\max\left\{1,\left\lfloor b_{\epsilon}^{\mathrm{rad}}\,n\log(en)\right\rfloor\right\}
\right\}.
\]
Then \(0<d_{\mathrm{rad}}\le d\).  If, in addition, \(N_{\epsilon}^{\mathrm{rad},0}\le n\), then
\[
d_{\mathrm{rad}}
\le \bar b_{\epsilon}^{\mathrm{rad}}n\log n.
\]
A source law on \(d_{\mathrm{rad}}\) cells carries, for every arm \(a\) and \emph{every} cell \(k\),
an arm-cell response distribution, zero-mass and empty arm-cells included; write
\(\mu_{ak}^{\mathrm{bin}}(P)\) for its mean, which agrees with \(E_P[B\mid A=a,X=k]\) whenever the
arm-cell probability is positive.  With that convention every summand below is defined on every
cell, and zero-mass cells contribute nothing because they enter weighted by \(p_k(P)=0\).  For a
control-zero one-arm source law \(P\) on \(d_{\mathrm{rad}}\) cells, write
\[
\psi(P):=
\sum_{k=1}^{d_{\mathrm{rad}}}
  p_k(P)\,\mu_{1k}^{\mathrm{bin}}(P).
\]
Under the same large-sample condition, every measurable uniformly bounded estimator \(\widehat\psi\) in the control-zero one-arm source experiment on \(d_{\mathrm{rad}}\) cells admits a source law \(P\) such that
\[
\frac{a_{\epsilon}^{\mathrm{rad}}}{2}
\left\{
\frac1n+\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}
\right\}
\le
E_{P^{\otimes n}}
\left[
\{\widehat\psi-\psi(P)\}^2
\right].
\]
The same constants also give, for every \(M,\sigma\in\mathbb R\),
\[
c_{\epsilon}^{\mathrm{rad}}M^2\sigma^2
\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}
\le
\left(\frac{M\sigma}{2}\right)^2
\frac{a_{\epsilon}^{\mathrm{rad}}}{2}
\left\{
\frac1n+
\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}
\right\}.
\]
Indeed, the cutoff gives \(n\ge3\) and \(2/b_{0,\epsilon}\le n\).  Hence
\[
1\le b_{0,\epsilon}n\log n=2b_{\epsilon}^{\mathrm{rad}}n\log n,
\qquad
1\le b_{\epsilon}^{\mathrm{rad}}n\log(en).
\]
The first inequality controls the \(\max\{1,\cdot\}\) guard in the source-range bound; together with \(\log(en)\le2\log n\) for \(n\ge3\), it yields \(d_{\mathrm{rad}}\le b_{0,\epsilon}n\log n\le b_{\epsilon}n\log n\).  The second inequality is the range input for \cref{obj:lem:capped-radial-transport-scale} with parameter \(b_{0,\epsilon}\), which gives the displayed transport scale.  The all-sample definition of \(d_{\mathrm{rad}}\) is retained below; the cutoff \(N_{\epsilon}^{\mathrm{rad},0}\le n\) is invoked for the source-range, source-risk, and large-sample transport-scale conclusions in this step. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exists_capped_radial_transport_package

\item
Use \cref{obj:lem:capped-exact-transport-package} to choose
\[
b_{\epsilon}^{\mathrm{ex}}>0,
\qquad
c_{\epsilon}^{\mathrm{ex}}>0,
\qquad
N_{\epsilon}^{\mathrm{ex}}\in\mathbb N
\]
and, for each \(n,d\ge1\), a source lower-risk level \(L_{\mathrm{ex}}(n,d)\) for the exact-homogeneity capped alphabet
\[
d_{\mathrm{ex}}
:=
\min\left\{
d,\max\left\{1,\left\lfloor b_{\epsilon}^{\mathrm{ex}}n^2\right\rfloor\right\}
\right\}.
\]
Thus \(0<d_{\mathrm{ex}}\le d\), the exact source estimator-wise lower bound holds at level \(L_{\mathrm{ex}}(n,d)\), and
\[
c_{\epsilon}^{\mathrm{ex}}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
\right\}
\le M^2L_{\mathrm{ex}}(n,d)
\]
for all \(M\).  Independently, \cref{obj:lem:scaled-binary-exact-lower-transfer-all-d} gives a constant \(c_{\epsilon}^{\mathrm{base}}>0\) such that, for every \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\),
\[
c_{\epsilon}^{\mathrm{base}}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exists_capped_exact_transport_package

\item
Let
\[
N_{\epsilon}^{\mathrm{rad}}:=\max\{N_{\epsilon}^{\mathrm{rad},0},1\},
\qquad
c_{\epsilon}^{\mathrm{small}}:=\frac{1}{800N_{\epsilon}^{\mathrm{rad}}},
\]
\[
c_{\epsilon}^{\mathrm{tr}}
:=
\min\{c_{\epsilon}^{\mathrm{rad}},c_{\epsilon}^{\mathrm{small}},c_{\epsilon}^{\mathrm{ex}}\},
\qquad
c_{\epsilon}
:=
\min\left\{\frac{\min\{c_{\epsilon}^{\mathrm{base}},c_{\epsilon}^{\mathrm{tr}}\}}{2},1\right\}.
\]
Then \(0<c_{\epsilon}\le1\), \(c_{\epsilon}\le c_{\epsilon}^{\mathrm{tr}}\), \(c_{\epsilon}\le c_{\epsilon}^{\mathrm{rad}}\), \(c_{\epsilon}\le c_{\epsilon}^{\mathrm{small}}\), \(c_{\epsilon}\le c_{\epsilon}^{\mathrm{ex}}\), and \(0<b_{\epsilon}^{\mathrm{rad}}\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

\item
Now fix \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\).  Use the two capped alphabets \(d_{\mathrm{rad}}\) and \(d_{\mathrm{ex}}\) defined above for this \((n,d)\), and embed each capped alphabet into \(\{1,\ldots,d\}\) by the first-coordinate padding map.  The radial source family is the full control-zero binary one-arm class on \(d_{\mathrm{rad}}\) cells, and the exact source family is the full exact-homogeneity binary class on \(d_{\mathrm{ex}}\) cells.  For each source law, take as full-data coupling the independent binary full-data lift: conditionally on the observed binary law, the missing potential response is drawn from the corresponding binary arm-cell outcome law.  These choices preserve the source full-data margins for the radial and exact source families.

For a radial source law \(P\), the independent coupling has treated potential margin
\[
E_P[B(1)\mid X=k]=E_P[B\mid A=1,X=k],
\]
so the source functional defined in Step 1 is equivalently
\[
\psi(P)=\sum_{k=1}^{d_{\mathrm{rad}}}p_k(P)\,E_P[B(1)\mid X=k].
\]
Set
\[
\lambda:=\frac{\sigma}{2},
\]
and let the channel be
\[
P\{B'(a)=1\mid B(a)\}
=
\frac12+\lambda\left(B(a)-\frac12\right),
\qquad
Y(a):=M\left(B'(a)-\frac12\right).
\]
Since \(0\le\sigma\le2\), the two channel probabilities are \(1/2-\sigma/4\) and \(1/2+\sigma/4\), and both lie in \([0,1]\).  The radial full law is the common two-potential-outcome Bernoulli channel applied to the independent source coupling and then scaled by \(b\mapsto M(b-1/2)\); the exact full law is the deterministic affine image \(Y(a)=M(B(a)-1/2)\) of the exact independent source coupling.

Padding preserves cell masses and propensities on the image and assigns zero mass outside it.  The affine full-data construction supplies consistency and conditional exchangeability; inherited propensities give overlap.  Since \(Y(a)\in[-M/2,M/2]\) and \(M\ge1\), the conditional mean normalization and second central moment requirements in \cref{obj:def:model-class} hold.  On every positive-mass padded radial source cell,
\[
E[Y(1)\mid X=k]
=
M\frac{\sigma}{2}
\left(E_P[B(1)\mid X=k]-\frac12\right),
\qquad
E[Y(0)\mid X=k]
=
-M\frac{\sigma}{4},
\]
and therefore
\[
\tau_k
=
M\frac{\sigma}{2}E_P[B(1)\mid X=k],
\qquad
\tau
=
M\frac{\sigma}{2}\psi(P).
\]
Thus
\[
|\tau_k-\tau|
\le
M\frac{\sigma}{2}
\le
\sigma M,
\]
so every radial embedded law belongs to \(\mathcal P_{d,\epsilon,M,\sigma}\).  Moreover, for all radial source laws \(P_0,P_1\), writing \(\tau(P_j)\) for the ATE of the embedded law,
\[
\tau(P_1)-\tau(P_0)
=
M\frac{\sigma}{2}\{\psi(P_1)-\psi(P_0)\}.
\]
The exact deterministic affine embedding has target slope \(M\) for the exact-homogeneity source functional. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radialPaddedAffineLaw_model

\item
Let \(\widehat\tau\) be any estimator taking values in \([-M,M]\).  The observed radial channel is the common Markov kernel that keeps the padded cell and treatment arm, applies the Bernoulli contraction to the observed binary response, and scales the result to \(M(B'-1/2)\).  For all radial source laws \(P_0,P_1\), this common kernel gives the product-sample data-processing bound
\[
\operatorname{TV}\!\left(Q_{\mathrm{rad},P_0}^{(n)},Q_{\mathrm{rad},P_1}^{(n)}\right)
\le
\operatorname{TV}\!\left(P_0^{\otimes n},P_1^{\otimes n}\right),
\]
where \(Q_{\mathrm{rad},P}^{(n)}\) is the observed product law after the radial embedding.

If \(\sigma=0\), the fixed-sample parametric radial source bound supplies a source law \(P\), and the radial lower bound asserted in the transfer certificate has left side zero; nonnegativity of squared error gives the required risk inequality.  Suppose \(\sigma>0\).  If \(N_{\epsilon}^{\mathrm{rad},0}\le n\), compose \(\widehat\tau\) with the product channel and set
\[
\widehat\psi_{\mathrm{rad}}(Z)
:=
\frac{
E\{\widehat\tau(W)\mid Z\}
}{
M\sigma/2
},
\]
where \(Z\) is the binary source sample and \(W\) is the channel output.  Since \(M\ge1\) and \(\sigma>0\), the divisor is positive, and the boundedness of \(\widehat\tau\) makes \(\widehat\psi_{\mathrm{rad}}\) uniformly bounded.  Jensen's inequality gives, for each source law \(P\),
\[
\left(\frac{M\sigma}{2}\right)^2
E_{P^{\otimes n}}
\left[
\{\widehat\psi_{\mathrm{rad}}-\psi(P)\}^2
\right]
\le
E_{Q_{\mathrm{rad},P}^{(n)}}
\left[
\{\widehat\tau-\tau(P)\}^2
\right].
\]
The source lower bound from Step 1 and the radial scale inequality then give some radial source law \(P\) such that
\[
c_{\epsilon}^{\mathrm{tr}}M^2\sigma^2
\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}
\le
E_{Q_{\mathrm{rad},P}^{(n)}}
\left[
\{\widehat\tau-\tau(P)\}^2
\right].
\]
If \(N_{\epsilon}^{\mathrm{rad},0}>n\), use the same all-sample capped alphabet \(d_{\mathrm{rad}}\) and the one-arm parametric source lower bound at level \(1/(200n)\).  Since \(1\le n\le N_{\epsilon}^{\mathrm{rad}}\), \cref{obj:lem:finite-sample-radial-transport-scale} gives
\[
c_{\epsilon}^{\mathrm{small}}M^2\sigma^2
\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}
\le
\left(\frac{M\sigma}{2}\right)^2\frac{1}{200n},
\]
and \(c_{\epsilon}^{\mathrm{tr}}\le c_{\epsilon}^{\mathrm{small}}\) yields the same estimator-wise radial bound.  Since \(c_{\epsilon}\le c_{\epsilon}^{\mathrm{tr}}\), the radial transfer certificate also holds with \(c_{\epsilon}\), using the cap, source family, padding map, embedding, independent full-data coupling, and channel fixed above. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radial_risk_of_kernel_transport

\item
The deterministic exact embedding is handled analogously.  For any bounded ambient estimator, compose it with the exact padding-and-scaling map and divide by \(M\).  Since \(M\ge1\), the affine slope is positive.  The exact source level \(L_{\mathrm{ex}}(n,d)\) and \(c_{\epsilon}^{\mathrm{tr}}\le c_{\epsilon}^{\mathrm{ex}}\) give, for every \(\widehat\tau\), some exact source law whose embedded risk is at least
\[
c_{\epsilon}^{\mathrm{tr}}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
\right\}.
\]
This supplies the exact half of the common transfer certificate; the radial half was established in Step 5. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exactRiskTransfer_of_affine_transport

\item
Because every radial embedded law belongs to \(\mathcal P_{d,\epsilon,M,\sigma}\), Step 5 implies
\[
c_{\epsilon}^{\mathrm{tr}}M^2\sigma^2
\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Together with the base lower bound from Step 2, put
\[
\mathfrak A_{n,d}:=
\frac1n+\min\left(1,\frac d{n^2}\right),
\qquad
\mathfrak R_{n,d,\sigma}:=
\sigma^2\min\left\{1,\frac{d^2}{n^2\log^2(en)}\right\}.
\]
Then
\[
c_{\epsilon}^{\mathrm{base}}M^2\mathfrak A_{n,d}
\le
\mathsf R_{n,d,\epsilon,M,\sigma},
\qquad
c_{\epsilon}^{\mathrm{tr}}M^2\mathfrak R_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Since \(\mathfrak A_{n,d}\) and \(\mathfrak R_{n,d,\sigma}\) are nonnegative,
\[
\frac{\min\{c_{\epsilon}^{\mathrm{base}},c_{\epsilon}^{\mathrm{tr}}\}}{2}
M^2(\mathfrak A_{n,d}+\mathfrak R_{n,d,\sigma})
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Finally \(c_{\epsilon}\le \min\{c_{\epsilon}^{\mathrm{base}},c_{\epsilon}^{\mathrm{tr}}\}/2\), so
\[
c_{\epsilon}M^2
\left\{
\frac1n+
\min\left(1,\frac d{n^2}\right)
+
\sigma^2\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The radial certificate asserted in the theorem is the package from Steps 4 and 5: the same \(b_{\epsilon}^{\mathrm{rad}}\), capped alphabet \(d_{\mathrm{rad}}\), source family, first-coordinate padding map, radial embedding, independent full-data coupling, common Bernoulli channel, product data-processing inequality, pairwise target identity, and estimator-wise radial lower bound. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.converseRate_lower_of_exact_transfer_and_handle
\end{enumerate}
\end{proof}
