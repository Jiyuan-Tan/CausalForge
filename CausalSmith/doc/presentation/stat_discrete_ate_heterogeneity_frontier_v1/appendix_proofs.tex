\begin{proof}[Proof of \cref{obj:prop:ate-identification}]
\begin{enumerate}
\item
For each cell \(k=1,\ldots,d\), set
\[
p_k=P(X=k);
\]
when \(d=0\), all sums below are empty.  When \(p_k>0\), let \(\pi_k=P(A=1\mid X=k)\).  The finite-cell law carries an arm-cell outcome law \(\nu_{ak}\) for every arm \(a\in\{0,1\}\) and every cell, including cells with \(p_k=0\), and its totalized mean is
\[
\mu_{ak}:=\int y\,d\nu_{ak}(y).
\]
For every supported cell and every Borel set \(\mathfrak A\subseteq\mathbb R\), the observed arm-cell factorization is
\[
P_O(X=k,A=a,Y\in\mathfrak A)
=
p_k
\begin{cases}
\pi_k, & a=1,\\
1-\pi_k, & a=0
\end{cases}
\nu_{ak}(\mathfrak A).
\]
If \(p_k>0\), \cref{obj:ass:overlap} and \(\epsilon>0\) make the displayed scalar positive for both arms.  Hence, on supported cells, \(\nu_{ak}\) is the observed conditional law of \(Y\) given \((A=a,X=k)\), and \(\mu_{ak}\) is the corresponding observed conditional mean.  If \(p_k=0\), then \(P_O(X=k)=0\); the totalized law \(\nu_{ak}\) and mean \(\mu_{ak}\) enter the finite formula below only through terms multiplied by \(p_k\). % lean: observed_arm_outcome_measure

\item
Fix \(a\in\{0,1\}\) and a supported cell \(k\), so \(p_k>0\).  By \cref{obj:ass:consistency}, on \(\{X=k,A=a\}\),
\[
Y(a)=Y
\quad\text{almost surely}.
\]
Together with the observed arm-cell factorization, this gives, for every Borel \(\mathfrak A\),
\[
P(X=k,A=a,Y(a)\in\mathfrak A)
=
p_k
\begin{cases}
\pi_k, & a=1,\\
1-\pi_k, & a=0
\end{cases}
\nu_{ak}(\mathfrak A).
\]
Applying \cref{obj:ass:conditional-exchangeability} in cell \(k\) to the potential-outcome event \(\{Y(a)\in\mathfrak A\}\) yields
\[
P(X=k,A=a,Y(a)\in\mathfrak A)\,P(X=k)
=
P(X=k,A=a)\,P(X=k,Y(a)\in\mathfrak A).
\]
Here \(P(X=k)=p_k\), and \(P(X=k,A=a)\) is \(p_k\pi_k\) when \(a=1\) and \(p_k(1-\pi_k)\) when \(a=0\).  The same overlap argument makes this arm-cell probability positive, so cancellation gives
\[
P(X=k,Y(a)\in\mathfrak A)=p_k\nu_{ak}(\mathfrak A).
\]
Equivalently,
\[
\mathcal L_P(Y(a);\,X=k)=p_k\,\nu_{ak},
\]
where the left side is the pushforward of the full-data law restricted to \(\{X=k\}\). % lean: full_cell_potential_measure

\item
The preceding restricted-law identity supplies the full-data integrability.  On supported cells, \cref{obj:ass:second-central-moment} gives integrability of \(y\mapsto (y-\mu_{ak})^2\) under \(\nu_{ak}\).  Thus \(y-\mu_{ak}\) is square-integrable, hence integrable, and adding the finite constant \(\mu_{ak}\) gives first-moment integrability of \(y\mapsto y\) under \(\nu_{ak}\).  The restricted-law identity transfers this integrability to \(Y(a)\) on each supported cell.  If \(p_k=0\), the full-data mass of \(\{X=k\}\) is \(0\), so the corresponding restricted measure is the zero measure.  Since there are finitely many cells,
\[
Y(1)\in L^1(P),\qquad Y(0)\in L^1(P).
\]
% lean: full_potential_integrable

\item
For every \(a\in\{0,1\}\) and every cell \(k\),
\[
\int_{\{X=k\}}Y(a)\,dP
=
p_k\int y\,d\nu_{ak}(y)
=
p_k\mu_{ak}.
\]
On supported cells this is the integral form of the restricted-law identity in the second step and the defining integral for \(\mu_{ak}\).  On zero-mass cells, the restricted measure is zero and \(p_k=0\), so both sides vanish. % lean: full_potential_cell_integral

\item
Apply the finite-fiber decomposition of the integral to the measurable finite-valued map \(X\).  Using the integrability just established,
\[
\int Y(1)\,dP
=
\sum_{k=1}^{d}\int_{\{X=k\}}Y(1)\,dP,
\qquad
\int Y(0)\,dP
=
\sum_{k=1}^{d}\int_{\{X=k\}}Y(0)\,dP.
\]
Therefore
\[
\begin{aligned}
\int\{Y(1)-Y(0)\}\,dP
&=
\sum_{k=1}^{d}
\left\{
\int_{\{X=k\}}Y(1)\,dP
-
\int_{\{X=k\}}Y(0)\,dP
\right\}\\
&=
\sum_{k=1}^{d}p_k(\mu_{1k}-\mu_{0k}).
\end{aligned}
\]
By \cref{obj:def:ate-functional},
\[
\tau(P)=\sum_{k=1}^{d}p_k(\mu_{1k}-\mu_{0k}),
\]
so
\[
\tau(P)
=
\int\{Y(1)-Y(0)\}\,dP.
\]
On supported cells, the first step identifies \(\mu_{ak}\) with \(E_P[Y\mid A=a,X=k]\); cells with \(p_k=0\) contribute \(0\) to the totalized sum.  Thus, when conditional expectations on zero-mass arm-cell events are chosen to equal the totalized means \(\mu_{ak}\), the same equality is written as
\[
\tau(P)
=
\int\{Y(1)-Y(0)\}\,dP
=
\sum_{k=1}^{d}p_k
\left\{
E_P[Y\mid A=1,X=k]-E_P[Y\mid A=0,X=k]
\right\}.
\]
The observed margin determines this finite-cell formula: \(p_k=P_O(X=k)\), and on every supported cell \cref{obj:ass:overlap} gives positive observed mass in both arms, so the observed conditional laws determine the two arm-cell means.  Zero-mass cells have weight \(p_k=0\). % lean: ate_identification
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:frontier-upper}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  By
\cref{obj:thm:robust-upper-construction-resolution-all-d}, choose
\(C_{\epsilon}^{0}>0\) and a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), such that
\[
N_{\epsilon}\le n,
\qquad
 d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n
\]
for all positive integers \(n,d\).  The same construction gives, for every
\(n,d\ge1\), every \(M\ge1\), and every \(0\le\sigma\le2\), measurable
\([-M,M]\)-valued estimators \(T_n^{\mathrm{poly}}\) and
\(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}^{0}M^2
\left\{
\frac1n+\sigma^2+\frac{d}{n^2}
\right\}.
\]
Set
\[
C_{\epsilon}:=\max\{1,C_{\epsilon}^{0}\}.
\]
Then \(C_{\epsilon}>0\) and \(C_{\epsilon}\ge C_{\epsilon}^{0}\), so both
displayed risk bounds hold with \(C_{\epsilon}\) in place of
\(C_{\epsilon}^{0}\). % lean: robust_upper_construction_resolution_all_d

\item For fixed admissible \(n,d,M,\sigma\), put
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2},
\qquad
r_{n,d,\sigma}:=\frac1n+\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}.
\]
The known-radius selector of \cref{obj:def:total-estimator} is the three-branch
estimator
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},
& h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,
& h_{n,d,\sigma}> \min\{1,u_{n,d}\}
  \text{ and } u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\end{cases}
\]
The first two branches are elements of \(\mathcal T_{n,M}\) by the construction
above.  The zero branch is measurable and takes values in \([-M,M]\) because
\(M\ge1\).  Hence \(\widehat\tau_n^{\star}\in\mathcal T_{n,M}\). % lean: totalSelector

\item Fix \(P\in\mathcal P_{d,\epsilon,M,\sigma}\).  On the collision branch,
\(h_{n,d,\sigma}\le \min\{1,u_{n,d}\}\), so
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=h_{n,d,\sigma}.
\]
Therefore
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+h_{n,d,\sigma}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
On the polynomial branch, \(u_{n,d}<\min\{1,h_{n,d,\sigma}\}\), so
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=u_{n,d}.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2\left(\frac1n+u_{n,d}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
It remains to consider the zero branch.  There
\[
h_{n,d,\sigma}> \min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then
\(h_{n,d,\sigma}>u_{n,d}\), while
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\); if \(h_{n,d,\sigma}\le1\) this gives
\(h_{n,d,\sigma}\le u_{n,d}<h_{n,d,\sigma}\), and if \(h_{n,d,\sigma}>1\) it
gives \(1\le u_{n,d}<1\).  Hence \(1\le u_{n,d}\).  The first zero-branch
inequality then gives \(1<h_{n,d,\sigma}\), so \(1\le h_{n,d,\sigma}\), and
\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}=1.
\]
Viewing the same law as a member of the unrestricted-radius class, the scale
bound in \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Since the selected estimator is identically zero on this branch,
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2
\le C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma},
\]
where the penultimate inequality uses \(C_{\epsilon}\ge1\) and \(n\ge1\).
Together the three branches prove the asserted risk bound for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\). % lean: frontier_upper_bound_all_d

\item Put \(c_{\epsilon}:=1\).  Then \(c_{\epsilon}>0\).  The preceding
all-alphabet bound applies to every positive \(n,d\), so it applies in
particular under the restricted-dimensional hypothesis
\[
d\le c_{\epsilon}\frac{n^2}{\log(en)}.
\]
The constants \(C_{\epsilon},c_{\epsilon}\) and the same polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\) therefore satisfy all claims of the theorem.
% lean: frontier_upper
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions}]
\begin{enumerate}
\item First record the logarithmic estimate used below.  If \(n\ge1\), then
\[
L_n=\log(en)=1+\log n .
\]
The standard bound \(\log n\le n^{1/2}/(1/2)=2\sqrt n\), together with
\(\log n\ge0\) and \(\sqrt n\ge1\), gives
\[
0\le L_n\le 1+2\sqrt n\le 3\sqrt n .
\]
Squaring yields
\[
L_n^2\le 9n .
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item We prove the all-alphabet comparison that supplies the constants.  For \(n,d\ge1\), put
\[
\eta_{n,d}:=\frac{d}{n^2},
\qquad
u_{n,d}:=\frac{d^2}{n^2L_n^2}.
\]
Since \(L_n\ge1\), both \(\eta_{n,d}\) and \(u_{n,d}\) are nonnegative.  At radius two the collision term is
\[
2^2+\eta_{n,d}=4+\eta_{n,d}\ge1,
\]
and therefore
\[
\frac1n+\min\{1,\min\{u_{n,d},4+\eta_{n,d}\}\}
=
\frac1n+\min\{1,u_{n,d}\}.
\]
At radius zero,
\[
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}
\le
\frac1n+\min\{1,\eta_{n,d}\},
\]
and the capped converse expression at radius zero is exactly the right-hand side.

It remains to compare the radius-two converse term
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\]
with
\[
\frac1n+\min\{1,u_{n,d}\}.
\]
If \(L_n^2\le d\), then
\[
\eta_{n,d}=\frac d{n^2}\le \frac{d^2}{n^2L_n^2}=u_{n,d},
\]
so
\[
\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}=\min\{1,\eta_{n,d}\}
\]
and
\[
\min\{1,\eta_{n,d}\}\le \min\{1,u_{n,d}\}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
5\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
If \(d<L_n^2\), the logarithmic estimate from the previous step gives
\[
d\le 9n,
\qquad
\eta_{n,d}\le \frac9n .
\]
Hence
\[
\frac1n+\min\{1,\eta_{n,d}\}\le \frac{10}{n}.
\]
The upper-frontier term at radius zero is at least \(1/n\), so
\[
\frac1{10}\left(\frac1n+\min\{1,\eta_{n,d}\}\right)
\le
\frac1n+\min\{1,\min\{u_{n,d},\eta_{n,d}\}\}.
\]
The remaining upper bound at radius zero follows from the displayed monotonicity inequality.  For radius two, nonnegativity gives the lower bound
\[
\frac1n+\min\{1,u_{n,d}\}
\le
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\},
\]
while \(\min\{1,\eta_{n,d}\}\le 9/n\) gives
\[
\frac1n+\min\{1,\eta_{n,d}\}+4\min\{1,u_{n,d}\}
\le
10\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
Thus, uniformly over all \(n,d\ge1\), the all-alphabet comparisons hold with
\[
c=\frac1{10},
\qquad
C=10 .
\]
The radius-two equality of attainable laws is exactly the consequence of \cref{obj:lem:scale-sanity}.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now fix \(\epsilon\in(0,1/2)\) and choose
\[
c_\epsilon=1,
\qquad
c=\frac1{10},
\qquad
C=10 .
\]
Then \(0<c_\epsilon\le1\) and \(0<c\le C\).  Suppose \(n,d\ge1\) and
\[
d\le c_\epsilon\frac{n^2}{L_n}=\frac{n^2}{L_n}.
\]
Because \(L_n\ge1\),
\[
d\le n^2,
\qquad
\frac d{n^2}\le1,
\qquad
\min\left\{1,\frac d{n^2}\right\}=\frac d{n^2}.
\]
Applying the all-alphabet comparison from the previous step and replacing the capped term by \(d/n^2\) gives
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\min\left\{u_{n,d},\frac d{n^2}\right\}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right),
\]
and
\[
c\left(\frac1n+\frac d{n^2}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}
\le
C\left(\frac1n+\frac d{n^2}\right).
\]
The class identity is the radius-two conclusion of \cref{obj:lem:scale-sanity}.  Finally, the all-alphabet radius-two calculation gives, for every \(n,d\ge1\),
\[
\frac1n+\min\{1,\min\{u_{n,d},4+d/n^2\}\}
=
\frac1n+\min\{1,u_{n,d}\},
\]
and its converse comparison gives
\[
c\left(\frac1n+\min\{1,u_{n,d}\}\right)
\le
\frac1n+\min\left\{1,\frac d{n^2}\right\}+4\min\{1,u_{n,d}\}
\le
C\left(\frac1n+\min\{1,u_{n,d}\}\right).
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:radius-channel-converse}]
\begin{enumerate}
\item Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).  Apply \cref{obj:thm:radius-channel-converse-all-d} to obtain constants \(\bar c_{\epsilon}\) and \(\bar b_{\epsilon}^{\mathrm{rad}}\) such that
\[
0<\bar c_{\epsilon}\le 1,
\qquad
0<\bar b_{\epsilon}^{\mathrm{rad}},
\]
and, for every admissible \(n,d,M,\sigma\),

\bar c_{\epsilon}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
+\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
Set \(c_{\epsilon}:=\bar c_{\epsilon}\) and
\(b_{\epsilon}^{\mathrm{rad}}:=\bar b_{\epsilon}^{\mathrm{rad}}\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

L:=\log(en).
\]
Since \(0<n\), we have \(n\ge1\), hence
\[
L=\log(e n)=1+\log n\ge1.
\]
The restricted range and \(c_{\epsilon}\le1\) give
\[
d\le c_{\epsilon}\frac{n^2}{L}
\le c_{\epsilon}n^2
\le n^2 .
\]
Because \(n>0\), division by \(n^2\) yields
\[
\frac d{n^2}\le1,
\qquad\text{so}\qquad
\min\left(1,\frac d{n^2}\right)=\frac d{n^2}.
\]
Substituting this identity into the all-alphabet bound displayed above gives
\[
\mathsf R_{n,d,\epsilon,M,\sigma}
\ge
c_{\epsilon}M^2
\left\{
\frac1n+\frac d{n^2}
+\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse

\item The radial certificate supplied by \cref{obj:thm:radius-channel-converse-all-d} uses the same constants \(c_{\epsilon}\) and \(b_{\epsilon}^{\mathrm{rad}}\).  Unpacking the certificate, there are an integer cap \(d_{\mathrm{rad}}\), a binary control-zero source class \(\mathcal S_{\mathrm{rad}}\), a padding map into the \(d\)-alphabet, an embedding \(\Psi\) that sends each source member into \(\mathcal P_{d,\epsilon,M,\sigma}\), and binary full-observation couplings such that the channel acts by
\[
B'(a)\mid B(a)
\sim
\operatorname{Bernoulli}\!\left\{
\frac12+\frac{\sigma}{2}\left(B(a)-\frac12\right)
\right\},
\qquad
Y(a):=M\{B'(a)-1/2\}.
\]
The source target in this certificate is the total treated-arm weighted functional
\[
\psi(P)
:=
\sum_{k=1}^{d_{\mathrm{rad}}}
p_k(P)\,\mu_{1k}^{\mathrm{bin}}(P),
\]
where \(\mu_{1k}^{\mathrm{bin}}(P)\) is the treated-arm binary mean coordinate carried by the source law in cell \(k\), defined for every source cell including zero-mass cells.  Thus, for all \(P_0,P_1\in\mathcal S_{\mathrm{rad}}\), the same certificate gives
\[
\tau(\Psi(P_1))-\tau(\Psi(P_0))
=
M\frac{\sigma}{2}
\left\{
\psi(P_1)-\psi(P_0)
\right\}.
\]
It also gives the estimator-wise lower bound: for every
\(\widehat\tau\in\mathcal T_{n,M}\), the class of total measurable \([-M,M]\)-valued estimators from \cref{obj:def:minimax-risk}, there is \(P\in\mathcal S_{\mathrm{rad}}\) such that
\[
c_{\epsilon}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\mathbb E_{\Psi(P)}
\left[
\{\widehat\tau-\tau(\Psi(P))\}^2
\right].
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:robust-upper-construction-resolution}]
Fix an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\).

\begin{enumerate}
\item First form the all-alphabet package.  By
\cref{obj:lem:continuous-ratio-polynomial-upper-all-d}, there are
\(C_{\mathrm{poly},\epsilon}>0\) and a polynomial handle
\[
\mathfrak h_\epsilon=(N_{\mathrm{cal},\epsilon},\rho_{\mathrm{cal},\epsilon}),
\qquad \rho_{\mathrm{cal},\epsilon}>0,
\]
such that, for all integers \(n,d\ge1\),
\[
N_{\mathrm{cal},\epsilon}\le n,\qquad
d\le \rho_{\mathrm{cal},\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n,
\]
and, for every \(M\ge1\), \(0\le\sigma\le2\), and
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
C_{\mathrm{poly},\epsilon}M^2
\left\{
\frac1n+
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}.
\]
The same cited construction gives measurability of \(T_n^{\mathrm{poly}}\) and
\[
T_n^{\mathrm{poly}}(s)\in[-M,M]
\qquad\text{for every }s\in\mathcal O_d^n .
\]

By \cref{obj:lem:continuous-occupancy-collision-upper-all-d}, there is
\(C_{\mathrm{col},\epsilon}>0\) such that, for every \(n,d\ge1\),
\(M\ge1\), \(0\le\sigma\le2\), and
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
C_{\mathrm{col},\epsilon}M^2
\left(
\frac1n+\sigma^2+\frac{d}{n^2}
\right).
\]
For the admissibility of the collision estimator, write its unclipped inner
statistic as the total sample function
\[
\Gamma_{\mathrm{col}}(s)
:=
\begin{cases}
\displaystyle
\frac{1}{R_n(s)}
\sum_{k=1}^d
N_k(s)I_k(s)
\left[
\mathbf 1\{N_{1k}(s)>0\}\frac{S_{1k}(s)}{N_{1k}(s)\vee 1}
-
\mathbf 1\{N_{0k}(s)>0\}\frac{S_{0k}(s)}{N_{0k}(s)\vee 1}
\right],
& R_n(s)>0,\\[1.2em]
0,& R_n(s)=0.
\end{cases}
\]
Here \(m\vee1\) denotes \(\max\{m,1\}\).  The displayed arm factors are total
functions of the sample and agree with the arm-cell means of
\cref{obj:def:collision-handle} on positive arm-cell counts; multiplication by
\(I_k\) selects exactly the cells where both arm-cell counts are positive.  Hence
\cref{obj:def:collision-handle} gives
\[
T_n^{\mathrm{col}}(s)=\max\{-M,\min(M,\Gamma_{\mathrm{col}}(s))\}.
\]
The counts, indicators, outcome totals, finite sums, totalized ratios, and the

\begin{proof}[Proof of \cref{obj:thm:frontier-upper-all-d}]
Fix \(0<\epsilon<1/2\).

\emph{Step 1. Common calibration and envelopes.}
By \cref{obj:thm:robust-upper-construction-resolution-all-d}, choose a polynomial handle
\((N_{\epsilon},\rho_{\epsilon})\), with \(\rho_{\epsilon}>0\), and a constant
\(\Gamma_{\epsilon}>0\).  Put
\[
C_{\epsilon}:=\max\{1,\Gamma_{\epsilon}\}.
\]
Then \(C_{\epsilon}>0\), \(\Gamma_{\epsilon}\le C_{\epsilon}\), and
\(1\le C_{\epsilon}\).  The selected handle carries the displayed calibration implication
for \(K_n\): for all integers \(n,d\ge1\),
\[
N_{\epsilon}\le n,
\qquad
d\le \rho_{\epsilon}n\log(en)
\quad\Longrightarrow\quad
2\le K_n .
\]
For fixed integers \(n,d\ge1\), \(M\ge1\), and \(0\le\sigma\le2\), define
\[
u_{n,d}:=\frac{d^2}{n^2\log^2(en)},
\qquad
h_{n,d,\sigma}:=\sigma^2+\frac{d}{n^2}.
\]
The construction also gives measurable \([-M,M]\)-valued estimators
\(T_n^{\mathrm{poly}}\) and \(T_n^{\mathrm{col}}\), and for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{poly}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+\min\{1,u_{n,d}\}\right)
\]
and
\[
E_{Q_P^{(n)}}\!\left[
\left(T_n^{\mathrm{col}}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.robust_upper_construction_resolution_all_d

\emph{Step 2. The selector is admissible.}
The known-radius rule in \cref{obj:def:total-estimator} is
\[
\widehat\tau_n^{\star}
=
\begin{cases}
T_n^{\mathrm{col}},&
h_{n,d,\sigma}\le \min\{1,u_{n,d}\},\\
T_n^{\mathrm{poly}},&
u_{n,d}<\min\{1,h_{n,d,\sigma}\},\\
0,&\text{otherwise}.
\end{cases}
\]
The first two branches are measurable and take values in \([-M,M]\) by
\cref{obj:thm:robust-upper-construction-resolution-all-d}.  The zero branch is measurable and
also takes values in \([-M,M]\) because \(M\ge1\).  Hence
\(\widehat\tau_n^{\star}\) is measurable and
\(\widehat\tau_n^{\star}(s)\in[-M,M]\) for every sample \(s\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.totalSelector

\emph{Step 3. Collision branch.}
Assume
\[
h_{n,d,\sigma}\le \min\{1,u_{n,d}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{col}}\), and
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=h_{n,d,\sigma}.
\]
Therefore, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+h_{n,d,\sigma}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
where the last inequality uses \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the
nonnegativity of \(M^2(1/n+h_{n,d,\sigma})\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 4. Polynomial branch.}
Assume the collision-branch condition fails and
\[
u_{n,d}<\min\{1,h_{n,d,\sigma}\}.
\]
Then \(\widehat\tau_n^{\star}=T_n^{\mathrm{poly}}\),
\[
\min\{1,u_{n,d}\}=u_{n,d},
\qquad
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=u_{n,d}.
\]
Hence, for every \(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
\Gamma_{\epsilon}M^2
\left(\frac1n+u_{n,d}\right)
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right),
\]
again because \(\Gamma_{\epsilon}\le C_{\epsilon}\) and the rate factor is nonnegative. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

\emph{Step 5. Zero branch.}
It remains to consider the branch in which
\[
h_{n,d,\sigma}>\min\{1,u_{n,d}\},
\qquad
u_{n,d}\ge \min\{1,h_{n,d,\sigma}\}.
\]
These two inequalities imply \(1\le u_{n,d}\): if \(u_{n,d}<1\), then the first inequality gives
\(h_{n,d,\sigma}>u_{n,d}\), while the second gives
\(\min\{1,h_{n,d,\sigma}\}\le u_{n,d}\), a contradiction whether
\(h_{n,d,\sigma}\le1\) or \(h_{n,d,\sigma}>1\).  With \(1\le u_{n,d}\), the first displayed
inequality gives \(1\le h_{n,d,\sigma}\), and consequently
\[
\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}=1.
\]
On this branch \(\widehat\tau_n^{\star}=0\).  Viewing the law of
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\) as a member of the unrestricted class obtained by
forgetting the radius condition, \cref{obj:lem:scale-sanity} gives
\[
|\tau(P)|\le M.
\]
Thus
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
=
\tau(P)^2
\le M^2.
\]
Since \(C_{\epsilon}\ge1\), \(M^2\ge0\), and \(n\ge1\),
\[
M^2
\le
C_{\epsilon}M^2\left(\frac1n+1\right)
=
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right).
\] % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_bound_all_d

Combining the three branches yields, for every
\(P\in\mathcal P_{d,\epsilon,M,\sigma}\),
\[
E_{Q_P^{(n)}}\!\left[
\left(\widehat\tau_n^{\star}-\tau(P)\right)^2
\right]
\le
C_{\epsilon}M^2
\left(\frac1n+\min\{1,\min(u_{n,d},h_{n,d,\sigma})\}\right)
=
C_{\epsilon}M^2 r_{n,d,\sigma}.
\]
Together with the measurability and range conclusion in Step 2, this proves the stated
selector guarantee. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.frontier_upper_all_d
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:endpoint-reductions-all-d}]
Take \(c=1/10\) and \(C=10\).  Fix integers \(n,d\ge1\), and write
\[
\Lambda_n:=\log(en),\qquad
u_{n,d}:=\frac{d^2}{n^2\Lambda_n^2},\qquad
v_{n,d}:=\frac d{n^2}.
\]
Also set
\[
\theta^{\mathrm{ex}}_{n,d}:=\frac1n+\min(1,v_{n,d}),
\qquad
\theta^{\mathrm{poly}}_{n,d}:=\frac1n+\min(1,u_{n,d}).
\]
The quantities \(1/n\), \(u_{n,d}\), and \(v_{n,d}\) are nonnegative.

\begin{enumerate}
\item Since \(n\ge1\),
\[
\Lambda_n
=\log(en)=1+\log n\ge1,
\]
and hence \(\Lambda_n^2>0\).  At radius two the collision term is
\[
2^2+v_{n,d}=4+v_{n,d}\ge1.
\]
Therefore
\[
\frac1n+\min\{1,u_{n,d},4+v_{n,d}\}
=\frac1n+\min(1,u_{n,d})
=\theta^{\mathrm{poly}}_{n,d}.
\]
This proves the endpoint identity, and also gives the two-sided comparison of the
radius-two upper frontier expression with \(\theta^{\mathrm{poly}}_{n,d}\), because
\(\theta^{\mathrm{poly}}_{n,d}\ge0\) and \(1/10\le1\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item We use the elementary logarithmic estimate
\[
\Lambda_n^2\le 9n.
\]
Indeed, \(\log n\le 2\sqrt n\), while \(\sqrt n\ge1\) and \(\log n\ge0\).  Thus
\[
\Lambda_n=1+\log n\le 1+2\sqrt n\le3\sqrt n,
\]
and squaring gives \(\Lambda_n^2\le9n\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.logEN_sq_le_nine_mul

\item First consider the case \(\Lambda_n^2\le d\).  Since \(d>0\),
\[
v_{n,d}=\frac d{n^2}
\le
\frac{d^2}{n^2\Lambda_n^2}
=u_{n,d}.
\]
Consequently
\[
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
=
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression is also exactly
\[
\frac1n+\min(1,v_{n,d})=\theta^{\mathrm{ex}}_{n,d}.
\]
For the radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and, using \(\min(1,v_{n,d})\le\min(1,u_{n,d})\),
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac1n+5\min(1,u_{n,d})
\le
5\theta^{\mathrm{poly}}_{n,d}.
\]
All required comparisons in this case follow from \(1/10\le1\le5\le10\).
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item Now consider the case \(d<\Lambda_n^2\).  By the logarithmic estimate,
\[
d\le9n,
\qquad\text{so}\qquad
v_{n,d}\le\frac9n.
\]
The radius-zero upper expression satisfies
\[
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\frac1n+\min(1,v_{n,d})
=
\theta^{\mathrm{ex}}_{n,d},
\]
and
\[
\theta^{\mathrm{ex}}_{n,d}
=\frac1n+\min(1,v_{n,d})
\le
\frac1n+\frac9n
=
\frac{10}{n}.
\]
Hence
\[
\frac1{10}\theta^{\mathrm{ex}}_{n,d}
\le
\frac1n
\le
\frac1n+\min\{1,u_{n,d},v_{n,d}\}
\le
\theta^{\mathrm{ex}}_{n,d}
\le
10\theta^{\mathrm{ex}}_{n,d}.
\]
The radius-zero converse expression again equals \(\theta^{\mathrm{ex}}_{n,d}\).  For the
radius-two converse expression,
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\ge
\theta^{\mathrm{poly}}_{n,d},
\]
and
\[
\frac1n+\min(1,v_{n,d})+4\min(1,u_{n,d})
\le
\frac{10}{n}+4\min(1,u_{n,d})
\le
10\theta^{\mathrm{poly}}_{n,d}.
\]
Together with the radius-two identity from the first step, this proves the stated
two-sided comparisons in the second case.
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.endpoint_reductions_all_d

\item It remains to identify the two radius-two classes.  By \cref{obj:lem:scale-sanity}, for

\begin{proof}[Proof of \cref{obj:thm:fixed-interior-tightness-and-shrinking-radius-gap-all-d}]
Fix \(0<\epsilon<1/2\).  The proof starts from the all-alphabet bracket in \cref{obj:thm:two-sided-minimax-bracket-all-d}; write
\[
\omega_{n,d,\sigma}
:=
\frac1n+\min\left(1,\frac d{n^2}\right)
+\sigma^2\min(1,u_{n,d})
\]
for the capped converse benchmark appearing there.

In the triangular region \(L_n^2<d<nL_n\), set \(a=1/n\), \(t=d/n^2\), \(u=u_{n,d}\), and \(s=\sigma^2\).  The two triangular inequalities give \(t\le u<1\), and the selector benchmark is
\[
q_{n,d,\sigma}=a+\min\{u,s+t\}.
\]
For nonnegative \(a,t,u,s\) with \(t\le u\), the lower bound follows by splitting according as \(u\le s+t\) or \(s+t<u\), while the upper bound follows by splitting according as \(u\le s\) or \(s<u\).  These two checks give
\[
\frac12\{a+t+\min(u,s)\}
\le a+\min\{u,s+t\}
\le a+t+\min(u,s).
\]
Thus
\[
\frac12\left\{b_{n,d}+\min(u_{n,d},\sigma^2)\right\}
\le q_{n,d,\sigma}
\le b_{n,d}+\min(u_{n,d},\sigma^2),
\]
and the product benchmark is, by definition,
\[
\ell_{n,d,\sigma}=b_{n,d}+\sigma^2u_{n,d}.
\]
% lean: selectorBenchmark_triangular

For every radius floor \(\underline\sigma>0\), put
\[
D_{\underline\sigma}:=\max\{1,\underline\sigma^{-2}\}.
\]
If \(\underline\sigma\le\sigma\le2\), then \(\sigma^2\ge\underline\sigma^2\), and the frontier and converse benchmarks satisfy
\[
r_{n,d,\sigma}
\le \frac1n+\min(1,u_{n,d})
\le D_{\underline\sigma}
\left\{\frac1n+\sigma^2\min(1,u_{n,d})\right\}
\le D_{\underline\sigma}\omega_{n,d,\sigma}.
\]
Combining this comparison with \cref{obj:thm:two-sided-minimax-bracket-all-d}, which gives
\[
c_0M^2\omega_{n,d,\sigma}
\le \mathsf R_{n,d,\epsilon,M,\sigma}
\le C_0M^2r_{n,d,\sigma},
\]
yields the fixed-radius constants \(c=c_0/D_{\underline\sigma}\) and \(C=C_0\):
\[
cM^2r_{n,d,\sigma}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}
\le
CM^2r_{n,d,\sigma}.
\]
% lean: fixed_radius_minimax_equivalence

For the matching elbows, the benchmark comparison used with the same bracket is
\[
r_{n,d,\sigma}\le (1+K)\omega_{n,d,\sigma}.
\]
When \(u_{n,d}\ge1\),
\[
r_{n,d,\sigma}\le \frac1n+\min\left\{1,\sigma^2+\frac d{n^2}\right\}\le \omega_{n,d,\sigma},
\]
where the last inequality is checked by the two cases \(d/n^2\le1\) and \(d/n^2>1\).  When \(u_{n,d}\le Kb_{n,d}\), the case \(d/n^2\le1\) gives
\[
r_{n,d,\sigma}\le \frac1n+u_{n,d}\le (1+K)b_{n,d}\le(1+K)\omega_{n,d,\sigma},
\]
and the case \(d/n^2>1\) gives \(r_{n,d,\sigma}\le n^{-1}+1\le\omega_{n,d,\sigma}\).  When \(\sigma^2\le Kb_{n,d}\), the case \(d/n^2\le1\) gives
\[
r_{n,d,\sigma}\le b_{n,d}+\sigma^2\le(1+K)b_{n,d}\le(1+K)\omega_{n,d,\sigma},
\]
and the case \(d/n^2>1\) is again \(r_{n,d,\sigma}\le n^{-1}+1\le\omega_{n,d,\sigma}\).  The all-alphabet bracket therefore gives the matching-elbow minimax equivalence with lower constant \(c_0/(1+K)\) and upper constant \(C_0\).
% lean: matching_elbow_minimax_equivalence

For \(n>0\), the logarithmic factor is positive.  Expanding \(u_{n,d}\le b_{n,d}\), multiplying by the positive denominator \(n^2L_n^2\), and collecting terms gives
\[
u_{n,d}\le b_{n,d}
\quad\Longleftrightarrow\quad
d^2\le nL_n^2+dL_n^2 .
\]
% lean: polynomialComponent_le_baseRate_iff

The sufficient elbows are the two displayed consequences of this algebra.  If \(d\le\sqrt n\,L_n\), then \(d^2\le nL_n^2\); if \(d\le L_n^2\), then \(d^2\le dL_n^2\).  In either case the exact algebraic elbow gives \(u_{n,d}\le b_{n,d}\).
% lean: polynomialComponent_le_baseRate_of_le_sqrt; polynomialComponent_le_baseRate_of_le_log_sq

For the residual-wedge implication, write along the sequence \(b_j=b_{j,d_j}\), \(u_j=u_{j,d_j}\), \(s_j=\sigma_j^2\), \(r_j=r_{j,d_j,\sigma_j}\), and \(\omega_j=\omega_{j,d_j,\sigma_j}\).  Since \(r_j/\omega_j\to\infty\), the matching-elbow comparison implies that, for every fixed \(K\ge0\), eventually
\[
u_j<1,\qquad Kb_j<u_j,\qquad Kb_j<s_j.
\]
Taking \(K=\eta^{-1}\) for arbitrary \(\eta>0\) gives \(b_j/u_j\to0\) and \(b_j/s_j\to0\).  To prove \(s_j\to0\), fix \(\alpha>0\), put \(\rho=\sqrt\alpha\), and set \(D=\max\{1,\rho^{-2}\}\).  Divergence makes \(r_j/\omega_j>D\) eventually; on the same tail, \(\sigma_j\ge\rho\) would give \(r_j\le D\omega_j\) by the fixed-radius comparison.  Hence \(0\le\sigma_j<\rho\) eventually, so \(s_j<\alpha\) eventually.

It remains to show \(u_j\to0\).  Fix \(\alpha>0\).  If \(\alpha\ge1\), the eventual inequality \(u_j<1\) gives \(u_j<\alpha\).  If \(0<\alpha<1\), use the eventual bounds \(b_j/u_j<1/2\), \(s_j<\alpha/2\), \(u_j<1\), and \(r_j/\omega_j>\alpha^{-1}\).  At any point on this tail with \(u_j\ge\alpha\), the inequality \(d_j/j^2\le b_j<u_j/2\) and \(s_j<u_j/2\) give \(s_j+d_j/j^2<u_j\) and \(d_j/j^2\le1\).  Consequently
\[
r_j=b_j+s_j,\qquad \omega_j=b_j+s_ju_j,
\]
and \(u_j\ge\alpha\) with \(\alpha<1\) implies \(r_j/\omega_j\le\alpha^{-1}\), a contradiction.  Thus \(u_j<\alpha\) eventually.  Together with the assumed eventual admissibility, these four limits are exactly the residual-wedge conditions.
% lean: benchmark_divergence_implies_residual_wedge

For the diagonal sequence \(d_n=n\) and \(\sigma_n=L_n^{-1/2}\), eventual admissibility follows from \(L_n\ge1\).  Direct simplification gives, for \(n\ge1\),
\[
b_{n,n}=\frac2n,\qquad
u_{n,n}=\frac1{L_n^2},\qquad
\sigma_n^2=\frac1{L_n}.
\]
The logarithmic facts \(L_n\to\infty\) and \(L_n^k/n\to0\) for each fixed \(k\) yield
\[
\frac{b_{n,n}}{u_{n,n}}=\frac{2L_n^2}{n}\to0,\qquad
u_{n,n}=\frac1{L_n^2}\to0,\qquad
\frac{b_{n,n}}{\sigma_n^2}=\frac{2L_n}{n}\to0,\qquad
\sigma_n^2=\frac1{L_n}\to0.
\]
Hence the diagonal sequence lies in the residual wedge.
% lean: diagonal_residual_wedge

Along the same sequence, the exact benchmark formulas are
\[
q_{n,n,L_n^{-1/2}}=\frac1n+\frac1{L_n^2},\qquad
\ell_{n,n,L_n^{-1/2}}=\frac2n+\frac1{L_n^3}.
\]
Since \(L_n^3/n\to0\), these formulas give eventually
\[
\frac13 L_n
\le
\frac{q_{n,n,L_n^{-1/2}}}{\ell_{n,n,L_n^{-1/2}}}
\le
2L_n.
\]
Thus the diagonal benchmark ratio is of order \(L_n\), and because \(L_n\to\infty\), the ratio diverges.
% lean: diagonal_benchmark_eventually_same_order; diagonal_benchmark_ratio_tendsto_atTop
\end{proof}

\le

=

\le
\mathsf R_{n,d,\epsilon,M,\sigma}.

C_{\epsilon}^{\mathrm{up}}M^2 r_{n,d,\sigma}
\le
C_{\epsilon}M^2 r_{n,d,\sigma}.

\[

\le
C_{\epsilon}M^2 r_{n,d,\sigma}.

\begin{proof}[Proof of \cref{obj:thm:published-binary-collision-comparison}]

\[
\min\{1,\min\{u_{n,d},h_{n,d,\sigma}\}\}
\le
\min\{u_{n,d},h_{n,d,\sigma}\}
\le
h_{n,d,\sigma}.

\[
h_{n,d_n,\sigma_n}
=
\sigma_n^2+\frac{d_n}{n^2}
>
0.
\]
Also \(u_{n,d_n}\ge0\), \(h_{n,d_n,\sigma_n}\ge0\), and \(1\ge0\).  Therefore
\[
0
\le
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}

\]
The same minimum inequalities, now using the left branch of the inner minimum, give

\le
\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}
\le
u_{n,d_n}.
\]
Dividing by the positive denominator \(h_{n,d_n,\sigma_n}\) yields, for all sufficiently large \(n\),
\[
0
\le
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}
{h_{n,d_n,\sigma_n}}
\le
\frac{u_{n,d_n}}{h_{n,d_n,\sigma_n}}.
\]
By hypothesis the right-hand side converges to \(0\), so the squeeze theorem gives
\[
\frac{\min\{1,\min\{u_{n,d_n},h_{n,d_n,\sigma_n}\}\}}
{h_{n,d_n,\sigma_n}}
\to0 .
\]
This proves the sequence comparison and completes the proof. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.published_binary_collision_comparison

% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.continuous_ratio_polynomial_upper_all_d
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:radius-channel-converse-all-d}]
Fix \(0<\epsilon<1/2\).  Apply \cref{obj:lem:zeng-binary-one-arm-lower} to obtain one-arm constants
\(a_{\mathrm{rad}},b_0,N_0\) with \(a_{\mathrm{rad}}>0\) and \(b_0>0\).  Put
\[
\tilde b_0:=\min\{b_0,4\},
\qquad
b_{\epsilon}^{\mathrm{rad}}:=\tilde b_0/2,
\qquad
\bar b_{\epsilon}^{\mathrm{rad}}:=b_0,
\qquad
c_{\mathrm{rad}}:=a_{\mathrm{rad}}\tilde b_0^2/128,
\]
and choose
\[
N_{\mathrm{rad},0}:=\max\{3,N_0,\lceil 2/\tilde b_0\rceil\}.
\]
Then
\[
0<a_{\mathrm{rad}},
\qquad
0<b_{\epsilon}^{\mathrm{rad}}<\bar b_{\epsilon}^{\mathrm{rad}},
\qquad
0<c_{\mathrm{rad}}.
\]
For every \(n,d\ge1\) with \(N_{\mathrm{rad},0}\le n\), the radial cap
\[
d_{\mathrm{rad}}
=
\min\{d,\max(1,\lfloor b_{\epsilon}^{\mathrm{rad}}n\log(en)\rfloor)\}
\]
is positive, is at most \(d\), and satisfies
\[
d_{\mathrm{rad}}
\le
\bar b_{\epsilon}^{\mathrm{rad}}n\log n .
\]
The same construction gives the source-risk interface used by the channel transport: for every measurable uniformly bounded source estimator \(\widehat\psi\) on the \(n\)-sample binary one-arm experiment over \(d_{\mathrm{rad}}\) cells, some control-zero source law \(P\) satisfies
\[
\frac{a_{\mathrm{rad}}}{2}
\left\{\frac1n+\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}\right\}
\le
E_{Q_P^{(n)}}
\left[
\{\widehat\psi-\psi(P)\}^2
\right].
\]
Finally, \cref{obj:lem:capped-radial-transport-scale}, applied with cap constant \(\tilde b_0=2b_{\epsilon}^{\mathrm{rad}}\), gives for every \(M\) and \(\sigma\)
\[
c_{\mathrm{rad}}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\left(\frac{M\sigma}{2}\right)^2
\frac{a_{\mathrm{rad}}}{2}
\left\{\frac1n+\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}\right\}.
\]
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exists_capped_radial_transport_package
Choose \(b_{\epsilon}^{\mathrm{ex}},c_{\mathrm{ex}}>0\) and \(N_{\mathrm{ex}}\) from \cref{obj:lem:capped-exact-transport-package}.  Let \(c_{\mathrm{base}}>0\) be the constant supplied by \cref{obj:lem:scaled-binary-exact-lower-transfer-all-d}. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exists_capped_exact_transport_package
% lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.scaled_binary_exact_lower_transfer_all_d

Set
\[
N_{\mathrm{rad}}:=\max\{N_{\mathrm{rad},0},1\},
\qquad
c_{\mathrm{small}}:=\frac{1}{800N_{\mathrm{rad}}},
\qquad
c_{\mathrm{tr}}:=\min\{c_{\mathrm{rad}},c_{\mathrm{small}},c_{\mathrm{ex}}\},
\]
and define
\[
c_\epsilon
:=
\min\left\{1,\frac12\min\{c_{\mathrm{base}},c_{\mathrm{tr}}\}\right\}.
\]
All constants in this display depend on \(\epsilon\) alone, and the displayed positivity gives \(0<c_\epsilon\le1\) and \(b_{\epsilon}^{\mathrm{rad}}>0\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radius_channel_converse_all_d

Fix \(n,d,M,\sigma\) with \(0<n\), \(0<d\), \(1\le M\), and \(0\le\sigma\le2\).  Form the two capped alphabets \(d_{\mathrm{rad}}\) and
\[
d_{\mathrm{ex}}
=
\min\{d,\max(1,\lfloor b_{\epsilon}^{\mathrm{ex}}n^2\rfloor)\}.
\]
Both caps are positive and at most \(d\).  Following \cref{obj:def:least-favorable-handle}, pad each source alphabet into the first target cells.  On the radial source, apply the hypothesis-independent Bernoulli channel
\[
B'(a)\mid B(a)
\sim
\operatorname{Bernoulli}\!\left\{\frac12+\frac{\sigma}{2}\left(B(a)-\frac12\right)\right\},
\qquad
Y(a)=M\{B'(a)-1/2\}.
\]
On the exact-homogeneity source, use the affine map \(Y(a)=M\{B(a)-1/2\}\).  The construction preserves the source cell masses on the padded cells, assigns zero mass outside them, realizes the independent full-data couplings, and gives the radial full-data channel of \cref{obj:synth_5}.  For a radial source law \(P\), write \(\eta_k=E_P[B(1)\mid A=1,X=k]\) on a positive source cell and \(\psi(P)=\sum_k p_k\eta_k\).  On each padded positive-mass cell the embedded means are
\[
\mu_{0k}=-\frac{M\sigma}{4},
\qquad
\mu_{1k}=\frac{M\sigma}{2}\left(\eta_k-\frac12\right),
\qquad
\tau_k=\mu_{1k}-\mu_{0k}=\frac{M\sigma}{2}\eta_k,
\]
and
\[
\tau(\Psi(P))=\frac{M\sigma}{2}\psi(P).
\]
Thus the overlap bounds are inherited from the source, the mean-normalization bounds in \cref{obj:ass:mean-normalization} follow from \(0\le\eta_k\le1\) and \(0\le\sigma\le2\), the conditional second-moment bound in \cref{obj:ass:second-central-moment} follows from \(Y(a)\in\{-M/2,M/2\}\), and the radius condition in \cref{obj:ass:approximate-homogeneity} follows from
\[
|\tau_k-\tau(\Psi(P))|
=
\frac{M\sigma}{2}|\eta_k-\psi(P)|
\le \sigma M .
\]
Together with consistency and conditional exchangeability from the independent full-data coupling, these checks place every radial embedding in \(\mathcal P_{d,\epsilon,M,\sigma}\) as defined in \cref{obj:def:model-class}.  Along the same embedding,
\[
\tau(\Psi(P_1))-\tau(\Psi(P_0))
=
M\frac{\sigma}{2}\{\psi(P_1)-\psi(P_0)\}.
\]
The exact affine embedding has target slope \(M\). % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.radialPaddedAffineLaw_model

The exact capped package supplies, for this \((n,d)\), a source level \(L_{\mathrm{ex}}\) such that every measurable exact-source estimator is defeated by some exactly homogeneous binary source law and
\[
c_{\mathrm{ex}}M^2
\left\{\frac1n+\min\left(1,\frac d{n^2}\right)\right\}
\le
M^2L_{\mathrm{ex}} .
\]
Since \(c_{\mathrm{tr}}\le c_{\mathrm{ex}}\), deterministic affine transport gives, for every \(\widehat\tau\in\mathcal T_{n,M}\), an exact source law \(P\) satisfying
\[
c_{\mathrm{tr}}M^2
\left\{\frac1n+\min\left(1,\frac d{n^2}\right)\right\}
\le
\operatorname{MSE}_{\Phi_M(P)}(\widehat\tau),
\]
where \(\operatorname{MSE}\) is the mean-squared error of \cref{obj:synth_7}.  This is the exact half of the common two-family transfer certificate. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.exactRiskTransfer_of_affine_transport

For the radial half, let \(K\) be the one-observation Markov kernel that maps a binary observed record \((X,A,B)\) to \((\mathrm{pad}(X),A,M(B'-1/2))\) using the Bernoulli channel above.  For every radial source law \(P\), the observed target law satisfies
\[
(\Psi(P))_O=K\circ P_O,
\]
and the target identity is
\[
\tau(\Psi(P))=\frac{M\sigma}{2}\psi(P).
\]
Consequently, whenever \(\sigma>0\), the nonzero scale \(M\sigma/2\), the preceding observed-law identity, and the source-risk interface imply the following transport rule: if for every measurable uniformly bounded \(\widehat\psi\) some source law \(P\) satisfies
\[
L\le
E_{Q_P^{(n)}}\left[\{\widehat\psi-\psi(P)\}^2\right]
\]
and if
\[
c\,M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\left(\frac{M\sigma}{2}\right)^2L,
\]
then every \(\widehat\tau\in\mathcal T_{n,M}\) has some radial source law \(P\) with
\[
c\,M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\operatorname{MSE}_{\Psi(P)}(\widehat\tau).
\]
This is the Markov-kernel comparison applied to the radial channel.  The finite-sample source interface used when the radial cutoff is unavailable is obtained from the one-arm parametric subexperiment.  In the binary one-arm experiment over \(d_{\mathrm{rad}}\) cells, take the two control-zero source laws whose treated functionals are \(1/2\) and \(1/2+\delta\), where \(\delta=(2/5)/\sqrt n\).  The usual product total-variation calculation gives distance at most \(1/2\) between their \(n\)-sample observed laws, so the standard two-point minimax inequality gives
\[
\frac{1}{100n}
\le
\inf_{\widetilde\psi}
\sup_{R}
E_{Q_R^{(n)}}
\left[
\{\widetilde\psi-\psi(R)\}^2
\right],
\]
where \(R\) ranges over the control-zero one-arm source laws on \(d_{\mathrm{rad}}\) cells.  Since \(1/(200n)\) is strictly below this minimax value, the hard-source extraction from the minimax infimum and supremum gives, for every measurable uniformly bounded source estimator \(\widehat\psi\), a control-zero source law \(P\) with
\[
\frac{1}{200n}
\le
E_{Q_P^{(n)}}
\left[
\{\widehat\psi-\psi(P)\}^2
\right].
\]
If \(\sigma=0\), the desired radial lower-bound term is zero; this finite-sample source law and nonnegativity of squared loss give the required estimator-wise inequality.  If \(\sigma>0\) and \(N_{\mathrm{rad},0}\le n\), the large-sample source-risk interface and the scale comparison displayed above give
\[
c_{\mathrm{tr}}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\left(\frac{M\sigma}{2}\right)^2
\frac{a_{\mathrm{rad}}}{2}
\left\{\frac1n+\frac{d_{\mathrm{rad}}^2}{n^2\log^2 n}\right\}.
\]
The transport rule therefore gives the radial estimator-wise lower bound in the large-sample branch.  If \(\sigma>0\) and \(n<N_{\mathrm{rad},0}\), then \(1\le n\le N_{\mathrm{rad}}\).  The finite-sample source interface just displayed, together with \cref{obj:lem:finite-sample-radial-transport-scale} and \(c_{\mathrm{tr}}\le c_{\mathrm{small}}\), gives the transported radial lower term in the small-sample branch.  Hence for every \(\widehat\tau\in\mathcal T_{n,M}\) there is a radial source law \(P\) with
\[
c_{\mathrm{tr}}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\operatorname{MSE}_{\Psi(P)}(\widehat\tau).
\]

Combining the radial and exact estimator-wise statements gives the common transfer certificate with constant \(c_{\mathrm{tr}}\).  Since \(c_\epsilon\le c_{\mathrm{tr}}\), the same certificate holds with \(c_\epsilon\).  The radial construction also carries the product data-processing certificate: writing \(\operatorname{TV}\) for total variation distance, for all radial source laws \(P_0,P_1\),
\[
\operatorname{TV}\!\left(Q_{\Psi(P_0)}^{(n)},Q_{\Psi(P_1)}^{(n)}\right)
\le
\operatorname{TV}\!\left(Q_{P_0}^{(n)},Q_{P_1}^{(n)}\right),
\]
because the left product laws are obtained from the right product laws by the coordinatewise common kernel \(K\).  Thus the cap identity, injective padding, source hardness, mass preservation, zero extension, full-data coupling, full-data channel law, membership in \(\mathcal P_{d,\epsilon,M,\sigma}\), radius bound, product data-processing bound, target scaling, and radial estimator-wise lower bound package into the radial certificate asserted in the theorem.

For the minimax lower bound, \cref{obj:lem:scaled-binary-exact-lower-transfer-all-d} gives
\[
c_{\mathrm{base}}M^2
\left\{\frac1n+\min\left(1,\frac d{n^2}\right)\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The radial half of the transfer certificate, after embedding into \(\mathcal P_{d,\epsilon,M,\sigma}\) and taking the minimax infimum in \cref{obj:def:minimax-risk}, gives
\[
c_{\mathrm{tr}}M^2\sigma^2
\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
For two nonnegative terms bounded above by the same risk in this way,
\[
\frac12\min\{c_{\mathrm{base}},c_{\mathrm{tr}}\}M^2
\left\{
\frac1n+\min\left(1,\frac d{n^2}\right)
+\sigma^2\min\left(1,\frac{d^2}{n^2\log^2(en)}\right)
\right\}
\le
\mathsf R_{n,d,\epsilon,M,\sigma}.
\]
The definition of \(c_\epsilon\) yields the displayed converse rate with constant \(c_\epsilon\), completing the proof. % lean: CausalSmith.Stat.DiscreteAteHeterogeneityFrontier.converseRate_lower_of_exact_transfer_and_handle
