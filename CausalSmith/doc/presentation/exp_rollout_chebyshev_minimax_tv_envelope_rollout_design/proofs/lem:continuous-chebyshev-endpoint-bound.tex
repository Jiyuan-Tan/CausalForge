\begin{proof}[Proof of \cref{obj:lem:continuous-chebyshev-endpoint-bound}]
Put
\[
\bar{x}=\frac{2}{q_{\max}}-1,
\qquad
\bar{\lambda}=\bar{x}+\sqrt{\bar{x}^2-1}.
\]
Since \(0<q_{\max}<1\) we have \(2/q_{\max}>2\), hence \(\bar{x}>1\); and since \(\sqrt{\bar{x}^2-1}\ge0\) this gives \(\bar{\lambda}\ge \bar{x}>1\). Choose
\[
C(q_{\max})=2,
\qquad
c(q_{\max})=\frac12(1-\bar{\lambda}^{-1})^2 .
\]
Both are positive: \(\bar{\lambda}>1\) gives \(0<\bar{\lambda}^{-1}<1\), so \(1-\bar{\lambda}^{-1}>0\). % lean: continuous_chebyshev_endpoint_bound

Now let \(\beta\ge1\) and \(0<q\le q_{\max}\), and write
\[
x_q=\frac2q-1,
\qquad
\lambda(q)=x_q+\sqrt{x_q^2-1}.
\]
Then \(q\le q_{\max}<1\), so exactly as above \(x_q>1\) and \(\lambda(q)>1\). Moreover \(q\le q_{\max}\) gives \(2/q_{\max}\le 2/q\) and therefore
\[
1\le \bar{x}\le x_q ,
\]
and on \([1,\infty)\) the map \(t\mapsto t+\sqrt{t^2-1}\) is nondecreasing, since it equals \(e^{\operatorname{arcosh} t}\) and both \(\operatorname{arcosh}\) and the exponential are nondecreasing. Hence
\[
\bar{\lambda}\le \lambda(q).
\] % lean: endpoint_lambda_mono

\emph{Chebyshev evaluation at the exterior point.}
Write \(x_q=\operatorname{cosh}a\) with \(a=\operatorname{arcosh}(x_q)\ge0\), which is legitimate because \(x_q\ge1\). Then \(T_\beta(\operatorname{cosh}a)=\operatorname{cosh}(\beta a)\), while \(e^{a}=x_q+\sqrt{x_q^2-1}=\lambda(q)\), so
\[
T_\beta(x_q)
=
\operatorname{cosh}(\beta a)
=
\frac{e^{\beta a}+e^{-\beta a}}{2}
=
\frac{\lambda(q)^\beta+\lambda(q)^{-\beta}}2 .
\] % lean: chebyshev_eval_eq_lambda_average

\emph{Upper bound.}
Let \(R\) be a real polynomial with \(\deg(R)\le\beta\) and \(|R(x)|\le1\) for \(x\in[-1,1]\). By \cref{obj:lem:chebyshev-exterior-extremal}, applied at the exterior point \(x_q>1\),
\[
|R(x_q)|\le T_\beta(x_q).
\] % lean: chebyshev_exterior_extremal
Also \(-1\in[-1,1]\), so \(|R(-1)|\le1\), and therefore
\[
|R(x_q)-R(-1)|
\le |R(x_q)|+|R(-1)|
\le T_\beta(x_q)+1.
\] % lean: continuous_chebyshev_endpoint_bound
Since \(\lambda(q)\ge1\) and \(\beta\ge1\) we have \(\lambda(q)^{-\beta}\le1\le\lambda(q)^{\beta}\), whence
\[
\frac{\lambda(q)^\beta+\lambda(q)^{-\beta}}2+1
\le
\frac{\lambda(q)^\beta+\lambda(q)^{\beta}}2+\lambda(q)^\beta
=
2\lambda(q)^\beta.
\] % lean: chebyshev_lambda_average_add_one_le
Combining the last three displays proves
\[
|R(x_q)-R(-1)|\le 2\lambda(q)^\beta
= C(q_{\max})\lambda(q)^\beta.
\]

\emph{Lower bound.}
The same evaluation formula and the algebraic identity
\[
\frac{\lambda(q)^\beta+\lambda(q)^{-\beta}}2-1
=
\frac12\left(1-\lambda(q)^{-\beta}\right)^2\lambda(q)^\beta
\]
(valid because \(\lambda(q)^\beta>0\); expand the right-hand side) reduce the claim to a monotonicity step. Since \(1<\bar{\lambda}\le\lambda(q)\) and \(\beta\ge1\),
\[
\bar{\lambda}\le\lambda(q)\le\lambda(q)^\beta ,
\qquad\text{hence}\qquad
\lambda(q)^{-\beta}\le\bar{\lambda}^{-1},
\]
so that \(0\le 1-\bar{\lambda}^{-1}\le 1-\lambda(q)^{-\beta}\) and therefore \((1-\bar{\lambda}^{-1})^2\le\left(1-\lambda(q)^{-\beta}\right)^2\). Multiplying by \(\tfrac12\lambda(q)^\beta>0\) gives
\[
T_\beta(x_q)-1
=
\frac{\lambda(q)^\beta+\lambda(q)^{-\beta}}2-1
\ge
\frac12(1-\bar{\lambda}^{-1})^2\lambda(q)^\beta
=
c(q_{\max})\lambda(q)^\beta.
\] % lean: chebyshev_lambda_average_sub_one_ge
This proves both assertions. % lean: continuous_chebyshev_endpoint_bound
\end{proof}
