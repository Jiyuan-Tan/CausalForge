\begin{proof}[Proof of \cref{obj:lem:block-prior-hellinger-bound}]
Let
\[
s:=B/2,
\qquad
\delta:=B\min\{(2H_{\beta,p})^{-1},(4\pi)^{-1}\}
  \min\left\{1,\sqrt{\frac{A_d}{m}}\right\},
\]
and, for an assignment \(z\), write
\[
r_b(z):=h_d(z_{V_b}) .
\]
For the two conditional block-coordinate densities define
\[
C_z^+(u):=\prod_{b=1}^{m} f_s\bigl(u_b-\delta r_b(z)\bigr),
\qquad
C_z^-(u):=\prod_{b=1}^{m} f_s\bigl(u_b+\delta r_b(z)\bigr),
\]
so that
\[
\pi_{+1}(z,u)=P_Z(z)C_z^+(u),
\qquad
\pi_{-1}(z,u)=P_Z(z)C_z^-(u).
\]
% lean: blockPrior_hellinger_le

Since \(P_Z(z)\ge0\),
\[
\bigl(\sqrt{\pi_{+1}(z,u)}-\sqrt{\pi_{-1}(z,u)}\bigr)^2
=
P_Z(z)\bigl(\sqrt{C_z^+(u)}-\sqrt{C_z^-(u)}\bigr)^2 .
\]
Applying product Fubini over the counting measure in \(z\) and Lebesgue measure in \(u\) gives
\[
H^2(\pi_{+1},\pi_{-1})
=
\sum_{z\in\{0,1\}^{V_n}}
P_Z(z)\,
H^2(C_z^+,C_z^-).
\]
% lean: blockPriorDensity_nonneg
% lean: blockPriorDensity_integrable
% lean: blockPrior_hellinger_le

We next bound the conditional Hellinger distance.  For fixed \(z\), set
\[
f_b(u):=f_s(u-\delta r_b(z)),
\qquad
g_b(u):=f_s(u+\delta r_b(z)).
\]
The coordinate densities are nonnegative, integrable, and normalized:
\[
f_b(u)\ge0,\qquad g_b(u)\ge0,\qquad
\int f_b(u)\,du=\int g_b(u)\,du=1 .
\]
% lean: cosSqDensity_nonneg
% lean: cosSqDensity_translate_integrable
% lean: cosSqDensity_translate_integral_one

Let
\[
\alpha_b:=\int_{\mathbb R}\sqrt{f_b(u)g_b(u)}\,du .
\]
The identity
\[
H^2(C_z^+,C_z^-)
=
2\left(1-\prod_{b=1}^{m}\alpha_b\right)
\]
follows by expanding
\((\sqrt{\prod_b f_b}-\sqrt{\prod_b g_b})^2\), using
\(\sqrt{\prod_b f_b g_b}=\prod_b\sqrt{f_b g_b}\), and applying finite-product Fubini.  Moreover \(0\le\alpha_b\le1\), where the upper bound is Cauchy--Schwarz together with the two normalization identities.  Therefore
\[
H^2(C_z^+,C_z^-)
\le
2\sum_{b=1}^{m}(1-\alpha_b).
\]
% lean: hellingerSqDensity_pi_eq_two_mul_one_sub_affinity
% lean: densityAffinity_pi
% lean: one_sub_prod_le_sum
% lean: conditional_block_hellinger_le

For any \(a,b\in\mathbb R\), the cosine translate affinity obeys
\[
1-\int_{\mathbb R}\sqrt{f_s(u-a)f_s(u-b)}\,du
\le
\frac{\pi^2(a-b)^2}{8s^2}.
\]
Indeed, if \(s\le |a-b|\), the left side is at most \(1\), while
\(\pi^2(a-b)^2/(8s^2)\ge1\).  If \(|a-b|<s\), order the shifts as \(x\le y\), put
\[
t:=y-x,
\qquad
q:=\frac{\pi t}{2s},
\]
and compute the overlap integral as
\[
\int\sqrt{f_s(u-x)f_s(u-y)}\,du
=
\left(1-\frac{t}{2s}\right)\cos q+\frac{\sin q}{\pi}.
\]
Since \(0\le q<\pi/2\) and \(q\le\tan q\), the right-hand side is at least \(\cos q\).  Thus
\[
1-\int\sqrt{f_s(u-x)f_s(u-y)}\,du
\le
1-\cos q
\le
\frac{q^2}{2}
=
\frac{\pi^2(x-y)^2}{8s^2},
\]
and symmetry gives the displayed bound. % lean: cosSqDensity_affinity_formula
% lean: cosSqDensity_affinity_defect_sharp

Applying this with \(a=\delta r_b(z)\) and \(b=-\delta r_b(z)\) yields
\[
1-\alpha_b
\le
\frac{\pi^2(2\delta r_b(z))^2}{8s^2}.
\]
Therefore, since \(s=B/2\),
\[
H^2(C_z^+,C_z^-)
\le
2\sum_{b=1}^{m}
\frac{\pi^2(2\delta r_b(z))^2}{8s^2}
=
\frac{4\pi^2\delta^2}{B^2}
\sum_{b=1}^{m} r_b(z)^2 .
\]
% lean: conditional_block_hellinger_le

Substituting the conditional bound into the outer sum gives
\[
H^2(\pi_{+1},\pi_{-1})
\le
\frac{4\pi^2\delta^2}{B^2}
\sum_{z\in\{0,1\}^{V_n}}P_Z(z)
\sum_{b=1}^{m}h_d(z_{V_b})^2 .
\]
% lean: blockPrior_hellinger_le

It remains to evaluate the Bernoulli average.  The restriction \(Z_{V_b}\) has the \(d\)-coordinate product Bernoulli law with probability \(p\), because \(P_Z\) is the product law from \cref{obj:ass:bernoulli-design}.  By the energy identity in \cref{obj:lem:block-energy-representer},
\[
\mathbb E\!\left[h_d(Z_{V_b})^2\right]=A_d^{-1}.
\]
Summing over the \(m\) blocks gives
\[
\sum_{z\in\{0,1\}^{V_n}}P_Z(z)
\sum_{b=1}^{m}h_d(z_{V_b})^2
=
\frac{m}{A_d}.
\]
% lean: E_blockAssignment
% lean: E_sum_blockRepresenter_sq
% lean: sum_assignmentMass_mul_sum_blockRepresenter_sq

Combining the last two displays,
\[
H^2(\pi_{+1},\pi_{-1})
\le
\frac{4\pi^2\delta^2}{B^2}\frac{m}{A_d}
=
\frac{4\pi^2m\delta^2}{B^2A_d}.
\]
This is the claimed bound. % lean: blockPrior_hellinger_le
\end{proof}
