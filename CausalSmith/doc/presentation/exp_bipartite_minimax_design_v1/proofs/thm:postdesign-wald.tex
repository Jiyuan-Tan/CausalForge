\begin{proof}[Proof of \cref{obj:thm:postdesign-wald}]
The eventual identity \(|O_n|=n\) implies that \(|O_n|\to\infty\); in particular \(|O_n|>0\) on an eventual tail.

\begin{enumerate}
\item Fix \(n\).  The admissibility condition \(\epsilon_n\in(0,1/2)\) and the feasibility floor give
\[
0<\epsilon_n\le p_{n,k}\le 1-\epsilon_n<1
\qquad(k\in I_n).
\]
This display supplies the assignment-probability hypothesis \(0<p_{n,k}<1\) in \cref{obj:thm:hetero-envelope}.  Together with \cref{obj:ass:independent-heterogeneous-bernoulli} and \cref{obj:ass:bounded-outcomes}, the second conclusion of \cref{obj:thm:hetero-envelope} gives
\[
\sigma^2_{G_n,p_n}(Y)\le V_{\mathrm{env}}(G_n,p_n).
\]
By \cref{obj:def:conservative-variance-estimator},
\(\widehat V_{\mathrm{cons}}(G_n,p_n)=V_{\mathrm{env}}(G_n,p_n)\), and hence
\[
\sigma^2_{G_n,p_n}(Y)\le \widehat V_{\mathrm{cons}}(G_n,p_n).
\]
Since \(n\) was arbitrary, this proves the variance-domination assertion for every \(n\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.hetero_envelope

\item By \cref{obj:ass:variance-nondegenerate} there are \(c>0\) and an eventual tail on which
\[
c\le \sigma^2_{G_n,p_n}(Y).
\]
Hence \(\sigma^2_{G_n,p_n}(Y)>0\) eventually.  The eventual identity
\(|O_n|=n\) also gives the positive normalization \(|O_n|>0\) eventually.
Under the assumptions listed in \cref{obj:thm:hetero-clt} -- outcome indexing,
assignment and interference, bounded outcomes, bounded degree, bounded overlap,
feasibility, optimality \(p_n=p_n^*(G_n)\), uniform positivity, and variance
nondegeneracy -- \cref{obj:thm:hetero-clt} gives, for every \(s\in\mathbb R\),
\[
\Pr_{p_n}\!\left(
\frac{\sqrt n\,[\widehat\tau_H(p_n)-\tau_n]}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}}\le s
\right)
\longrightarrow \Phi(s).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.hetero_clt

\item Write
\[
W_n=
\frac{\sqrt n\,[\widehat\tau_H(p_n)-\tau_n]}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}},
\qquad
z=z_{1-\alpha_{\mathrm{cov}}/2}.
\]
Since \(z\ge0\), the event \(\{W_n\le -z\}\) is contained in \(\{W_n\le z\}\), and the finite-design probability split gives
\[
\Pr_{p_n}\!\left(-z<W_n\le z\right)
=\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right).
\]
The convergence supplied by \cref{obj:thm:hetero-clt} at \(s=z\) and \(s=-z\), together with
\(\Phi(-z)=1-\Phi(z)\) and the calibration \(\Phi(z)=1-\alpha_{\mathrm{cov}}/2\), yields
\[
\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right)
\longrightarrow
\Phi(z)-\Phi(-z)
=1-\alpha_{\mathrm{cov}} .
\]
On the eventual tail where \(|O_n|\), \(n\), and \(\sigma^2_{G_n,p_n}(Y)\) are positive, the strict interval event is contained in the Wald coverage event.  Indeed, if
\(-z<W_n\le z\), then \(|W_n|\le z\), so
\[
\sqrt n\,|\widehat\tau_H(p_n)-\tau_n|
\le z\sqrt{\sigma^2_{G_n,p_n}(Y)}.
\]
Dividing by \(\sqrt n>0\) and using the variance-domination bound
\(\sigma^2_{G_n,p_n}(Y)\le \widehat V_{\mathrm{cons}}(G_n,p_n)\), monotonicity of the square root, and \(z\ge0\), gives
\[
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z
\sqrt{\frac{\sigma^2_{G_n,p_n}(Y)}{n}}
\le
z
\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}.
\]
Consequently, on that tail,
\[
\Pr_{p_n}\!\left(
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}
\right)
\ge
\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right).
\]
Taking lower limits and using the displayed convergence gives
\[
1-\alpha_{\mathrm{cov}}
\le
\liminf_{n\to\infty}
\Pr_{p_n}\!\left(
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z_{1-\alpha_{\mathrm{cov}}/2}
\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}
\right),
\]
which is the claimed coverage bound.
% lean: Causalean.Experimentation.DesignBased.conservative_wald_liminf_of_studentized_cdf
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.postdesign_wald
\end{enumerate}
\end{proof}
