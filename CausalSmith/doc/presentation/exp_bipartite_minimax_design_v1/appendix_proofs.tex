\begin{proof}[Proof of \cref{obj:prop:homogeneous-reduction}]
Recall the three covariance loads. For \(i,j\in O_n\) and an assignment-probability vector \(p=(p_k)_{k\in I_n}\) with \(0<p_k<1\),
\[
r_{ij}^1(G_n,p)
=\mathbf 1\{|S_{ij}(G_n)|>0\}
\left(\prod_{k\in S_{ij}(G_n)}p_k^{-1}-1\right),
\qquad
r_{ij}^0(G_n,p)
=\mathbf 1\{|S_{ij}(G_n)|>0\}
\left(\prod_{k\in S_{ij}(G_n)}(1-p_k)^{-1}-1\right),
\]
\[
r_{ij}^{10}(G_n)=\mathbf 1\{|S_{ij}(G_n)|>0\}.
\]
Throughout the proof, \(p\) denotes the probability vector and \(\rho\in(0,1)\) denotes the common homogeneous coordinate, so \(p_k=\rho\) for every \(k\in I_n\). We write \(\rho^{-\ell}\) as shorthand for \((\rho^{\ell})^{-1}\), and similarly for \((1-\rho)^{-\ell}\). We also adopt throughout the reciprocal convention
\[
1/0=0,
\]
so that the symbol \(1/n\), with \(n=\lvert O_n\rvert\), is defined for every finite outcome-unit set: it is the ordinary reciprocal when \(O_n\neq\varnothing\) and equals \(0\) when \(O_n=\varnothing\). Every identity displayed below is asserted under this convention, so the degenerate case \(O_n=\varnothing\) needs no separate hypothesis.

\begin{enumerate}
\item For every \(k\in I_n\), homogeneity and \(\rho\in(0,1)\) give
\[
0<p_k=\rho<1,
\]
so all exposure probabilities \(\pi_i^1(p)=\prod_{k\in N_i(G_n)}p_k\) and \(\pi_i^0(p)=\prod_{k\in N_i(G_n)}(1-p_k)\) are strictly positive and the reciprocals above are defined.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.homogeneous_reduction

\item Fix \(i,j\in O_n\). If \(\lvert S_{ij}(G_n)\rvert>0\), substituting \(p_k=\rho\) into the displayed definition of the treated load yields
\[
r_{ij}^1(G_n,p)
=\prod_{k\in S_{ij}(G_n)}\rho^{-1}-1
=(\rho^{\lvert S_{ij}(G_n)\rvert})^{-1}-1
=\rho^{-\lvert S_{ij}(G_n)\rvert}-1,
\]
because the product has \(\lvert S_{ij}(G_n)\rvert\) equal factors. If \(\lvert S_{ij}(G_n)\rvert=0\), the indicator in that definition vanishes and \(r_{ij}^1(G_n,p)=0\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.homogeneous_reduction

\item The same split applies to the control load, with \(\rho\) replaced by \(1-\rho\). When
\(\lvert S_{ij}(G_n)\rvert>0\),
\[
r_{ij}^0(G_n,p)
=\prod_{k\in S_{ij}(G_n)}(1-\rho)^{-1}-1
=((1-\rho)^{\lvert S_{ij}(G_n)\rvert})^{-1}-1
=(1-\rho)^{-\lvert S_{ij}(G_n)\rvert}-1;
\]
when \(\lvert S_{ij}(G_n)\rvert=0\), it equals \(0\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.homogeneous_reduction

\item It remains to prove the variance-scale identity. Write, for \(i\in O_n\),
\[
\delta_i^1=Y_i^1-\mu_1,\qquad \delta_i^0=Y_i^0-\mu_0,
\qquad
A_i=\frac{T_i(Z)}{\pi_i^1(p)}-1,
\qquad
B_i=\frac{C_i(Z)}{\pi_i^0(p)}-1,
\]
for the centered potential-outcome deviations and the centered inverse-probability weights, so that \cref{obj:def:first-order-linearization} reads \(\eta_i(p,Z)=A_i\delta_i^1-B_i\delta_i^0\). Under \cref{obj:ass:independent-heterogeneous-bernoulli} the coordinates of \(Z\) are independent, so
\[
\mathbb E_p\!\left[T_i(Z)\right]=\prod_{k\in N_i(G_n)}p_k=\pi_i^1(p),
\qquad
\mathbb E_p\!\left[C_i(Z)\right]=\prod_{k\in N_i(G_n)}(1-p_k)=\pi_i^0(p),
\]
both strictly positive by the coordinate bounds above; hence \(\mathbb E_p[A_i]=\mathbb E_p[B_i]=0\) and
\[
\mathbb E_p\!\left[\eta_i(p,Z)\right]=0 .
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.linScore_mean_zero

Unfolding \cref{obj:def:variance-scale} and applying the finite-design variance formula to the linear combination \(\sum_{i\in O_n}\frac1n\eta_i(p,Z)\) gives
\[
\sigma_{G_n,p}^2(Y)
= n\sum_{i\in O_n}\sum_{j\in O_n}
 \frac1n\cdot\frac1n\,
 \operatorname{Cov}_p\!\left(\eta_i(p,Z),\eta_j(p,Z)\right).
\]
The centering just proved turns each covariance into the corresponding raw pair moment. If \(O_n\neq\varnothing\), then \(n>0\) and the scalar identity \(n\cdot(1/n)^2=1/n\) applies; if \(O_n=\varnothing\), both the displayed right-hand side and the double sum below are empty, hence \(0\), and the two sides again agree under the reciprocal convention. In either case
\[
\sigma_{G_n,p}^2(Y)
=\frac1n\sum_{i\in O_n}\sum_{j\in O_n}
\mathbb E_p\!\left[\eta_i(p,Z)\eta_j(p,Z)\right].
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.varScale_pair_moments

Next evaluate those pair moments. For any random variables \(U,V\) with \(\mathbb E_p[U]=\theta_U\neq0\) and \(\mathbb E_p[V]=\theta_V\neq0\),
\(\mathbb E_p[(U/\theta_U-1)(V/\theta_V-1)]=\mathbb E_p[UV]/(\theta_U\theta_V)-1\). Independence gives
\(\mathbb E_p[T_i(Z)T_j(Z)]=\prod_{k\in N_i(G_n)\cup N_j(G_n)}p_k\), while
\(\pi_i^1(p)\pi_j^1(p)=\bigl(\prod_{k\in N_i(G_n)\cup N_j(G_n)}p_k\bigr)\bigl(\prod_{k\in S_{ij}(G_n)}p_k\bigr)\), so
\[
\mathbb E_p\!\left[A_iA_j\right]
=\prod_{k\in S_{ij}(G_n)}p_k^{-1}-1
=r_{ij}^1(G_n,p).
\]
The same argument with \(1-p_k\) in place of \(p_k\) gives
\[
\mathbb E_p\!\left[B_iB_j\right]
=\prod_{k\in S_{ij}(G_n)}(1-p_k)^{-1}-1
=r_{ij}^0(G_n,p),
\]
with the empty-overlap case reducing to \(1-1=0\), the value selected by the indicators in \(r_{ij}^1\) and \(r_{ij}^0\). For the mixed moments, if \(S_{ij}(G_n)\neq\varnothing\), then \(T_i(Z)C_j(Z)=0\) identically, since a shared intervention cannot be both treated and untreated, so \(\mathbb E_p[A_iB_j]=-1\). If \(S_{ij}(G_n)=\varnothing\), then the two exposure events involve disjoint assignment coordinates, and independence gives \(\mathbb E_p[T_i(Z)C_j(Z)]=\pi_i^1(p)\pi_j^0(p)\), hence \(\mathbb E_p[A_iB_j]=0\). In both cases, and likewise with the roles of the two outcome units exchanged,
\[
\mathbb E_p\!\left[A_iB_j\right]=\mathbb E_p\!\left[B_iA_j\right]=-r_{ij}^{10}(G_n).
\]
Expanding \(\eta_i(p,Z)\eta_j(p,Z)=A_iA_j\delta_i^1\delta_j^1-A_iB_j\delta_i^1\delta_j^0-B_iA_j\delta_i^0\delta_j^1+B_iB_j\delta_i^0\delta_j^0\) and inserting the four displayed moments gives
\[
\mathbb E_p\!\left[\eta_i(p,Z)\eta_j(p,Z)\right]
=r_{ij}^1(G_n,p)\delta_i^1\delta_j^1
+r_{ij}^0(G_n,p)\delta_i^0\delta_j^0
+r_{ij}^{10}(G_n)\left(\delta_i^1\delta_j^0+\delta_i^0\delta_j^1\right).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.linScore_pair_moment

Finally, \(S_{ij}(G_n)=S_{ji}(G_n)\), so \(r_{ij}^{10}(G_n)=r_{ji}^{10}(G_n)\); relabelling \((i,j)\mapsto(j,i)\) in the double sum gives
\[
\sum_{i\in O_n}\sum_{j\in O_n}r_{ij}^{10}(G_n)\delta_i^0\delta_j^1
=\sum_{i\in O_n}\sum_{j\in O_n}r_{ij}^{10}(G_n)\delta_i^1\delta_j^0 .
\]
The two mixed contributions therefore combine into a single doubled term. Substituting the pair-moment identity into the pair expansion of \(\sigma_{G_n,p}^2(Y)\) yields
\[
\sigma_{G_n,p}^2(Y)
=
\frac{1}{n}\sum_{i\in O_n}\sum_{j\in O_n}
\left[
r_{ij}^1(G_n,p)(Y_i^1-\mu_1)(Y_j^1-\mu_1)
+r_{ij}^0(G_n,p)(Y_i^0-\mu_0)(Y_j^0-\mu_0)
+2r_{ij}^{10}(G_n)(Y_i^1-\mu_1)(Y_j^0-\mu_0)
\right],
\]
as asserted.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.varScale_homogeneous_formula
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:hetero-envelope}]
Recall the three covariance loads: for \(i,j\in O_n\),
\[
r_{ij}^1(G_n,p)
=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}p_k^{-1}-1\right),
\quad
r_{ij}^0(G_n,p)
=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}(1-p_k)^{-1}-1\right),
\]
\[
r_{ij}^{10}(G_n)=\mathbf 1\{|S_{ij}(G_n)|>0\},
\]
all well defined under \(0<p_k<1\).

\begin{enumerate}
\item Temporarily write
\[
A_i=\frac{T_i(Z)}{\pi_i^1(p)}-1,
\qquad
B_i=\frac{C_i(Z)}{\pi_i^0(p)}-1.
\]
Then \cref{obj:def:first-order-linearization} gives
\[
\eta_i(p,Z)=A_i(Y_i^1-\mu_1)-B_i(Y_i^0-\mu_0).
\]
Under \cref{obj:ass:independent-heterogeneous-bernoulli}, the assignment coordinates are independent, and \(0<p_k<1\) makes the exposure probabilities positive.  Hence
\[
\mathbb E_p[T_i(Z)]=\prod_{k\in N_i(G_n)}p_k=\pi_i^1(p),
\qquad
\mathbb E_p[T_i(Z)T_j(Z)]
=\prod_{k\in N_i(G_n)\cup N_j(G_n)}p_k,
\]
because \(T_i(Z)T_j(Z)\) is the indicator that every intervention in
\(N_i(G_n)\cup N_j(G_n)\) is treated.  For random variables \(U,V\) with
\(\alpha=\mathbb E_p[U]\neq0\) and \(\beta=\mathbb E_p[V]\neq0\),
\[
\mathbb E_p\!\left[\left(\frac{U}{\alpha}-1\right)\left(\frac{V}{\beta}-1\right)\right]
=\frac{\mathbb E_p[UV]}{\alpha\beta}-1.
\]
Applying this identity to \(T_i(Z)\) and \(T_j(Z)\) gives
\[
\mathbb E_p[A_iA_j]
=
\frac{\prod_{k\in N_i(G_n)\cup N_j(G_n)}p_k}{\pi_i^1(p)\pi_j^1(p)}-1.
\]
The last ratio is matched to \(r_{ij}^1(G_n,p)\) by the two overlap branches.  If
\(S_{ij}(G_n)\neq\varnothing\), then the nonzero factors satisfy
\[
\frac{\prod_{k\in N_i(G_n)\cup N_j(G_n)}p_k}{\pi_i^1(p)\pi_j^1(p)}
=
\prod_{k\in S_{ij}(G_n)}p_k^{-1},
\]
so \(\mathbb E_p[A_iA_j]=\prod_{k\in S_{ij}(G_n)}p_k^{-1}-1=r_{ij}^1(G_n,p)\).  If
\(S_{ij}(G_n)=\varnothing\), then \(N_i(G_n)\) and \(N_j(G_n)\) are disjoint, the union product factors as
\(\pi_i^1(p)\pi_j^1(p)\), and \(\mathbb E_p[A_iA_j]=0=r_{ij}^1(G_n,p)\).  Therefore
\[
\mathbb E_p[A_iA_j]=r_{ij}^1(G_n,p).
\]
The control-control calculation first gives
\[
\mathbb E_p[B_iB_j]
=
\frac{\prod_{k\in N_i(G_n)\cup N_j(G_n)}(1-p_k)}{\pi_i^0(p)\pi_j^0(p)}-1.
\]
It has the same two branches with \(1-p_k\) in place of \(p_k\): on the nonempty-overlap branch the ratio equals
\(\prod_{k\in S_{ij}(G_n)}(1-p_k)^{-1}\), while on the empty-overlap branch the disjoint union product factors and the centered ratio is zero.  Thus
\[
\mathbb E_p[B_iB_j]=r_{ij}^0(G_n,p).
\]
For the mixed moment, if \(S_{ij}(G_n)\neq\varnothing\), then a shared intervention is required to be both treated and untreated in \(T_i(Z)C_j(Z)\), so \(T_i(Z)C_j(Z)=0\) for every assignment and
\(\mathbb E_p[A_iB_j]=-1\).  If \(S_{ij}(G_n)=\varnothing\), the two exposure events use disjoint assignment coordinates, so
\(\mathbb E_p[T_i(Z)C_j(Z)]=\pi_i^1(p)\pi_j^0(p)\) and \(\mathbb E_p[A_iB_j]=0\).  Thus
\[
\mathbb E_p[A_iB_j]=-r_{ij}^{10}(G_n).
\]
The same argument with \(i\) and \(j\) exchanged, using \(S_{ij}(G_n)=S_{ji}(G_n)\), gives
\[
\mathbb E_p[B_iA_j]=-r_{ij}^{10}(G_n).
\]
Expanding the product of the two linearization summands,
\[
\begin{aligned}
\eta_i(p,Z)\eta_j(p,Z)
={}&A_iA_j(Y_i^1-\mu_1)(Y_j^1-\mu_1)
-A_iB_j(Y_i^1-\mu_1)(Y_j^0-\mu_0)\\
&{}-B_iA_j(Y_i^0-\mu_0)(Y_j^1-\mu_1)
+B_iB_j(Y_i^0-\mu_0)(Y_j^0-\mu_0),
\end{aligned}
\]
and inserting the four moment identities yields
\[
\begin{aligned}
\mathbb E_p\!\left[\eta_i(p,Z)\eta_j(p,Z)\right]
={}&
r_{ij}^1(G_n,p)(Y_i^1-\mu_1)(Y_j^1-\mu_1)
+r_{ij}^0(G_n,p)(Y_i^0-\mu_0)(Y_j^0-\mu_0)\\
&{}+r_{ij}^{10}(G_n)\!
\left\{(Y_i^1-\mu_1)(Y_j^0-\mu_0)+(Y_i^0-\mu_0)(Y_j^1-\mu_1)\right\}.
\end{aligned}
\]
% lean: linScore_pair_moment

\item The one-score centered-ratio identities give \(\mathbb E_p[A_i]=\mathbb E_p[B_i]=0\), hence \(\mathbb E_p[\eta_i(p,Z)]=0\) for every \(i\).  Starting from \cref{obj:def:variance-scale}, the finite-design variance of the linear combination \(n^{-1}\sum_i\eta_i(p,Z)\) expands as a double sum of covariances, and the zero means identify those covariances with joint moments.  The empty-outcome branch is evaluated separately, where both finite sums and the scaled variance term are zero; on the nonempty branch, this is the ordinary pair average.  Thus
\[
\sigma^2_{G_n,p}(Y)
=\frac1n\sum_{i\in O_n}\sum_{j\in O_n}
\mathbb E_p\!\left[\eta_i(p,Z)\eta_j(p,Z)\right].
\]
Substituting the preceding pair-moment identity and combining the two mixed double sums by the symmetry \(S_{ij}(G_n)=S_{ji}(G_n)\) gives
\[
\begin{aligned}
\sigma^2_{G_n,p}(Y)
=
\frac1n\sum_{i\in O_n}\sum_{j\in O_n}
\bigl\{&
r_{ij}^1(G_n,p)(Y_i^1-\mu_1)(Y_j^1-\mu_1)
+r_{ij}^0(G_n,p)(Y_i^0-\mu_0)(Y_j^0-\mu_0)\\
&{}+2r_{ij}^{10}(G_n)(Y_i^1-\mu_1)(Y_j^0-\mu_0)
\bigr\}.
\end{aligned}
\]
% lean: varScale_pair_moments
% lean: varScale_homogeneous_formula

\item By \cref{obj:ass:bounded-outcomes}, \(|Y_i^1|\le1\) and \(|Y_i^0|\le1\) for every \(i\).  Applying the triangle inequality to the finite-population means gives
\[
|\mu_1|\le1,
\qquad
|\mu_0|\le1,
\]
with the empty-population case evaluating both means as \(0\).  Therefore, for every \(i\in O_n\),
\[
|Y_i^1-\mu_1|\le2,
\qquad
|Y_i^0-\mu_0|\le2.
\]
The overlap loads are nonnegative under \(0<p_k<1\).  Indeed, when \(S_{ij}(G_n)=\varnothing\) all three loads are \(0\); when \(S_{ij}(G_n)\neq\varnothing\), the factors \(p_k^{-1}\) and \((1-p_k)^{-1}\) are at least \(1\), so the corresponding products minus \(1\) are nonnegative, and \(r_{ij}^{10}(G_n)=1\).  Thus
\[
r_{ij}^1(G_n,p)\ge0,
\qquad
r_{ij}^0(G_n,p)\ge0,
\qquad
r_{ij}^{10}(G_n)\ge0.
\]
% lean: centeredY1_abs_le_two
% lean: centeredY0_abs_le_two
% lean: r1_nonneg
% lean: r0_nonneg
% lean: r10_nonneg

\item For a nonnegative weight \(r\) and real numbers \(u,v\) with \(|u|\le2\) and \(|v|\le2\),
\[
ruv\le r|u||v|\le4r.
\]
Applying this bound to the treated, control, and mixed terms using the bounds and nonnegativity just established gives, for every ordered pair \((i,j)\),
\[
\begin{aligned}
& r_{ij}^1(G_n,p)(Y_i^1-\mu_1)(Y_j^1-\mu_1)
+r_{ij}^0(G_n,p)(Y_i^0-\mu_0)(Y_j^0-\mu_0)\\
&\qquad{}+2r_{ij}^{10}(G_n)(Y_i^1-\mu_1)(Y_j^0-\mu_0)\\
&\hspace{2cm}\le
4\left\{
r_{ij}^1(G_n,p)+r_{ij}^0(G_n,p)+2r_{ij}^{10}(G_n)
\right\}.
\end{aligned}
\]
% lean: mul_centered_pair_le_four_mul_of_nonneg
% lean: pair_kernel_le_envelope_kernel

\item Summing the pairwise inequality over \(i,j\in O_n\) and multiplying by the nonnegative inverse \(n^{-1}\) gives
\[
\begin{aligned}
\sigma^2_{G_n,p}(Y)
&\le
\frac1n\sum_{i\in O_n}\sum_{j\in O_n}
4\left\{
r_{ij}^1(G_n,p)+r_{ij}^0(G_n,p)+2r_{ij}^{10}(G_n)
\right\}\\
&=V_{\mathrm{env}}(G_n,p),
\end{aligned}
\]
by \cref{obj:def:graph-envelope}.
% lean: hetero_envelope
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:convex-design}]
Throughout write \(m_n=|I_n|\) and \(S_{ij}=S_{ij}(G_n)\), and interpret the factors \(n^{-1}\) in \cref{obj:def:graph-envelope,obj:def:kkt-gradient} as \((|O_n|)^{-1}\), with the total reciprocal convention \(0^{-1}=0\).  Thus when \(O_n=\varnothing\) the corresponding finite sums and scaled gradients are zero, and otherwise the notation is ordinary division by the outcome count.  Recall that the three covariance loads entering \cref{obj:def:graph-envelope} are
\[
r_{ij}^1(G_n,p)
=\mathbf 1\{|S_{ij}|>0\}\left(\prod_{k\in S_{ij}}p_k^{-1}-1\right),
\qquad
r_{ij}^0(G_n,p)
=\mathbf 1\{|S_{ij}|>0\}\left(\prod_{k\in S_{ij}}(1-p_k)^{-1}-1\right),
\]
\[
r_{ij}^{10}(G_n)=\mathbf 1\{|S_{ij}|>0\}.
\]
Note that every \(p\in\mathcal P_{n,B_n,\epsilon}\) satisfies \(0<\epsilon\le p_k\le1-\epsilon<1\), so all coordinate reciprocals in these loads are defined on the feasible class.

\begin{enumerate}
\item First suppose that \(I_n\) is nonempty and define the constant design
\[
p_k=\frac{B_n}{m_n}\qquad(k\in I_n).
\]
Since \(m_n>0\), the budget bounds \(m_n\epsilon\le B_n\le m_n(1-\epsilon)\) give
\[
\epsilon\le \frac{B_n}{m_n}\le 1-\epsilon,
\]
and \(0<\epsilon<1/2\) makes this interval a subset of \([0,1]\).  Thus \(p\) satisfies the probability and positivity requirements, and its budget is \(\sum_{k\in I_n}p_k=m_n(B_n/m_n)=B_n\).  If instead \(I_n\) is empty, the two budget inequalities read \(0\le B_n\le0\) and force \(B_n=0\), so the unique empty design is feasible.  Hence \(\mathcal P_{n,B_n,\epsilon}\neq\varnothing\).
% lean: feasibleSet_nonempty

\item The feasible class is closed.  Indeed, under \(0<\epsilon<1/2\), \cref{obj:def:feasible-designs} describes it as the intersection of the coordinatewise closed probability constraints \(0\le p_k\le1\), the coordinatewise closed bounds \(\epsilon\le p_k\le1-\epsilon\), and the closed affine hyperplane \(\sum_{k\in I_n}p_k=B_n\); each of these is the preimage of a closed set under a continuous coordinate map or under the continuous finite sum.  It is contained in the finite product \([0,1]^{I_n}\), which is compact; a closed subset of a compact set is compact, so \(\mathcal P_{n,B_n,\epsilon}\) is compact.
% lean: feasibleSet_compact

\item Let \(p,q\in\mathcal P_{n,B_n,\epsilon}\), and let \(\theta,\theta'\ge0\) with \(\theta+\theta'=1\).  Coordinatewise,
\[
\epsilon=\theta\epsilon+\theta'\epsilon
 \le \theta p_k+\theta'q_k
 \le \theta(1-\epsilon)+\theta'(1-\epsilon)=1-\epsilon,
\]
and by the same convex combination \(0\le \theta p_k+\theta'q_k\le1\).  Its budget is
\[
\sum_{k\in I_n}(\theta p_k+\theta'q_k)
 =\theta\sum_{k\in I_n}p_k+\theta'\sum_{k\in I_n}q_k
 =\theta B_n+\theta'B_n=B_n.
\]
Thus \(\theta p+\theta'q\in\mathcal P_{n,B_n,\epsilon}\), proving convexity of the feasible class.
% lean: feasibleSet_convex

\item The envelope is continuous on the feasible class.  Fix \(i,j\in O_n\).  If \(S_{ij}\neq\varnothing\), then the maps \(p\mapsto\prod_{k\in S_{ij}}p_k\) and \(p\mapsto\prod_{k\in S_{ij}}(1-p_k)\) are continuous, and the bounds \(\epsilon\le p_k\le1-\epsilon\) with \(0<\epsilon<1/2\) keep both products nonzero on \(\mathcal P_{n,B_n,\epsilon}\); reciprocals of nonvanishing continuous functions are continuous, so \(r_{ij}^1(G_n,\cdot)\) and \(r_{ij}^0(G_n,\cdot)\) are continuous there.  If \(S_{ij}=\varnothing\), both are identically zero.  Adding the constant \(2r_{ij}^{10}(G_n)\), summing over the finitely many pairs \((i,j)\), and multiplying by the fixed scalar \(4(|O_n|)^{-1}\) shows that \(V_{\mathrm{env}}(G_n,\cdot)\) is continuous on \(\mathcal P_{n,B_n,\epsilon}\).
% lean: varEnvelope_continuousOn_feasible

\item For each nonempty \(S_{ij}\), the maps
\[
p\longmapsto
\left(\prod_{k\in S_{ij}}p_k\right)^{-1},
\qquad
p\longmapsto
\left(\prod_{k\in S_{ij}}(1-p_k)\right)^{-1}
\]
are convex on \(\mathcal P_{n,B_n,\epsilon}\).  Indeed, on that class every \(p_k\) and every \(1-p_k\) is strictly positive, so
\[
\left(\prod_{k\in S_{ij}}p_k\right)^{-1}
=\exp\left(\sum_{k\in S_{ij}}-\log p_k\right),
\qquad
\left(\prod_{k\in S_{ij}}(1-p_k)\right)^{-1}
=\exp\left(\sum_{k\in S_{ij}}-\log(1-p_k)\right).
\]
Each \(p\mapsto-\log p_k\) is convex on the feasible class as a convex one-dimensional function composed with a linear coordinate map; likewise \(p\mapsto-\log(1-p_k)\), with the affine coordinate map \(p\mapsto1-p_k\).  Finite sums of convex functions are convex, and \(\exp\) is convex and nondecreasing, so composing preserves convexity.  Subtracting the constant \(1\) gives convexity of \(r^1_{ij}(G_n,\cdot)\) and \(r^0_{ij}(G_n,\cdot)\); when \(S_{ij}=\varnothing\) both are constant, hence convex.  Consequently each summand
\[
r^1_{ij}(G_n,p)+r^0_{ij}(G_n,p)+2r^{10}_{ij}(G_n)
\]
is convex on the feasible class.  Finite summation preserves convexity, and so does multiplication by the nonnegative scalar \(4(|O_n|)^{-1}\).  Thus \(V_{\mathrm{env}}(G_n,\cdot)\) is convex on \(\mathcal P_{n,B_n,\epsilon}\).
% lean: prod_inv_convexOn_feasible
% lean: prod_one_sub_inv_convexOn_feasible

\item The preceding nonemptiness and compactness conclusions, together with the continuity of \(V_{\mathrm{env}}(G_n,\cdot)\) on \(\mathcal P_{n,B_n,\epsilon}\), put the envelope in the finite-dimensional extreme-value theorem.  Hence it attains its minimum on \(\mathcal P_{n,B_n,\epsilon}\).  Denote one minimizer by \(p_n^*(G_n)\).  Then
\[
V_{\mathrm{env}}(G_n,p_n^*(G_n))
\le V_{\mathrm{env}}(G_n,q)
\]
for every \(q\in\mathcal P_{n,B_n,\epsilon}\).
% lean: IsCompact.exists_isMinOn

\item It remains to establish the multiplier statement, and for that we first identify the coordinate derivatives of the envelope.  Let \(\chi:\mathbb R\to\mathbb R\) be the standard smooth transition function with \(\chi(t)=0\) for \(t\le0\) and \(\chi(t)=1\) for \(t\ge1\).  Define the floored reciprocal
\[
R_\epsilon(x)=
\begin{cases}
\chi\!\left(\dfrac{x-\epsilon/4}{\epsilon/4}\right)x^{-1},&x\neq0,\\[0.6em]
0,&x=0.
\end{cases}
\]
Because \(\chi((x-\epsilon/4)/(\epsilon/4))\) is identically zero on a neighborhood of the singular point \(x=0\), this function is continuously differentiable on all of \(\mathbb R\).  If \(x\ge\epsilon/2\), then \((x-\epsilon/4)/(\epsilon/4)\ge1\), so \(R_\epsilon(x)=x^{-1}\).  Replace every reciprocal factor in the two overlap kernels by this \(R_\epsilon\), obtaining
\[
\widetilde r^1_{ij}(p)=\mathbf 1\{|S_{ij}|>0\}\left(\prod_{k\in S_{ij}}R_\epsilon(p_k)-1\right),
\qquad
\widetilde r^0_{ij}(p)=\mathbf 1\{|S_{ij}|>0\}\left(\prod_{k\in S_{ij}}R_\epsilon(1-p_k)-1\right),
\]
\[
\widetilde V(p)=4(|O_n|)^{-1}\sum_{i,j\in O_n}
\left\{\widetilde r^1_{ij}(p)+\widetilde r^0_{ij}(p)+2r^{10}_{ij}(G_n)\right\}.
\]
Then \(\widetilde V\) is continuously differentiable on all of \(\mathbb R^{I_n}\), and
\[
\widetilde V(p)=V_{\mathrm{env}}(G_n,p)
\qquad\text{whenever }p_k\ge\epsilon/2\text{ and }1-p_k\ge\epsilon/2\text{ for every }k\in I_n .
\]
Transporting \(\widetilde V/4\) along a bijection between \(I_n\) and \(\{1,\ldots,m_n\}\) to Euclidean coordinates gives a differentiable objective on \(\mathbb R^{m_n}\).

For a feasible \(p\), a coordinate \(k\), and a shared neighborhood \(S_{ij}\), differentiation along the \(k\)-th coordinate line gives
\[
\left.\frac{d}{dt}\right|_{t=0}
 \prod_{\ell\in S_{ij}}
 \bigl(p_\ell+\mathbf 1\{\ell=k\}t\bigr)^{-1}
=
\begin{cases}
-\left(\prod_{\ell\in S_{ij}}p_\ell^{-1}\right)p_k^{-1},
   &k\in S_{ij},\\
0,&k\notin S_{ij},
\end{cases}
\]
and
\[
\left.\frac{d}{dt}\right|_{t=0}
 \prod_{\ell\in S_{ij}}
 \bigl(1-p_\ell-\mathbf 1\{\ell=k\}t\bigr)^{-1}
=
\begin{cases}
\left(\prod_{\ell\in S_{ij}}(1-p_\ell)^{-1}\right)(1-p_k)^{-1},
   &k\in S_{ij},\\
0,&k\notin S_{ij}.
\end{cases}
\]
When \(k\notin S_{ij}\) the product does not depend on \(t\), and when \(k\in S_{ij}\) only the factor indexed by \(k\) varies, contributing \(\frac{d}{dt}(p_k+t)^{-1}|_{t=0}=-p_k^{-2}\) and \(\frac{d}{dt}(1-p_k-t)^{-1}|_{t=0}=(1-p_k)^{-2}\), respectively.  Subtracting the constant \(1\) leaves these derivatives unchanged, so the same formulas hold for the treated and control overlap loads, including the case \(S_{ij}=\varnothing\), where both derivatives vanish.  Summing over \((i,j)\) and multiplying by \((|O_n|)^{-1}\) gives exactly \cref{obj:def:kkt-gradient}:
\[
\left.\frac{d}{dt}\right|_{t=0}
\frac{V_{\mathrm{env}}\bigl(G_n,p+t e_k\bigr)}{4}
=g_k(G_n,p).
\]
Since \(\epsilon\le p_k\le1-\epsilon\), every \(|t|<\epsilon/2\) keeps \(p+te_k\) in the region where \(\widetilde V\) agrees with \(V_{\mathrm{env}}(G_n,\cdot)\).  Therefore the \(k\)-th Euclidean partial derivative of the transported objective at any feasible \(p\) is \(g_k(G_n,p)\).
% lean: deriv_varEnvelope_div_four_coord_line
% lean: envelopeGrad_eq_gradient_varEnvelopeExt

\item Fix any minimizer \(p_n^*(G_n)\).  In Euclidean coordinates, impose the equality constraint
\[
h(x)=\sum_{u=1}^{m_n}x_u-B_n=0
\]
and the \(2m_n\) inequality constraints
\[
c_u^-(x)=x_u-\epsilon\ge0,
\qquad
c_u^+(x)=1-\epsilon-x_u\ge0
\qquad(u=1,\ldots,m_n).
\]
The correspondence between the Euclidean points satisfying these constraints and \(\mathcal P_{n,B_n,\epsilon}\) is exact.  The displayed agreement of \(\widetilde V\) with \(V_{\mathrm{env}}(G_n,\cdot)\) on the feasible box shows that the transported smooth objective agrees on that set with \(V_{\mathrm{env}}(G_n,\cdot)/4\); hence the Euclidean image \(x^*\) of \(p_n^*(G_n)\) is a local minimum of the transported objective subject to those constraints.  The objective is differentiable, the Euclidean domain is all of \(\mathbb R^{m_n}\), and \(h,c_u^-,c_u^+\) are affine, so the affine linear constraint qualification holds at \(x^*\).

The finite-dimensional first-order necessary condition under this affine constraint qualification states: if \(x^*\) is a local minimum of a differentiable \(f\) subject to \(h(x)=0\) and \(c_j(x)\ge0\) with all constraints affine, then there exist \(\lambda_n\in\mathbb R\) and \(\nu_j\ge0\) with
\[
\nabla f(x^*)-\lambda_n\nabla h(x^*)-\sum_j\nu_j\nabla c_j(x^*)=0
\qquad\text{and}\qquad
\nu_jc_j(x^*)=0\quad\text{for every }j.
\]
Write \(\nu^-_{k,n}\) for the multiplier of \(c^-_u\) and \(\nu^+_{k,n}\) for the multiplier of \(c^+_u\) at the coordinate \(u\) corresponding to \(k\in I_n\).  Since \(\nabla h\) is the all-ones vector, \(\nabla c_u^-=e_u\) and \(\nabla c_u^+=-e_u\), the \(k\)-th coordinate of the stationarity condition reads, using the coordinate-gradient identity just established, whose right-hand side is the score in \cref{obj:def:kkt-gradient},
\[
g_k(G_n,p_n^*(G_n))
-\lambda_n-\nu^-_{k,n}+\nu^+_{k,n}=0,
\]
or equivalently
\[
g_k(G_n,p_n^*(G_n))
=\lambda_n-\nu^+_{k,n}+\nu^-_{k,n}.
\]
Dual feasibility gives \(\nu^+_{k,n}\ge0\) and \(\nu^-_{k,n}\ge0\).  Complementary slackness for the upper and lower constraints gives \(\nu^+_{k,n}c_u^+(x^*)=0\) and \(\nu^-_{k,n}c_u^-(x^*)=0\), equivalently
\[
\nu^+_{k,n}\bigl((p_n^*(G_n))_k-(1-\epsilon)\bigr)=0,
\qquad
\nu^-_{k,n}\bigl(\epsilon-(p_n^*(G_n))_k\bigr)=0.
\]
This holds for every \(k\in I_n\), as required.
% lean: convex_design
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:hetero-clt}]
Write \(p_n=p_n^*(G_n)\).  The proof uses the feasibility, positivity, assignment, interference, graph-regularity, and variance hypotheses attached to this sequence.  Throughout, put
\[
K(\epsilon,\bar d)=\epsilon^{-1}\max\{1,\epsilon^{-(\bar d-1)}\}.
\]

\begin{enumerate}
\item Outcome indexing gives \(|O_n|=n\) eventually.  Hence \(|O_n|\to\infty\), and, on this tail, \(\sqrt{|O_n|}=\sqrt n\) and \(|O_n|^{-1/2}=n^{-1/2}\).  All limit statements below may therefore be proved after discarding the finitely many earlier stages.
% lean: hetero_clt

\item Uniform positivity gives \(\epsilon_0>0\) with \(\epsilon_0\le\epsilon_n\) eventually.  Since each \(\epsilon_n\in(0,1/2)\), the denominator-kernel monotonicity bound gives
\[
K(\epsilon_n,\bar d)\le K(\epsilon_0,\bar d)
\qquad\text{eventually.}
\]
Together with \(|O_n|\to\infty\), this fixed upper bound implies
\[
\frac{K(\epsilon_n,\bar d)}{|O_n|}\longrightarrow0.
\]
% lean: denominatorKernelBound_div_card_tendsto_zero

\item Define the two centered numerators and the two centered denominator ratios,
\[
U_1(Z)=\sum_{i\in O_n}\frac{T_i(Z)}{\pi_i^1(p_n)}\left(Y_i^1-\mu_1\right),
\qquad
U_0(Z)=\sum_{i\in O_n}\frac{C_i(Z)}{\pi_i^0(p_n)}\left(Y_i^0-\mu_0\right),
\]
\[
R_{1n}=|O_n|^{-1/2}U_1(Z),\qquad
R_{0n}=|O_n|^{-1/2}U_0(Z),
\]
\[
Q_{1n}=|O_n|^{-1}D_1(p_n,Z)-1,\qquad
Q_{0n}=|O_n|^{-1}D_0(p_n,Z)-1.
\]

The scaled numerators are bounded in probability.  Indeed, \(\mathbb E_{D_n}[T_i(Z)/\pi_i^1(p_n)]=1\) and \(\mu_1=|O_n|^{-1}\sum_iY_i^1\) give \(\mathbb E_{D_n}[U_1(Z)]=0\), hence \(\mathbb E_{D_n}[R_{1n}]=0\); and expanding the variance into pair covariances, using \cref{obj:ass:bounded-outcomes} and the row bound of \cref{obj:ass:bounded-overlap-dependency} together with the kernel bound supplied by \cref{obj:ass:bounded-outcome-degree} and the eventual floor \(\epsilon_0\le\epsilon_n\), one gets eventually
\[
\operatorname{Var}_{D_n}(R_{1n})\le 4\bar D K(\epsilon_0,\bar d),
\qquad
\operatorname{Var}_{D_n}(R_{0n})\le 4\bar D K(\epsilon_0,\bar d).
\]
A sequence with mean zero and uniformly bounded variance is bounded in probability, by Chebyshev's inequality.
% lean: treatNumerator_scaled_boundedInProb
% lean: ctrlNumerator_scaled_boundedInProb

The centered denominator ratios vanish in probability.  The treated denominator has mean \(|O_n|\), so \(\mathbb E_{D_n}[Q_{1n}]=0\).  The same covariance expansion gives
\[
\operatorname{Var}_{D_n}(D_1(p_n,Z))
\le |O_n|\bar D K(\epsilon_n,\bar d),
\]
and therefore
\[
\operatorname{Var}_{D_n}(Q_{1n})
\le\frac{\bar D K(\epsilon_n,\bar d)}{|O_n|}\longrightarrow0
\]
by the denominator-kernel rate displayed above.  Chebyshev's inequality then yields
\[
Q_{1n}\xrightarrow{\Pr}0,
\qquad
Q_{0n}\xrightarrow{\Pr}0.
\]
% lean: treatDenominatorRatioCentered_tendstoInProb_zero
% lean: ctrlDenominatorRatioCentered_tendstoInProb_zero
A product of a sequence bounded in probability with one vanishing in probability vanishes in probability, so
\[
2|R_{1n}||Q_{1n}|+2|R_{0n}||Q_{0n}|
\xrightarrow{\Pr}0.
\]

For all sufficiently large \(n\), split according as
\[
D_1(p_n,Z)\ge\frac{|O_n|}{2}
\quad\text{and}\quad
D_0(p_n,Z)\ge\frac{|O_n|}{2}.
\]
When both inequalities hold, both denominators are positive, the Hájek estimator in \cref{obj:def:hetero-hajek-estimator} takes its ratio form in both arms, and expanding each ratio around its denominator mean gives the capped remainder bound
\[
\left|
\sqrt{|O_n|}\{\widehat\tau_H(p_n)-\tau_n\}
-|O_n|^{-1/2}\sum_{i\in O_n}\eta_i(p_n,Z)
\right|
\le
2|R_{1n}||Q_{1n}|+2|R_{0n}||Q_{0n}| .
\]
Here \cref{obj:ass:bipartite-interference} identifies the observed outcome \(Y_i^{\mathrm{obs}}\) with \(Y_i^1\) on \(\{T_i(Z)=1\}\) and with \(Y_i^0\) on \(\{C_i(Z)=1\}\), so that the two arm numerators are \(U_1(Z)+\mu_1D_1(p_n,Z)\) and \(U_0(Z)+\mu_0D_0(p_n,Z)\), and the cap \(D_a(p_n,Z)\ge|O_n|/2\) bounds the factor \(|O_n|/D_a(p_n,Z)\) by \(2\).
% lean: hajek_remainder_capped_bound
If instead \(D_1(p_n,Z)<|O_n|/2\), then \(|O_n|^{-1}D_1(p_n,Z)<1/2\), so \(Q_{1n}<-1/2\) and \(|Q_{1n}|>1/2\); likewise for the control arm.  Thus, for every \(\delta>0\), the probability that the absolute remainder exceeds \(\delta\) is bounded by the probability of this denominator-ratio bad event plus
\[
\Pr_{D_n}\!\left(
2|R_{1n}||Q_{1n}|+2|R_{0n}||Q_{0n}|\ge\delta
\right),
\]
and both terms converge to zero.
% lean: remainder_tendstoInProb_zero

\item Consequently, for every \(\delta>0\) the probability that the absolute remainder is at least \(\delta\) tends to zero, that is
\[
\sqrt{|O_n|}\{\widehat\tau_H(p_n)-\tau_n\}
-|O_n|^{-1/2}\sum_{i\in O_n}\eta_i(p_n,Z)
\xrightarrow{\Pr}0.
\]
Replacing \(|O_n|\) by \(n\) on the outcome-indexing tail gives the first conclusion.
% lean: heteroDenominatorTightness_discharged
% lean: hetero_linearization_of_denominator_tightness

\item Set \(X_{n,i}=\eta_i(p_n,Z)\) for \(i\in O_n\), and declare \(i\) and \(j\) adjacent when \(i=j\) or \(S_{ij}(G_n)\neq\varnothing\).  This is a dependency graph for \(\{X_{n,i}\}\): the summand \(\eta_i(p_n,Z)\) of \cref{obj:def:first-order-linearization} is assembled from \(T_i(Z)\) and \(C_i(Z)\), together with \(\pi_i^1(p_n),\pi_i^0(p_n),Y_i^1,Y_i^0,\mu_1,\mu_0\), so it depends on \(Z\) only through the coordinates in \(N_i(G_n)\).  If no outcome in \(A\subseteq O_n\) is adjacent to any outcome in \(B\subseteq O_n\), then \(\bigcup_{i\in A}N_i(G_n)\) and \(\bigcup_{j\in B}N_j(G_n)\) are disjoint, and \cref{obj:ass:independent-heterogeneous-bernoulli} makes the two assignment-coordinate blocks independent.  The closed neighborhood at \(i\) is contained in \(\{i\}\cup\{j:S_{ij}(G_n)\neq\varnothing\}\), so \cref{obj:ass:bounded-overlap-dependency} gives cardinality at most \(\lceil\bar D\rceil+1\).
% lean: linScoreDepGraph
% lean: linScoreDepGraph_degree_le

These summands have mean zero, because \(\mathbb E_{D_n}[T_i(Z)/\pi_i^1(p_n)]=\mathbb E_{D_n}[C_i(Z)/\pi_i^0(p_n)]=1\).
% lean: linScore_mean_zero
Moreover \cref{obj:ass:bounded-outcomes} gives \(|Y_i^1-\mu_1|\le2\) and \(|Y_i^0-\mu_0|\le2\), while \cref{obj:def:feasible-designs}, \cref{obj:ass:bounded-outcome-degree}, and the eventual floor \(\epsilon_0\le\epsilon_n\) give
\[
\left(\pi_i^1(p_n)\right)^{-1}\le K(\epsilon_0,\bar d)+1,
\qquad
\left(\pi_i^0(p_n)\right)^{-1}\le K(\epsilon_0,\bar d)+1 .
\]
Thus \(|T_i(Z)/\pi_i^1(p_n)-1|\le K(\epsilon_0,\bar d)+2\) and likewise for the control arm, and the eventual uniform bound
\[
|X_{n,i}|\le 4\{K(\epsilon_0,\bar d)+2\}
\]
holds for every \(i\in O_n\) and every assignment.
% lean: linScore_abs_le_uniform_floor

Finally, with
\[
v_n=\mathbb E_{D_n}\!\left[\left(\sum_{i\in O_n}X_{n,i}\right)^2\right],
\]
expanding the square into pair moments and comparing with \cref{obj:def:variance-scale} gives
\[
v_n=
\sum_{i\in O_n}\sum_{j\in O_n}
\mathbb E_{D_n}\!\left[\eta_i(p_n,Z)\eta_j(p_n,Z)\right]
=|O_n|\,\sigma^2_{G_n,p_n}(Y).
\]
% lean: linScore_sum_square_integral_eq_card_mul_varScale
\cref{obj:ass:variance-nondegenerate} supplies \(c>0\) with \(\sigma^2_{G_n,p_n}(Y)\ge c\) eventually.  Hence \(v_n\ge c|O_n|\) eventually, and on the outcome-indexing tail this is the variance-growth condition \(v_n\ge cn\).  Applying \cref{obj:lem:bounded-degree-dependency-clt} with \(D=\lceil\bar D\rceil+1\), \(M=4\{K(\epsilon_0,\bar d)+2\}\), and this \(c\) yields, for every \(s\in\mathbb R\),
\[
\Pr_{D_n}\!\left(\frac{\sum_{i\in O_n}\eta_i(p_n,Z)}{\sqrt{v_n}}\le s\right)
\longrightarrow
\Pr\{N(0,1)\le s\}.
\]
Since \(\sqrt{v_n}=\sqrt{|O_n|}\sqrt{\sigma^2_{G_n,p_n}(Y)}\), the statistic inside is
\(\bigl(|O_n|^{-1/2}\sum_i\eta_i(p_n,Z)\bigr)\big/\sqrt{\sigma^2_{G_n,p_n}(Y)}\), so
\[
\Pr_{D_n}\!\left(
\frac{|O_n|^{-1/2}\sum_{i\in O_n}\eta_i(p_n,Z)}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}}\le s
\right)
\longrightarrow
\Pr\{N(0,1)\le s\}.
\]
% lean: hetero_linscore_clt_of_depgraph

\item The difference between the studentized Hájek statistic and the studentized linear-score statistic is
\[
\frac{
\sqrt{|O_n|}\{\widehat\tau_H(p_n)-\tau_n\}
-|O_n|^{-1/2}\sum_{i\in O_n}\eta_i(p_n,Z)}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}}.
\]
By \cref{obj:ass:variance-nondegenerate}, eventually \(\sqrt{\sigma^2_{G_n,p_n}(Y)}\ge\sqrt c\).  Hence, for every \(\delta>0\), the probability that the absolute value of the displayed difference is at least \(\delta\) is eventually bounded above by
\[
\Pr_{D_n}\!\left(
\left|
\sqrt{|O_n|}\{\widehat\tau_H(p_n)-\tau_n\}
-|O_n|^{-1/2}\sum_{i\in O_n}\eta_i(p_n,Z)
\right|
\ge\delta\sqrt c
\right),
\]
which tends to zero by the linearization established above.  The converging-together principle in CDF form then transfers the standard-normal CDF limit from the studentized linear-score statistic to the studentized Hájek statistic: for each \(x\), sandwich the event for the Hájek statistic between the corresponding linear-score events at \(x-\delta\) and \(x+\delta\), up to the small-difference event, and let \(\delta\downarrow0\) using continuity of the standard-normal CDF.  Therefore, for every \(s\in\mathbb R\),
\[
\Pr_{D_n}\!\left(
\frac{\sqrt{|O_n|}\{\widehat\tau_H(p_n)-\tau_n\}}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}}\le s
\right)
\longrightarrow\Pr\{N(0,1)\le s\}.
\]
Replacing \(|O_n|\) by \(n\) on the outcome-indexing tail proves the second conclusion.
% lean: finiteDesign_cdf_converging_together
% lean: hetero_studentized_slutsky
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:postdesign-wald}]
The eventual identity \(|O_n|=n\) implies that \(|O_n|\to\infty\); in particular \(|O_n|>0\) on an eventual tail.

\begin{enumerate}
\item Fix \(n\).  The admissibility condition \(\epsilon_n\in(0,1/2)\) and the feasibility floor give
\[
0<\epsilon_n\le p_{n,k}\le 1-\epsilon_n<1
\qquad(k\in I_n).
\]
This display supplies the assignment-probability hypothesis \(0<p_{n,k}<1\) in \cref{obj:thm:hetero-envelope}.  Together with \cref{obj:ass:independent-heterogeneous-bernoulli} and \cref{obj:ass:bounded-outcomes}, the second conclusion of \cref{obj:thm:hetero-envelope} gives
\[
\sigma^2_{G_n,p_n}(Y)\le V_{\mathrm{env}}(G_n,p_n).
\]
By \cref{obj:def:conservative-variance-estimator},
\(\widehat V_{\mathrm{cons}}(G_n,p_n)=V_{\mathrm{env}}(G_n,p_n)\), and hence
\[
\sigma^2_{G_n,p_n}(Y)\le \widehat V_{\mathrm{cons}}(G_n,p_n).
\]
Since \(n\) was arbitrary, this proves the variance-domination assertion for every \(n\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.hetero_envelope

\item By \cref{obj:ass:variance-nondegenerate} there are \(c>0\) and an eventual tail on which
\[
c\le \sigma^2_{G_n,p_n}(Y).
\]
Hence \(\sigma^2_{G_n,p_n}(Y)>0\) eventually.  The eventual identity
\(|O_n|=n\) also gives the positive normalization \(|O_n|>0\) eventually.
Under the assumptions listed in \cref{obj:thm:hetero-clt} -- outcome indexing,
assignment and interference, bounded outcomes, bounded degree, bounded overlap,
feasibility, optimality \(p_n=p_n^*(G_n)\), uniform positivity, and variance
nondegeneracy -- \cref{obj:thm:hetero-clt} gives, for every \(s\in\mathbb R\),
\[
\Pr_{p_n}\!\left(
\frac{\sqrt n\,[\widehat\tau_H(p_n)-\tau_n]}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}}\le s
\right)
\longrightarrow \Phi(s).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.hetero_clt

\item Write
\[
W_n=
\frac{\sqrt n\,[\widehat\tau_H(p_n)-\tau_n]}
{\sqrt{\sigma^2_{G_n,p_n}(Y)}},
\qquad
z=z_{1-\alpha_{\mathrm{cov}}/2}.
\]
Since \(z\ge0\), the event \(\{W_n\le -z\}\) is contained in \(\{W_n\le z\}\), and the finite-design probability split gives
\[
\Pr_{p_n}\!\left(-z<W_n\le z\right)
=\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right).
\]
The convergence supplied by \cref{obj:thm:hetero-clt} at \(s=z\) and \(s=-z\), together with
\(\Phi(-z)=1-\Phi(z)\) and the calibration \(\Phi(z)=1-\alpha_{\mathrm{cov}}/2\), yields
\[
\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right)
\longrightarrow
\Phi(z)-\Phi(-z)
=1-\alpha_{\mathrm{cov}} .
\]
On the eventual tail where \(|O_n|\), \(n\), and \(\sigma^2_{G_n,p_n}(Y)\) are positive, the strict interval event is contained in the Wald coverage event.  Indeed, if
\(-z<W_n\le z\), then \(|W_n|\le z\), so
\[
\sqrt n\,|\widehat\tau_H(p_n)-\tau_n|
\le z\sqrt{\sigma^2_{G_n,p_n}(Y)}.
\]
Dividing by \(\sqrt n>0\) and using the variance-domination bound
\(\sigma^2_{G_n,p_n}(Y)\le \widehat V_{\mathrm{cons}}(G_n,p_n)\), monotonicity of the square root, and \(z\ge0\), gives
\[
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z
\sqrt{\frac{\sigma^2_{G_n,p_n}(Y)}{n}}
\le
z
\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}.
\]
Consequently, on that tail,
\[
\Pr_{p_n}\!\left(
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}
\right)
\ge
\Pr_{p_n}\!\left(W_n\le z\right)-\Pr_{p_n}\!\left(W_n\le -z\right).
\]
Taking lower limits and using the displayed convergence gives
\[
1-\alpha_{\mathrm{cov}}
\le
\liminf_{n\to\infty}
\Pr_{p_n}\!\left(
\left|\tau_n-\widehat\tau_H(p_n)\right|
\le
z_{1-\alpha_{\mathrm{cov}}/2}
\sqrt{\frac{\widehat V_{\mathrm{cons}}(G_n,p_n)}{n}}
\right),
\]
which is the claimed coverage bound.

\begin{proof}[Proof of \cref{obj:thm:dispersion-certificate-unbounded}]
Fix \(\epsilon\in(0,1/2)\), \(c_{\mathrm{disp}}>0\), and
\(C_{\mathrm{disp}}\geq 1\).  Put
\[
t_n=n+1,\qquad d_n=t_n^2,\qquad
\rho=\frac{\epsilon+1/2}{2},\qquad
\varphi=\frac{3\rho-1/2}{2},
\qquad
R(x)=x^{-1}+(1-x)^{-1},
\]
so that \(\epsilon<\rho<1/2\) and \(\epsilon<\varphi<1/2\), both by direct substitution of \(0<\epsilon<1/2\).  Recall also the objects the argument evaluates: for \(i,j\in O_n\),
\[
r_{ij}^1(G_n,p)=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}p_k^{-1}-1\right),
\quad
r_{ij}^0(G_n,p)=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}(1-p_k)^{-1}-1\right),
\]
\[
r_{ij}^{10}(G_n)=\mathbf 1\{|S_{ij}(G_n)|>0\}.
\]

\begin{enumerate}
\item Let \(I_n\) be the disjoint union of a core block of \(d_n\)
interventions and a filler block of \(2d_n\) interventions, so \(m_n=3d_n\).  Let \(O_n\)
consist of \(d_n\) core outcomes together with \(t_n\) filler outcomes for
each filler intervention, so
\[
|O_n|=d_n+2d_nt_n .
\]
Each core outcome is adjacent to every core intervention, while each filler outcome is adjacent only to its designated filler intervention; take all potential-outcome schedules to be zero.  Explicitly, \(N_i(G_n)\) is the whole core block when \(i\) is a core outcome, and \(N_{(f,r)}(G_n)=\{f\}\) when \((f,r)\) is the \(r\)-th filler outcome of the filler intervention \(f\).  Counting the outcomes adjacent to a given intervention,
\[
s_k=d_n\ \ (k\text{ core}),
\qquad
s_k=t_n\ \ (k\text{ filler}).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_sdeg_core
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_sdeg_filler

\item Set \(B_n=m_n\rho\).  Since
\(\epsilon<\rho<1/2\), one has
\(\epsilon\leq\rho\leq1-\epsilon\), and multiplying by \(m_n\geq0\),
\[
m_n\epsilon\leq B_n\leq m_n(1-\epsilon).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionBudget_admissible

\item Summing the two degree values over the \(d_n\) core and \(2d_n\) filler interventions, the degree energy is
\[
\sum_{k\in I_n}s_k^2
=d_n\,d_n^2+2d_n\,t_n^2
=d_n^3+2d_n^2 ,
\]
using \(t_n^2=d_n\); since \(d_n\geq1\) this is strictly positive.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_degree_energy
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_degree_energy_pos

\item Since \(d_n\to\infty\) and \(c_{\mathrm{disp}}>0\), for all
sufficiently large \(n\),
\[
1\leq c_{\mathrm{disp}}d_n .
\]
Fix such an \(n\).  If \(k\) is a core intervention, then \(s_k^2=d_n^2\) and, multiplying the displayed inequality by \(d_n^2>0\),
\[
s_k^2=d_n^2
 \leq c_{\mathrm{disp}}d_n^3
 \leq c_{\mathrm{disp}}
 \bigl(d_n^3+2d_n^2\bigr)
 =c_{\mathrm{disp}}\sum_{l\in I_n}s_l^2 .
\]
If \(k\) is a filler intervention, then \(s_k^2=t_n^2=d_n\), and since \(d_n\geq1\),
\[
s_k^2=d_n\leq d_n^2\leq c_{\mathrm{disp}}d_n^3
 \leq c_{\mathrm{disp}}\sum_{l\in I_n}s_l^2 .
\]
Thus the degree-dispersion condition holds eventually.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionD_tendsto_atTop
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionCertificateUnbounded

\item Every intervention has the same strictly positive \(h\)-weight.  Indeed, the ordered pairs \((i,j)\) with \(k\in S_{ij}(G_n)\) are: for a core \(k\), the \(d_n^2\) core-core pairs, each with \(|S_{ij}(G_n)|=d_n\); and for a filler \(k=f\), the \(t_n^2\) pairs of filler outcomes both carrying the label \(f\), each with \(|S_{ij}(G_n)|=1\).  A core outcome and a filler outcome share no intervention, and two filler outcomes with distinct labels share none.  Hence
\[
h_k(G_n)=\frac{d_n^2\cdot d_n^{-1}}{|O_n|}=\frac{d_n}{|O_n|}
\quad(k\text{ core}),
\qquad
h_k(G_n)=\frac{t_n^2\cdot1}{|O_n|}=\frac{d_n}{|O_n|}
\quad(k\text{ filler}),
\]
again by \(t_n^2=d_n\).  Consequently, for every \(k,l\in I_n\),
\[
h_k(G_n)=h_l(G_n)=\frac{d_n}{|O_n|}>0
\qquad\text{and hence}\qquad
h_k(G_n)\leq C_{\mathrm{disp}}h_l(G_n),
\]
the last step because \(C_{\mathrm{disp}}\geq1\).  So the \(h\)-weight-ratio condition holds for every \(n\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_hWeight_core
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_hWeight_filler
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_hWeight_eq
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionExperiment_hWeight_pos

\item \textit{The surrogate design is homogeneous.}  The equal-weight calculation above makes the surrogate objective of \cref{obj:def:surrogate-design} a positive multiple of a separable barrier sum,
\[
\sum_{k\in I_n}h_k(G_n)\left\{p_k^{-1}+(1-p_k)^{-1}\right\}
=\frac{d_n}{|O_n|}\sum_{k\in I_n}R(p_k).
\]
For \(0<r<1\), the barrier \(R\) satisfies
\[
R(x)-R(r)-R'(r)(x-r)
=
\frac{(x-r)^2\{r^2+2rx-2r-x+1\}}
{r^2x(1-r)^2(1-x)}
\qquad(0<x<1).
\]
The denominator is positive.  The bracket is also positive: if \(r\leq1/2\), then
\[
r^2+2rx-2r-x+1=r^2+(1-2r)(1-x)>0,
\]
and if \(1/2<r\), then
\[
r^2+2rx-2r-x+1=(1-r)^2+(2r-1)x>0.
\]
Therefore
\[
R(r)+R'(r)(x-r)\leq R(x)
\qquad(0<x<1),
\]
and equality holds exactly when \(x=r\), since the displayed gap then has positive denominator and positive bracket, leaving only the factor \((x-r)^2\) to vanish.
Summing the tangent inequality over \(k\in I_n\) at \(r=\rho\) and using the budget \(\sum_{k\in I_n}p_k=B_n=m_n\rho\), the linear terms cancel and
\[
m_nR(\rho)\leq\sum_{k\in I_n}R(p_k)
\qquad\text{for every }p\in\mathcal P_{n,B_n,\epsilon},
\]
with equality only if \(p_k=\rho\) for every \(k\).  The homogeneous vector \(p^{\mathrm{hom}}_k=\rho\) is feasible, because \(\epsilon<\rho<1/2<1-\epsilon\) and \(\sum_{k\in I_n}\rho=m_n\rho=B_n\).  Hence the surrogate minimizer is unique and
\[
p_n^{\deg}(G_n)=p^{\mathrm{hom}} .
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_surrogateObjective_eq_weighted_sum
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.reciprocalBarrier_sum_unique_minimizer
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_surrogateDesign_eq_homogeneous

\item \textit{Two envelope evaluations.}  For any design \(p\), \cref{obj:def:graph-envelope} splits over the outcome blocks.  A core-core ordered pair has \(S_{ij}(G_n)\) equal to the whole core block, of size \(d_n\); a pair of filler outcomes with the same label \(f\) has \(S_{ij}(G_n)=\{f\}\); all remaining ordered pairs have \(S_{ij}(G_n)=\varnothing\) and contribute \(0\).  There are \(d_n^2\) pairs of the first kind and \(2d_nt_n^2\) of the second.  At the homogeneous design \(p^{\mathrm{hom}}\) this gives
\[
\frac{V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})}{4}
=\frac{1}{|O_n|}
\left\{
d_n^2\left(\rho^{-d_n}+(1-\rho)^{-d_n}\right)
+2d_nt_n^2R(\rho)
\right\},
\]
since a core-core pair contributes \((\rho^{-d_n}-1)+((1-\rho)^{-d_n}-1)+2=\rho^{-d_n}+(1-\rho)^{-d_n}\) and a same-label filler pair contributes \((\rho^{-1}-1)+((1-\rho)^{-1}-1)+2=R(\rho)\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionHomogeneous_envelope

For comparison, let \(q\) be the design assigning \(1/2\) to every core intervention and \(\varphi\) to every filler intervention.  It is feasible: \(\epsilon<\varphi<1/2\) and \(\epsilon<1/2<1-\epsilon\) give the box constraints, and
\[
\sum_{k\in I_n}q_k=d_n\cdot\frac12+2d_n\varphi
=d_n\left(\frac12+3\rho-\frac12\right)=3d_n\rho=B_n .
\]
The same block split, with a core-core pair now contributing \(2^{d_n}+2^{d_n}=2^{d_n+1}\), gives
\[
\frac{V_{\mathrm{env}}(G_n,q)}{4}
=\frac{1}{|O_n|}
\left\{
d_n^2\,2^{d_n+1}
+2d_nt_n^2R(\varphi)
\right\}.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionComparisonDesign_feasible
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionComparison_envelope

\item \textit{The envelope minimum is positive.}  The bounded-outcome-degree condition holds here with \(\bar d=d_n\), since core outcomes have degree \(d_n\) and filler outcomes degree \(1\).  Hence the lower half of the sandwich in \cref{obj:thm:surrogate-certificate}, applied at the envelope-optimal design \(p_n^*(G_n)\), gives
\[
0<\frac{d_n}{|O_n|}\sum_{k\in I_n}R\bigl((p_n^*(G_n))_k\bigr)
=\sum_{k\in I_n}h_k(G_n)\left\{(p_n^*(G_n))_k^{-1}+(1-(p_n^*(G_n))_k)^{-1}\right\}
\leq\frac{V_{\mathrm{env}}(G_n,p_n^*(G_n))}{4},
\]
the strict positivity because \(d_n/|O_n|>0\), \(I_n\neq\varnothing\), and \(R>0\) on \((0,1)\).  Thus
\(V_{\mathrm{env}}(G_n,p_n^*(G_n))>0\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_envMin_pos

\item \textit{Lower bound on the ratio.}  The positivity just proved selects the positive branch of \cref{obj:def:approximation-ratio}, and the surrogate-design calculation identifies its numerator with the homogeneous envelope.  Since \(q\) is feasible, \(V_{\mathrm{env}}(G_n,p_n^*(G_n))\le V_{\mathrm{env}}(G_n,q)\), and \(V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})\ge0\), so
\[
\alpha_{\mathrm{cert}}(G_n)
=\frac{V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})}{V_{\mathrm{env}}(G_n,p_n^*(G_n))}
\geq
\frac{V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})}{V_{\mathrm{env}}(G_n,q)} .
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_approxRatio_ge_envelopeRatio
Substituting the two envelope evaluations above, cancelling the common factor \(|O_n|^{-1}\), and using \(2d_nt_n^2=2d_n^2\) so that \(d_n^2\) cancels as well,
\[
\frac{V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})}{V_{\mathrm{env}}(G_n,q)}
=\frac{\rho^{-d_n}+(1-\rho)^{-d_n}+2R(\rho)}
      {2^{d_n+1}+2R(\varphi)}
\geq
\frac{\rho^{-d_n}}
 {2^{d_n+1}+2R(\varphi)},
\]
the inequality because \((1-\rho)^{-d_n}\ge0\) and \(R(\rho)>0\).  Hence
\[
\alpha_{\mathrm{cert}}(G_n)\geq
\frac{\rho^{-d_n}}
 {2^{d_n+1}+2R(\varphi)} .
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_approxRatio_lower_bound

\item Since \(2^{d_n}\to\infty\), eventually
\(2R(\varphi)\leq2^{d_n}\), so the denominator is at most \(2^{d_n+1}+2^{d_n}=3\cdot2^{d_n}\) and
\[
\alpha_{\mathrm{cert}}(G_n)\geq
\frac{\rho^{-d_n}}{3\cdot2^{d_n}}
=\frac13(2\rho)^{-d_n}.
\]
Finally, \(\rho<1/2\), so \((2\rho)^{-1}>1\); as \(d_n\to\infty\), the
right-hand side tends to \(+\infty\).  Therefore
\(\alpha_{\mathrm{cert}}(G_n)\to\infty\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersion_geometric_lower_bound
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.dispersionApproxRatio_tendsto_atTop
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:heterogeneity-separation}]
Throughout, \(e_k\) is the \(k\)-th coordinate vector on \(\mathbb R^{I_n}\), and we write
\[
\eta_{\mathrm{box}}=\min\{\rho-\epsilon,\,1-\epsilon-\rho\}>0,
\]
which is positive because \(\epsilon<\rho<1-\epsilon\).  Recall from \cref{obj:def:kkt-gradient} the summation formula defining the envelope gradient score \(g_k(G_n,p)\).  Throughout we use the reciprocal convention
\[
1/0=0,
\]
so that the normalizing factor \(1/n\), with \(n=\lvert O_n\rvert\), is defined for every finite outcome-unit set: it is the ordinary reciprocal when \(O_n\neq\varnothing\) and equals \(0\) when \(O_n=\varnothing\).  Under this convention the gradient formula of \cref{obj:def:kkt-gradient} and the identities derived from it below hold verbatim, with no separate nonemptiness hypothesis.

\begin{enumerate}
\item Suppose that the homogeneous-point gradient scores are not all equal.  First, \(p^{\mathrm{hom}}\in\mathcal P_{n,B_n,\epsilon}\): each coordinate equals \(\rho\in(\epsilon,1-\epsilon)\subseteq[0,1]\), and
\[
\sum_{k\in I_n}p^{\mathrm{hom}}_k=m_n\rho=B_n.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.homogeneous_feasible

We next show that \(p^{\mathrm{hom}}\) is not a minimizer.  Otherwise, \cref{obj:thm:convex-design} supplies a budget multiplier \(\lambda_n\) and nonnegative upper- and lower-box multipliers \(\nu^+_{k,n}\) and \(\nu^-_{k,n}\) satisfying
\[
g_k(G_n,p^{\mathrm{hom}})
=\lambda_n-\nu^+_{k,n}+\nu^-_{k,n},
\]
together with the complementary-slackness identities
\[
\nu^+_{k,n}\bigl(p^{\mathrm{hom}}_k-(1-\epsilon)\bigr)=0,
\qquad
\nu^-_{k,n}\bigl(\epsilon-p^{\mathrm{hom}}_k\bigr)=0.
\]
Since \(p^{\mathrm{hom}}_k=\rho\) and \(\epsilon<\rho<1-\epsilon\), both factors multiplying the respective multipliers are nonzero.  Thus \(\nu^+_{k,n}=\nu^-_{k,n}=0\) for every \(k\), so all gradient scores equal \(\lambda_n\), a contradiction.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.homogeneous_not_minimizer_of_gradient_spread

Choose \(a\) attaining the maximum and \(b\) attaining the minimum of
\(k\mapsto g_k(G_n,p^{\mathrm{hom}})\), which exist because \(I_n\) is finite and nonempty.  If these extrema were equal, all scores would be equal, contrary to hypothesis.  Hence
\[
\Delta_g
=g_a(G_n,p^{\mathrm{hom}})-g_b(G_n,p^{\mathrm{hom}})
>0.
\]
For these extremal indices, set
\[
d=e_b-e_a,
\qquad
\gamma(s)=\frac14V_{\mathrm{env}}(G_n,p^{\mathrm{hom}}+sd).
\]
The segment \(p^{\mathrm{hom}}+sd\) is feasible for \(s\in[-\eta_{\mathrm{box}},\eta_{\mathrm{box}}]\).  Indeed \(|d_k|\le1\), so for \(0\le s\le\eta_{\mathrm{box}}\) every coordinate satisfies
\[
\epsilon\le\rho-s\le\rho+sd_k\le\rho+s\le1-\epsilon
\]
by the two bounds defining \(\eta_{\mathrm{box}}\); the budget is preserved because
\[
\sum_{k\in I_n}d_k=1-1=0,
\]
so \(\sum_k(p^{\mathrm{hom}}_k+sd_k)=B_n\).  For \(s<0\) the same argument applies with the direction reversed.  Moreover, on \(0\le s\le\eta_{\mathrm{box}}\),
\[
p^{\mathrm{hom}}_k+sd_k\ge\rho-s\ge\epsilon\ge\epsilon/2,
\qquad
1-(p^{\mathrm{hom}}_k+sd_k)\ge1-\rho-s\ge\epsilon\ge\epsilon/2.
\]
On this region the envelope agrees with the globally twice continuously differentiable cutoff extension obtained by replacing each reciprocal factor by a \(C^2\) reciprocal cutoff that agrees with \(x^{-1}\) for \(x\ge\epsilon/2\).  Consequently \(\gamma\) is twice continuously differentiable on \([0,\eta_{\mathrm{box}}]\) and differentiable at \(0\).  The line-calculus identity for \(V_{\mathrm{env}}(G_n,\cdot)/4\) along \(d\) gives
\[
\gamma'(0)=\sum_{k\in I_n}d_k\,g_k(G_n,p^{\mathrm{hom}})
=g_b(G_n,p^{\mathrm{hom}})-g_a(G_n,p^{\mathrm{hom}})
=-\Delta_g.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.envelopeLineC2Data_holds

Define the directional second-order modulus for this same pair by
\[
L_{ab}
=
\sup_{q\in\mathcal P_{n,B_n,\epsilon}}
\left.
\frac{d^2}{dt^2}
\left\{\frac14V_{\mathrm{env}}(G_n,q+t(e_b-e_a))\right\}
\right|_{t=0}.
\]
The supremum is finite: the feasible class is compact by \cref{obj:thm:convex-design}, and the displayed directional curvature is continuous in the base point through the same \(C^2\) cutoff extension.  For \(0\le t\le\eta_{\mathrm{box}}\), the shifted base point \(p^{\mathrm{hom}}+td\) is feasible, and the second derivative of \(\gamma\) at \(t\) is precisely the directional curvature at that base point.  Hence
\[
\gamma''(t)\le L_{ab}
\qquad(0\le t\le\eta_{\mathrm{box}}).
\]
Convexity of \(V_{\mathrm{env}}(G_n,\cdot)\) on \(\mathcal P_{n,B_n,\epsilon}\), supplied by \cref{obj:thm:convex-design}, makes \(\gamma\) convex on the feasible line segment.  Thus the curvature at the homogeneous base point is nonnegative, and since that curvature is one of the feasible values bounded by the displayed supremum,
\[
0\le L_{ab}.
\]

Define
\[
s_*= 
\begin{cases}
\eta_{\mathrm{box}}, & L_{ab}=0,\\
\min\{\eta_{\mathrm{box}},\,\Delta_g/L_{ab}\}, & L_{ab}\ne0 .
\end{cases}
\]
Then \(s_*\in[0,\eta_{\mathrm{box}}]\).  The one-dimensional second-order descent bound applied to \(\gamma\), with initial slope at most \(-\Delta_g\) and second derivative bounded above by \(L_{ab}\) on the segment, gives
\[
\gamma(0)-\gamma(s_*)\ge \frac{\Delta_g}{2}s_*.
\]
The design \(q=p^{\mathrm{hom}}+s_*d\) is feasible by the segment argument above.  Therefore, for every
\(p_n^*(G_n)\in\operatorname{argmin}_{p\in\mathcal P_{n,B_n,\epsilon}}
V_{\mathrm{env}}(G_n,p)\),
\[
V_{\mathrm{env}}(G_n,p_n^*(G_n))
\le V_{\mathrm{env}}(G_n,q).
\]
Multiplying the preceding descent inequality by \(4\), and using \(4\gamma(0)=V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})\) and \(4\gamma(s_*)=V_{\mathrm{env}}(G_n,q)\), gives
\[
V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})
-
V_{\mathrm{env}}(G_n,p_n^*(G_n))
\ge
V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})
-
V_{\mathrm{env}}(G_n,q)
\ge 2\Delta_gs_* ,
\]
which is the asserted bound after substituting \(\eta_{\mathrm{box}}\) and \(s_*\).

Finally, \(p_n^*(G_n)\neq p^{\mathrm{hom}}\), since otherwise
\(p^{\mathrm{hom}}\) would be a minimizer.  Optimality and feasibility of \(p^{\mathrm{hom}}\) give
\(V_{\mathrm{env}}(G_n,p_n^*(G_n))\le V_{\mathrm{env}}(G_n,p^{\mathrm{hom}})\); equality would again make \(p^{\mathrm{hom}}\) a minimizer.  Thus
\[
V_{\mathrm{env}}(G_n,p_n^*(G_n))
<
V_{\mathrm{env}}(G_n,p^{\mathrm{hom}}).
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.envelope_segment_descent_gap

\item Now suppose that \(|N_i(G_n)|=1\) for every \(i\in O_n\), and fix \(k\in I_n\).  If \(k\in S_{ij}(G_n)\), then \(S_{ij}(G_n)\) is a nonempty subset of \(N_i(G_n)\), and hence has cardinality one.  Since \(p^{\mathrm{hom}}\) is constant, every nonzero summand in the gradient formula of \cref{obj:def:kkt-gradient}, evaluated at \(p^{\mathrm{hom}}\), equals
\[
-\rho^{-1}\rho^{-1}+(1-\rho)^{-1}(1-\rho)^{-1}
=((1-\rho)^2)^{-1}-(\rho^2)^{-1}.
\]
For the counting term, the intervention-side degree is
\[
s_k=\left|\{i\in O_n:\ k\in N_i(G_n)\}\right|.
\]
The condition \(k\in S_{ij}(G_n)\) is exactly the conjunction \(k\in N_i(G_n)\) and \(k\in N_j(G_n)\).  Hence the ordered pairs \((i,j)\) contributing to the sum are precisely
\[
\{i\in O_n:\ k\in N_i(G_n)\}\times
\{j\in O_n:\ k\in N_j(G_n)\},
\]
and their number is \(s_k^2\).  Substitution into the gradient formula gives, for \(\rho\in(0,1)\) and \(n=\lvert O_n\rvert\),
\[
g_k(G_n,p^{\mathrm{hom}})
=
\frac{s_k^2}{n}\left((1-\rho)^{-2}-\rho^{-2}\right),
\]
which is the displayed formula.  When \(O_n\neq\varnothing\) both sides are read in the ordinary way; when \(O_n=\varnothing\) the defining double sum is empty and \(s_k=0\) for every \(k\), so both sides are \(0\) under the reciprocal convention fixed above.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.shared_card_eq_one_of_singleton
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.sum_indicator_shared_eq_sdeg_sq_mul
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.singleton_envelopeGrad_eq_sdeg_sq

\item Retain the singleton-neighborhood condition and suppose that
\(s_a^2\neq s_b^2\) for some \(a,b\in I_n\).  Put
\[
C=((1-\rho)^2)^{-1}-(\rho^2)^{-1}.
\]
This scalar is nonzero.  Indeed, if \(C=0\), then
\((1-\rho)^2=\rho^2\); thus either \(1-\rho=\rho\), contradicting
\(\rho\neq1/2\), or \(1-\rho=-\rho\), which would imply \(1=0\).

Also \(n=\lvert O_n\rvert>0\).  Otherwise \(O_n\) would be empty, so \(\{i\in O_n:\ k\in N_i(G_n)\}\) would be empty and \(s_k=0\) for every \(k\), making \(s_a^2\neq s_b^2\) impossible.  Hence \(C/n\neq0\), and the formula of the preceding step gives
\[
g_a(G_n,p^{\mathrm{hom}})-g_b(G_n,p^{\mathrm{hom}})
=\frac{C}{n}\left(s_a^2-s_b^2\right)\neq0,
\]
so that
\[
g_a(G_n,p^{\mathrm{hom}})\neq g_b(G_n,p^{\mathrm{hom}}).
\]
The conclusions of the first step therefore apply.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.singleton_degree_spread_to_gradient_spread
\end{enumerate}
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.heterogeneity_separation
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:surrogate-certificate}]
Put
\[
A(q):=\sum_{k\in I_n}h_k(G_n)\{q_k^{-1}+(1-q_k)^{-1}\},
\qquad
C:=\max\{1,\epsilon^{-(\bar d-1)}\}.
\]
Then \(C\geq 1\), hence \(C\geq0\).  Recall also the three covariance loads, for \(i,j\in O_n\):
\[
r_{ij}^1(G_n,q)=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}q_k^{-1}-1\right),
\quad
r_{ij}^0(G_n,q)=\mathbf 1\{|S_{ij}(G_n)|>0\}\left(\prod_{k\in S_{ij}(G_n)}(1-q_k)^{-1}-1\right),
\]
\[
r_{ij}^{10}(G_n)=\mathbf 1\{|S_{ij}(G_n)|>0\}.
\]

\begin{enumerate}
\item By \cref{obj:thm:convex-design}, whose positivity-margin and feasible-budget hypotheses are exactly those assumed here, the envelope-optimal design
\(p_n^*(G_n)\) is feasible and satisfies
\[
V_{\mathrm{env}}(G_n,p_n^*(G_n))
 \leq V_{\mathrm{env}}(G_n,q)
\quad\text{for every }q\in\mathcal P_{n,B_n,\epsilon}.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.optimalDesign_feasible_minimizes

\item On \(\mathcal P_{n,B_n,\epsilon}\), every \(q_k\) and \(1-q_k\) lies in \([\epsilon,1-\epsilon]\), hence is strictly
positive.  Thus \(A\) is continuous there, being a finite sum of reciprocals of nonvanishing continuous
coordinate functions.  Nonemptiness and compactness from \cref{obj:thm:convex-design}
therefore yield a feasible surrogate design \(p_n^{\deg}(G_n)\) such that
\[
A(p_n^{\deg}(G_n))\leq A(q)
\quad\text{for every }q\in\mathcal P_{n,B_n,\epsilon}.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.surrogateDesign_feasible_minimizes

\item Fix \(q\in\mathcal P_{n,B_n,\epsilon}\), and write
\[
K_{ij}:=r_{ij}^1(G_n,q)+r_{ij}^0(G_n,q)+2r_{ij}^{10}(G_n).
\]
Substituting the definition of \(h_k(G_n)\) into \(A(q)\) and exchanging the order of the finite summations,
\[
A(q)=\sum_{i\in O_n}\sum_{j\in O_n}\sum_{k\in S_{ij}(G_n)}
 n^{-1}|S_{ij}(G_n)|^{-1}
 \{q_k^{-1}+(1-q_k)^{-1}\},
\]
and \cref{obj:def:graph-envelope} gives
\[
\frac{V_{\mathrm{env}}(G_n,q)}4
=\sum_{i\in O_n}\sum_{j\in O_n}n^{-1}K_{ij}.
\]
For a fixed pair \((i,j)\), let \(S=S_{ij}(G_n)\).  If \(S=\varnothing\), then \(K_{ij}=0\) and the corresponding
triple-sum contribution to \(A(q)\) is an empty sum.  Otherwise, since \(0<q_k<1\) for each \(k\in S\), the two reciprocal factors
\(q_k^{-1}\) and \((1-q_k)^{-1}\) are at least one.  Hence the product of the remaining treated reciprocal factors is at least one, so
\[
q_k^{-1}\leq\prod_{\ell\in S}q_\ell^{-1}\qquad(k\in S),
\]
and averaging over \(k\in S\) gives
\[
|S|^{-1}\sum_{k\in S}q_k^{-1}\leq\prod_{k\in S}q_k^{-1}.
\]
The same argument applied to the control reciprocal factors gives
\[
|S|^{-1}\sum_{k\in S}(1-q_k)^{-1}\leq\prod_{k\in S}(1-q_k)^{-1}.
\]
For nonempty \(S\), the displayed definitions of the loads give
\[
K_{ij}=\left(\prod_{k\in S}q_k^{-1}-1\right)+\left(\prod_{k\in S}(1-q_k)^{-1}-1\right)+2
=\prod_{k\in S}q_k^{-1}+\prod_{k\in S}(1-q_k)^{-1}.
\]
Adding the preceding two inequalities yields
\[
|S|^{-1}\sum_{k\in S}\{q_k^{-1}+(1-q_k)^{-1}\}\leq K_{ij}.
\]
Multiplication by \(n^{-1}\geq0\) and summation over \((i,j)\), with the empty-shared-neighborhood pairs contributing zero on both sides, prove
\[
A(q)\leq \frac{V_{\mathrm{env}}(G_n,q)}4.
\]
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.pairAverage_le_envelopeKernel

\item For the reverse bound, retain \(S\neq\varnothing\).  Since
\(S\subseteq N_i(G_n)\), \cref{obj:ass:bounded-outcome-degree}
gives \(|S|\leq d_i\le\bar d\).  Moreover, the feasibility bounds \(\epsilon\le q_k\le1-\epsilon\) give
\[
0\leq q_k^{-1}\leq\epsilon^{-1},
\qquad
0\leq (1-q_k)^{-1}\leq\epsilon^{-1} .
\]
After deleting any single index \(k\in S\), each treated or control reciprocal product has \(|S|-1\) factors and is therefore bounded by
\[
(\epsilon^{-1})^{|S|-1}
=\epsilon^{-(|S|-1)}
 \leq \epsilon^{-(\bar d-1)}
 \leq C,
\]
the first inequality because \(0<\epsilon\le1\) makes \(t\mapsto\epsilon^{-t}\) nondecreasing and \(|S|\le\bar d\).
For nonnegative reciprocal factors whose product after removal of any one factor is at
most \(C\), the full product is at most \(C\) times the factors' average: for each \(k\in S\),
\[
\prod_{\ell\in S}q_\ell^{-1}
=q_k^{-1}\prod_{\ell\in S\setminus\{k\}}q_\ell^{-1}
\le Cq_k^{-1},
\]
and averaging over \(k\in S\) gives the treated reciprocal claim.  Applying the same argument to the control reciprocal factors gives
\[
\prod_{k\in S}q_k^{-1}
 \leq C|S|^{-1}\sum_{k\in S}q_k^{-1},
\qquad
\prod_{k\in S}(1-q_k)^{-1}
 \leq C|S|^{-1}\sum_{k\in S}(1-q_k)^{-1}.
\]
Adding these and using the preceding identity for \(K_{ij}\),
\[
K_{ij}\leq
C|S|^{-1}\sum_{k\in S}\{q_k^{-1}+(1-q_k)^{-1}\}.
\]
If \(S=\varnothing\), then \(K_{ij}=0\) by the displayed load definitions, and the associated triple-sum contribution on the
right-hand side of the final summed inequality is empty.  Multiplying the nonempty-pair inequality by
\(n^{-1}\geq0\), summing over all nonempty \((i,j)\), and adding the empty pairs with zero contribution on both sides, the displayed
expansions above yield
\[
\frac{V_{\mathrm{env}}(G_n,q)}4\leq C\,A(q).
\]
This proves the asserted sandwich for every feasible \(q\).
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.envelopeKernel_le_certificate_pairAverage

\item Let \(V_*:=V_{\mathrm{env}}(G_n,p_n^*(G_n))\).  If \(V_*>0\), the
two sandwich bounds and the minimizing property of \(p_n^{\deg}(G_n)\) established above imply
\[
\frac{V_{\mathrm{env}}\!\left(G_n,p_n^{\deg}(G_n)\right)}4
 \leq C\,A(p_n^{\deg}(G_n))
 \leq C\,A(p_n^*(G_n))
 \leq C\,\frac{V_*}{4},
\]
the middle step because \(p_n^*(G_n)\) is feasible and \(C\ge0\).  Thus
\[
V_{\mathrm{env}}\!\left(G_n,p_n^{\deg}(G_n)\right)\leq C V_*,
\]
and division by \(V_*>0\), together with \cref{obj:def:approximation-ratio},
gives \(\alpha_{\mathrm{cert}}(G_n)\leq C\).  If \(V_*\not>0\), that
definition assigns \(\alpha_{\mathrm{cert}}(G_n)=1\), which is at most
\(C\).  The claimed approximation-ratio bound follows.
% lean: CausalSmith.Experimentation.BipartiteMinimaxDesign.surrogate_certificate
\end{enumerate}
\end{proof}
