\begin{proof}[Proof of \cref{obj:lem:weighted-simplex-active-set}]
Throughout \(M=2m>0\) and every \(\beta_i>0\).

\emph{Part A: \(\kappa>0\).}

\emph{Step A1 (an admissible pair exists).}  Define the threshold function
\[
G(\lambda)=\sum_{i\in\mathcal I}\frac{\bigl(\max\{\lambda-\alpha_i,0\}\bigr)^2}{\beta_i},
\qquad\lambda\in\mathbb R,
\]
which is continuous because it is a finite sum of continuous functions of \(\lambda\).  Let \(\alpha_{\min}=\min_i\alpha_i\) and \(\alpha_{\max}=\max_i\alpha_i\), attained at indices \(k_{\min}\) and \(k_{\max}\).  At \(\lambda=\alpha_{\min}\) every positive part vanishes, so
\[
G(\alpha_{\min})=0\le\kappa^2 .
\]
At \(\lambda_{+}=\alpha_{\max}+\kappa\sqrt{\beta_{k_{\max}}}\) the single summand at \(k_{\max}\) already equals
\[
\frac{\bigl(\kappa\sqrt{\beta_{k_{\max}}}\bigr)^2}{\beta_{k_{\max}}}=\kappa^2,
\]
and all summands are nonnegative, so \(G(\lambda_{+})\ge\kappa^2\).  Since \(\alpha_{\min}\le\lambda_{+}\), the intermediate value theorem yields \(\lambda\in[\alpha_{\min},\lambda_{+}]\) with \(G(\lambda)=\kappa^2\).  Put
\[
S=\{i\in\mathcal I:\alpha_i<\lambda\}.
\]
Off \(S\) the positive part \(\max\{\lambda-\alpha_i,0\}\) vanishes, so
\[
\sum_{i\in S}\frac{(\lambda-\alpha_i)^2}{\beta_i}=G(\lambda)=\kappa^2 ,
\]
and \(S\) is nonempty (otherwise the left side would be \(0\ne\kappa^2\)).  By construction \(\alpha_i<\lambda\) on \(S\) and \(\lambda\le\alpha_j\) off \(S\), so \((S,\lambda)\) is admissible. % lean: exists_admissible

\emph{Step A2 (the active-set point lies in \(\Delta_M\)).}  Write
\[
D=\sum_{h\in S}\frac{\lambda-\alpha_h}{\beta_h}.
\]
Each summand is strictly positive (\(\lambda>\alpha_h\) on \(S\), \(\beta_h>0\)) and \(S\ne\emptyset\), so \(D>0\).  Hence each \(t_i^\star=M\bigl((\lambda-\alpha_i)/\beta_i\bigr)/D\) with \(i\in S\) is nonnegative --- indeed strictly positive --- while \(t_j^\star=0\) off \(S\); and
\[
\sum_{i\in\mathcal I}t_i^\star
=\frac{M}{D}\sum_{i\in S}\frac{\lambda-\alpha_i}{\beta_i}
=\frac{M}{D}\cdot D=M .
\]
Thus \(t^\star\in\Delta_M\), and moreover
\[
t_i^\star>0\iff i\in S .
\]
% lean: activeSetPoint_mem % lean: activeSetPoint_pos_iff

\emph{Step A3 (norm and stationarity at \(t^\star\)).}  Using \(t_i^\star=0\) off \(S\) and the admissibility equation,
\[
\sum_{i\in\mathcal I}\beta_i(t_i^\star)^2
=\frac{M^2}{D^2}\sum_{i\in S}\beta_i\frac{(\lambda-\alpha_i)^2}{\beta_i^2}
=\frac{M^2}{D^2}\sum_{i\in S}\frac{(\lambda-\alpha_i)^2}{\beta_i}
=\Bigl(\frac{M\kappa}{D}\Bigr)^2 ,
\]
so, writing \(N=\sqrt{\sum_i\beta_i(t_i^\star)^2}\),
\[
N=\frac{M\kappa}{D}>0 .
\]
% lean: activeSetPoint_normSq
Consequently, for every \(i\in S\),
\[
\frac{\kappa\,\beta_i t_i^\star}{N}
=\frac{\kappa\,\beta_i}{N}\cdot\frac{M(\lambda-\alpha_i)}{\beta_iD}
=\frac{\kappa M}{ND}(\lambda-\alpha_i)
=\lambda-\alpha_i ,
\]
which is the stationarity relation on the active support.

\emph{Step A4 (the optimal value).}  Multiplying the stationarity relation of Step A3 by \(t_i^\star\), summing over \(i\in S\) (equivalently over all of \(\mathcal I\), since \(t^\star\) vanishes off \(S\)), and using \(\sum_i\beta_i(t_i^\star)^2=N^2\) and \(\sum_it_i^\star=M\),
\[
\sum_{i\in\mathcal I}\alpha_it_i^\star+\kappa N
=\sum_{i\in\mathcal I}\alpha_it_i^\star+\frac{\kappa}{N}\sum_{i\in\mathcal I}\beta_i(t_i^\star)^2
=\lambda\sum_{i\in\mathcal I}t_i^\star=M\lambda ,
\]
i.e.
\[
\Phi(t^\star)=M\lambda .
\]
% lean: activeSetPoint_value

\emph{Step A5 (strict optimality).}  Let \(s\in\Delta_M\) with \(s\ne t^\star\), and write \(N(s)=\sqrt{\sum_i\beta_is_i^2}\).  Since \(\beta\) is strictly positive, \(\sum_is_i=\sum_it_i^\star=M\ne0\) and \(s\ne t^\star\), the strict weighted Cauchy--Schwarz inequality gives
\[
\sum_{i\in\mathcal I}\beta_it_i^\star s_i<N\,N(s).
\]
% lean: weighted_cs_simplex_strict
Subtracting \(\sum_i\beta_i(t_i^\star)^2=N^2\) from both sides and multiplying by \(\kappa/N>0\),
\[
\frac{\kappa}{N}\sum_{i\in\mathcal I}\beta_it_i^\star\,(s_i-t_i^\star)
<\kappa\bigl(N(s)-N\bigr).
\]
By the stationarity relation of Step A3 (and \(t_j^\star=0\) off \(S\)) the left side equals \(\sum_{i\in S}(\lambda-\alpha_i)(s_i-t_i^\star)\), so
\[
\Phi(s)-\Phi(t^\star)
=\sum_{i\in\mathcal I}\alpha_i(s_i-t_i^\star)+\kappa\bigl(N(s)-N\bigr)
>\sum_{i\in\mathcal I}\alpha_i(s_i-t_i^\star)
+\sum_{i\in S}(\lambda-\alpha_i)(s_i-t_i^\star).
\]
Setting \(\gamma_i=\lambda\) for \(i\in S\) and \(\gamma_i=\alpha_i\) for \(i\notin S\), the right-hand side is \(\sum_i\gamma_i(s_i-t_i^\star)\); since \(\gamma_it_i^\star=\lambda t_i^\star\) for every \(i\) (on \(S\) by definition of \(\gamma\), off \(S\) because \(t_i^\star=0\)) and \(\sum_it_i^\star=M\), it equals
\[
\sum_{i\in\mathcal I}\gamma_is_i-\lambda M .
\]
Finally \(\gamma_i\ge\lambda\) for every \(i\) --- with equality on \(S\) and by admissibility \(\alpha_j\ge\lambda\) off \(S\) --- and \(s_i\ge0\), so \(\sum_i\gamma_is_i\ge\lambda\sum_is_i=\lambda M\) and the last display is nonnegative.  Hence
\[
\Phi(t^\star)<\Phi(s)
\qquad\text{for every }s\in\Delta_M\text{ with }s\ne t^\star .
\]
% lean: activeSetPoint_strict_min

\emph{Step A6 (uniqueness of \((S,\lambda)\)).}  Let \((S',\lambda')\) be another admissible pair, with active-set point \(t^{\star\prime}\).  Steps A2 and A5 apply to it as well, so both \(t^\star\) and \(t^{\star\prime}\) are strict minimizers of \(\Phi\) over \(\Delta_M\); if they differed, each would have strictly smaller value than the other, which is impossible.  Hence \(t^\star=t^{\star\prime}\).  By the support identity of Step A2,
\[
i\in S\iff t_i^\star>0\iff t_i^{\star\prime}>0\iff i\in S',
\]
so \(S=S'\).  By Step A4, \(M\lambda=\Phi(t^\star)=\Phi(t^{\star\prime})=M\lambda'\), and \(M>0\) gives \(\lambda=\lambda'\). % lean: activeSetPoint_strict_min % lean: activeSetPoint_pos_iff % lean: activeSetPoint_value

\emph{Part B: \(\kappa=0\).}  Here \(\Phi(t)=\sum_i\alpha_it_i\) is linear.  Fix \(k\) with \(\alpha_k=\min_j\alpha_j\).

If \(t\in\Delta_M\) minimizes \(\Phi\) over \(\Delta_M\), compare it with the vertex of \(\Delta_M\) that places all the mass on coordinate \(k\), i.e.\ the point \(w\) with \(w_k=M\) and \(w_j=0\) for \(j\ne k\), whose value is \(\Phi(w)=M\alpha_k\):
\[
\sum_{i\in\mathcal I}\alpha_it_i\le M\alpha_k=\alpha_k\sum_{i\in\mathcal I}t_i,
\qquad\text{i.e.}\qquad
\sum_{i\in\mathcal I}(\alpha_i-\alpha_k)t_i\le0 .
\]
Every summand is nonnegative (\(\alpha_i\ge\alpha_k\), \(t_i\ge0\)), so every summand vanishes; hence \(t_i\ne0\) forces \(\alpha_i=\alpha_k\le\alpha_j\) for every \(j\).  That is exactly membership in the exposed \(\alpha\)-minimizing face.

Conversely, if \(t\in\Delta_M\) is supported only on \(\alpha\)-minimizing coordinates, then \(\alpha_it_i=\alpha_kt_i\) for every \(i\) and
\[
\Phi(t)=\alpha_k\sum_{i\in\mathcal I}t_i=M\alpha_k ,
\]
whereas every \(s\in\Delta_M\) satisfies \(\Phi(s)=\sum_i\alpha_is_i\ge\alpha_k\sum_is_i=M\alpha_k\).  So \(t\) is a minimizer.  The two implications give the stated equivalence, and the minimizer set is precisely the exposed \(\alpha\)-minimizing face of \(\Delta_M\). % lean: kappa_zero_face
\end{proof}
