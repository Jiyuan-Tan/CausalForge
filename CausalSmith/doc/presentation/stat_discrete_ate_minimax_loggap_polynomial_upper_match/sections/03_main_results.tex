\section{Main results}

The main theorem gives the fixed-interior minimax characterization for the risk in \cref{obj:def:minimax-risk}. The statement uses the same overlap-restricted experiment class, ATE functional, and hybrid estimator introduced in \cref{obj:def:experiment-class,obj:def:ate-functional,obj:def:hybrid-estimator-handle}. For each fixed overlap parameter in the interior range \(0<\epsilon<1/2\), it identifies the mean-squared-error scale
\[
r_{n,d}
=
\frac1n+\frac{d^2}{n^2(\log n)^2}
\]
through matching lower and upper bounds in the displayed calibration range.

\begin{theoremv}{thm:sharp-minimax-fixed-interior}[Sharp fixed-interior minimax rate]*
For an overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\), there exist constants \(a_\epsilon,\rho_\epsilon,C_\epsilon>0\) and an integer \(N_\epsilon<\infty\) such that the following statements hold. Let
\[
r_{n,d}=\frac{1}{n}+\frac{d^2}{n^2(\log n)^2}.
\]
\begin{itemize}
\item \textbf{(Sharp sandwich.)} For every \(n,d\), every single-observation law \(P\), and every sample law \(P^{\otimes n}\in\mathcal E_{n,d,\epsilon}\) as in \cref{obj:def:experiment-class}, if \(d>0\), \(n\ge N_\epsilon\), and
\[
d\le \rho_\epsilon n\log n,
\]
then the minimax mean-squared-error risk \(\mathsf R_{n,d,\epsilon}\) from \cref{obj:def:minimax-risk} and the hybrid estimator \(\widehat\tau_n^{\mathrm{hyb}}\) from \cref{obj:def:hybrid-estimator-handle} satisfy
\[
a_\epsilon r_{n,d}
\le
\mathsf R_{n,d,\epsilon}
\le
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
E_P\!\left[\left(\widehat\tau_n^{\mathrm{hyb}}-\tau(P)\right)^2\right]
\le
C_\epsilon r_{n,d}.
\]

\item \textbf{(Parametric interior.)} For every \(K>0\), there exists \(C_K>0\) such that, for every \(n,d\), every single-observation law \(P\), and every sample law \(P^{\otimes n}\in\mathcal E_{n,d,\epsilon}\), if \(d>0\), \(n\ge N_\epsilon\),
\[
d\le K\sqrt n\log n,
\qquad
d\le \rho_\epsilon n\log n,
\]
then
\[
\frac{a_\epsilon}{n}
\le
\mathsf R_{n,d,\epsilon}
\le
\frac{C_K}{n}.
\]

\item \textbf{(Consistency threshold.)} For any sequences \(n_j,d_j\) with \(n_j\to\infty\), with \(d_j>0\) eventually, and with
\[
d_j\le \rho_\epsilon n_j\log n_j
\]
eventually, one has
\[
\mathsf R_{n_j,d_j,\epsilon}\to 0
\quad\Longleftrightarrow\quad
\frac{d_j}{n_j\log n_j}\to 0.
\]
\end{itemize}
\end{theoremv}

\Cref{obj:thm:sharp-minimax-fixed-interior} has three immediate implications for the finite-alphabet observational problem. First, the large-alphabet term is \(d^2/\{n^2(\log n)^2\}\), matching the logarithmic scale suggested by the lower-bound phenomenon in \citet{ZengBalakrishnanHanKennedy2024}. Second, the ordinary \(1/n\) parametric rate persists throughout the interior regime \(d=O(\sqrt n\log n)\), subject to the theorem's fixed-overlap calibration range. Third, along admissible sequences in that range, consistency is characterized by \(d=o(n\log n)\).

The intuition is that the risk separates into a sampling-variance component and a sparse-cell approximation component. Heavy categories contribute through stabilized empirical ratios, while light categories are handled by the factorial-moment polynomial branch in \cref{obj:def:hybrid-estimator-handle}. The Chebyshev construction follows the approximation-theoretic approach used for nonsmooth large-alphabet functionals \citep{JiaoVenkatHanWeissman2015,WuYang2016,HanJiaoWeissman2020,CaiLow2011}; here it is applied to the cell contribution \(\phi\) and summed across unrestricted covariate categories.

The second theorem records the operational and endpoint consequences of the same construction. It states that the hybrid estimator has a finite arithmetic implementation with fixed numerical tuning, gives its fixed-interior envelope, combines it with a centered estimator near the randomized endpoint, and brackets the endpoint risk.

\begin{theoremv}{thm:overlap-adaptive-universal-hybrid}[Universal hybrid envelope]
With \(\widehat\tau_n^{\mathrm{hyb}}\) as in \cref{obj:def:hybrid-estimator-handle} and \(\mathsf R_{n,d,\epsilon}\) as in \cref{obj:def:minimax-risk}, the following conclusions hold.
\begin{itemize}
\item \textbf{Computability.} There is a constant \(K\in\mathbb N\), \(K>0\), such that for every \(n,d\in\mathbb N\) there is a finite real-arithmetic program which reads the split count vector, returns \(\widehat\tau_n^{\mathrm{hyb}}\) exactly on every sample \(O_1,\ldots,O_n\), and uses at most \(K d M(n)^4\) arithmetic and comparison operations.
\item \textbf{Fixed-interior calibration.} For every overlap parameter \(\epsilon\) with \(0<\epsilon<1/2\), there are constants \(C_\epsilon,\rho_\epsilon>0\) and \(N_\epsilon\in\mathbb N\) such that, for all \(n,d\in\mathbb N\) with \(n>0\), \(d>0\), \(N_\epsilon\le n\), and \(d\le \rho_\epsilon n\log n\),
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
E_P\!\left[(\widehat\tau_n^{\mathrm{hyb}}-\tau(P))^2\right]
\le
C_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}.
\]
For the deterministic estimator \(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}\) that equals \(\widehat\tau_n^{\mathrm{hyb}}\) when
\[
C_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}
\le
\frac1n+4(1/2-\epsilon)^2
\]
and equals \(\widehat\tau_{\mathrm{ctr}}\) otherwise, the same \(n,d,\epsilon\) satisfy
\[
\mathsf R_{n,d,\epsilon}
\le
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
E_P\!\left[(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}-\tau(P))^2\right],
\]
and
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
E_P\!\left[(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}-\tau(P))^2\right]
\le
\max\{C_\epsilon,4\}
\left[
\frac1n+
\min\left\{\frac{d^2}{n^2(\log n)^2},(1/2-\epsilon)^2\right\}
\right].
\]
\item \textbf{Centered randomization bound.} For every overlap parameter \(\epsilon\) with \(0<\epsilon\le 1/2\) and every \(n,d\in\mathbb N\) with \(n>0\) and \(d>0\), the centered estimator
\[
\widehat\tau_{\mathrm{ctr}}
=
\frac1n\sum_{i=1}^n
2(2A_i-1)\left(Y_i-\frac12\right)
\]
satisfies
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
E_P\!\left[(\widehat\tau_{\mathrm{ctr}}-\tau(P))^2\right]
\le
\frac1n+4(1/2-\epsilon)^2.
\]
\item \textbf{Endpoint bracket.} For every \(n,d\in\mathbb N\) with \(n>0\) and \(d>0\),
\[
\frac{1}{100n}\le \mathsf R_{n,d,1/2}\le \frac1n.
\]
\end{itemize}
\end{theoremv}

\Cref{obj:thm:overlap-adaptive-universal-hybrid} makes the estimator in \cref{obj:def:hybrid-estimator-handle} an explicit finite-sample object: for each \(n\) and \(d\), it is computed from split counts using a number of arithmetic and comparison operations of order \(dM(n)^4\). The same fixed numerical tuning applies across fixed interior overlap classes; the constants in the risk envelope depend on \(\epsilon\), while the construction itself is the one already displayed in \cref{obj:def:hybrid-estimator-handle}.

The centered randomization bound explains the endpoint behavior. As \(\epsilon\) approaches \(1/2\), treatment assignment is close to randomized within each category, and the centered linear estimator has worst-case risk bounded by \(1/n+4(1/2-\epsilon)^2\). The selected estimator compares this bound with the fixed-interior hybrid envelope and obtains the combined rate
\[
\frac1n+
\min\left\{\frac{d^2}{n^2(\log n)^2},(1/2-\epsilon)^2\right\}
\]
up to the displayed constant. At the exact randomized endpoint \(\epsilon=1/2\), the minimax risk is bracketed at the parametric scale \(1/n\), uniformly over positive \(d\).
