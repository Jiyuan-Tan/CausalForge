\begin{proof}[Proof of \cref{obj:lem:one-category-bernoulli-lower}]
Fix \(n,d\in\mathbb N\) with \(n>0\) and \(d>0\), and put
\[
\delta=\frac{2/5}{\sqrt n}.
\]
Since \(\sqrt n\ge1\),
\[
0\le\delta\le\frac25,
\qquad
\delta^2=\frac{4}{25n},
\qquad
\frac12+\delta\le\frac{9}{10}\le1 .
\]
% lean: one_category_bernoulli_lower (hdelta0, hdeltasq, hdeltaU)

\textbf{Step 1: the two competing laws.}
Define two laws \(P_0,P_1\in\mathsf{Prob}(\{1,\ldots,d\}\times\{0,1\}^2)\) as in \cref{obj:def:finite-alphabet-probability-laws} by prescribing, for every category \(k\in\{1,\ldots,d\}\), the four-cell vector \(q_k=(q_{0k0},q_{0k1},q_{1k0},q_{1k1})\) of \cref{obj:def:finite-alphabet-probability-laws}, whose entries are the cell masses \(q_{aky}\) of \((X,A,Y)=(k,a,y)\) listed in the treatment--outcome order \((a,y)=(0,0),(0,1),(1,0),(1,1)\):
\[
q_k(P_0)=\left(\frac{1}{4d},\ \frac{1}{4d},\ \frac{1}{4d},\ \frac{1}{4d}\right),
\qquad
q_k(P_1)=\left(\frac{1}{4d},\ \frac{1}{4d},\ \frac{1}{2d}\left(\frac12-\delta\right),\ \frac{1}{2d}\left(\frac12+\delta\right)\right).
\]
Both prescriptions are legitimate laws: all coordinates are nonnegative because \(0\le\delta\le2/5<1/2\), and in each case the \(4d\) masses sum to
\(d\cdot\{1/(4d)+1/(4d)+1/(2d)\}=1\).
Equivalently, under both laws \(X\) is uniform on \(\{1,\ldots,d\}\) and \(A\mid X=k\sim\mathsf{Bernoulli}(1/2)\) in the sense of \cref{obj:def:bernoulli-law}; the control arm has \(Y\mid A=0,X=k\sim\mathsf{Bernoulli}(1/2)\) under both laws, while the treated arm has \(Y\mid A=1,X=k\sim\mathsf{Bernoulli}(1/2)\) under \(P_0\) and \(Y\mid A=1,X=k\sim\mathsf{Bernoulli}(1/2+\delta)\) under \(P_1\).
% lean: endpointParametricLaw, endpointParametricPertLaw, endpointParametricLaw_jointMass, endpointParametricPertLaw_jointMass, Parametric.validDGP_null, Parametric.validDGP_pert

\textbf{Step 2: both product laws lie in the class.}
For \(j\in\{0,1\}\) and every \(k\), summing the four displayed coordinates and the two treated coordinates gives
\[
p_k=\sum_{a,y\in\{0,1\}}q_{aky}(P_j)=\frac1d>0,
\qquad
\pi_k=\frac{q_{1k0}(P_j)+q_{1k1}(P_j)}{p_k}=\frac{1/(2d)}{1/d}=\frac12 .
\]
Hence \(1/2\le\pi_k\le1-1/2\) for every category, so each \(P_j\) satisfies \cref{obj:ass:overlap} with overlap constant \(\epsilon=1/2\), and therefore
\[
P_0^{\otimes n}\in\mathcal E_{n,d,1/2},
\qquad
P_1^{\otimes n}\in\mathcal E_{n,d,1/2}
\]
in the sense of \cref{obj:def:experiment-class}.
% lean: endpointParametricLaw_overlap, endpointParametricPertLaw_overlap, ExperimentClass

\textbf{Step 3: the two estimand values.}
The conditional outcome means of \cref{obj:def:finite-alphabet-probability-laws} are read off the same coordinates: for every \(k\),
\[
\mu_{0k}(P_j)=\frac{q_{0k1}(P_j)}{q_{0k0}(P_j)+q_{0k1}(P_j)}=\frac12\quad(j=0,1),
\qquad
\mu_{1k}(P_0)=\frac{1/(4d)}{1/(2d)}=\frac12,
\qquad
\mu_{1k}(P_1)=\frac{(1/(2d))(1/2+\delta)}{1/(2d)}=\frac12+\delta .
\]
Substituting into the weighted-regression form of \cref{obj:def:ate-functional} and using \(p_k=1/d\),
\[
\tau(P_0)=\sum_{k=1}^{d}p_k\{\mu_{1k}(P_0)-\mu_{0k}(P_0)\}=0,
\qquad
\tau(P_1)=\sum_{k=1}^{d}\frac1d\,\delta=\delta .
\]
% lean: endpointParametricLaw_ate, endpointParametricPertLaw_ate, ateFunctional_eq_weighted_regression

\textbf{Step 4: the two sample laws are statistically indistinguishable.}
For probability laws \(\mu,\nu\) on a common finite alphabet \(\mathcal Z\) with \(\nu\) charging every point, write
\[
\chi^2(\mu\|\nu)=\sum_{z\in\mathcal Z}\frac{\{\mu(z)-\nu(z)\}^2}{\nu(z)},
\qquad
\operatorname{TV}(\mu,\nu)=\sup_{A\subseteq\mathcal Z}|\mu(A)-\nu(A)| .
\]
The laws \(P_0\) and \(P_1\) differ only in the two treated cells of each category, where the masses differ by \(\pm\delta/(2d)\), and \(P_0\) puts mass \(1/(4d)\) on each of those cells; summing over the \(2d\) treated cells gives the single-observation value
\[
\chi^2(P_1\|P_0)
=2d\cdot\frac{\{\delta/(2d)\}^2}{1/(4d)}
=2\delta^2
=\frac{(1/2)\,\delta^2}{(1/2)(1-1/2)} .
\]
Because the \(n\) observations are independent and identically distributed under each product law, the shifted divergence tensorizes, and \(1+x\le e^{x}\) gives
\[
1+\chi^2\!\left(P_1^{\otimes n}\,\big\|\,P_0^{\otimes n}\right)
=\left\{1+\chi^2(P_1\|P_0)\right\}^{n}
=\left(1+2\delta^2\right)^{n}
\le
\exp\!\left(2n\delta^2\right).
\]
With \(\delta^2=4/(25n)\) the exponent is calibrated below \(\log 2\):
\[
2n\delta^2=\frac{8}{25}\le\log 2 ,
\]
since \(\log2>0.693>8/25\). Hence \(1+\chi^2(P_1^{\otimes n}\|P_0^{\otimes n})\le2\), that is,
\[
\chi^2\!\left(P_1^{\otimes n}\,\big\|\,P_0^{\otimes n}\right)\le1 ,
\]
and the Cauchy--Schwarz comparison \(\operatorname{TV}(\mu,\nu)\le\tfrac12\sqrt{\chi^2(\mu\|\nu)}\) yields
\[
\operatorname{TV}\!\left(P_0^{\otimes n},P_1^{\otimes n}\right)\le\frac12\sqrt1=\frac12 .
\]
% lean: Parametric.one_add_chiSqDiv_obsPert_obsNull, Parametric.chiSqDiv_productLaw_le_one, Parametric.tvDist_productLaw_le_half

\textbf{Step 5: two-point testing bound.}
Let \(\widehat\tau\) be any measurable estimator based on \(O_1,\ldots,O_n\), as in \cref{obj:def:minimax-risk}. By Step 3 the two estimand values are separated at radius \(\delta/2\):
\[
2\cdot\frac{\delta}{2}=\delta=\left|\tau(P_0)-\tau(P_1)\right| .
\]
No number is within \(\delta/2\) of both \(\tau(P_0)\) and \(\tau(P_1)\), so the two events \(\{|\widehat\tau-\tau(P_0)|<\delta/2\}\) and \(\{|\widehat\tau-\tau(P_1)|<\delta/2\}\) are disjoint; comparing the two laws on these events and using the bound of Step 4 gives
\[
\max_{j=0,1}
P_j^{\otimes n}\!\left(\left|\widehat\tau-\tau(P_j)\right|\ge\frac{\delta}{2}\right)
\ge
\frac{1-\operatorname{TV}\!\left(P_0^{\otimes n},P_1^{\otimes n}\right)}{2}
\ge\frac{1-1/2}{2}
=\frac14 .
\]
% lean: Causalean.Stat.two_point_lower_bound_of_tvDist_le, Causalean.Stat.real_two_point_lower_bound

\textbf{Step 6: from miss probability to mean squared error.}
For each \(j=0,1\), the miss event coincides with
\(\left\{(\widehat\tau-\tau(P_j))^2\ge(\delta/2)^2\right\}\),
and the squared loss is nonnegative, so integrating it over that event gives
\[
\left(\frac{\delta}{2}\right)^2
P_j^{\otimes n}\!\left(\left|\widehat\tau-\tau(P_j)\right|\ge\frac{\delta}{2}\right)
\le
E_{P_j^{\otimes n}}\!\left[(\widehat\tau-\tau(P_j))^2\right].
\]
Multiplying the bound of Step 5 by \((\delta/2)^2\ge0\) and using \(\delta^2=4/(25n)\),
\[
\max_{j=0,1}
E_{P_j^{\otimes n}}\!\left[(\widehat\tau-\tau(P_j))^2\right]
\ge
\left(\frac{\delta}{2}\right)^2\cdot\frac14
=
\frac{\delta^2}{16}
=
\frac{1}{100n}.
\]
% lean: one_category_bernoulli_lower (hmse0, hmse1, htwo)

\textbf{Step 7: passing to the minimax risk.}
By Step 2 both \(P_0^{\otimes n}\) and \(P_1^{\otimes n}\) are members of \(\mathcal E_{n,d,1/2}\), and for a fixed estimator the risks over the class are bounded above uniformly, because the sample space is finite and \(|\tau(P)|\le1\) for every law in the class. Hence the worst-case risk of \(\widehat\tau\) dominates each of the two risks in Step 6:
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,1/2}}
E_{P^{\otimes n}}\!\left[(\widehat\tau-\tau(P))^2\right]
\ge
\max_{j=0,1}
E_{P_j^{\otimes n}}\!\left[(\widehat\tau-\tau(P_j))^2\right]
\ge
\frac{1}{100n}.
\]
The estimator \(\widehat\tau\) was an arbitrary measurable estimator, so taking the infimum over all such estimators in \cref{obj:def:minimax-risk} gives
\[
\mathsf R_{n,d,1/2}\ge\frac{1}{100n},
\]
which is the claim.
% lean: mse_le_estimator_abs_sum_bound, worstCaseMSE, minimaxRisk, one_category_bernoulli_lower

\end{proof}
