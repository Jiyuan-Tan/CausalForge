\begin{proof}[Proof of \cref{obj:lem:heavy-residual-bound}]
For an observation array \(z=(O_i)_{i\in I}\), with \(O_i=(X_i,A_i,Y_i)\), write
\[
N_k(z)=\sum_{i\in I}\mathbf 1\{X_i=k\},
\qquad
N_{ak}(z)=\sum_{i\in I}\mathbf 1\{X_i=k,A_i=a\},
\]
and
\[
N_{ak1}(z)=\sum_{i\in I}\mathbf 1\{X_i=k,A_i=a,Y_i=1\}.
\]
Let \(m=|I|\), and put
\[
C_{ak}(z)
=
N_k(z)\frac{\mathbf 1\{N_{ak}(z)>0\}}{N_{ak}(z)},
\qquad
\xi_{ak,i}(z)
=
\mathbf 1\{X_i=k,A_i=a\}\{Y_i-\mu_{ak}\}.
\]
With the convention that the displayed ratio is zero when \(N_{ak}(z)=0\), the residual can be written as
\[
\mathcal R_a(H)
=
\sum_{k\in H}C_{ak}(z)\sum_{i\in I}\xi_{ak,i}(z),
\]
because \(\sum_i\xi_{ak,i}=N_{ak1}(z)-\mu_{ak}N_{ak}(z)\).

Expand the square as a double sum over pairs \((k,i)\) and \((l,j)\). If \(i\ne j\), then the factor
\(C_{ak}(z)C_{al}(z)\xi_{al,j}(z)\) is unchanged when only the outcome coordinate of observation \(i\) is replaced; conditioning on all other coordinates and using the centered arm-category residual gives integral zero. If \(i=j\) and \(k\ne l\), then the two category indicators cannot both hold for the same observation, so the product of residuals is identically zero. Hence only the diagonal terms remain:
\[
\mathbb E_{P^{\otimes I}}\!\left[\mathcal R_a(H)^2\right]
=
\sum_{k\in H}\sum_{i\in I}
\mathbb E_{P^{\otimes I}}\!\left[C_{ak}(z)^2\xi_{ak,i}(z)^2\right].
\]
% lean: integral_fixedRatioResidual_sq_eq_diagonal

For each fixed \(k\) and \(i\), the multiplier \(C_{ak}(z)^2\) is nonnegative and depends only on the category and treatment coordinates of the array. The one-coordinate conditional Bernoulli variance bound gives
\[
\mathbb E_{P^{\otimes I}}\!\left[C_{ak}(z)^2\xi_{ak,i}(z)^2\right]
\le
\mathbb E_{P^{\otimes I}}\!\left[
C_{ak}(z)^2\mathbf 1\{X_i=k,A_i=a\}
\right].
\]
Therefore, for each fixed \(k\),
\[
\begin{aligned}
\sum_{i\in I}
\mathbb E_{P^{\otimes I}}\!\left[C_{ak}(z)^2\xi_{ak,i}(z)^2\right]
&\le
\mathbb E_{P^{\otimes I}}\!\left[
\sum_{i\in I}C_{ak}(z)^2\mathbf 1\{X_i=k,A_i=a\}
\right]\\
&=
\mathbb E_{P^{\otimes I}}\!\left[
N_k(z)^2\frac{\mathbf 1\{N_{ak}(z)>0\}}{N_{ak}(z)}
\right].
\end{aligned}
\]
% lean: integral_ratioResidual_diagonal_le
% lean: sum_tupleRatioCoeff_sq_armIndicator_eq

Let \(\rho_{ak}=P(A=a\mid X=k)\), so \(\rho_{1k}=\pi_k\) and \(\rho_{0k}=1-\pi_k\). For \(k\in H\), the positive-mass hypothesis gives \(p_k>0\). Overlap gives \(\rho_{ak}\ge\epsilon\), and \(\rho_{ak}\le1\) because it is an arm probability. Put
\[
C_k=\{X=k\},
\qquad
R_{ak}=\{X=k,A=a\}.
\]
Then \(R_{ak}\subset C_k\), and under the one-observation law
\[
P(C_k)=p_k,
\quad
P(R_{ak})=p_k\rho_{ak},
\quad
P(C_k\setminus R_{ak})=p_k(1-\rho_{ak}),
\quad
P(C_k^c)=1-p_k.
\]
Partition the product sample space according to
\[
U=\{i:O_i\in C_k\},
\qquad
V=\{i:O_i\in R_{ak}\}.
\]
The nested relation forces \(V\subset U\). For fixed \(U,V\) with \(V\subset U\), independence gives the product probability
\[
(p_k\rho_{ak})^{|V|}\{p_k(1-\rho_{ak})\}^{|U|-|V|}(1-p_k)^{m-|U|},
\]
and the integrand equals
\[
|U|^2\frac{\mathbf 1\{|V|>0\}}{|V|}.
\]
Summing over all such pairs and then collecting first by \(t=|U|\) and then by \(s=|V|\) gives
\[
\begin{aligned}
&\mathbb E_{P^{\otimes I}}\!\left[
N_k^2\frac{\mathbf 1\{N_{ak}>0\}}{N_{ak}}
\right] \\
&\quad=
\sum_{t=0}^{m}{m\choose t}p_k^t(1-p_k)^{m-t}t^2
\sum_{s=0}^{t}{t\choose s}\rho_{ak}^{\,s}(1-\rho_{ak})^{t-s}
\frac{\mathbf 1\{s>0\}}{s}.
\end{aligned}
\]

It remains to bound the inner reciprocal-count sum. For every integer \(s\ge0\), with the left side interpreted as zero at \(s=0\),
\[
\frac{\mathbf 1\{s>0\}}{s}\le \frac{2}{s+1}.
\]
Since \(0<\rho_{ak}\le1\), the binomial weights are nonnegative, and therefore
\[
\begin{aligned}
&\sum_{s=0}^{t}{t\choose s}\rho_{ak}^{\,s}(1-\rho_{ak})^{t-s}
\frac{\mathbf 1\{s>0\}}{s}\\
&\quad\le
2\sum_{s=0}^{t}{t\choose s}\rho_{ak}^{\,s}(1-\rho_{ak})^{t-s}\frac{1}{s+1}\\
&\quad=
\frac{2}{(t+1)\rho_{ak}}
\sum_{s=0}^{t}{t+1\choose s+1}\rho_{ak}^{\,s+1}(1-\rho_{ak})^{t-s}\\
&\quad=
\frac{2\{1-(1-\rho_{ak})^{t+1}\}}{(t+1)\rho_{ak}}
\le
\frac{2}{(t+1)\rho_{ak}}.
\end{aligned}
\]
Here the middle equality uses
\({t\choose s}/(s+1)={t+1\choose s+1}/(t+1)\), and the final inequality uses \((1-\rho_{ak})^{t+1}\ge0\). Hence, using \(t^2/(t+1)\le t\) and the binomial first moment,
\[
\begin{aligned}
\mathbb E_{P^{\otimes I}}\!\left[
N_k^2\frac{\mathbf 1\{N_{ak}>0\}}{N_{ak}}
\right]
&\le
\sum_{t=0}^{m}{m\choose t}p_k^t(1-p_k)^{m-t}\frac{2t}{\rho_{ak}}\\
&=
\frac{2mp_k}{\rho_{ak}}
\le
\frac{2mp_k}{\epsilon}.
\end{aligned}
\]
% lean: ate_nested_count_ratio_integral_le

Summing over \(k\in H\) gives
\[
\mathbb E_{P^{\otimes I}}\!\left[\mathcal R_a(H)^2\right]
\le
\sum_{k\in H}\frac{2mp_k}{\epsilon}
=
\frac{2m}{\epsilon}\sum_{k\in H}p_k .
\]
% lean: integral_fixedRatioResidual_sq_le
\end{proof}
