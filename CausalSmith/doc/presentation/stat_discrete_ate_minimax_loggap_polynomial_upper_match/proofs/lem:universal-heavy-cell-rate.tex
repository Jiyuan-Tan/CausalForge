\begin{proof}[Proof of \cref{obj:lem:universal-heavy-cell-rate}]
Throughout, \(L=\log(en)\), \(m_0=|I_0|=\lfloor n/2\rfloor\) and \(m_1=|I_1|=n-\lfloor n/2\rfloor\) are as in \cref{obj:def:hybrid-estimator-handle}, \(p_k\) is the mass of category \(k\), \(\mu_{ak}=\mathbb E_P[Y\mid X=k,A=a]\), \(q_k\) is the four-cell vector of category \(k\), and
\[
\pi_{ak}=\Pr\nolimits_P(A=a\mid X=k),
\qquad\text{so}\qquad
\pi_{1k}=\pi_k,\quad \pi_{0k}=1-\pi_k .
\]
By \cref{obj:ass:overlap}, \(\pi_{ak}\ge\epsilon\) for both arms \(a\in\{0,1\}\) whenever \(p_k>0\).

\textbf{Step 0: the constants.}
Apply the numerical (\(t=256\), \(K=4\)) part of \cref{obj:lem:pilot-sandwich} with growth constant \(c=1\).  It supplies \(C>0\) and a cutoff \(N_{\mathrm{bad}}\) such that, writing
\[
\mathcal B
=
\left\{\widehat{\mathcal H}\not\subseteq\{k:p_k\ge B_-\}\right\}
\cup
\left\{\widehat{\mathcal L}\not\subseteq\{k:p_k\le B_+\}\right\},
\qquad
B_-=\frac{256L}{2m_0},\quad B_+=\frac{512L}{m_0},
\]
for the pilot heavy set at threshold \(256\), one has
\[
\Pr\nolimits_{P^{\otimes n}}(\mathcal B)\le Cn^{-4}
\qquad\text{whenever } n\ge N_{\mathrm{bad}}\text{ and } d\le nL .
\]
Set
\[
\kappa_\epsilon=\frac{16}{\epsilon}+12+\frac1{\epsilon^4},\qquad
C_{\mathrm{bad}}=4C,\qquad
C_\epsilon=4\kappa_\epsilon+C_{\mathrm{bad}},
\]
\[
\rho_\epsilon=1,\qquad
N_\epsilon=\max\{8,N_0,N_{\mathrm{bad}}\},
\]
with \(N_0\) the calibration cutoff of \cref{obj:def:hybrid-estimator-handle}.  Since \(\epsilon>0\) and \(C>0\), both \(C_\epsilon>0\) and \(\rho_\epsilon>0\).
% lean: pilot_sandwich_256, heavy_bad_pilot_rate, universal_heavy_cell_rate

Fix \(n,d,P,\mu_n\) satisfying the three hypotheses.  Membership in \(\mathcal E_{n,d,\epsilon}\) gives \(\mu_n=P^{\otimes n}\) and overlap at \(\epsilon\), so all integrals below are against \(P^{\otimes n}\).  From \(n\ge N_\epsilon\) we get \(n\ge8\), \(n\ge N_0\) and \(n\ge N_{\mathrm{bad}}\); and \(\log n\le L\) turns the dimension hypothesis \(d\le\rho_\epsilon n\log n=n\log n\) into \(d\le nL\), so the pilot bound above is in force.  Since \(n\ge 8\),
\[
m_1\ge \frac n2\ge 4,\qquad 2m_0\le n,\qquad 4(m_1-2)\ge n,\qquad L=1+\log n>0 .
\]
Abbreviate the heavy error by
\[
\mathcal D=\widehat T_{\mathcal H}-\sum_{k\in\widehat{\mathcal H}_n}\phi(q_k),
\]
and split the integral over \(\mathcal B\) and its complement,
\[
\int\mathcal D^2\,dP^{\otimes n}
=
\int_{\mathcal B}\mathcal D^2\,dP^{\otimes n}
+
\int_{\mathcal B^c}\mathcal D^2\,dP^{\otimes n}.
\]
% lean: integral_add_compl, splitSize_one_eq, splitSize_zero_eq, logScale

\textbf{Step 1: the pilot-bad event.}
Both \(\widehat T_{\mathcal H}\) and its target lie in \([-1,1]\).  For the estimator, each of the two empirical arm ratios in \(\widehat h_k\) lies in \([0,1]\), being a count divided by a larger count (or \(0\) when the arm count vanishes), so their difference has modulus at most one and
\[
|\widehat h_k|\le \frac{N^{(1)}_k}{m_1},
\qquad\text{hence}\qquad
|\widehat T_{\mathcal H}|
\le\sum_{k\in\widehat{\mathcal H}_n}\frac{N^{(1)}_k}{m_1}
\le\sum_{k=1}^d\frac{N^{(1)}_k}{m_1}
=\frac{m_1}{m_1}=1,
\]
the last equality because every index of \(I_1\) is counted in exactly one category.
% lean: empirical_arm_ratio_mem_unitInterval, abs_empiricalRatioCell_le, sum_splitCategoryCount, abs_heavyContribution_le_one

For the target, evaluating \cref{obj:def:cell-functional} at the four cell masses of category \(k\) gives
\[
\phi(q_k)=p_k(\mu_{1k}-\mu_{0k}),
\qquad\text{so}\qquad
|\phi(q_k)|\le p_k ,
\]
because \(\mu_{1k},\mu_{0k}\in[0,1]\); summing and using \(\sum_{k=1}^dp_k=1\) gives \(\left|\sum_{k\in\widehat{\mathcal H}_n}\phi(q_k)\right|\le1\).
% lean: cellPhi_cellVector_eq_weighted, abs_cellPhi_cellVector_le_mass_unconditional, sum_cellMass_eq_one, abs_targetHeavy_le_one

Consequently \(|\mathcal D|\le2\), so \(\mathcal D^2\le4\) pointwise and
\[
\int_{\mathcal B}\mathcal D^2\,dP^{\otimes n}
\le
4\Pr\nolimits_{P^{\otimes n}}(\mathcal B)
\le
4Cn^{-4}
=
C_{\mathrm{bad}}n^{-4}.
\]
% lean: heavy_component_sq_le_four, heavy_bad_pilot_setIntegral_le, heavy_bad_pilot_rate

\textbf{Step 2: the selected categories are heavy in population off \(\mathcal B\).}
Because \(n\ge N_0\), \cref{obj:def:hybrid-estimator-handle} selects \(\widehat{\mathcal H}_n=\{k:N^{(0)}_k>256L\}\), and since the pilot counts \(N^{(0)}_k\) are integers this is the same set as \(\{k:N^{(0)}_k>\lfloor256L\rfloor\}\), i.e.\ exactly the pilot heavy set \(\widehat{\mathcal H}\) at threshold \(t=256\) appearing in \(\mathcal B\).  Hence, off \(\mathcal B\),
\[
p_k\ge B_-=\frac{256L}{2m_0}>0
\qquad\text{for every }k\in\widehat{\mathcal H}_n .
\]
% lean: heavyCells_eq_pilotHeavyAt_of_cutoff_le, heavy_cell_mass_lower_of_good_pilot

\textbf{Step 3: arm decomposition for a deterministic category set.}
For a deterministic \(H\subseteq\{1,\ldots,d\}\) and an arm \(a\in\{0,1\}\), define
\[
\widehat\theta_a(H)=\sum_{k\in H}\frac{N^{(1)}_k}{m_1}\cdot\frac{N^{(1)}_{ak1}}{N^{(1)}_{ak}}\mathbf1\{N^{(1)}_{ak}>0\},
\qquad
\bar\theta_a(H)=\sum_{k\in H}\frac{N^{(1)}_k}{m_1}\mu_{ak},
\qquad
\theta_a(H)=\sum_{k\in H}p_k\mu_{ak}.
\]
By the definition of \(\widehat h_k\) in \cref{obj:def:hybrid-estimator-handle} and by \(\phi(q_k)=p_k(\mu_{1k}-\mu_{0k})\) from Step 1,
\[
\sum_{k\in H}\widehat h_k=\widehat\theta_1(H)-\widehat\theta_0(H),
\qquad
\sum_{k\in H}\phi(q_k)=\theta_1(H)-\theta_0(H),
\]
so \((x_1-x_0)^2\le2x_1^2+2x_0^2\) applied to \(x_a=\widehat\theta_a(H)-\theta_a(H)\) gives, pointwise in the sample,
\[
\left(\sum_{k\in H}\widehat h_k-\sum_{k\in H}\phi(q_k)\right)^2
\le
2\left(\widehat\theta_1(H)-\theta_1(H)\right)^2
+2\left(\widehat\theta_0(H)-\theta_0(H)\right)^2 .
\]
% lean: fixedHeavyArmContribution, fixedEmpiricalMassArm, fixedTargetArm, fixedHeavyContribution_eq_arm_sub, fixedTargetHeavy_eq_arm_sub, fixedHeavy_error_sq_le_two_arms

\textbf{Step 4: the fixed-set arm bound.}
We claim the following \emph{fixed-set arm bound}: for every deterministic \(H\) with \(p_k\ge B_-\) for all \(k\in H\), and every \(a\in\{0,1\}\),
\[
\int\left(\widehat\theta_a(H)-\theta_a(H)\right)^2dP^{\otimes n}
\le
\frac{8\sum_{k\in H}p_k}{m_1\epsilon}
+\frac{6}{m_1}
+4\left(\frac{|H|}{u_n^2B_-}\right)^2,
\qquad
u_n=\frac{m_1-2}{2}\,\epsilon .
\]
Splitting about \(\bar\theta_a(H)\) with \((x+y)^2\le2x^2+2y^2\),
\[
\left(\widehat\theta_a(H)-\theta_a(H)\right)^2
\le
2\left(\widehat\theta_a(H)-\bar\theta_a(H)\right)^2
+2\left(\bar\theta_a(H)-\theta_a(H)\right)^2 .
\]
% lean: fixedHeavyArm_error_sq_le_noise_mass

\emph{(4a) The empirical-mass term.}  Put \(g_{a,H}(O)=\mu_{aX}\mathbf1\{X\in H\}\), a function of one observation with values in \([0,1]\).  Grouping the observations of \(I_1\) by category,
\[
\bar\theta_a(H)=\frac1{m_1}\sum_{i\in I_1}g_{a,H}(O_i),
\qquad
\theta_a(H)=\mathbb E_P\left[g_{a,H}(O)\right],
\]
so \(\bar\theta_a(H)-\theta_a(H)\) is the centered sample mean of an iid bounded score over the \(m_1\) observations of \(I_1\), whence
\[
\int\left(\bar\theta_a(H)-\theta_a(H)\right)^2dP^{\otimes n}
\le
\frac{\mathbb E_P\left[g_{a,H}(O)^2\right]}{m_1}
\le
\frac1{m_1},
\]
the second inequality because \(0\le g_{a,H}\le1\).
% lean: fixedMassScore, fixedMassScore_eq, fixedMassScore_mem_unitInterval, integral_fixedMassScore_eq, fixedEmpiricalMassArm_sub_target_eq_score, integral_fixedEmpiricalMassArm_sub_target_sq_le, iid_mean_sq_le_fintype

\emph{(4b) Exact decomposition of the ratio noise.}  Set
\[
\eta_{ak}(O)=\mathbf1\{X=k,\,A=a\}\left(Y-\mu_{ak}\right),
\qquad
w_{ak}=\frac{N^{(1)}_k}{N^{(1)}_{ak}}\mathbf1\{N^{(1)}_{ak}>0\},
\]
and
\[
R_a(H)=\sum_{k\in H}w_{ak}\sum_{i\in I_1}\eta_{ak}(O_i),
\qquad
\Delta_a(H)=\sum_{k\in H}\mu_{ak}N^{(1)}_k\mathbf1\{N^{(1)}_{ak}=0\}.
\]
Since \(\sum_{i\in I_1}\eta_{ak}(O_i)=N^{(1)}_{ak1}-\mu_{ak}N^{(1)}_{ak}\), a term-by-term comparison of the two cases \(N^{(1)}_{ak}>0\) and \(N^{(1)}_{ak}=0\) gives the pointwise identity
\[
\widehat\theta_a(H)-\bar\theta_a(H)
=
\frac{R_a(H)-\Delta_a(H)}{m_1},
\qquad\text{hence}\qquad
\left(\widehat\theta_a(H)-\bar\theta_a(H)\right)^2
\le
\frac{2R_a(H)^2}{m_1^2}+\frac{2\Delta_a(H)^2}{m_1^2}.
\]
% lean: fixedRatioResidual, tupleRatioCoeff, armOutcomeResidual, fixedMissingOutcomeBias, missingIndexCount, fixedHeavyArm_noise_decomposition

\emph{(4c) The residual term.}  All cross terms in the expansion of \(R_a(H)^2\) integrate to zero: the weights \(w_{ak}\) depend on the sample only through the labels \((X_i,A_i)_{i\in I_1}\), and conditionally on those labels the residuals \(\eta_{ak}(O_i)\) are independent across \(i\) with mean zero, while for \(k\ne l\) the product \(\eta_{ak}(O_i)\eta_{al}(O_i)\) vanishes identically because one observation lies in one category.  Only the diagonal survives, and \(|Y-\mu_{ak}|\le1\) gives \(\eta_{ak}(O_i)^2\le\mathbf1\{X_i=k,A_i=a\}\), so that
\[
\sum_{i\in I_1}w_{ak}^2\,\eta_{ak}(O_i)^2
\le
w_{ak}^2N^{(1)}_{ak}
=
\frac{\left(N^{(1)}_k\right)^2}{N^{(1)}_{ak}}\mathbf1\{N^{(1)}_{ak}>0\}.
\]
The nested-count second-moment identity for a multinomial category and its nested arm, evaluated with the category probability \(p_k\) and arm probability \(\pi_{ak}\) and then bounded using \(\pi_{ak}\ge\epsilon\) from \cref{obj:ass:overlap}, yields
\[
\int\frac{\left(N^{(1)}_k\right)^2}{N^{(1)}_{ak}}\mathbf1\{N^{(1)}_{ak}>0\}\,dP^{\otimes n}
\le
\frac{2m_1p_k}{\pi_{ak}}
\le
\frac{2m_1p_k}{\epsilon},
\]
which requires \(p_k>0\) and is available here since \(p_k\ge B_->0\) on \(H\).  Summing over \(k\in H\),
\[
\int R_a(H)^2\,dP^{\otimes n}
\le
\frac{2m_1}{\epsilon}\sum_{k\in H}p_k .
\]
% lean: integral_fixedRatioResidual_sq_eq_diagonal, integral_ratioResidual_diagonal_le, ate_nested_count_ratio_integral_le, integral_fixedRatioResidual_sq_le

\emph{(4d) The missing-arm term.}  Since \(0\le\mu_{ak}\le1\),
\[
|\Delta_a(H)|\le\mathcal N_a(H):=\sum_{k\in H}N^{(1)}_k\mathbf1\{N^{(1)}_{ak}=0\},
\qquad\text{so}\qquad
\Delta_a(H)^2\le\mathcal N_a(H)^2 .
\]
The second moment of \(\mathcal N_a(H)\) is an exact multinomial expression in the masses \(p_k\) and arm probabilities \(\pi_{ak}\): its diagonal part is
\(m_1p_k(1-\pi_{ak})(1-p_k\pi_{ak})^{m_1-1}+m_1(m_1-1)\left\{p_k(1-\pi_{ak})\right\}^2(1-p_k\pi_{ak})^{m_1-2}\)
and its \(k\ne l\) part is
\(m_1(m_1-1)p_k(1-\pi_{ak})p_l(1-\pi_{al})(1-p_k\pi_{ak}-p_l\pi_{al})^{m_1-2}\).
Bounding \(1-\pi_{ak}\le1\), \(m_1(m_1-1)\le m_1^2\), and \((1-p_k\pi_{ak})^{m_1-2}\le e^{-(m_1-2)\epsilon p_k}\) by \(\pi_{ak}\ge\epsilon\), and writing \(f_k=p_ke^{-u_np_k}\) with \(u_n=(m_1-2)\epsilon/2\), the diagonal is at most \(m_1p_k+m_1^2f_k^2\) and each cross term at most \(m_1^2f_kf_l\), so that
\[
\int\mathcal N_a(H)^2\,dP^{\otimes n}
\le
m_1\sum_{k\in H}p_k+m_1^2\left(\sum_{k\in H}f_k\right)^2
\le
m_1+m_1^2\left(\sum_{k\in H}f_k\right)^2 ,
\]
using \(\sum_{k\in H}p_k\le1\).  Finally \(s^2\le e^{s}\) for \(s\ge0\), applied with \(s=u_np_k\), gives \(f_k=p_ke^{-u_np_k}\le1/(u_n^2p_k)\le1/(u_n^2B_-)\) for each \(k\in H\), whence
\[
\sum_{k\in H}f_k\le\frac{|H|}{u_n^2B_-},
\qquad\text{and so}\qquad
\int\Delta_a(H)^2\,dP^{\otimes n}
\le
m_1+m_1^2\left(\frac{|H|}{u_n^2B_-}\right)^2 .
\]
% lean: fixedMissingCount, abs_fixedMissingOutcomeBias_le, fixedMissingOutcomeBias_sq_le, integral_fixedMissingCount_sq_eq, missing_diag_envelope, missing_cross_envelope, sum_mass_mul_exp_neg_mul_le, integral_fixedMissingCount_sq_le

\emph{(4e) Assembling the fixed-set arm bound.}  Integrating the pointwise bound of (4b) and inserting (4c) and (4d),
\begin{align*}
\int\left(\widehat\theta_a(H)-\bar\theta_a(H)\right)^2dP^{\otimes n}
&\le
\frac{2}{m_1^2}\cdot\frac{2m_1}{\epsilon}\sum_{k\in H}p_k
+\frac{2}{m_1^2}\left\{m_1+m_1^2\left(\frac{|H|}{u_n^2B_-}\right)^2\right\}\\
&=
\frac{4\sum_{k\in H}p_k}{m_1\epsilon}+\frac2{m_1}+2\left(\frac{|H|}{u_n^2B_-}\right)^2 .
\end{align*}
Doubling this, adding twice the bound \(1/m_1\) of (4a), and collecting terms gives exactly the fixed-set arm bound.  (Note \(m_1\ge4>2\), so \(u_n>0\).)
% lean: integral_fixedHeavyArm_noise_sq_le, integral_fixedHeavyArm_error_sq_le

\textbf{Step 5: the fixed-set envelope is at the target rate.}
Write
\[
V=\kappa_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}.
\]
We claim that the right-hand side of the fixed-set arm bound is at most \(V\) for every \(H\).  Indeed:
\[
\frac{8\sum_{k\in H}p_k}{m_1\epsilon}\le\frac{8}{m_1\epsilon}\le\frac{16}{\epsilon n},
\qquad
\frac{6}{m_1}\le\frac{12}{n},
\]
using \(\sum_{k\in H}p_k\le1\) and \(m_1\ge n/2\).  For the third term, \(2m_0\le n\) gives
\[
B_-=\frac{256L}{2m_0}\ge\frac{256L}{n},
\qquad
u_n=\frac{m_1-2}{2}\epsilon\ge\frac{n\epsilon}{8}
\quad\text{(from }4(m_1-2)\ge n\text{)},
\]
hence \(u_n^2B_-\ge(n\epsilon/8)^2\cdot256L/n=4n\epsilon^2L\).  With \(|H|\le d\) and \(0<\log n\le L\),
\[
4\left(\frac{|H|}{u_n^2B_-}\right)^2
\le
4\left(\frac{d}{4n\epsilon^2L}\right)^2
=
\frac{1}{4\epsilon^4}\cdot\frac{d^2}{n^2L^2}
\le
\frac1{\epsilon^4}\cdot\frac{d^2}{n^2(\log n)^2}.
\]
Adding the three bounds,
\[
\frac{8\sum_{k\in H}p_k}{m_1\epsilon}+\frac{6}{m_1}+4\left(\frac{|H|}{u_n^2B_-}\right)^2
\le
\left(\frac{16}\epsilon+12\right)\frac1n+\frac1{\epsilon^4}\cdot\frac{d^2}{n^2(\log n)^2}
\le
\kappa_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}=V,
\]
the last step because both \(16/\epsilon+12\) and \(1/\epsilon^4\) are at most \(\kappa_\epsilon\) and both rate terms are nonnegative.  In particular \(V\ge0\).
% lean: heavy_fixed_envelope_rate, minimaxRate

\textbf{Step 6: transfer to the random heavy set.}
Fix an arm \(a\) and let \(\mathfrak E=\{H\subseteq\{1,\ldots,d\}:p_k\ge B_-\text{ for all }k\in H\}\).  By Step 2, off \(\mathcal B\) the realized set \(\widehat{\mathcal H}_n\) belongs to \(\mathfrak E\), so pointwise
\[
\mathbf1_{\mathcal B^c}\left(\widehat\theta_a(\widehat{\mathcal H}_n)-\theta_a(\widehat{\mathcal H}_n)\right)^2
\le
\sum_{H\in\mathfrak E}\mathbf1\{\widehat{\mathcal H}_n=H\}\left(\widehat\theta_a(H)-\theta_a(H)\right)^2 .
\]
For each fixed \(H\), the event \(\{\widehat{\mathcal H}_n=H\}\) is a function of the pilot observations \((O_i)_{i\in I_0}\) alone, while \(\left(\widehat\theta_a(H)-\theta_a(H)\right)^2\) is a function of the estimation observations \((O_i)_{i\in I_1}\) alone; under \(P^{\otimes n}\) these two disjoint index blocks are independent, so the fiber integral factorizes,
\[
\int\mathbf1\{\widehat{\mathcal H}_n=H\}\left(\widehat\theta_a(H)-\theta_a(H)\right)^2dP^{\otimes n}
=
\Pr\nolimits_{P^{\otimes n}}\!\left(\widehat{\mathcal H}_n=H\right)
\int\left(\widehat\theta_a(H)-\theta_a(H)\right)^2dP^{\otimes n}.
\]
Every \(H\in\mathfrak E\) satisfies the eligibility hypothesis of Step 4, so by Steps 4 and 5 the second factor is at most \(V\).  Summing over \(H\in\mathfrak E\) and using \(\sum_{H\in\mathfrak E}\Pr(\widehat{\mathcal H}_n=H)\le1\) together with \(V\ge0\),
\[
\int_{\mathcal B^c}\left(\widehat\theta_a(\widehat{\mathcal H}_n)-\theta_a(\widehat{\mathcal H}_n)\right)^2dP^{\otimes n}\le V
\qquad (a\in\{0,1\}).
\]
% lean: infinite_fixedHeavyArm_fiber_factorization, oneShot_integral_estimate_restrict_pilot, heavy_good_arm_setIntegral_le

\textbf{Step 7: the pilot-good integral.}
Applying the pointwise bound of Step 3 with the random set \(H=\widehat{\mathcal H}_n\) — legitimate because that bound holds for every realized set — and integrating over \(\mathcal B^c\),
\[
\int_{\mathcal B^c}\mathcal D^2\,dP^{\otimes n}
\le
2\int_{\mathcal B^c}\left(\widehat\theta_1(\widehat{\mathcal H}_n)-\theta_1(\widehat{\mathcal H}_n)\right)^2dP^{\otimes n}
+2\int_{\mathcal B^c}\left(\widehat\theta_0(\widehat{\mathcal H}_n)-\theta_0(\widehat{\mathcal H}_n)\right)^2dP^{\otimes n}
\le
4V .
\]
% lean: fixedHeavy_error_sq_le_two_arms, heavyContribution_eq_fixed, targetHeavy_eq_fixed

\textbf{Step 8: conclusion.}
Since \(n\ge1\),
\[
n^{-4}\le n^{-1}\le\frac1n+\frac{d^2}{n^2(\log n)^2},
\]
so combining Step 1 with Step 7 in the decomposition of Step 0,
\[
\int\mathcal D^2\,dP^{\otimes n}
\le
C_{\mathrm{bad}}n^{-4}+4V
\le
\left(C_{\mathrm{bad}}+4\kappa_\epsilon\right)\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}
=
C_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\},
\]
which, since \(\mu_n=P^{\otimes n}\), is the asserted bound.
% lean: universal_heavy_cell_rate
\end{proof}
