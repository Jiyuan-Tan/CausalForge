\begin{proof}[Proof of \cref{obj:lem:combined-upper-envelope}]
Fix \(n,d\in\mathbb N\) with \(0<n\), \(N_\epsilon\le n\) and \(d\le\rho_\epsilon n\log n\), and abbreviate
\[
a=\frac1n,\qquad
u=\frac{d^2}{n^2(\log n)^2},\qquad
v=\left(\frac12-\epsilon\right)^2,\qquad
K_\epsilon=\max\{C_\epsilon,4\},
\]
so that the hybrid rate is \(a+u\) and the asserted envelope is \(K_\epsilon\{a+\min(u,v)\}\). Because \(n>0\), we have \(a>0\); \(u\ge0\) as a ratio of squares; and \(v\ge0\) as a square. Also \(C_\epsilon\le K_\epsilon\) and \(4\le K_\epsilon\) by the definition of the maximum.

\textbf{Step 1: the centered estimator.} With \(\widehat\tau_{\mathrm{ctr}}\) as in \cref{obj:def:centered-estimator},
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau_{\mathrm{ctr}}-\tau(P)\right)^2\right]
\le a+4v .
\]
If \(\mathcal E_{n,d,\epsilon}\) is empty the supremum is \(0\), and \(0\le a+4v\) because \(a>0\) and \(v\ge0\). Otherwise, for each \(P^{\otimes n}\in\mathcal E_{n,d,\epsilon}\), \cref{obj:lem:near-randomization-linear-upper}, applied with \(\mu_n=P^{\otimes n}\) and the overlap constant \(\epsilon\), gives
\[
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau_{\mathrm{ctr}}-\tau(P)\right)^2\right]
\le
\frac1n+4\left(\frac12-\epsilon\right)^2
= a+4v ,
\]
and taking the supremum over the class preserves the bound.
% lean: near_randomization_linear_upper, centeredEstimator, worstCaseMSE

\textbf{Step 2: the selected estimator.} We claim
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}-\tau(P)\right)^2\right]
\le
\min\left\{C_\epsilon(a+u),\,a+4v\right\}.
\]
By \cref{obj:def:selected-estimator} the selector splits on the comparison \(C_\epsilon(a+u)\le a+4v\), so we treat the two cases separately.

If \(C_\epsilon(a+u)\le a+4v\), then \(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}=\widehat\tau_n^{\mathrm{hyb}}\), and the same comparison identifies the minimum,
\[
\min\left\{C_\epsilon(a+u),\,a+4v\right\}=C_\epsilon(a+u).
\]
Since \(0<n\), \(N_\epsilon\le n\) and \(d\le\rho_\epsilon n\log n\), the assumed hybrid bound applies and yields
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau_n^{\mathrm{hyb}}-\tau(P)\right)^2\right]
\le
C_\epsilon\left\{\frac1n+\frac{d^2}{n^2(\log n)^2}\right\}
=C_\epsilon(a+u),
\]
which is the claim in this case.

If instead \(C_\epsilon(a+u)>a+4v\), then \(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}=\widehat\tau_{\mathrm{ctr}}\), and the reversed comparison \(a+4v\le C_\epsilon(a+u)\) identifies the minimum,
\[
\min\left\{C_\epsilon(a+u),\,a+4v\right\}=a+4v,
\]
so the claim is exactly the bound of Step 1.
% lean: selectedEstimator, hHybrid, min_eq_left, min_eq_right

\textbf{Step 3: comparison with the minimax risk.} The sample \(O_1,\ldots,O_n\) takes values in the finite set \((\{1,\ldots,d\}\times\{0,1\}^2)^n\), so every real-valued function of it, in particular \(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}\), is measurable and is therefore one of the candidates over which the infimum in \cref{obj:def:minimax-risk} is taken. That infimum is over a set of reals bounded below by \(0\): for a fixed measurable \(\widehat\tau\), each \(\mathbb E_{P^{\otimes n}}[(\widehat\tau-\tau(P))^2]\) is nonnegative as the integral of a square, and the supremum over an empty class is \(0\). Hence
\[
\mathsf R_{n,d,\epsilon}
\le
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}-\tau(P)\right)^2\right].
\]
% lean: minimaxRisk, measurable_of_finite, ciInf_le

\textbf{Step 4: the minimum is below the envelope.} It remains to show
\[
\min\left\{C_\epsilon(a+u),\,a+4v\right\}
\le
K_\epsilon\{a+\min(u,v)\}.
\]
Suppose first \(u\le v\), so \(\min(u,v)=u\). Since \(a+u\ge0\) and \(C_\epsilon\le K_\epsilon\),
\[
\min\left\{C_\epsilon(a+u),\,a+4v\right\}
\le C_\epsilon(a+u)
\le K_\epsilon(a+u)
= K_\epsilon\{a+\min(u,v)\}.
\]
Suppose instead \(v<u\), so \(\min(u,v)=v\). From \(4\le K_\epsilon\) we get both \(a\le K_\epsilon a\) (as \(1\le4\le K_\epsilon\) and \(a\ge0\)) and \(4v\le K_\epsilon v\) (as \(v\ge0\)); adding the two and factoring gives
\[
a+4v\le K_\epsilon a+K_\epsilon v=K_\epsilon(a+v),
\]
whence
\[
\min\left\{C_\epsilon(a+u),\,a+4v\right\}
\le a+4v
\le K_\epsilon(a+v)
= K_\epsilon\{a+\min(u,v)\}.
\]
% lean: le_max_left, le_max_right, mul_le_mul_of_nonneg_right, min_le_left, min_le_right

Chaining Step 2 with Step 4 and substituting back the definitions of \(a,u,v\) gives
\[
\sup_{P^{\otimes n}\in\mathcal E_{n,d,\epsilon}}
\mathbb E_{P^{\otimes n}}\!\left[\left(\widehat\tau^{\mathrm{sel}}_{C_\epsilon,\epsilon}-\tau(P)\right)^2\right]
\le
K_\epsilon\left[
\frac1n+
\min\left\{\frac{d^2}{n^2(\log n)^2},\left(\frac12-\epsilon\right)^2\right\}
\right],
\]
which is the asserted bound; combined with Step 3 it also bounds \(\mathsf R_{n,d,\epsilon}\) by the same envelope.
% lean: combined_upper_envelope

\end{proof}
