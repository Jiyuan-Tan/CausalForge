\begin{proof}[Proof of \cref{obj:lem:factorial-product-moment}]
Let \(\mathcal C=\{0,1\}\times\{0,1\}\), let \(m=m_1\), and write
\[
q_{ay}=q_{aky},\qquad r_{ay}=r(a,y),\qquad s_{ay}=s(a,y),
\qquad |r|=\sum_{(a,y)\in\mathcal C}r_{ay},\quad |s|=\sum_{(a,y)\in\mathcal C}s_{ay}.
\]
Adjoin one extra label \(\varnothing\) for observations outside category \(k\).  For the \(j\)-th observation in the estimation split define
\[
Z_j=
\begin{cases}
(a,y),& (X_j,A_j,Y_j)=(k,a,y),\\
\varnothing,& X_j\ne k,
\end{cases}
\qquad
\alpha_{ay}=q_{ay},\quad
\alpha_{\varnothing}=1-\sum_{(a,y)\in\mathcal C}q_{ay}.
\]
The variables \(Z_1,\ldots,Z_m\) are iid on \(\mathcal C\cup\{\varnothing\}\) with masses \(\alpha\).  Extend \(r,s\) by setting \(r_{\varnothing}=s_{\varnothing}=0\).  If \(N_b=\#\{j:Z_j=b\}\), then
\[
\mathfrak m_{k,r}
=
\frac{\prod_{b\in\mathcal C\cup\{\varnothing\}}(N_b)_{r_b}}{(m)_{|r|}},
\qquad
\mathfrak m_{k,s}
=
\frac{\prod_{b\in\mathcal C\cup\{\varnothing\}}(N_b)_{s_b}}{(m)_{|s|}} .
\]
This is just the split-count notation of \cref{obj:def:sample-splits}, with the outside-category exponent equal to zero. % lean: factorialMonomial_trunc_eq_iidCount

For an overlap choice
\[
H=(H_b)_{b\in\mathcal C\cup\{\varnothing\}},
\qquad
0\le H_b\le \min\{r_b,s_b\},
\]
set
\[
|H|=\sum_b H_b,\qquad
t_b(H)=r_b+s_b-H_b,\qquad
A_H=\prod_b {r_b\choose H_b}{s_b\choose H_b}H_b! .
\]
The cellwise falling-factorial product identity gives
\[
\prod_b (N_b)_{r_b}\prod_b (N_b)_{s_b}
=
\sum_H
A_H\prod_b (N_b)_{t_b(H)} .
\]
% lean: multinomialFactorialCount_mul

Taking expectations and using the factorial moment formula for multinomial counts,
\[
\mathbb E\!\left[
\mathfrak m_{k,r}\mathfrak m_{k,s}
\right]
=
\sum_H
A_H
\frac{(m)_{|r|+|s|-|H|}}{(m)_{|r|}(m)_{|s|}}
\prod_b \alpha_b^{\,r_b+s_b-H_b}.
\]
% lean: integral_normalized_multinomialFactorialCount_mul_sample

The normalization ratio satisfies, for every such \(H\),
\[
\frac{(m)_{|r|+|s|-|H|}}{(m)_{|r|}(m)_{|s|}}
\le
\frac{\mathrm e}{m^{|H|}} .
\]
Indeed, if \(|s|=0\) then also \(|r|=0\), and the bound is immediate.  Otherwise put \(u=m-|s|\).  The assumptions \(4|s|^2\le m\) and \(|r|\le |s|\) imply \(u>0\) and \(2|s|^2\le u\).  Also
\[
(m)_{|r|+|s|-|H|}\le m^{|r|+|s|-|H|},\qquad
(m)_{|r|}\ge u^{|r|},\qquad
(m)_{|s|}\ge u^{|s|}.
\]
Hence the ratio is at most
\[
\frac{m^{|r|+|s|-|H|}}{u^{|r|+|s|}}
=
\frac{(m/u)^{|r|+|s|}}{m^{|H|}} .
\]
Since \(m/u=1+|s|/u\),
\[
(|r|+|s|)\log(m/u)
\le
(|r|+|s|)\frac{|s|}{u}
\le 1,
\]
and therefore \((m/u)^{|r|+|s|}\le \mathrm e\). % lean: factorial_ratio_bound

Because all masses \(\alpha_b\) are nonnegative,
\[
\mathbb E\!\left[
\mathfrak m_{k,r}\mathfrak m_{k,s}
\right]
\le
\mathrm e
\sum_H
A_H\,m^{-|H|}
\prod_b \alpha_b^{\,r_b+s_b-H_b}.
\]
% lean: normalized_multinomial_joint_moment_bound

The remaining overlap sum factors by coordinates:
\[
\sum_H
A_H\,m^{-|H|}
\prod_b \alpha_b^{\,r_b+s_b-H_b}
=
\prod_b
\sum_{h=0}^{\min(r_b,s_b)}
{r_b\choose h}{s_b\choose h}h!\,m^{-h}
\alpha_b^{\,r_b+s_b-h}.
\]
% lean: overlap_sum_eq_prod

For each \(b\) and each \(0\le h\le\min\{r_b,s_b\}\), one has \(h\le s_b\) and
\[
{r_b\choose h}h!=(r_b)_h\le r_b^h\le |r|^h\le |s|^h .
\]
The assumption \(m_1>0\) gives \(m>0\), and \(\alpha_b\ge0\).  After rewriting
\(\alpha_b^{r_b+s_b-h}=\alpha_b^{r_b}\alpha_b^{s_b-h}\), the preceding coefficient bound gives
\[
{r_b\choose h}{s_b\choose h}h!\,m^{-h}
\alpha_b^{\,r_b+s_b-h}
\le
\alpha_b^{r_b}{s_b\choose h}
\left(\frac{|s|}{m}\right)^h
\alpha_b^{s_b-h}.
\]
The right-hand summands are nonnegative for every \(0\le h\le s_b\), so enlarging the summation range and applying the binomial theorem yields
\[
\sum_{h=0}^{\min(r_b,s_b)}
{r_b\choose h}{s_b\choose h}h!\,m^{-h}
\alpha_b^{\,r_b+s_b-h}
\le
\alpha_b^{r_b}
\sum_{h=0}^{s_b}{s_b\choose h}
\left(\frac{|s|}{m}\right)^h
\alpha_b^{s_b-h}
=
\alpha_b^{r_b}
\left(\alpha_b+\frac{|s|}{m}\right)^{s_b}.
\]
% lean: cell_overlap_sum_le

Multiplying these coordinatewise bounds gives
\[
\mathbb E\!\left[
\mathfrak m_{k,r}\mathfrak m_{k,s}
\right]
\le
\mathrm e
\prod_{b}\alpha_b^{r_b}
\prod_b\left(\alpha_b+\frac{|s|}{m}\right)^{s_b}.
\]
The \(\varnothing\)-coordinate contributes \(1\), since \(r_{\varnothing}=s_{\varnothing}=0\).  Replacing \(b\) by \((a,y)\in\mathcal C\) and \(m\) by \(m_1\) yields
\[
\mathbb E_{P^{\otimes n}}\!\left[\mathfrak m_{k,r}\mathfrak m_{k,s}\right]
\le
\mathrm e
\prod_{(a,y)}q_{aky}^{\,r_{ay}}
\prod_{(a,y)}
\left(q_{aky}+\frac{|s|}{m_1}\right)^{s_{ay}} .
\]
% lean: integral_factorialMonomial_mul_trunc_le
\end{proof}
