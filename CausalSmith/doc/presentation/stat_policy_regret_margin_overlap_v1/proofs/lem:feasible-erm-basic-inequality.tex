\begin{proof}[Proof of \cref{obj:lem:feasible-erm-basic-inequality}]
Throughout, fix a realized sample and abbreviate the observed foldwise scores and the empirical criterion of \cref{obj:def:feasible-erm} by
\[
\Gamma_i=\Gamma_q\!\left(O_i;\widehat\eta^{(-k(i))}\right),
\qquad
\widehat V_{n,q}(\pi)=\frac1n\sum_{i=1}^n\mathbf 1\{\pi(X_i)=1\}\,\Gamma_i .
\]
Note that \(\widehat V_{n,q}(\pi)\) depends on \(\pi\) only through the finitely many values \(\pi(X_1),\ldots,\pi(X_n)\).

\emph{Step 1: a \(1/n\)-near maximizer over the skeleton exists.}  Since each indicator is \(0\) or \(1\),
\[
\widehat V_{n,q}(\pi_j)
=\frac1n\sum_{i=1}^n\mathbf 1\{\pi_j(X_i)=1\}\,\Gamma_i
\le\frac1n\sum_{i=1}^n|\Gamma_i|
\qquad\text{for every }j\ge1,
\]
so the set of skeleton scores \(\{\widehat V_{n,q}(\pi_j):j\ge1\}\) is nonempty and bounded above; write \(S=\sup_{j\ge1}\widehat V_{n,q}(\pi_j)\), which is finite.  As \(1/n>0\) we have \(S-1/n<S\), so by the definition of a supremum there exists an index \(j\) with \(\widehat V_{n,q}(\pi_j)>S-1/n\), that is,
\[
\widehat V_{n,q}(\pi_{j'})\le \widehat V_{n,q}(\pi_j)+\frac1n
\qquad\text{for every }j'\ge1 .
\]
The set of such indices is therefore a nonempty set of positive integers and has a least element, which is the index \(j_n\) selected in \cref{obj:def:feasible-erm}; consequently
\[
\widehat V_{n,q}(\pi_{j'})\le \widehat V_{n,q}(\widehat\pi_n)+\frac1n
\qquad\text{for every }j'\ge1 .
\]
% lean: feasibleERM_nearSet_nonempty

\emph{Step 2: the estimator is \(\Pi\)-valued and measurable.}  Every enumerated policy \(\pi_j\) belongs to \(\Pi\) by \cref{obj:ass:policy-class}, so \(\widehat\pi_n=\pi_{j_n}\in\Pi\).  For measurability, the event \(\{j_n=j\}\) is exactly the event that \(j\) is a \(1/n\)-near maximizer and no smaller index is:
\[
\{j_n=j\}
=
N_j\cap\left(\text{complement of }N_1\right)\cap\ldots\cap\left(\text{complement of }N_{j-1}\right),
\qquad
N_k=\left\{\widehat V_{n,q}(\pi_{j'})\le \widehat V_{n,q}(\pi_k)+\tfrac1n\ \text{ for every }j'\ge1\right\}.
\]
Each score \(\widehat V_{n,q}(\pi_j)\) is a measurable function of the sample, because \(\pi_j\) and the plug-in nuisance functions are measurable; hence each \(N_k\) is a countable intersection of measurable score comparisons, and \(\{j_n=j\}\) is a finite intersection of such sets and their complements, so it is measurable for every \(j\).  Since the index set is countable, \(j_n\) is a measurable integer-valued map, and therefore the map from the sample to \(R_P(\widehat\pi_n)=R_P(\pi_{j_n})\) is measurable, so the risk of \(\widehat\pi_n\) is a well-defined integral. % lean: feasible_erm_basic_inequality

\emph{Step 3: from the skeleton to every comparator in \(\Pi\).}  Let \(\pi^b\in\Pi\).  By \cref{obj:ass:policy-class} there is a sequence \((\pi_{j_t})_{t\ge1}\) from \(\Pi_0\) converging pointwise to \(\pi^b\).  Convergence of a \(\{0,1\}\)-valued sequence at a point means it is eventually equal to the limit there, so for each of the \(n\) covariate values \(X_1,\ldots,X_n\) there is an index beyond which the values agree; taking the largest of these \(n\) indices gives a single \(t\) with
\[
\pi_{j_t}(X_i)=\pi^b(X_i)\qquad\text{for all }i=1,\ldots,n .
\]
Since \(\widehat V_{n,q}\) depends on the policy only through those \(n\) values,
\[
\widehat V_{n,q}(\pi_{j_t})=\widehat V_{n,q}(\pi^b).
\]
Applying the near-maximality of Step 1 to the index \(j_t\) gives
\[
\widehat V_{n,q}(\pi^b)
=\widehat V_{n,q}(\pi_{j_t})
\le
\widehat V_{n,q}(\widehat\pi_n)+\frac1n .
\]
As \(\pi^b\in\Pi\) was arbitrary, taking the supremum over \(\Pi\) gives the first assertion,
\[
\widehat V_{n,q}(\widehat\pi_n)\ge\sup_{\pi\in\Pi}\widehat V_{n,q}(\pi)-\frac1n .
\]
% lean: feasible_erm_basic_inequality

\emph{Step 4: the comparator form.}  For any two policies, the definition of \(\widehat V_{n,q}\) gives the identity
\[
\widehat V_{n,q}(\widehat\pi_n)-\widehat V_{n,q}(\pi^b)
=
\frac1n\sum_{i=1}^n\left\{\mathbf 1\{\widehat\pi_n(X_i)=1\}-\mathbf 1\{\pi^b(X_i)=1\}\right\}\Gamma_i
=
P_n\!\left[(\widehat\pi_n-\pi^b)\Gamma_q\right],
\]
so the inequality of Step 3 is exactly
\[
P_n\!\left[(\widehat\pi_n-\pi^b)\Gamma_q\right]\ge -\frac1n .
\]
% lean: feasible_erm_basic_inequality
Finally, under \cref{obj:ass:optimal-in-class} the oracle policy satisfies \(\pi^\star_P\in\Pi\), so the comparator inequality applies with \(\pi^b=\pi^\star_P\). % lean: feasible_erm_basic_inequality
\end{proof}
